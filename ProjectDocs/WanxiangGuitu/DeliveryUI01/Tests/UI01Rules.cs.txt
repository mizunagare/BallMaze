using System;using System.IO;using System.Linq;using System.Reflection;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;
public static class UI01Rules {
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")??throw new Exception("explicit root");static CombatSession s;static PlayerHudPresenter hud;static InteractionResolver r;static ForestRegionRuntime region;static Keyboard kb;static int phase,readyFrame;static int weaponIndex=-1;static double deadline;static LootPickup far,close;static string expected;static float remaining;static int samples;static double timing;static int scanStart;static float sampleAt;
 static void Need(bool b,string why){if(!b)throw new Exception(why);}static void Mark(string v){File.AppendAllText(Path.Combine(Root,"ui01-results.txt"),v+"\n");Debug.Log("UI01_"+v);}static void Step(){phase++;InputSystem.QueueStateEvent(kb,new KeyboardState());}
 public static void Run(){deadline=EditorApplication.timeSinceStartup+180;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void AtFerry(){var x=s.Flow.ExitConfigs().Single(e=>e.repairRequired);s.hero.RestorePosition(x.position+Vector3.back*1.5f,0);s.orbit.RestoreView(-90,28);foreach(var e in s.enemies)e.enabled=false;}
 static void NextRun(){s.Flow.OpenAbandon();Need(s.Flow.ConfirmAbandon()&&s.Flow.CommitSettlement()&&s.Flow.AcknowledgeSettlement(),"controlled abandon");Need(s.Flow.Depart(AdventureMode.ForestExploration),"next departure");AtFerry();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("UI01 timeout phase="+phase);if(!EditorApplication.isPlaying||Time.time<.3f&&s==null)return;if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(Path.Combine(Root,"ui01-save"),false);region=s.GetComponent<ForestRegionRuntime>()??s.gameObject.AddComponent<ForestRegionRuntime>();region.LayoutFactory=_=>ForestLayout.Generate(418);s.hero.Inventory.Add(MaterialKind.Wood,2);s.hero.Inventory.Add(MaterialKind.Stone,1);Need(s.Flow.Depart(AdventureMode.ForestExploration),"UI departure");AtFerry();hud=s.GetComponent<PlayerHudPresenter>();r=s.hero.Inventory.Interactions;kb=InputSystem.AddDevice<Keyboard>();readyFrame=Time.frameCount+3;InputSystem.settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;InputSystem.settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;far=LootPickup.Spawn(MaterialKind.Wood,1,s.hero.transform.position+Vector3.right*1.9f+Vector3.up*.5f);expected=s.Flow.CurrentExit.id;return;}
 var b=s.hero.Inventory;var f=s.Flow;
 if(phase==0){if(Time.frameCount<readyFrame||hud.VisibleEId==null)return;Need(hud.VisibleEId==expected&&hud.VisiblePrompt.Contains("修复")&&hud.VisiblePrompt.Contains("2/2"),"display does not show actual paid nearest action");InputSystem.QueueStateEvent(kb,new KeyboardState(Key.E,Key.H));phase++;return;}
 if(phase==1){if(!f.CurrentExit.repaired)return;Need(r.LastExecutedId==expected&&b.Count(MaterialKind.Wood)==0&&f.ActiveRun.phase==RunPhase.Running,"E/H duplicate or wrong target charged");Mark("U03_VISIBLE_PAID_EH_TARGET_REAL_INPUT_ONCE_PASS");close=LootPickup.Spawn(MaterialKind.Wood,1,s.hero.transform.position+Vector3.up*.5f+Vector3.right*.05f);Step();return;}
 if(phase==2){if(hud.VisibleEId!=close.InteractionId)return;Need(hud.VisibleHId==expected&&hud.VisiblePrompt.Contains("木材"),"E loot/H exit prompt");InputSystem.QueueStateEvent(kb,new KeyboardState(Key.E));phase++;return;}
 if(phase==3){if(!close.Collected)return;Need(f.ActiveRun.phase==RunPhase.Running&&r.LastExecutedId==close.InteractionId,"E unexpectedly executed H");Mark("U02_U04_VISIBLE_E_LOOT_H_EXIT_REAL_INPUT_PASS");InputSystem.QueueStateEvent(kb,new KeyboardState(Key.H));phase++;return;}
 if(phase==4){if(f.ActiveRun.phase!=RunPhase.Extracting)return;close=LootPickup.Spawn(MaterialKind.Wood,1,s.hero.transform.position+Vector3.up*.5f+Vector3.right*.05f);Step();return;}
 if(phase==5){if(hud.VisibleEId!=close.InteractionId)return;Need(hud.VisibleSecondary.Contains("[E]")&&hud.VisibleSecondary.Contains("木材")&&hud.VisibleSecondary.Contains("[H]"),"extraction hides nearest E action");InputSystem.QueueStateEvent(kb,new KeyboardState(Key.E));phase=51;return;}
 if(phase==51){if(!close.Collected)return;Need(f.ActiveRun.phase==RunPhase.Extracting&&r.LastExecutedId==close.InteractionId,"displayed E during extraction executed wrong action");remaining=f.ActiveRun.extractionRemaining;hud.SetDetails("帮助");InputSystem.QueueStateEvent(kb,new KeyboardState());phase=50;return;}
 if(phase==50){Need(Mathf.Abs(remaining-f.ActiveRun.extractionRemaining)<.05f&&!r.ExecuteDisplayed(true),"panel input/pause leaked");hud.SetDetails("");InputSystem.QueueStateEvent(kb,new KeyboardState());phase=6;return;}

 if(phase==6){s.hero.Health.Damage(9);Need(f.ActiveRun.phase==RunPhase.Running&&f.CurrentExit.repaired,"hit did not cancel channel or lost repair");Mark("U07_ACTUAL_DAMAGE_CANCEL_AND_HELP_PAUSE_PASS");NextRun();b.Add(MaterialKind.Wood,2);b.Add(MaterialKind.Stone,1);close=LootPickup.Spawn(MaterialKind.Wood,1,s.hero.transform.position+Vector3.up*.5f+Vector3.right*.05f);r.Refresh();r.Present();var token=r.DisplayedE;Need(close.PickUp(b),"stale pickup setup");int wood=b.Count(MaterialKind.Wood);Need(!r.ExecuteDisplayed(false)&&b.Count(MaterialKind.Wood)==wood&&!f.CurrentExit.repaired,"expired E switched to fee");Mark("U02_STALE_VISIBLE_TARGET_NEVER_SWITCHES_TO_PAID_ACTION_PASS");Step();return;}
 if(phase==7){Need(s.hero.Dodge.TryDodge(s.hero.transform.forward),"rolling prompt setup");r.Refresh();string rolling=(string)typeof(PlayerHudPresenter).GetMethod("OfferText",BindingFlags.NonPublic|BindingFlags.Instance).Invoke(hud,new object[]{r.E,"E"});Need(rolling.Contains("翻滚")&&!rolling.Contains("缺材料"),"ready materials mislabelled when rolling");s.hero.Dodge.ResetDodge();s.Saves.WriteValidationHook=p=>{if(p=="before-publish")throw new IOException("UI01 before-publish");};Need(!f.RepairExit(f.CurrentExit.id),"prepublish accepted");s.Saves.WriteValidationHook=null;b.NotifyPickup(MaterialKind.Wood,1);b.NotifyPickup(MaterialKind.Wood,2);Need(b.Notifications.Text.Contains("+3"),"pickup merging");Step();return;}
 if(phase==8){Need(hud.CriticalMessage.Contains("修复未生效")&&s.Saves.SaveFailed&&!f.CurrentExit.repaired,"save error hidden or fake repaired");s.Saves.WriteValidationHook=p=>{if(p=="after-publish")throw new IOException("UI01 after-publish callback");};Need(f.RepairExit(f.CurrentExit.id),"published repair recovery");s.Saves.WriteValidationHook=null;Need(!s.Saves.SaveFailed&&f.CurrentExit.repaired,"published confirmed still reported as failed");Mark("U05_U06_U11_FAILURE_PRIORITY_ROLLBACK_PUBLISHED_CONFIRM_PASS");s.hero.GetComponent<AdventureCombat>().RestoreCooldown(4);s.hero.GetComponent<AdventureLoadout>().SelectMain(s.hero.GetComponent<AdventureLoadout>().items.First(i=>i.type==GearType.Staff));Step();return;}
 if(phase==9){if(weaponIndex==-1){Need(hud.SkillIconKey=="skill_staff"&&hud.SkillMask>.3f,"staff skill icon");weaponIndex=0;}
 if(weaponIndex>0)Need(hud.SkillIconKey=="skill_"+((GearType)(weaponIndex-1)).ToString().ToLowerInvariant(),"four weapon icon mismatch");
 Need(s.hero.GetComponent<AdventureCombat>().SkillRemaining>3,"switch reset cooldown");if(weaponIndex<4){Need(s.hero.GetComponent<AdventureLoadout>().SelectMain(s.hero.GetComponent<AdventureLoadout>().items.First(i=>(int)i.type==weaponIndex)),"select weapon");weaponIndex++;return;}
hud.SetDetails("帮助");b.OpenPage(PlayerInventory.Page.Character);Need(hud.Detail==""&&b.IsOpen&&Time.timeScale==0,"help -> bag focus");Step();return;}
 if(phase==10){Need(!r.ExecuteDisplayed(false),"inventory interaction penetrated");b.SetOpen(false);Step();return;}
 if(phase==11){Need(Time.timeScale>0&&!s.hero.GameplayPaused,"bag close stuck paused");s.hero.Progression.GrantSkillChoice();Need(!r.ExecuteDisplayed(true),"skill choice input penetrated");s.hero.Progression.SelectChoice(0);Step();return;}
 if(phase==12){Need(UnityEngine.Object.FindObjectsByType<PlayerHudPresenter>(FindObjectsSortMode.None).Length==1&&UnityEngine.Object.FindObjectsByType<InteractionResolver>(FindObjectsSortMode.None).Length==1,"duplicate HUD/input");Need(s.Saves.SaveProgress(false)&&s.Saves.LoadProgress(false),"HUD recovery");Mark("U08_U09_U14_U15_ICON_COOLDOWN_FOCUS_SINGLETON_RECOVERY_PASS");s.hero.RestorePosition(new Vector3(0,.1f,182),0);Step();return;}
 if(phase==13){if(hud.ContextVisible)return;Need(!hud.VisiblePrompt.Contains("检查点"),"debug leaked");Need(!s.hero.Inventory.Notifications.Text.Contains("已到营地"),"old camp notification leaked into run");Mark("U01_EMPTY_NO_CONTEXT_OR_DEBUG_PASS");scanStart=r.ScanCount;sampleAt=Time.unscaledTime;phase++;return;}
 if(phase==14){timing+=hud.LastUpdateMicroseconds;if(++samples<120)return;Mark("PERF_RENDER_ON presenter_us_mean="+(timing/samples).ToString("F2")+" resolver_scans="+(r.ScanCount-scanStart)+" elapsed="+(Time.unscaledTime-sampleAt).ToString("F2"));s.GetComponentInChildren<Canvas>().enabled=false;samples=0;timing=0;phase++;return;}
 if(phase==15){timing+=hud.LastUpdateMicroseconds;if(++samples<120)return;Mark("PERF_CANVAS_OFF_PRESENTER_STILL_UPDATES us_mean="+(timing/samples).ToString("F2"));s.GetComponentInChildren<Canvas>().enabled=true;Mark("RULES_PASS controlled API and virtual keyboard, AI disabled; human NOT_RUN");EditorApplication.update-=Tick;InputSystem.RemoveDevice(kb);EditorApplication.Exit(0);}
 }catch(Exception e){if(s!=null)s.Saves.WriteValidationHook=null;Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
