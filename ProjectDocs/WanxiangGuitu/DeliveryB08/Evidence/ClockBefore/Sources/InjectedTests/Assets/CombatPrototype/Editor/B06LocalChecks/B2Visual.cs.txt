using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class B2Visual {
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT");static CombatSession s;static PlayerHudPresenter hud;static EditorWindow view;static int index,frame;static double deadline,next;static string capture;static bool prepared;
 static void Need(bool v,string m){if(!v)throw new Exception(m);}
 public static void Run(){Need(ValidationStorage.Root(Environment.GetCommandLineArgs(),true)==ValidationStorage.ValidateRoot(Root),"preplay isolation");ValidationStorage.Record("pre-play",Root);deadline=EditorApplication.timeSinceStartup+180;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();view?.Repaint();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("B2 visual timeout "+index);if(!EditorApplication.isPlaying)return;
 if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||!s.Saves.IsInitialized){s=null;return;}s.hero.Inventory.RestoreInventory(new int[6],0,true,0);Need(CraftingTargets.Select(s.hero,GearType.Sword,GearQuality.Red),"target");var region=s.GetComponent<ForestRegionRuntime>()??s.gameObject.AddComponent<ForestRegionRuntime>();region.LayoutFactory=_=>ForestLayout.Generate(418);Need(s.Flow.Depart(AdventureMode.ForestExploration),"depart");foreach(var enemy in s.enemies)enemy.enabled=false;hud=s.GetComponent<PlayerHudPresenter>();return;}
 if(EditorApplication.timeSinceStartup<next)return;
 if(capture!=null){if(Time.frameCount<=frame||!File.Exists(capture))return;capture=null;index++;prepared=false;}
 if(index==8){File.WriteAllText(Path.Combine(Root,"b2-visual-result.txt"),"PASS 8 actual controlled captures; NPC disabled, domain setup; NOT native UI or human\n");EditorApplication.update-=Tick;EditorApplication.Exit(0);return;}
 int width=index<4?1280:1920,height=index<4?720:1080,mode=index%4;
 if(!prepared){hud.SetDetails("");view=UI01Before.Size(width,height);hud.SetUiScale(mode==3?1.5f:1);if(mode==1||mode==3)hud.SetDetails("目标");if(mode==2)hud.SetDetails("路线");prepared=true;next=EditorApplication.timeSinceStartup+.7;return;}
 Need(Screen.width==width&&Screen.height==height,"capture resolution");Canvas.ForceUpdateCanvases();var target=s.GetComponentsInChildren<UnityEngine.UI.Text>(true).First(t=>t.name=="TargetText");if(mode==0)Need(target.preferredHeight<=target.rectTransform.rect.height+.1f,"target HUD text clipped");
 string label=$"b2-{width}x{height}-{mode}";capture=Path.Combine(Root,label+".png");frame=Time.frameCount;ScreenCapture.CaptureScreenshot(capture);File.AppendAllText(Path.Combine(Root,"b2-visual.jsonl"),JsonUtility.ToJson(new Receipt{file=label+".png",width=Screen.width,height=Screen.height,frame=frame,detail=hud.Detail,scale=hud.UiScale,text=CraftingTargets.Details(s)})+"\n");next=EditorApplication.timeSinceStartup+.3;
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
 [Serializable]class Receipt{public string file,detail,text;public int width,height,frame;public float scale;}
}
