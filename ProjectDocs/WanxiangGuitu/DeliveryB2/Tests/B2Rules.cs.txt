using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class B2Rules {
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")??throw new Exception("explicit root required");
 static CombatSession s;static ForestRegionRuntime region;static int phase,index;static bool resume;static double deadline;static string receipt;
 static void Need(bool ok,string why){if(!ok)throw new Exception(why);}static void Mark(string text){Debug.Log("B2_"+text);File.AppendAllText(Path.Combine(Root,"b2-results.txt"),text+"\n");}
 public static void Run(){Start(false);}public static void Resume(){Start(true);}
 static void Start(bool continuing){resume=continuing;ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+120;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("B2 timeout phase "+phase);if(!EditorApplication.isPlaying)return;
 if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||Time.time<.3f){s=null;return;}s.Saves.InitializeStorage(Path.Combine(Root,"b2-save"),resume);region=s.GetComponent<ForestRegionRuntime>()??s.gameObject.AddComponent<ForestRegionRuntime>();
  if(resume){Need(s.Saves.LoadProgress(false),"new-process load");var expected=File.ReadAllLines(Path.Combine(Root,"b2-expected.txt"));Need(s.Flow.Profile.profileId==expected[0]&&s.Flow.ActiveRun.runId==expected[1]&&s.Flow.ActiveRun.regionInstanceId==expected[2]&&s.Flow.ActiveRun.seed.ToString()==expected[3],"identity/seed changed");Need(CraftingTargets.Name(s.Flow.Profile.craftingTarget)==expected[4]&&JsonUtility.ToJson(region.Layout)==expected[5],"target/layout changed");Need(string.Join(",",s.Flow.ActiveRun.earnedRemaining)==expected[6],"ledger changed");Need((s.Flow.ExitConfigs().Single(e=>e.id=="exit:ferry").repairReceipt??"")==expected[7],"receipt changed");Mark("NEW_PROCESS_TARGET_LAYOUT_RECEIPT_LEDGER_PASS");Finish();return;}
  s.hero.Inventory.RestoreInventory(new int[6],0,true,0);var stock=new int[6];stock[3]=3;s.Camp.Restore(s.Camp.Items.ToArray(),stock);Need(CraftingTargets.Select(s.hero,GearType.Sword,GearQuality.White),"select and save");
  var random=JsonUtility.ToJson(UnityEngine.Random.state);var needs=CraftingTargets.Needs(s);Need(needs.Single(n=>n.kind==MaterialKind.Wood).Missing==0&&needs.Single(n=>n.kind==MaterialKind.Ore).Missing==2,"warehouse/bag deficits");Need(random==JsonUtility.ToJson(UnityEngine.Random.state),"query consumed RNG");
  var advanced=new CraftingTarget{type=GearType.Sword,quality=GearQuality.Red};s.Flow.Profile.craftingTarget=advanced;Need(CraftingTargets.Needs(s).Single(n=>n.kind==MaterialKind.BossCore).required>2,"locked prerequisite costs omitted");Need(CraftingTargets.Select(s.hero,GearType.Sword,GearQuality.White),"restore selection");var savedText=File.ReadAllText(s.Saves.SavePath);s.Saves.WriteValidationHook=point=>{if(point=="before-publish")throw new IOException("B2_SYNTHETIC_TARGET_WRITE_FAILURE");};Need(!CraftingTargets.Select(s.hero,GearType.Sword,GearQuality.Red),"save failure hidden");Need(File.ReadAllText(s.Saves.SavePath)==savedText&&s.Saves.SaveFailed,"target failure overwrote disk");s.Saves.WriteValidationHook=null;Need(s.Saves.LoadProgress(false),"target failure restore");Need(s.Flow.Profile.craftingTarget.quality==GearQuality.White,"failed choice survived restart");Mark("TARGET_SAVE_FAILURE_DISK_PRESERVED_RELOAD_PASS");Mark("REAL_COSTS_WAREHOUSE_UNLOCK_CHAIN_RNG_QUERY_PASS");
  region.LayoutFactory=_=>ForestLayout.Generate(418);Need(s.Flow.Depart(AdventureMode.ForestExploration),"depart");Need(region.Layout.revision==3&&region.Layout.points.Select(p=>p.siteKind).Distinct().Count()==7,"new site schema");foreach(var enemy in s.enemies)enemy.enabled=false;phase=1;return;
 }
 if(phase==1){var woodland=region.Layout.points.Single(p=>p.siteKind==ForestSiteKind.Woodland);var mine=region.Layout.points.Single(p=>p.siteKind==ForestSiteKind.Mineral);Need(CraftingTargets.Available(s,woodland).Contains(MaterialKind.Wood)&&CraftingTargets.Available(s,mine).Contains(MaterialKind.Ore),"actual gathering layout");Need(CraftingTargets.Route(region.Layout,region.Layout.entry,mine.id).Contains(mine.label),"route graph");var before=s.Flow.ActiveRun.earnedRemaining.ToArray();var old=region.Layout;old.revision=2;Need(ForestRewards.Site(s,s.Events.Points.First().entityId)==null,"legacy rewards changed");old.revision=3;Need(before.SequenceEqual(s.Flow.ActiveRun.earnedRemaining),"query granted loot");s.Flow.Profile.craftingTarget=new CraftingTarget{type=GearType.Sword,quality=GearQuality.Red};var rng=JsonUtility.ToJson(UnityEngine.Random.state);Need(CraftingTargets.Recommendations(s).Contains("无核心来源"),"fake core recommendation");Need(rng==JsonUtility.ToJson(UnityEngine.Random.state),"recommendation consumed RNG");Need(CraftingTargets.Compact(s).Length<40,"HUD summary unbounded");Need(CraftingTargets.Select(s.hero,GearType.Sword,GearQuality.White),"restore target after core query");Mark("SITE_LAYOUT_REAL_ROUTES_LEGACY_REWARD_GATE_PASS");phase=2;return;}
 if(phase==2){var sites=region.Layout.points.Where(p=>new[]{ForestSiteKind.Woodland,ForestSiteKind.Mineral,ForestSiteKind.Ruins,ForestSiteKind.Outpost,ForestSiteKind.Lakeside}.Contains(p.siteKind)).ToArray();if(index<sites.Length){var site=sites[index++];var chest=s.Events.Points.Single(p=>p.entityId==site.id+":chest");s.hero.RestorePosition(chest.transform.position+Vector3.back*1.6f+Vector3.up*.1f,0);Physics.SyncTransforms();var previous=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).ToArray();var ledger=s.Flow.ActiveRun.earnedRemaining.ToArray();Need(chest.Interact(s.hero),"open chest "+site.label);Need(!chest.Interact(s.hero),"double chest reward");var drops=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Except(previous).ToArray();Need(drops.Where(l=>l.Gear==null).Select(l=>l.Kind).OrderBy(k=>k).SequenceEqual(ForestRewards.ChestMaterials(site.siteKind).OrderBy(k=>k)),"wrong actual reward "+site.label);Need(ledger.SequenceEqual(s.Flow.ActiveRun.earnedRemaining),"chest auto claimed");if(site.siteKind==ForestSiteKind.Ruins)Need(drops.Any(l=>l.Gear!=null),"ruins gear missing");Mark("ACTUAL_CHEST_GROUND_REWARD_ONCE_"+site.siteKind+"_PASS");return;}var elite=s.enemies.First(e=>e.archetype==EnemyArchetype.Elite&&e.GetComponent<EncounterGuard>()?.poi=="poi:4");var existing=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None);elite.Health.Damage(100000);elite.DropLootOnce();var bonusDrops=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Except(existing).ToArray();int crystals=bonusDrops.Where(l=>l.Gear==null&&l.Kind==MaterialKind.CrystalShard).Sum(l=>l.Amount);Need(crystals>=elite.crystalAmount+2&&crystals<=elite.crystalAmount+4,"outpost actual extra reward");int count=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Length;elite.DropLootOnce();Need(count==UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Length,"elite duplicate reward");for(int choice=0;choice<20&&s.hero.Progression.PendingChoices>0;choice++){s.hero.Progression.BeginChoice();s.hero.Progression.SelectChoice(0);}Need(s.hero.Progression.PendingChoices==0,"controlled skill choices");Mark("OUTPOST_ELITE_REAL_DROP_ONCE_PASS controlled kill");phase=3;return;}
 if(phase==3){var forest=region.Layout.points.Single(p=>p.siteKind==ForestSiteKind.Woodland);foreach(var node in UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).Where(n=>n.kind==MaterialKind.Wood))node.Restore(0);Need(CraftingTargets.Available(s,forest).Contains(MaterialKind.Wood),"unclaimed ground loot omitted");var wood=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).First(l=>l.Gear==null&&l.Kind==MaterialKind.Wood&&Vector3.Distance(l.transform.position,forest.position)<10);s.hero.RestorePosition(wood.transform.position-Vector3.up*.5f,0);Need(wood.PickUp(s.hero.Inventory),"pickup");Need(s.Flow.ActiveRun.earnedRemaining[3]>0,"pickup source ledger");Need(!CraftingTargets.Available(s,forest).Contains(MaterialKind.Wood),"depleted woodland still recommends wood");Mark("DEPLETED_AND_UNCLAIMED_SOURCE_QUERY_PASS");var stone=s.hero.Inventory.Count(MaterialKind.Stone);if(stone<1)s.hero.Inventory.Add(MaterialKind.Stone,1);var ferry=s.Flow.ExitConfigs().Single(e=>e.repairRequired);s.hero.RestorePosition(ferry.position+Vector3.back,0);Need(s.Flow.RepairExit(ferry.id),"repair");receipt=s.Flow.ExitConfigs().Single(e=>e.id==ferry.id).repairReceipt;var cost=s.hero.Inventory.Count(MaterialKind.Wood);Need(!s.Flow.RepairExit(ferry.id)&&s.hero.Inventory.Count(MaterialKind.Wood)==cost,"duplicate charge");Need(s.Saves.SaveProgress(false),"save B2 world");var snapshot=JsonUtility.ToJson(region.Capture());Need(s.Saves.LoadProgress(false),"same-process restore");Need(JsonUtility.ToJson(region.Capture())==snapshot,"restore layout changed");foreach(var enemy in s.enemies)enemy.enabled=false;
  var hud=s.GetComponent<PlayerHudPresenter>();hud.SetDetails("目标");ScreenCapture.CaptureScreenshot(Path.Combine(Root,"b2-target-details.png"));Mark("PICKUP_RECEIPT_SAVE_RESTORE_PASS");phase=4;return;}
 if(phase==4){var hud=s.GetComponent<PlayerHudPresenter>();hud.SetDetails("");s.Flow.OpenAbandon();Need(s.Flow.ConfirmAbandon()&&s.Flow.CommitSettlement()&&s.Flow.AcknowledgeSettlement(),"abandon settlement");Need(CraftingTargets.Valid(s.Flow.Profile.craftingTarget),"target lost in settlement");Need(s.Flow.Depart(AdventureMode.ForestExploration),"next forest run");foreach(var enemy in s.enemies)enemy.enabled=false;Need(s.Saves.SaveProgress(false),"final checkpoint");File.WriteAllLines(Path.Combine(Root,"b2-expected.txt"),new[]{s.Flow.Profile.profileId,s.Flow.ActiveRun.runId,s.Flow.ActiveRun.regionInstanceId,s.Flow.ActiveRun.seed.ToString(),CraftingTargets.Name(s.Flow.Profile.craftingTarget),JsonUtility.ToJson(region.Capture()),string.Join(",",s.Flow.ActiveRun.earnedRemaining),s.Flow.ExitConfigs().Single(e=>e.id=="exit:ferry").repairReceipt??""});Mark("ABANDON_UNIQUE_SETTLEMENT_TARGET_REDEPART_PASS");phase=5;return;}
 if(phase==5)Finish();
 }catch(Exception e){if(s!=null)s.Saves.WriteValidationHook=null;Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
 static void Finish(){Mark("PASS controlled rules; enemies disabled; NOT native UI");EditorApplication.update-=Tick;EditorApplication.Exit(0);}
}
