using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEngine.AI;using UnityEditor;using UnityEditor.SceneManagement;
public static class R1Rules {
 static CombatSession s;static int seed=1;static bool resume;static double deadline;static HashSet<int> actual=new HashSet<int>();
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT");
 static void Need(bool v,string m){if(!v)throw new Exception(m);}static void Mark(string m){File.AppendAllText(Root+"/r1-results.txt",m+"\n");Debug.Log("R1_"+m);}
 public static void Run(){Start(false);}public static void Resume(){Start(true);}
 static void Start(bool load){resume=load;ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+500;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static float Length(Vector3[] corners){float result=0;for(int i=1;i<corners.Length;i++)result+=Vector3.Distance(corners[i-1],corners[i]);return result;}
 static bool SegmentHits(Vector3[] corners,Bounds box){box.Expand(-.3f);for(int i=1;i<corners.Length;i++){var a=corners[i-1];var delta=corners[i]-a;if(box.IntersectRay(new Ray(a,delta.normalized),out var distance)&&distance<delta.magnitude)return true;}return false;}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("R1 rules deadline");if(!EditorApplication.isPlaying||Time.time<.5f)return;if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(Root+"/r1-rules-save",false);if(resume){Need(s.Saves.LoadProgress(false),"legacy new process load");var loaded=s.GetComponent<ForestRegionRuntime>();Need(loaded.Layout.revision==4&&JsonUtility.ToJson(loaded.Capture())==File.ReadAllText(Root+"/r1-old-expected.txt"),"revision4 changed on fresh process");Mark("REVISION4_NEW_PROCESS_ORIGINAL_GEOMETRY_PASS");Finish();return;}Need(s.Flow.Depart(AdventureMode.ForestExploration),"depart");File.WriteAllText(Root+"/r1-nav.csv","requested,actual,poi,status,shortMeters,safeMeters,cornerCount\n");}
 var region=s.GetComponent<ForestRegionRuntime>();var pure=ForestLayout.Generate(seed);pure.Validate();Need(pure.revision==5&&pure.terrain.version==1,"terrain version");Need(JsonUtility.ToJson(pure)==JsonUtility.ToJson(ForestLayout.Generate(seed)),"same seed non deterministic");region.GeneratePlayable(seed);var map=region.Capture();foreach(var e in s.enemies)e.enabled=false;Need(map.revision==5,"new revision");
 string rng=JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams),global=JsonUtility.ToJson(UnityEngine.Random.state);var queries=new List<ForestRouteResult>();
 foreach(var poi in map.points){var route=region.NavigationRoute(map.entry,poi.id);var safe=region.NavigationRoute(map.entry,poi.id,true);Need(route.reachable,"POI unreachable "+poi.id);Need(Mathf.Abs(route.distance-Length(route.corners))<.01f,"wrong nav distance");var lake=new Bounds(map.terrain.lakeCenter+Vector3.up, new Vector3(map.terrain.lakeSize.x,10,map.terrain.lakeSize.z));var ridge=new Bounds(map.terrain.ridgeCenter+Vector3.up,map.terrain.ridgeSize+Vector3.up*5);Need(!SegmentHits(route.corners,lake)&&!SegmentHits(route.corners,ridge),"path through solid/water");Need(route.corners.All(p=>map.WorldBounds.Contains(p)),"path outside bounds");queries.Add(route);File.AppendAllText(Root+"/r1-nav.csv",seed+","+map.seed+","+poi.id+","+route.status+","+route.distance+","+(safe.reachable?safe.distance:-1)+","+route.corners.Length+"\n");}
 foreach(var exit in map.exits)Need(region.TryInteractionApproach(exit,out var p)&&region.PathComplete(map.entry,p),"exit "+exit.id);
 foreach(var node in map.nodes)Need(region.TryGround(node.position,4,out var p)&&region.PathComplete(map.entry,p),"resource "+node.id);
 foreach(var enemy in map.encounters)Need(region.PathComplete(map.entry,enemy.home),"encounter "+enemy.id);
 int offroad=0;for(int x=-80;x<=80;x+=10)for(int z=165;z<=320;z+=10){var p=new Vector3(x,map.terrain.Height(map,new Vector3(x,0,z)),z);if(map.terrain.InLake(p,4)||map.terrain.InRidge(p,4)||map.routes.Min(e=>ForestTerrain.FlatSegment(p,map.points.Single(q=>q.id==e.from).position,map.points.Single(q=>q.id==e.to).position))<9)continue;if(region.TryGround(p,1,out var hit)&&ForestTerrain.FlatDistance(hit,p)<1&&region.PathComplete(map.entry,hit))offroad++;}Need(offroad>=20,"forest became fixed-lane maze");
 // Both a path through the live outpost warning area and a genuine avoidance path.
 var outpost=region.NavigationRoute(map.entry,"poi:4");Need(outpost.reachable&&outpost.nearElite,"actual outpost warning path absent");var safeAltar=region.NavigationRoute(map.entry,"poi:5",true);Need(safeAltar.reachable&&!safeAltar.nearElite,"actual elite bypass absent");
 s.hero.RestorePosition(map.entry,0);region.InvalidateRoutes();CraftingTargets.Recommendations(s);int before=region.RouteQueries;for(int i=0;i<20;i++)CraftingTargets.Recommendations(s);Need(region.RouteQueries==before,"read-only HUD repeated full path queries");Need(rng==JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams)&&global==JsonUtility.ToJson(UnityEngine.Random.state),"query consumes RNG");
 var target=region.Layout.points.Single(p=>p.id=="poi:1");var saved=target.position;target.position=new Vector3(5000,0,5000);region.InvalidateRoutes();Need(!region.NavigationRoute(map.entry,target.id).reachable,"unreachable target valid");Need(!CraftingTargets.Recommendations(s).Split('\n').Any(line=>line.StartsWith(target.label+"（")),"unreachable recommended");target.position=saved;region.InvalidateRoutes();
 File.AppendAllText(Root+"/r1-terrain.jsonl",JsonUtility.ToJson(map)+"\n");File.AppendAllText(Root+"/r1-open-space.csv",seed+","+map.seed+","+offroad+"\n");actual.Add(map.seed);seed++;
 if(actual.Count<30){Need(seed<=61,"30 legal actual seeds not reached");return;}
 Mark("30_ACTUAL_SEEDS_TERRAIN_NAV_OPEN_SPACE_QUERY_CACHE_RNG_PASS");
 var legacy=JsonUtility.FromJson<ForestLayout>(File.ReadAllText(Root+"/r1-legacy-layout4.json"));region.Build(legacy,false);s.hero.RestorePosition(legacy.entry,0);s.Flow.ActiveRun.contentRevision="B06-B4";s.Flow.ActiveRun.generationVersion=6;Need(s.Saves.SaveProgress(false),"legacy save");File.WriteAllText(Root+"/r1-old-expected.txt",JsonUtility.ToJson(region.Capture()));Need(s.Saves.LoadProgress(false),"legacy same process load");Need(JsonUtility.ToJson(region.Capture())==File.ReadAllText(Root+"/r1-old-expected.txt"),"legacy relayout");Mark("REVISION4_ORIGINAL_LAYOUT_AND_GEOMETRY_PASS");Finish();
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
 static void Finish(){EditorApplication.update-=Tick;EditorApplication.Exit(0);}
}
