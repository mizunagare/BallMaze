using System;using System.IO;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
// Editor-only controlled test. Never shipped or claimed as formal Player UI input.
public static class P01Confirmation {
 static double deadline;static string root;static bool done;
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);if(root!=ValidationStorage.ValidateRoot(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")))throw new Exception("root mismatch");deadline=EditorApplication.timeSinceStartup+90;EditorSceneManager.OpenScene(GameEntry.TitlePath);EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("confirmation timeout");var e=GameEntry.Instance;if(!EditorApplication.isPlaying||e==null||e.Info==null||done)return;done=true;if(!e.Info.canContinue)throw new Exception("Synthetic ready fixture required");string original=SaveCatalog.Fingerprint(e.CurrentDirectory),bytes=File.ReadAllText(Path.Combine(e.CurrentDirectory,"save.json"));e.AskNew();string sentinel=Path.Combine(e.CurrentDirectory,"confirmation-change.txt");File.WriteAllText(sentinel,"PLAYER01 synthetic concurrent collection change, no game state fabricated");string changed=SaveCatalog.Fingerprint(e.CurrentDirectory);bool refused=!e.ConfirmNew();bool preserved=File.ReadAllText(Path.Combine(e.CurrentDirectory,"save.json"))==bytes&&SaveCatalog.Fingerprint(e.CurrentDirectory)==changed;bool noGame=UnityEngine.Object.FindFirstObjectByType<CombatSession>()==null;var error=GameEntry.Error;File.Delete(sentinel);bool restored=SaveCatalog.Fingerprint(e.CurrentDirectory)==original;var result=new Result{scope="Editor controlled GameEntry methods; NOT formal Player/native input",refused=refused,changedFingerprint=original!=changed,saveBytesPreserved=preserved,noGameplay=noGame,syntheticSentinelRemoved=restored,error=error};File.WriteAllText(Path.Combine(root,"confirmation-result.json"),JsonUtility.ToJson(result,true));if(!refused||!preserved||!noGame||!restored||!result.changedFingerprint)throw new Exception("new-game confirmation protection failure");EditorApplication.update-=Tick;EditorApplication.Exit(0);}catch(Exception x){Debug.LogException(x);EditorApplication.Exit(1);}}
 [Serializable]class Result{public string scope,error;public bool refused,changedFingerprint,saveBytesPreserved,noGameplay,syntheticSentinelRemoved;}
}
