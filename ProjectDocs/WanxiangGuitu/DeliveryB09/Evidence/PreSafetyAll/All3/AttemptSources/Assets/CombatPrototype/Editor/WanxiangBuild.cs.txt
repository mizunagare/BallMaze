using System;
using System.IO;
using UnityEditor;
using UnityEditor.Build.Reporting;
public static class WanxiangBuild {
 [MenuItem("万象归途/构建/Windows 正式版")]
 public static void Windows(){Build(Path.GetFullPath("Builds/WanxiangGuitu/WanxiangGuitu.exe"));}
 public static void ValidateBuild(){string root=Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")??throw new Exception("Explicit local results directory required");Build(Path.Combine(root,"Release","WanxiangGuitu.exe"));EditorApplication.Exit(0);}
 static void Build(string output){Directory.CreateDirectory(Path.GetDirectoryName(output));var report=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{GameEntry.TitlePath,FantasySceneMenu.DefaultScene},locationPathName=output,target=BuildTarget.StandaloneWindows64,options=BuildOptions.None});var s=report.summary;string text="result="+s.result+"\nerrors="+s.totalErrors+"\nwarnings="+s.totalWarnings+"\nbytes="+s.totalSize+"\nduration="+s.totalTime+"\nproduct="+PlayerSettings.productName+"\ncompany="+PlayerSettings.companyName+"\n";File.WriteAllText(Path.Combine(Path.GetDirectoryName(output),"BuildSummary.txt"),text);if(s.result!=BuildResult.Succeeded)throw new Exception("万象归途 Windows build failed: "+s.result);UnityEngine.Debug.Log("WANXIANG_RELEASE_BUILD_PASS "+text);}
}
