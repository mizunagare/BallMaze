using System;using UnityEngine;using UnityEngine.UI;using UnityEngine.EventSystems;using UnityEngine.InputSystem.UI;
public sealed class TitleMenu:MonoBehaviour {
 GameEntry entry;RectTransform canvas,panel;Font font;string page="主菜单";GamePreferences preferences;Text error;public string Page=>page;public Canvas Canvas=>canvas.GetComponent<Canvas>();
 void Start(){entry=GetComponent<GameEntry>();if(GameEntry.ExplicitGameplay)return;font=Resources.GetBuiltinResource<Font>("LegacyRuntime.ttf");var g=new GameObject("万象归途 · 标题界面",typeof(RectTransform),typeof(Canvas),typeof(CanvasScaler),typeof(GraphicRaycaster));g.transform.SetParent(transform,false);canvas=g.GetComponent<RectTransform>();Canvas.renderMode=RenderMode.ScreenSpaceOverlay;Canvas.sortingOrder=100;var scale=g.GetComponent<CanvasScaler>();scale.uiScaleMode=CanvasScaler.ScaleMode.ScaleWithScreenSize;scale.referenceResolution=new Vector2(1280,720);scale.matchWidthOrHeight=1;
 if(EventSystem.current==null){var es=new GameObject("标题输入",typeof(EventSystem),typeof(InputSystemUIInputModule));es.transform.SetParent(transform,false);es.GetComponent<InputSystemUIInputModule>().AssignDefaultActions();}ScenePresentation.TitleBackdrop(transform);preferences=GamePreferences.Load(entry.PreferencesPath);Show("主菜单");}
 RectTransform Box(string name,Transform parent,Vector2 pos,Vector2 size,Color color){var g=new GameObject(name,typeof(RectTransform),typeof(Image));g.transform.SetParent(parent,false);var r=g.GetComponent<RectTransform>();r.anchorMin=r.anchorMax=r.pivot=new Vector2(0,1);r.anchoredPosition=pos;r.sizeDelta=size;g.GetComponent<Image>().color=color;g.GetComponent<Image>().raycastTarget=false;return r;}
 Text Label(string value,Vector2 pos,Vector2 size,int pts=20){var r=Box(value,panel,pos,size,Color.clear);Destroy(r.GetComponent<Image>());var t=r.gameObject.AddComponent<Text>();t.font=font;t.text=value;t.fontSize=pts;t.color=new Color(.91f,.94f,.85f);t.raycastTarget=false;t.horizontalOverflow=HorizontalWrapMode.Wrap;t.verticalOverflow=VerticalWrapMode.Truncate;return t;}
 Button Button(string label,int row,Action action,bool enabled=true){var r=Box(label,panel,new Vector2(34,-(180+row*56)),new Vector2(252,44),new Color(.12f,.23f,.21f,.96f));var image=r.GetComponent<Image>();image.raycastTarget=true;var b=r.gameObject.AddComponent<Button>();b.targetGraphic=image;var colors=b.colors;colors.highlightedColor=new Color(.65f,.84f,.66f);colors.pressedColor=new Color(.95f,.7f,.28f);colors.disabledColor=new Color(.35f,.35f,.35f);b.colors=colors;b.interactable=enabled;b.onClick.AddListener(()=>action());var t=Label(label,new Vector2(48,-(187+row*56)),new Vector2(224,34));return b;}
 void Detail(string text){Label(text,new Vector2(345,-175),new Vector2(520,300),20);}
 public void Show(string next){page=next;if(panel!=null){panel.gameObject.SetActive(false);Destroy(panel.gameObject);}panel=Box("入口面板",canvas,new Vector2(0,0),new Vector2(930,620),new Color(.025f,.08f,.075f,.9f));panel.anchorMin=panel.anchorMax=new Vector2(.5f,.5f);panel.pivot=new Vector2(.5f,.5f);panel.anchoredPosition=Vector2.zero;panel.GetComponent<Image>().raycastTarget=true;Label("万象归途",new Vector2(34,-28),new Vector2(800,64),42).color=new Color(1,.78f,.38f);Label("林海之中，寻回归途",new Vector2(38,-103),new Vector2(800,36),20);error=Label(GameEntry.Error,new Vector2(34,-535),new Vector2(850,68),17);error.color=new Color(1,.65f,.4f);
 if(next=="主菜单"){Button("继续游戏",0,()=>{if(!entry.Continue())Show("主菜单");},entry.Info?.canContinue==true);Button("新游戏",1,()=>{entry.AskNew();Show("新游戏确认");});Button("读取存档",2,()=>{entry.Refresh();Show("读取存档");});Button("设置",3,()=>Show("设置"));Button("退出游戏",4,()=>{entry.Exit();Show("主菜单");});Detail(entry.Info?.Summary??"正在检查自动存档…");}
 else if(next=="新游戏确认"){Detail("建立独立的新档案。\n将重置：角色、Run、随身物品、永久仓库、配方、徽章和教程。\n保留：音量/画面/输入设置；旧存档完整集合移入独立归档（含备份和掉落分页），不与新档混合。\n取消不会写入新进度。");Button("确认建立新档案",0,()=>{if(!entry.ConfirmNew())Show("新游戏确认");});Button("取消，保留当前进度",1,()=>{entry.CancelNew();Show("主菜单");});}
 else if(next=="读取存档"){Detail("当前仅有一个自动保存档，不是多槽列表。\n\n"+entry.Info.Summary);Button("读取所示自动档",0,()=>{if(!entry.ReadSelected())Show("读取存档");},entry.Info.canContinue);Button("重新检查",1,()=>{entry.Refresh();Show("读取存档");});Button("明确选择新版",2,()=>{entry.ChooseCurrent();Show("读取存档");},entry.Info.status==SaveAvailability.Conflict);Button("返回",4,()=>Show("主菜单"));}
 else if(next=="设置"){Detail("已有功能\n主音量、鼠标灵敏度、HUD大小与窗口分辨率。\n固定键位：WASD移动 / 鼠1攻击 / F技能 / V切换 / 空格翻滚 / E交互 / H撤离。\n本版没有按键重绑定，不提供假选项。");Button("音量 "+preferences.volume.ToString("0%"),0,()=>{preferences.volume=preferences.volume>=.99f?0:Mathf.Min(1,preferences.volume+.25f);Show("设置");});Button("灵敏度 "+preferences.sensitivity.ToString("0.00"),1,()=>{preferences.sensitivity=preferences.sensitivity>=.29f?.06f:preferences.sensitivity+.06f;Show("设置");});Button("HUD "+preferences.uiScale.ToString("0%"),2,()=>{preferences.uiScale=preferences.uiScale>=1.49f?1:preferences.uiScale+.25f;Show("设置");});Button((preferences.fullScreen?"全屏 ":"窗口 ")+preferences.width+"×"+preferences.height,3,()=>{if(preferences.width==1280){preferences.width=1920;preferences.height=1080;}else{preferences.width=1280;preferences.height=720;preferences.fullScreen=!preferences.fullScreen;}Show("设置");});Button("应用并保存",4,()=>{try{preferences.Save(entry.PreferencesPath);Show("主菜单");}catch(Exception e){error.text="设置未保存："+e.Message;}});Button("取消返回",5,()=>{preferences=GamePreferences.Load(entry.PreferencesPath);Show("主菜单");});}
 }
 void Update(){if(panel!=null)foreach(var b in panel.GetComponentsInChildren<Button>())if(GameEntry.Loading)b.interactable=false;}
 void OnDestroy(){if(font!=null)Destroy(font);}
}
