using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class B08Rules {
 static CombatSession s;static string mode;static int phase;static double deadline;static string Root=>ValidationStorage.ValidateRoot(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT"));
 static void Need(bool b,string m){if(!b)throw new Exception(m);}static void Mark(string m){File.AppendAllText(Root+"/b08-rules.txt",m+"\n");}
 public static void Run(){Start("run");}public static void Resume(){Start("resume");}public static void FinalResume(){Start("final");}public static void CompleteResume(){Start("complete");}
 static void Start(string m){mode=m;ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+240;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Freeze(){foreach(var e in s.enemies){e.enabled=false;if(e.GetComponent<ForestBossBrain>()!=null)e.GetComponent<ForestBossBrain>().enabled=false;}}
 static string Snapshot()=>JsonUtility.ToJson(Onboarding.State(s))+"|"+s.Flow.Profile.profileId+"|"+s.Flow.ActiveRun?.runId+"|"+s.Flow.ActiveRun?.regionInstanceId+"|"+s.Flow.ActiveRun?.seed+"|"+JsonUtility.ToJson(s.Flow.ActiveRun?.randomStreams)+"|"+string.Join(",",s.Flow.Profile.materials)+"|"+string.Join(",",s.hero.Inventory.Materials.OrderBy(x=>x.material).Select(x=>x.material+":"+x.amount))+"|"+string.Join(",",s.hero.GetComponent<AdventureLoadout>().items.Select(x=>x.id).OrderBy(x=>x))+"|"+string.Join(",",s.Flow.ActiveRun?.earnedRemaining??new int[0]);
 static void Finish(){Need(s.Saves.SaveProgress(false),"final checkpoint");File.WriteAllText(Root+"/b08-expected.txt",Snapshot());Mark("PASS "+mode+" "+Onboarding.State(s).stage);EditorApplication.update-=Tick;EditorApplication.Exit(0);}
 static void Facility(string id){Need(s.Camp.Explore(),"explore");s.hero.RestorePosition(s.Camp.FacilityPosition(id)+Vector3.back*.7f,0);Need(s.Camp.InteractFacility(id),"actual facility "+id+" "+s.Camp.FacilityReason(id));}
 static void Harvest(MaterialKind kind){var node=UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).Where(r=>r.kind==kind&&r.Health>0).OrderBy(r=>Vector3.Distance(r.transform.position,s.hero.transform.position)).First();s.hero.RestorePosition(node.transform.position+Vector3.back*1.3f,0);var old=Onboarding.State(s).stage;int count=s.hero.Inventory.Count(kind);node.Hit(node.maxHealth,s.hero);Need(Onboarding.State(s).stage==old&&s.hero.Inventory.Count(kind)==count,"unpicked harvest advanced");var loot=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Where(l=>!l.Collected&&l.Gear==null&&l.Kind==kind).OrderBy(l=>Vector3.Distance(l.transform.position,node.transform.position)).First();Need(loot.PickUp(s.hero.Inventory),"real resource pickup "+kind);Need(s.hero.Inventory.Count(kind)>count&&s.Flow.ActiveRun.earnedRemaining[(int)kind]>0,"pickup ownership/ledger");}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("B08 rule deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.5f)return;
  if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.autoSaveInterval=3600;s.Saves.InitializeStorage(Root+"/b08-save",mode!="run");Freeze();if(mode!="run"){Need(Snapshot()==File.ReadAllText(Root+"/b08-expected.txt"),"new process tutorial/identity/ledger/gear changed");Mark("EXACT_NEW_PROCESS_"+mode);if(mode=="complete"){Need(!Onboarding.Active(s)&&Onboarding.State(s).stage==TutorialStage.Complete,"completed tutorial replay");Finish();return;}if(mode=="final"){string old=Onboarding.State(s).firstRunId;Need(Onboarding.State(s).stage==TutorialStage.DepartAgain,"manufacture restore");Need(s.Flow.Depart(AdventureMode.ForestExploration)&&s.Flow.ActiveRun.runId!=old&&!Onboarding.Active(s)&&Onboarding.State(s).stage==TutorialStage.Complete,"actual new run completes tutorial");Freeze();Finish();return;}Need(Onboarding.State(s).stage==TutorialStage.FreeExit,"in-flight stage");var region=s.GetComponent<ForestRegionRuntime>();var exit=s.Flow.ExitConfigs().First(e=>!e.repairRequired);Need(region.TryInteractionApproach(exit,out var approach),"free approach");s.hero.RestorePosition(approach,0);Need(s.Flow.BeginExtraction(exit.id),"start extraction");s.Flow.CancelExtraction("controlled cancel");Need(Onboarding.State(s).stage==TutorialStage.FreeExit,"cancel advanced tutorial");Need(s.Flow.BeginExtraction(exit.id),"restart normal countdown");phase=1;return;}
   Need(Onboarding.Active(s)&&Onboarding.State(s).stage==TutorialStage.Equipment,"fresh tutorial");string profile=JsonUtility.ToJson(s.Flow.Profile),rng=JsonUtility.ToJson(UnityEngine.Random.state);for(int i=0;i<10;i++){Onboarding.Details(s);Onboarding.Compact(s);}Need(profile==JsonUtility.ToJson(s.Flow.Profile)&&rng==JsonUtility.ToJson(UnityEngine.Random.state),"help mutated state");var q=new HudNotificationQueue();q.Message("制造");q.SetIcon("craft");Need(q.IconKey=="craft","craft category");q.Pickup(MaterialKind.Wood,1);Need(q.IconKey=="bag","pickup category");Mark("PURE_HELP_PROFILE_RNG_AND_ICONS");
   s.Flow.Profile.hasTutorial=false;s.Flow.Profile.tutorial=null;Need(s.Saves.SaveProgress(false)&&s.Saves.LoadProgress(false)&&!Onboarding.Active(s)&&Onboarding.State(s)==null,"old v9 optional state");Need(Onboarding.Enable(s),"explicit old camp opt-in");Need(s.Camp.Explore(),"camp walk");Need(!s.hero.GetComponent<AdventureCombat>().Strike(),"camp attack leakage");var foe=s.enemies.First();var savedFoe=new SavedEnemy{position=foe.transform.position,yaw=foe.transform.eulerAngles.y,health=foe.Health.Current};foe.RestoreState(new SavedEnemy{position=s.hero.transform.position+Vector3.forward,health=foe.Health.Current});float hp=s.hero.Health.Current;foe.ReleaseAttack();Need(s.hero.Health.Current==hp,"camp enemy attack leakage");foe.RestoreState(savedFoe);s.hero.RestorePosition(s.Camp.FacilityPosition("camp:equipment"),0);s.Saves.WriteValidationHook=p=>{if(p=="before-publish")throw new IOException("B08_EXPECTED_SAVE_FAILURE");};Need(!s.Saves.SaveProgress(false),"save failure fixture");string before=JsonUtility.ToJson(Onboarding.State(s));Need(!s.Camp.InteractFacility("camp:equipment")&&before==JsonUtility.ToJson(Onboarding.State(s)),"blocked facility advanced");s.Saves.WriteValidationHook=null;Need(s.Saves.SaveProgress(false),"retry save");
   s.Saves.WriteValidationHook=p=>{if(p=="before-publish")throw new IOException("B08_EXPECTED_SETTING_FAILURE");};Need(!Onboarding.Skip(s)&&Onboarding.Active(s),"skip prepublish rollback");s.Saves.WriteValidationHook=null;Need(s.Saves.SaveProgress(false),"settings retry");s.Saves.WriteValidationHook=p=>{if(p=="after-publish")throw new IOException("B08_EXPECTED_AFTER_PUBLISH");};Need(Onboarding.Skip(s)&&!Onboarding.Active(s)&&!s.Saves.SaveFailed,"setting postpublish reconciliation");s.Saves.WriteValidationHook=null;Need(Onboarding.Enable(s),"opt in again");Mark("SAVE_FAILURE_BLOCKS_AND_SETTINGS_RECONCILE");
   Facility("camp:equipment");Need(Onboarding.State(s).stage==TutorialStage.Warehouse,"equipment real interaction");Facility("camp:workbench");Need(Onboarding.State(s).stage==TutorialStage.Warehouse,"out of order advanced");Facility("camp:warehouse");Facility("camp:workbench");Need(Onboarding.State(s).stage==TutorialStage.Departure,"camp progression");Need(s.Flow.Depart(AdventureMode.ForestExploration),"forest departure");Freeze();Need(Onboarding.State(s).stage==TutorialStage.Move,"first run binding");s.hero.MoveCharacter(Vector3.forward*2.5f);s.hero.MoveCharacter(Vector3.forward*.8f);Need(Onboarding.State(s).stage==TutorialStage.Harvest,"real movement");Harvest(MaterialKind.Wood);Need(Onboarding.State(s).stage==TutorialStage.Pickup,"harvest pickup");Harvest(MaterialKind.Stone);Harvest(MaterialKind.Ore);Need(Onboarding.State(s).stage==TutorialStage.FreeExit,"materials ready");Mark("REAL_MOVEMENT_HARVEST_PICKUP_AND_LEDGER");Finish();return;
  }
  if(phase==1){if(!s.Flow.HasResult)return;Need(s.Flow.Settlement.outcome==AdventureOutcome.Extracted&&s.Flow.Settlement.keptMaterials.Sum()>0&&Onboarding.State(s).stage==TutorialStage.Settlement,"real terminal");Need(s.Flow.CommitSettlement(),"commit");string p=JsonUtility.ToJson(s.Flow.Profile);Need(s.Flow.CommitSettlement()&&p==JsonUtility.ToJson(s.Flow.Profile),"duplicate settlement");Need(s.Flow.AcknowledgeSettlement()&&Onboarding.State(s).stage==TutorialStage.Withdraw,"ack");foreach(var c in CraftingCosts.Gear(GearQuality.White))Need(s.Camp.TransferMaterial(c.material,c.amount,false),"actual withdraw");Need(Onboarding.State(s).stage==TutorialStage.Manufacture,"actual cost available");int gear=s.hero.GetComponent<AdventureLoadout>().items.Count;s.Saves.WriteValidationHook=p=>{if(p=="after-publish")throw new IOException("B08_EXPECTED_CRAFT_AFTER_PUBLISH");};Need(s.Camp.Manufacture(GearType.Sword,GearQuality.White)&&s.hero.GetComponent<AdventureLoadout>().items.Count==gear+1&&Onboarding.State(s).stage==TutorialStage.DepartAgain,"actual manufacture");s.Saves.WriteValidationHook=null;Need(!s.Saves.SaveFailed&&string.IsNullOrEmpty(s.Flow.Error),"manufacture not reconciled");Mark("REAL_SETTLEMENT_STORE_WITHDRAW_MANUFACTURE");Finish();}
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
