using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class Aim01PhysicsProbe {
 static string root;static double deadline;static List<string> logs=new List<string>();
 [Serializable]class Row{public string type,id;public Vector3 query,closest;public Bounds bounds;public bool convex;}
 [Serializable]class Report{public Row[] rows;public string[] physicsLogs;}
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+120;EditorSceneManager.playModeStartScene=null;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void CaptureLog(string condition,string stack,LogType type){if(condition.Contains("ClosestPoint"))logs.Add(type+":"+condition);}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("physics probe startup deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f)return;var s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;EditorApplication.update-=Tick;s.Saves.InitializeStorage(root+"/Store",false);s.Saves.enabled=false;AdventureRandom.SetNextSeedForValidation(418);if(!s.Flow.Depart(AdventureMode.ForestExploration))throw new Exception(s.Flow.Error);var colls=new[]{s.enemies.First().GetComponent<Collider>(),UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).First(n=>n.kind==MaterialKind.Stone&&n.gameObject.activeInHierarchy).GetComponent<Collider>(),UnityEngine.Object.FindObjectsByType<MeshCollider>(FindObjectsSortMode.None).First(c=>!c.convex)};var rows=new List<Row>();Application.logMessageReceived+=CaptureLog;foreach(var c in colls){var q=c.bounds.center+Vector3.forward*3;rows.Add(new Row{type=c.GetType().Name,id=c.name,bounds=c.bounds,query=q,closest=c.ClosestPoint(q),convex=c is MeshCollider m&&m.convex});}Application.logMessageReceived-=CaptureLog;s.Saves.SuspendEntry();File.WriteAllText(root+"/physics-probe.json",JsonUtility.ToJson(new Report{rows=rows.ToArray(),physicsLogs=logs.ToArray()},true));EditorApplication.Exit(0);}catch(Exception e){EditorApplication.update-=Tick;File.WriteAllText(root+"/exception.txt",e.ToString());EditorApplication.Exit(1);}}
}
