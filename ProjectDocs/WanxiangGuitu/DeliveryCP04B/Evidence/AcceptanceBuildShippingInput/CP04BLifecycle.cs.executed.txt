using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEngine.SceneManagement;
public static class CP04BLifecycle {
 [Serializable]public class HP{public int frame;public float before,after,applied;} [Serializable]public class Row{public string name,status,detail;public int frame;}
 [Serializable]public class Result{public Row[] rows;}
 static object F(object x,string n)=>x.GetType().GetField(n,BindingFlags.Instance|BindingFlags.NonPublic|BindingFlags.Public).GetValue(x);static List<Row> rows=new List<Row>();
 static void C(bool ok,string name,string detail=""){rows.Add(new Row{name=name,status=ok?"PASS":"FAIL",detail=detail,frame=Time.frameCount});if(!ok)CP04BCases.ExitCode=1;}
 static AdventureProjectile Own(ThirdPersonHero h)=>UnityEngine.Object.FindObjectsByType<AdventureProjectile>(FindObjectsSortMode.None).FirstOrDefault(p=>(ThirdPersonHero)F(p,"hero")==h);
 public static IEnumerator Run(CombatSession s,ThirdPersonHero h,AdventureCombat combat,string root,bool death){
 CP04BCases.Context(death?"death-cleanup":"pause-switch-expiry"); h.Health.Observed+=x=>{if(x.kind==HealthObservationKind.Damage)File.AppendAllText(root+"/hero-damage.jsonl",JsonUtility.ToJson(new HP{frame=x.frame,before=x.before,after=x.after,applied=x.applied})+"\n");}; combat.ResetCombat();var gear=h.GetComponent<AdventureLoadout>();gear.mainWeapon=gear.items.First(x=>x.InBag&&x.Weapon&&x.type==GearType.Staff).id;gear.reserveWeapon=gear.items.First(x=>x.InBag&&x.Weapon&&x.type==GearType.Bow).id;gear.Refresh();yield return null;
 if(death){var source=s.Waves.CreateEncounter(s.transform,"cp04b-lifecycle-source",h.transform.position+Vector3.forward*6,EnemyArchetype.Boss);source.enabled=false;var brain=source.GetComponent<BossBrain>();if(brain)brain.enabled=false;float incoming=ArmorMitigation.Incoming(source.damage,h.Defense);int n=0;while(h.Health.Current>incoming&&n++<100){EnemyProjectile.Fire(source,h.transform.position+Vector3.up*1.1f+Vector3.forward*.8f,h.transform.position+Vector3.up*1.1f,source.damage,10,source.projectileMaterial);float end=Time.time+.18f;while(Time.time<end)yield return null;}
 C(!h.Health.Dead&&h.Health.Current<=incoming,"death setup from actual enemy projectiles only");C(combat.Strike(),"death released shot accepted");float limit=Time.time+1;while(combat.PendingAttackCount>0&&Time.time<limit)yield return null;C(Own(h)!=null,"released player projectile before lethal impact");EnemyProjectile.Fire(source,h.transform.position+Vector3.up*1.1f+Vector3.forward*.8f,h.transform.position+Vector3.up*1.1f,source.damage,10,source.projectileMaterial);float until=Time.time+.8f;while(Time.time<until)yield return null;C(h.Health.Dead&&Own(h)==null,"real death disposes released projectile");
 }else{
 C(combat.Strike(),"pause released shot accepted");float limit=Time.time+1;while(combat.PendingAttackCount>0&&Time.time<limit)yield return null;var p=Own(h);C(p!=null,"real player projectile released");if(p!=null){h.Inventory.SetOpen(true);yield return null;var position=p.transform.position;float life=(float)F(p,"life");float wait=Time.realtimeSinceStartup+.3f;while(Time.realtimeSinceStartup<wait)yield return null;C(p!=null&&p.transform.position==position&&(float)F(p,"life")==life,"pause freezes position lifetime collision progression");h.Inventory.SetOpen(false);for(int i=0;i<3;i++)yield return null;C(p!=null&&p.transform.position!=position,"resume continues released projectile");C(combat.SwitchWeapon(),"switch after release accepted");yield return null;C(p!=null,"released projectile independent of new weapon");float expire=Time.time+2.7f;while(Time.time<expire)yield return null;C(p==null,"normal lifetime expires released shot");}
 CP04BCases.Context("scene-unload");combat.ResetCombat();yield return null;C(combat.Strike(),"scene unload shot accepted");limit=Time.time+1;while(combat.PendingAttackCount>0&&Time.time<limit)yield return null;var flying=Own(h);C(flying!=null,"scene owns actual released projectile");var scene=h.gameObject.scene;s.Saves.SuspendEntry();var empty=SceneManager.CreateScene("CP04B unload observer");SceneManager.SetActiveScene(empty);var op=SceneManager.UnloadSceneAsync(scene);while(!op.isDone)yield return null;for(int i=0;i<3;i++)yield return null;C(flying==null&&h==null,"real owner scene unload destroys projectile and owner");
 }
 File.WriteAllText(root+"/lifecycle.json",JsonUtility.ToJson(new Result{rows=rows.ToArray()},true));
 }
}
