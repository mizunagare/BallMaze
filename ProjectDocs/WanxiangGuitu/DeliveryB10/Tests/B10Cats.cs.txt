using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
// Controlled pre/post art regression, not native input or human acceptance.
public static class B10Cats {
 static CombatSession s;static EditorWindow view;static int phase,frame,page;static double deadline;
 static string Root=>ValidationStorage.ValidateRoot(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT"));
 public static void Run(){if(ValidationStorage.Root(Environment.GetCommandLineArgs(),true)!=Root)throw new Exception("B10 root must be set before Play");view=UI01Before.Size(1280,720);ShaderUtil.allowAsyncCompilation=false;deadline=EditorApplication.timeSinceStartup+180;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();view?.Repaint();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("B10 cat audit timeout phase="+phase);if(!EditorApplication.isPlaying)return;if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||!s.Saves.IsInitialized){s=null;return;}s.Camp.Explore();var actor=s.hero.GetComponent<FantasyActorVisual>();File.WriteAllText(Root+"/b10-cat-rig-before.txt",string.Join("\n",actor.model.GetComponentsInChildren<Transform>(true).Select(t=>t.name+" local="+t.localPosition+" model="+actor.model.transform.InverseTransformPoint(t.position)+" scale="+t.lossyScale)));frame=Time.frameCount;return;}if(phase==0&&Time.frameCount>frame+30){ScreenCapture.CaptureScreenshot(Root+"/b10-cat-camp.png");phase=1;frame=Time.frameCount;return;}if(phase==1&&Time.frameCount>frame+4){var cat=s.hero.GetComponent<CatActorVisual>();if(cat==null||!cat.Ready)throw new Exception("B10_BLACK_CAT_NOT_INTEGRATED");page=0;s.hero.Inventory.OpenPage((PlayerInventory.Page)page);frame=Time.frameCount;phase=2;return;}if(phase==2&&Time.frameCount>frame+12){if(!CatInventoryPresenter.Instance.Visible)throw new Exception("B10_UGUI_INVENTORY_MISSING");ScreenCapture.CaptureScreenshot(Root+"/b10-cat-page-"+page+".png");frame=Time.frameCount;phase=3;return;}if(phase==3&&Time.frameCount>frame+4){page++;if(page<7){s.hero.Inventory.OpenPage((PlayerInventory.Page)page);frame=Time.frameCount;phase=2;return;}File.WriteAllText(Root+"/b10-cat-result.txt","PASS actual cat rig rendering and seven uGUI pages; controlled, not native or human acceptance\n");EditorApplication.update-=Tick;EditorApplication.Exit(0);}}catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
