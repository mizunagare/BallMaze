using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEngine.Profiling;using UnityEditor;using UnityEditor.SceneManagement;using UnityEditorInternal;using UnityEditor.Profiling;
public static class BRProbeEditor {
 [Serializable]class Allocation{public int index;public long bytes;}
 [Serializable]class Scope{public string marker,thread;public int frame;public double cpuMs;public Allocation[] allocations;}
 [Serializable]class Report{public Scope[] scopes;public string[] errors;}
 static HashSet<int> read=new HashSet<int>();static List<Scope> scopes=new List<Scope>();static List<string> errors=new List<string>();static string root;static double deadline;static bool started,oldEnabled,oldBinary,oldDriver,oldProfileEditor,saved;static int frame=-1;static EditorWindow view;
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+900;oldEnabled=UnityEngine.Profiling.Profiler.enabled;oldBinary=UnityEngine.Profiling.Profiler.enableBinaryLog;oldDriver=ProfilerDriver.enabled;oldProfileEditor=ProfilerDriver.profileEditor;ProfilerDriver.profileEditor=false;ProfilerDriver.enabled=true;UnityEngine.Profiling.Profiler.enableBinaryLog=false;UnityEngine.Profiling.Profiler.enabled=true;File.WriteAllText(root+"/profiler-state.json","driver="+ProfilerDriver.enabled+" engine="+UnityEngine.Profiling.Profiler.enabled+" editor="+ProfilerDriver.profileEditor);ProfilerDriver.SetAreaEnabled(ProfilerArea.CPU,true);view=UI01Before.Size(1280,720);EditorSceneManager.playModeStartScene=null;EditorSceneManager.OpenScene(GameEntry.ForestPath);EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Capture(){int last=ProfilerDriver.lastFrameIndex;for(int f=Math.Max(ProfilerDriver.firstFrameIndex,last-5);f<last;f++){if(read.Contains(f))continue;read.Add(f);using(var data=ProfilerDriver.GetRawFrameDataView(f,0)){if(!data.valid)continue;for(int i=0;i<data.sampleCount;i++){string name=data.GetSampleName(i);if(!name.StartsWith("CP04BR.Scope."))continue;int end=i+data.GetSampleChildrenCountRecursive(i);var values=new List<Allocation>();for(int j=i+1;j<=end&&j<data.sampleCount;j++)if(data.GetSampleName(j)=="GC.Alloc"){try{values.Add(new Allocation{index=j,bytes=data.GetSampleMetadataAsLong(j,0)});}catch(Exception e){errors.Add("metadata "+f+"/"+j+" "+e.GetType().Name);}}scopes.Add(new Scope{marker=name,frame=f,thread=data.threadName,cpuMs=data.GetSampleTimeMs(i),allocations=values.ToArray()});}}}}
 static void Tick(){try{Capture();EditorApplication.QueuePlayerLoopUpdate();view?.Repaint();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("bounded BR probe deadline");if(!EditorApplication.isPlaying||Time.frameCount==frame)return;frame=Time.frameCount;if(!started){var s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||!s.Saves.IsInitialized||frame<6)return;BRProbe.Begin(s,root);started=true;}BRProbe.Tick();if(BRProbe.ScopesComplete&&!saved){Capture();File.WriteAllText(root+"/profile-save-result.txt",ProfilerDriver.SaveProfile(root+"/profiler.data").ToString());saved=true;}if(BRProbe.Done){Capture();File.WriteAllText(root+"/allocation-events.json",JsonUtility.ToJson(new Report{scopes=scopes.ToArray(),errors=errors.ToArray()},true));EditorApplication.update-=Tick;UnityEngine.Profiling.Profiler.enabled=oldEnabled;UnityEngine.Profiling.Profiler.enableBinaryLog=oldBinary;ProfilerDriver.enabled=oldDriver;ProfilerDriver.profileEditor=oldProfileEditor;EditorApplication.Exit(BRProbe.ExitCode);}}catch(Exception e){File.WriteAllText(root+"/exception.txt",e.ToString());EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
