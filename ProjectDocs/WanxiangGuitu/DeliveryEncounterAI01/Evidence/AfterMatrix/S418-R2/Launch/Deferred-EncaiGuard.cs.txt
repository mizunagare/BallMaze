using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using UnityEngine;using UnityEngine.AI;using UnityEditor;using UnityEditor.SceneManagement;
public static class EncaiGuard{
 static string root;static CombatSession s;static ThirdPersonHero h;static IEnumerator test;static int frame=-1;static double deadline;static List<Row> rows=new List<Row>();
 [Serializable]class Row{public string name,status,detail;}[Serializable]class Report{public string scope="controlled real generated forest, real guard/AI/physics, explicit target poses; not normal AI/native/manual";public Row[] rows;}
 static void C(bool b,string name,string detail=""){rows.Add(new Row{name=name,status=b?"PASS":"FAIL",detail=detail});}
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+220;EditorSceneManager.playModeStartScene=null;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("bounded guard deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f||frame==Time.frameCount)return;frame=Time.frameCount;if(test==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;h=s.hero;s.Saves.InitializeStorage(root+"/Store",false);s.Saves.enabled=false;test=Cases();}if(!test.MoveNext())End();}catch(Exception e){C(false,"exception",e.ToString());End();}}
 static void End(){EditorApplication.update-=Tick;Time.captureDeltaTime=0;Time.timeScale=1;if(h!=null)h.HudPanelOpen=false;if(s!=null)s.Saves.SuspendEntry();File.WriteAllText(root+"/guard.json",JsonUtility.ToJson(new Report{rows=rows.ToArray()},true));EditorApplication.Exit(rows.Any(r=>r.status=="FAIL")?1:0);}
 static IEnumerator Wait(float seconds){float end=Time.time+seconds;while(Time.time<end)yield return null;}
 static IEnumerator Cases(){Time.captureDeltaTime=.02f;AdventureRandom.SetNextSeedForValidation(418);C(s.Flow.Depart(AdventureMode.ForestExploration),"actual seed418 forest");h.enabled=false;s.orbit.enabled=false;s.Flow.BeginLegacyRestore();var region=s.GetComponent<ForestRegionRuntime>();var all=s.enemies.ToArray();var e=all.First(x=>x.archetype==EnemyArchetype.Melee&&x.GetComponent<EncounterGuard>()!=null);var guard=e.GetComponent<EncounterGuard>();var friends=all.Where(x=>x.GetComponent<EncounterGuard>()?.poi==guard.poi).ToArray();foreach(var a in all)if(a!=e)a.enabled=false;
 var path=new NavMeshPath();Vector3 target=guard.home;bool found=false;foreach(var d in new[]{Vector3.forward,Vector3.right,Vector3.back,Vector3.left})if(NavMesh.SamplePosition(guard.home+d*7,out var p,2,NavMesh.AllAreas)&&NavMesh.CalculatePath(guard.home,p.position,NavMesh.AllAreas,path)&&path.status==NavMeshPathStatus.PathComplete){target=p.position;found=true;break;}C(found,"actual reachable target near encounter");h.RestorePosition(target+Vector3.up*.1f,0);e.target=h;Physics.SyncTransforms();var start=e.transform.position;float end=Time.time+4;bool engaged=false;while(Time.time<end){engaged|=guard.Engaged;yield return null;}C(engaged,"real LOS detection engages");C(friends.All(x=>x.GetComponent<EncounterGuard>().Engaged),"same POI alert propagation without replacing guard");C(Vector3.Distance(start,e.transform.position)>.5f,"real NavMesh chase moves","distance="+Vector3.Distance(start,e.transform.position));
 var far=region.Layout.points.Select(p=>p.position).Where(p=>Vector3.Distance(p,guard.home)>36).First();h.RestorePosition(far+Vector3.up*.1f,0);Physics.SyncTransforms();bool returning=false;end=Time.time+18;while(Time.time<end){returning|=guard.State==2;if(returning&&guard.State==0)break;yield return null;}C(returning,"beyond31 target leash enters return");C(guard.State==0&&Vector3.Distance(e.transform.position,guard.home)<1.25f,"actual return reaches home","state="+guard.State+" distance="+Vector3.Distance(e.transform.position,guard.home));
 foreach(var a in all){a.enabled=false;var b=a.GetComponent<ForestBossBrain>();if(b!=null)b.enabled=false;}h.Health.ResetHealth();h.HudPanelOpen=false;PlayerInput.Focus(true);
 var shooter=all.First(x=>x.archetype==EnemyArchetype.Ranged);shooter.target=h;var shot=EnemyProjectile.Fire(shooter,shooter.transform.position+Vector3.up*2,h.transform.position+Vector3.up,shooter.damage,shooter.projectileSpeed,shooter.projectileMaterial);var pos=shot.transform.position;float life=(float)typeof(EnemyProjectile).GetField("remainingLife",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic).GetValue(shot);h.HudPanelOpen=true;var wait=Wait(1);while(wait.MoveNext())yield return null;C(shot!=null&&shot.transform.position==pos,"paused actual projectile stays still");C(shot!=null&&Mathf.Approximately(life,(float)typeof(EnemyProjectile).GetField("remainingLife",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic).GetValue(shot)),"paused projectile lifetime frozen");h.HudPanelOpen=false;yield return null;C(shot!=null&&shot.transform.position!=pos,"resume projectile continues");shooter.gameObject.SetActive(false);yield return null;yield return null;C(shot==null,"disabled owner removes released projectile");
 shooter.gameObject.SetActive(true);shooter.target=h;shot=EnemyProjectile.Fire(shooter,shooter.transform.position+Vector3.up*2,h.transform.position+Vector3.up,shooter.damage,shooter.projectileSpeed,shooter.projectileMaterial);EnemyProjectile.Clear(s.gameObject.scene);yield return null;C(shot==null,"scene-scoped projectile clear");C(!e.HasAttackSlot(Time.time)||!e.WindingUp,"read-only slot query remains safe with inactive peers");
 }
}
