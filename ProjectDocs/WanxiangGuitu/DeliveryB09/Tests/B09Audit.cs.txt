using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class B09Audit {
 static CombatSession s;static int phase,frame;static EditorWindow view;static double deadline;static string Root=>ValidationStorage.ValidateRoot(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT"));
 public static void Run(){if(ValidationStorage.Root(Environment.GetCommandLineArgs(),true)!=Root)throw new Exception("B09 pre-play root mismatch");view=UI01Before.Size(1280,720);ShaderUtil.allowAsyncCompilation=false;deadline=EditorApplication.timeSinceStartup+180;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();view.Repaint();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("audit deadline");if(!EditorApplication.isPlaying)return;
 if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||!s.Saves.IsInitialized){s=null;return;}var v=s.hero.GetComponent<FantasyActorVisual>();var skins=v.model.GetComponentsInChildren<SkinnedMeshRenderer>(true);var a=v.model.GetComponentInChildren<Animation>(true);string text="MODEL="+v.model.name+"\nSKINNED="+skins.Length+"\nBONES="+skins.Sum(x=>x.bones.Length)+"\nANIMATOR="+v.model.GetComponentsInChildren<Animator>(true).Length+"\n";if(a!=null)foreach(AnimationState st in a){var b=AnimationUtility.GetCurveBindings(st.clip);text+=st.name+" length="+st.length+" legacy="+st.clip.legacy+" bindings="+b.Length+" resolved="+b.Count(x=>string.IsNullOrEmpty(x.path)||a.transform.Find(x.path)!=null)+"\n";}File.WriteAllText(Root+"/b09-model-audit.txt",text);s.Camp.Explore();frame=Time.frameCount;return;}
 if(phase==0&&Time.frameCount>frame+30){ScreenCapture.CaptureScreenshot(Root+"/b09-before-camp.png");phase=1;frame=Time.frameCount;return;}
 if(phase==1&&Time.frameCount>frame+3&&File.Exists(Root+"/b09-before-camp.png")){EditorApplication.update-=Tick;EditorApplication.Exit(0);}
 }catch(Exception e){Debug.LogException(e);EditorApplication.Exit(1);}}
}
public static class B09Probe {public static void Run(){if(typeof(CombatSession).Assembly.GetType("GameEntry")==null||typeof(CombatSession).Assembly.GetType("SaveCatalog")==null)throw new Exception("B09_MISSING_FORMAL_ENTRY_AND_SAVE_CATALOG");EditorApplication.Exit(0);}}
