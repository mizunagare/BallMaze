using System;using System.IO;using System.Reflection;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEditor.Build.Profile;
public static class PrivacyARealNegatives {
 public static bool Inject;
 [Serializable]class Row{public string name,status,error;public bool marker;}
 [Serializable]class Result{public Row[] rows;}
 public static void Run(){var rows=new List<Row>();var p=AssetDatabase.LoadAssetAtPath<BuildProfile>(ReleaseDiagnosticsGuard.ProfilePath);var f=BindingFlags.Public|BindingFlags.NonPublic|BindingFlags.Instance;var platform=typeof(BuildProfile).GetProperty("platformBuildProfile",f).GetValue(p);var prop=platform.GetType().GetProperty("insightsSettingsContainer",f);object container=prop.GetValue(platform);var state=container.GetType().GetProperty("buildProfileEngineDiagnosticsState",f);var old=state.GetValue(container);string root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);
 foreach(string name in new[]{"WrongProfile","Enabled","ProjectSettings","OldOutput","NativeReenabled"}){string exe=root+"/"+name+"/Game.exe";Directory.CreateDirectory(Path.GetDirectoryName(exe));if(name=="OldOutput")File.WriteAllText(exe,"old");PrivacyARealNegatives.Inject=name=="NativeReenabled";bool failed=false;string error="";try{BuildProfile.SetActiveBuildProfile(name=="WrongProfile"?null:p);if(name=="Enabled"||name=="ProjectSettings"){state.SetValue(container,Enum.Parse(state.PropertyType,name));prop.SetValue(platform,container);}var report=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{GameEntry.BootPath,GameEntry.TitlePath,GameEntry.ForestPath},target=BuildTarget.StandaloneWindows64,locationPathName=exe,options=BuildOptions.None});failed=report.summary.result!=UnityEditor.Build.Reporting.BuildResult.Succeeded;error=report.summary.result.ToString();}catch(Exception e){failed=true;error=e.ToString();}finally{PrivacyARealNegatives.Inject=false;NativeReenableProbe.Set(false);state.SetValue(container,old);prop.SetValue(platform,container);BuildProfile.SetActiveBuildProfile(p);}bool marker=File.Exists(exe+".release.json");rows.Add(new Row{name=name,status=failed&&!marker?"PASS":"FAIL",error=error,marker=marker});}
 File.WriteAllText(root+"/real-negatives.json",JsonUtility.ToJson(new Result{rows=rows.ToArray()},true));EditorApplication.Exit(rows.Exists(x=>x.status=="FAIL")?1:0);
 }
}
