using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;
public static class Input01Rules {
 static string Root=>ValidationStorage.ValidateRoot(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT"));static List<string> results=new List<string>();
 static void Need(bool b,string name){if(!b)throw new Exception(name);results.Add(name);}
 static PlayerBinding[] Change(PlayerAction a,string key,string second=""){var v=PlayerBindings.Defaults();var b=v.Single(x=>x.action==a.ToString());b.primary=key;b.secondary=second;return v;}
 public static void Run(){try{
  Need(ValidationStorage.Root(Environment.GetCommandLineArgs(),true)==Root,"pre-launch isolation");
  Need(PlayerBindings.Validate(PlayerBindings.Defaults())=="","I01 defaults including R mutually exclusive contexts");
  var path=Root+"/preferences.json";var p=GamePreferences.Load(path);Need(p.volume==1&&p.inputBindings==null&&!File.Exists(path),"I01 missing stays missing");
  File.WriteAllText(path,"{\"version\":1,\"width\":1920,\"height\":1080,\"volume\":0.35,\"sensitivity\":0.18,\"uiScale\":1.25,\"futureText\":\"ignored explicitly\"}");var old=File.ReadAllBytes(path);p=GamePreferences.Load(path);Need(p.volume==.35f&&p.width==1920&&p.uiScale==1.25f&&p.inputBindings==null&&old.SequenceEqual(File.ReadAllBytes(path)),"I01 old preference fields retained; load read-only; unknown JSON metadata ignored");
  Need(PlayerBindings.Validate(Change(PlayerAction.Skill,"E")).Contains("冲突"),"I04 same-context conflict rejected");Need(PlayerBindings.Validate(Change(PlayerAction.Skill,"Escape"))!="","I04 Escape reserved");Need(PlayerBindings.Validate(Change(PlayerAction.Skill,"F9"))!="","I04 F9 reserved no developer unlock");Need(PlayerBindings.Validate(Change(PlayerAction.Skill,"K","K"))!="","I04 duplicate alias rejected");Need(PlayerBindings.Validate(Change(PlayerAction.Forward,"MouseMiddle"))!="","I04 directional keyboard only");Need(PlayerBindings.Validate(Change(PlayerAction.Skill,"MouseMiddle"))=="","I04 discrete mouse allowed");Need(PlayerBindings.Validate(Change(PlayerAction.Skill,"MouseRight"))!="","I04 camera RMB reserved");Need(PlayerBindings.Validate(Change(PlayerAction.Skill,"Ctrl+K"))!="","I04 chord rejected");
  var draft=Change(PlayerAction.Skill,"K","L");Need(PlayerInput.Apply(draft),"valid custom apply");draft.Single(x=>x.action=="Skill").primary="J";Need(PlayerInput.Key(PlayerAction.Skill)=="K / L","I06 effective detached from editor copy");var snap=PlayerInput.Snapshot();snap[0].primary="UpArrow";Need(PlayerInput.Key(PlayerAction.Forward)=="W","I06 returned snapshot detached");
  p.inputBindings=Change(PlayerAction.Skill,"K","L");p.Save(path);Need(GamePreferences.Load(path).inputBindings.Single(x=>x.action=="Skill").primary=="K"&&p.volume==.35f&&File.Exists(path+".bak"),"I07 persisted bindings and backup preserve other options");
  string disk=File.ReadAllText(path),effective=PlayerInput.Key(PlayerAction.Skill);p=GamePreferences.Load(path);p.inputBindings=Change(PlayerAction.Skill,"J");p.BeforePublish=()=>throw new IOException("INPUT01 prepublish fault");bool failed=false;try{p.Save(path);}catch(IOException){failed=true;}Need(failed&&File.ReadAllText(path)==disk&&PlayerInput.Key(PlayerAction.Skill)==effective,"I07 before-publish failure keeps disk and effective bindings");
  p.BeforePublish=null;p.Save(path);Need(PlayerInput.Key(PlayerAction.Skill)=="J","I07 retry publishes once");
  p=GamePreferences.Load(path);File.AppendAllText(path," ");failed=false;try{p.Save(path);}catch(IOException){failed=true;}Need(failed&&File.ReadAllText(path).EndsWith(" "),"I07 external preference change is not overwritten");
  File.WriteAllText(path,"synthetic damaged preference");p=GamePreferences.Load(path);Need(p.Warning!=""&&p.inputBindings==null&&File.ReadAllText(path)=="synthetic damaged preference","I07 corrupt visible fallback without auto-write");
  var bad=new GamePreferences{volume=.4f,inputBindings=Change(PlayerAction.Skill,"E")};File.WriteAllText(path,JsonUtility.ToJson(bad));disk=File.ReadAllText(path);p=GamePreferences.Load(path);Need(p.Warning!=""&&p.volume==.4f&&p.inputBindings==null&&File.ReadAllText(path)==disk,"I07 conflict fallback retains audio and original file");
  bad.inputBindings=Change(PlayerAction.Skill,"K");bad.inputBindings[0].action="UnknownFutureAction";File.WriteAllText(path,JsonUtility.ToJson(bad));p=GamePreferences.Load(path);Need(p.Warning.Contains("未知")&&p.inputBindings==null,"I07 unknown action visible, not silently dropped");
  File.WriteAllText(path,"{\"version\":2}");Need(GamePreferences.Load(path).Warning!=""&&File.ReadAllText(path)=="{\"version\":2}","I07 unsupported preference version original retained");
  Need(PlayerInput.Apply(PlayerBindings.Defaults())&&PlayerInput.Key(PlayerAction.Interact)=="E"&&PlayerInput.Key(PlayerAction.Inventory)=="Tab"&&PlayerInput.Key(PlayerAction.Items)=="B","I01 default E H Tab B semantics available");
  File.WriteAllText(Root+"/input01-rules.json",JsonUtility.ToJson(new Report{status="PASS",checks=results.ToArray()},true));EditorApplication.Exit(0);
 }catch(Exception e){Debug.LogException(e);File.WriteAllText(Root+"/input01-rules-failure.txt",e.ToString());EditorApplication.Exit(1);}}
 [Serializable]class Report{public string status;public string[] checks;}
}
