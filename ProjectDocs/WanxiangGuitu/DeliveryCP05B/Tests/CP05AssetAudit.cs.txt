using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;
public static class CP05AssetAudit {
 [Serializable]class Row{public string path,guid,type;public int missing,hero,enemy;}
 [Serializable]class Result{public Row[] before,after;}
 static Row[] Snapshot(){string[] paths={"Assets/CombatPrototype/ThirdPersonHero.cs","Assets/CombatPrototype/AdventureCombat.cs","Assets/CombatPrototype/FantasyActorVisual.cs","Assets/CombatPrototype/AdventureLoadout.cs","Assets/CombatPrototype/Prefabs/Hero.prefab","Assets/CombatPrototype/Resources/CP05Training/Hero.prefab","Assets/CombatPrototype/Resources/CP05Training/Enemy.prefab"};return paths.Select(p=>{var r=new Row{path=p,guid=AssetDatabase.AssetPathToGUID(p)};if(p.EndsWith(".cs")){var script=AssetDatabase.LoadAssetAtPath<MonoScript>(p);r.type=script?.GetClass()?.FullName??"NULL";}else{var g=AssetDatabase.LoadAssetAtPath<GameObject>(p);r.missing=g==null?-1:g.GetComponentsInChildren<Component>(true).Count(c=>c==null);r.hero=g==null?0:g.GetComponents<ThirdPersonHero>().Length;r.enemy=g==null?0:g.GetComponents<CombatEnemy>().Length;}return r;}).ToArray();}
 public static void Run(){var root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);var before=Snapshot();foreach(var r in before)AssetDatabase.ImportAsset(r.path,ImportAssetOptions.ForceUpdate);AssetDatabase.Refresh(ImportAssetOptions.ForceUpdate);var after=Snapshot();File.WriteAllText(root+"/asset-audit.json",JsonUtility.ToJson(new Result{before=before,after=after},true));EditorApplication.Exit(after.Any(r=>r.type=="NULL"||r.missing!=0)?1:0);}
}
