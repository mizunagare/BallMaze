using System;using System.IO;using System.Linq;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEngine.Profiling;using UnityEditor;using UnityEditor.Profiling;using UnityEditorInternal;using UnityEditor.SceneManagement;
public static class CP05ScopedPerf {
 [Serializable]class Sample{public string marker,kind;public double stopwatchMs;}
 [Serializable]class Raw{public string marker,thread;public int frame;public double cpuMs;public long[] bytes;}
 [Serializable]class Report{public string scope="Edit-mode actual camera/selection/native Solve queries; NOT Player or whole-frame battle performance",status,unity;public Sample[] samples;public Raw[] raw;public string[] errors;public bool calibrated;}
 static string root;static ThirdPersonHero hero;static CombatEnemy enemy;static TargetAssist assist;static OrbitCamera orbit;static Camera camera;static MethodInfo late;static int step,tail,serial;static double deadline;static bool oldDriver,oldEditor,oldEngine,oldBinary;static EditorWindow view;static List<Sample> samples=new List<Sample>();static List<Raw> raw=new List<Raw>();static List<string> errors=new List<string>();static HashSet<int> processed=new HashSet<int>();static byte[][] keep=new byte[2][];
 static void Awake(MonoBehaviour b){b.GetType().GetMethod("Awake",BindingFlags.Instance|BindingFlags.NonPublic)?.Invoke(b,null);}
 public static void Run(){try{root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+180;var scene=EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Additive);UnityEngine.SceneManagement.SceneManager.SetActiveScene(scene);var g=new GameObject("Scoped query owner");hero=g.AddComponent<ThirdPersonHero>();Awake(g.GetComponent<CombatHealth>());Awake(hero);Awake(hero.Dodge);Awake(g.GetComponent<AdventureCombat>());assist=g.GetComponent<TargetAssist>();Awake(assist);var t=new GameObject("Scoped query target");enemy=t.AddComponent<CombatEnemy>();Awake(t.GetComponent<CombatHealth>());Awake(enemy);enemy.entityId="perf-target";enemy.transform.position=Vector3.forward*6;var cc=t.GetComponent<CharacterController>();cc.center=Vector3.up;cc.height=2;cc.radius=.4f;var cam=new GameObject("Scoped camera",typeof(Camera),typeof(OrbitCamera));camera=cam.GetComponent<Camera>();camera.fieldOfView=65;camera.nearClipPlane=.05f;camera.aspect=1280f/720;orbit=cam.GetComponent<OrbitCamera>();orbit.target=g.transform;hero.cameraTransform=cam.transform;late=typeof(OrbitCamera).GetMethod("LateUpdate",BindingFlags.Instance|BindingFlags.NonPublic);Pose();for(int i=0;i<8;i++)assist.Choose(g.transform.position+Vector3.up*1.25f+Vector3.forward*.7f,40);if(assist.Selected!=enemy)throw new Exception("Actual selection fixture not ready");
 oldDriver=ProfilerDriver.enabled;oldEditor=ProfilerDriver.profileEditor;oldEngine=Profiler.enabled;oldBinary=Profiler.enableBinaryLog;Profiler.enableBinaryLog=false;ProfilerDriver.profileEditor=true;ProfilerDriver.enabled=true;ProfilerDriver.SetAreaEnabled(ProfilerArea.CPU,true);Profiler.enabled=true;view=UI01Before.Size(1280,720);EditorApplication.update+=Tick;
 }catch(Exception e){Fail(e);}}
 static void Pose(){camera.transform.position=new Vector3(0,4,-5);camera.transform.LookAt(enemy.transform.position+Vector3.up);Physics.SyncTransforms();}
 static void Measure(string kind,Action action){string marker="CP05Scoped."+(++serial).ToString("D4");Profiler.BeginSample(marker);long before=System.Diagnostics.Stopwatch.GetTimestamp();action();long end=System.Diagnostics.Stopwatch.GetTimestamp();Profiler.EndSample();samples.Add(new Sample{marker=marker,kind=kind,stopwatchMs=(end-before)*1000.0/System.Diagnostics.Stopwatch.Frequency});}
 static void Capture(){int last=ProfilerDriver.lastFrameIndex;for(int f=Math.Max(ProfilerDriver.firstFrameIndex,last-5);f<last;f++){if(processed.Contains(f))continue;using(var d=ProfilerDriver.GetRawFrameDataView(f,0)){if(!d.valid)continue;processed.Add(f);for(int i=0;i<d.sampleCount;i++){string name=d.GetSampleName(i);if(!name.StartsWith("CP05Scoped."))continue;int end=i+d.GetSampleChildrenCountRecursive(i);var allocations=new List<long>();for(int j=i+1;j<=end&&j<d.sampleCount;j++)if(d.GetSampleName(j)=="GC.Alloc"){try{allocations.Add(d.GetSampleMetadataAsLong(j,0));}catch(Exception e){errors.Add(e.ToString());}}raw.Add(new Raw{marker=name,thread=d.threadName,frame=f,cpuMs=d.GetSampleTimeMs(i),bytes=allocations.ToArray()});}}}}
 static void Tick(){try{Capture();EditorApplication.QueuePlayerLoopUpdate();view?.Repaint();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("Bounded scoped profiler deadline");if(step<24){int mode=step%3;if(mode==0)Measure("zero",()=>{});if(mode==1)Measure("positive-65536",()=>keep[0]=new byte[65536]);if(mode==2){keep[1]=new byte[65536];Measure("release-reference",()=>keep[1]=null);GC.Collect();}step++;return;}if(step<72){Pose();int n=(step-24)/16;if(n==0)Measure("camera-reflection-inclusive",()=>late.Invoke(orbit,null));else if(n==1)Measure("target-query",()=>assist.Choose(hero.transform.position+Vector3.up*1.25f+Vector3.forward*.7f,40));else Measure("native-free-solve",()=>CombatAim.Solve(hero,hero.transform.position+Vector3.up*1.25f+Vector3.forward*.7f,"read-only query fixture",40));step++;return;}if(++tail<8)return;Capture();bool complete=raw.Count==samples.Count&&raw.Select(x=>x.marker).Distinct().Count()==samples.Count;bool calibrated=complete&&errors.Count==0&&samples.All(s=>{var r=raw.SingleOrDefault(x=>x.marker==s.marker);return r!=null&&(s.kind=="positive-65536"?r.bytes.Sum()>=65536:s.kind=="zero"||s.kind=="release-reference"?r.bytes.Sum()==0:true);});File.WriteAllText(root+"/scoped-perf.json",JsonUtility.ToJson(new Report{status=calibrated?"MEASURED":"NOT_RUN",unity=Application.unityVersion,calibrated=calibrated,samples=samples.ToArray(),raw=raw.ToArray(),errors=errors.ToArray()},true));Restore();EditorApplication.Exit(calibrated?0:3);}catch(Exception e){Fail(e);}}
 static void Restore(){EditorApplication.update-=Tick;ProfilerDriver.enabled=oldDriver;ProfilerDriver.profileEditor=oldEditor;Profiler.enabled=oldEngine;Profiler.enableBinaryLog=oldBinary;}
 static void Fail(Exception e){if(root!=null)File.WriteAllText(root+"/exception.txt",e.ToString());Restore();EditorApplication.Exit(1);}
}
