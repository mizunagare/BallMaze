using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
// Controlled calibration of health events, not AI wins or native interaction.
public static class S2R1BoundaryRules {
 static CombatSession s;static CombatEnemy enemy;static EditorWindow view;static int phase,frames;static double deadline,pausedClock;static float waitAt;
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT");static void Need(bool x,string m){if(!x)throw new Exception(m);}static void Mark(string m){File.AppendAllText(Root+"/s2r1-boundaries.txt",m+"\n");}
 public static void Run(){ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+180;view=UI01Before.Size(1280,720);EditorSceneManager.OpenScene(GameEntry.ForestPath);EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();Application.runInBackground=true;view?.Repaint();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("calibration deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.5f)return;
 if(phase==0){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||!s.Saves.IsInitialized)return;s.Saves.InitializeStorage(Root+"/BoundaryPlayer",false);AdventureRandom.SetNextSeedForValidation(1);Need(s.Flow.Depart(AdventureMode.ForestExploration),"controlled depart");s.hero.enabled=false;foreach(var e in s.enemies){e.enabled=false;if(e.GetComponent<ForestBossBrain>()!=null)e.GetComponent<ForestBossBrain>().enabled=false;if(e.GetComponent<EncounterGuard>()!=null)e.GetComponent<EncounterGuard>().enabled=false;}enemy=s.enemies.First(e=>e.archetype==EnemyArchetype.Melee);var h=enemy.Health;var global=JsonUtility.ToJson(UnityEngine.Random.state);var random=JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams);float start=h.Current,cumulative=h.TotalDamageTaken;int notifications=0,deaths=0;HealthObservation last=default;Action<HealthObservation> count=x=>{notifications++;if(x.deathTransition)deaths++;last=x;};h.Observed+=count;
 using(var o=new S2R1Observation(s,enemy,Root+"/calibration-same-frame")){h.Damage(0);Need(notifications==0,"zero damage notified");h.Damage(start/4);h.Heal(start/8);h.Damage(start/4);Need(Mathf.Abs((h.TotalDamageTaken-cumulative)-start/2)<.001f,"heal canceled cumulative loss");h.Damage(99999);Need(h.Dead&&deaths==1&&o.Data.hits==3&&!o.Data.firstHitKill&&o.Data.ttk==0,"same-frame ordered hits/overkill/death");int countBefore=notifications;h.Damage(1);o.Target(last);Need(notifications==countBefore&&deaths==1&&o.Data.hits==3,"duplicate death repeated measurement");Need(Mathf.Abs((h.TotalDamageTaken-cumulative)-start*1.125f)<.001f,"overkill counted requested amount");}
 h.Observed-=count;float total=h.TotalDamageTaken;h.ResetHealth();Need(h.TotalDamageTaken==total,"ResetHealth changed cumulative observation semantics");using(var o=new S2R1Observation(s,enemy,Root+"/calibration-reset")){h.Damage(1);h.ResetHealth();Need(o.Data.status=="censored"&&o.Data.ttk<0,"reset fabricated complete TTK");}
 h.ResetHealth();using(var o=new S2R1Observation(s,enemy,Root+"/calibration-restore")){h.Damage(1);h.RestoreHealth(h.Current);Need(o.Data.status=="censored","restore boundary not censored");}h.ResetHealth();using(var o=new S2R1Observation(s,enemy,Root+"/calibration-first-hit-kill")){h.Damage(h.Current+1);Need(o.Data.status=="killed"&&o.Data.firstHitKill&&o.Data.hits==1&&o.Data.ttk==0,"first hit kill zero is legal, not infinite DPS");}h.ResetHealth();
 bool afterFault=false;Action<HealthObservation> fault=x=>{throw new InvalidOperationException("S2R1_EXPECTED_OBSERVER_FAULT");};Action<HealthObservation> later=x=>afterFault=true;h.Observed+=fault;h.Observed+=later;float hp=h.Current;Debug.Log("S2R1_EXPECTED_OBSERVER_FAULT_BEGIN");h.Damage(1);h.Observed-=fault;h.Observed-=later;Need(h.Current<hp&&afterFault,"subscriber fault changed damage or blocked subsequent observer");h.ResetHealth();
 var go=new GameObject("isolated unload calibration only");var plain=go.AddComponent<CombatHealth>();int unload=0;plain.Observed+=x=>{if(x.kind==HealthObservationKind.Unload)unload++;};UnityEngine.Object.DestroyImmediate(go);Need(unload==1,"unload transition missing");
Need(global==JsonUtility.ToJson(UnityEngine.Random.state)&&random==JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams),"observation/calibration changed gameplay/global RNG");Mark("HEALTH_BOUNDARIES_RNG_PASS_CONTROLLED");waitAt=Time.time+.05f;phase=3;return;}
 if(phase==3){if(Time.time<waitAt)return; int heroEvents=0;Action<HealthObservation> heroCount=x=>{if(x.kind==HealthObservationKind.Damage)heroEvents++;};s.hero.Health.Observed+=heroCount;Need(s.hero.Dodge.TryDodge(Vector3.right),"actual dodge start");float heroHp=s.hero.Health.Current;s.hero.Health.Damage(10);s.hero.Health.Observed-=heroCount;Need(s.hero.Health.Current==heroHp&&heroEvents==0,"invulnerable damage produced applied hit");Mark("IFRAME_PASS_CONTROLLED");waitAt=Time.time+.55f;phase=1;return;}
 if(phase==1){if(Time.time<waitAt)return;s.hero.Inventory.SetOpen(true);Need(Time.timeScale==0&&s.hero.GameplayPaused,"actual pause");pausedClock=Time.timeAsDouble;frames=0;phase=2;return;}
 if(phase==2){if(++frames<4)return;Need(Time.timeAsDouble==pausedClock,"pause advanced activity clock");s.hero.Inventory.SetOpen(false);Mark("ACTUAL_PAUSE_ACTIVITY_CLOCK_PASS");EditorApplication.update-=Tick;EditorApplication.Exit(0);}
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
