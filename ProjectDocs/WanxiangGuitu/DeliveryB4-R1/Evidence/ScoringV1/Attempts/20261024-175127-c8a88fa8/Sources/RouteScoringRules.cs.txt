using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
[Serializable] public class RouteEvaluationSet { public RouteEvaluation[] evaluations; }
public static class RouteScoringRules {
 static CombatSession s;static double deadline;static bool resume;
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT");
 static void Need(bool value,string message){if(!value)throw new Exception(message);}
 static string Data(RouteEvaluation[] rows)=>JsonUtility.ToJson(new RouteEvaluationSet{evaluations=rows});
 static void Mark(string text){File.AppendAllText(Root+"/score-results.txt",text+"\n");Debug.Log("SCORE_"+text);}
 public static void Run(){Start(false);}public static void Resume(){Start(true);}
 static void Start(bool load){resume=load;ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+300;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static string State()=>JsonUtility.ToJson(s.Flow.ActiveRun)+JsonUtility.ToJson(s.Flow.Profile)+JsonUtility.ToJson(s.GetComponent<ForestRegionRuntime>().Capture())+string.Join(",",Enumerable.Range(0,6).Select(i=>s.hero.Inventory.Count((MaterialKind)i)))+"/"+UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Length+"/"+s.Saves.HasSave+"/"+s.Flow.Revision;
 static void Freeze(){foreach(var enemy in s.enemies)enemy.enabled=false;}
 static RouteEvaluation Find(string id)=>RouteScoring.Evaluate(s).Single(r=>r.poiId==id);
 static void Log(string name){File.AppendAllText(Root+"/score-components.jsonl","{\"case\":\""+name+"\",\"data\":"+Data(RouteScoring.Evaluate(s))+"}\n");}
 static void Finite(RouteEvaluation[] rows){foreach(var row in rows)foreach(var f in typeof(RouteEvaluation).GetFields().Where(f=>f.FieldType==typeof(float))){float v=(float)f.GetValue(row);Need(!float.IsNaN(v)&&!float.IsInfinity(v),"nonfinite "+row.poiId+"/"+f.Name);}}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("scoring deadline");if(!EditorApplication.isPlaying||Time.time<.4f)return;s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(Root+"/score-save",false);
  if(resume){Need(s.Saves.LoadProgress(false),"score new process load");Freeze();var actual=Data(RouteScoring.Evaluate(s));Need(actual==File.ReadAllText(Root+"/score-checkpoint.txt"),"score components changed after fresh process");Mark("NEW_PROCESS_ALL_COMPONENTS_EXACT_PASS");Finish();return;}
  Need(s.Flow.Depart(AdventureMode.ForestExploration),"depart");Freeze();var region=s.GetComponent<ForestRegionRuntime>();var config=RouteScoringConfig.Read();File.WriteAllText(Root+"/score-configuration.json",JsonUtility.ToJson(config,true));
  var invalid=RouteScoringConfig.Read();invalid.distanceReference=0;bool refused=false;try{invalid.Validate();}catch(InvalidDataException){refused=true;}Need(refused,"zero reference accepted");
  // Explicit controlled state construction; never claimed to be normal AI or native UI.
  int changed=0;foreach(int seed in new[]{1,17,29}){region.GeneratePlayable(seed);Freeze();s.hero.RestorePosition(region.Layout.entry,0);s.Flow.Profile.craftingTarget=null;s.Flow.ActiveRun.goal=ExpeditionGoal.Gather;region.InvalidateRoutes();
   var before=State();var rng=JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams);var global=JsonUtility.ToJson(UnityEngine.Random.state);string first=Data(RouteScoring.Evaluate(s));int queries=region.RouteQueries,approaches=region.ApproachQueries;
   for(int i=0;i<8;i++)Need(first==Data(RouteScoring.Evaluate(s)),"same state score changed");Need(queries==region.RouteQueries&&approaches==region.ApproachQueries,"repeat query recalculated navigation/approach");region.InvalidateRoutes();Need(first==Data(RouteScoring.Evaluate(s)),"fresh path changed score");Need(before==State()&&rng==JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams)&&global==JsonUtility.ToJson(UnityEngine.Random.state),"read query mutated world/RNG");
   var streams=AdventureData.Clone(s.Flow.ActiveRun.randomStreams);var random=UnityEngine.Random.state;RouteScoring.Evaluate(s);var rewardA=ForestRewards.Chest(ForestSiteKind.Ruins,s);string afterA=JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams);s.Flow.ActiveRun.randomStreams=AdventureData.Clone(streams);UnityEngine.Random.state=random;var rewardB=ForestRewards.Chest(ForestSiteKind.Ruins,s);Need(string.Join(",",rewardA.Select(v=>v.material+":"+v.amount))==string.Join(",",rewardB.Select(v=>v.material+":"+v.amount))&&afterA==JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams),"query changed actual reward result/stream");s.Flow.ActiveRun.randomStreams=streams;UnityEngine.Random.state=random;
   var near=RouteScoringConfig.Read();near.targetWeight=near.yieldWeight=near.extractionWeight=near.noveltyWeight=near.destinationRiskWeight=near.pathRiskWeight=near.eliteRiskWeight=0;near.distanceWeight=1;
   string ranked=string.Join(",",RouteScoring.Ranked(s).Select(r=>r.poiId)),distanceOnly=string.Join(",",RouteScoring.Ranked(s,near).Select(r=>r.poiId));if(ranked!=distanceOnly)changed++;
   File.AppendAllText(Root+"/score-order.csv",seed+","+region.Layout.seed+",\""+ranked+"\",\""+distanceOnly+"\"\n");Log("seed_"+seed);Finite(RouteScoring.Evaluate(s));
  }Need(changed>0,"score has no effect on recommendation order");Mark("FIXED_SEEDS_REPEAT_FRESH_PATH_CACHE_WORLD_RNG_ACTUAL_REWARD_AND_ORDER_PASS changed="+changed);
  region.GeneratePlayable(1);Freeze();s.hero.RestorePosition(region.Layout.entry,0);s.Flow.Profile.craftingTarget=null;s.Flow.ActiveRun.goal=ExpeditionGoal.Gather;region.InvalidateRoutes();
  s.hero.Inventory.Clear();for(int i=0;i<6;i++)s.Camp.Materials[i]=0;s.Flow.Profile.craftingTarget=new CraftingTarget{type=GearType.Sword,quality=GearQuality.White};Need(CraftingTargets.Needs(s).Sum(n=>n.Missing)==5,"white recipe gap fixture changed");Need(Mathf.Abs(Find("poi:1").targetMatch-.4f)<.00001f&&Mathf.Abs(Find("poi:2").targetMatch-.6f)<.00001f,"real recipe shortage weighting incorrect");Log("white_recipe_known_gap");s.Flow.Profile.craftingTarget=null;
  var wood=region.Layout.points.Single(p=>p.id=="poi:1");s.hero.RestorePosition(wood.position,0);region.InvalidateRoutes();var zero=Find(wood.id);Need(zero.reachable&&zero.navigationDistance<.01f&&zero.distanceCost<.0001f,"zero-distance handling");Finite(new[]{zero});Log("zero_distance");s.hero.RestorePosition(region.Layout.points.Single(p=>p.id=="poi:4").position,0);region.InvalidateRoutes();Need(Find("poi:4").risk.pathElites>0,"stationary elite proximity omitted");
  s.hero.RestorePosition(region.Layout.entry,0);var old=wood.position;wood.position=new Vector3(5000,0,5000);region.InvalidateRoutes();var missing=Find(wood.id);Need(!missing.reachable&&!missing.eligible&&missing.navigationDistance==-1&&missing.exclusionReasons.Contains("NO_PATH"),"unreachable scored");Log("no_path");wood.position=old;region.InvalidateRoutes();
  var exits=UnityEngine.Object.FindObjectsByType<ExtractionPoint>(FindObjectsSortMode.None).Where(p=>p.forest).ToArray();foreach(var p in exits)p.gameObject.SetActive(false);Need(RouteScoring.Evaluate(s).All(r=>!r.eligible&&(!r.reachable||r.exclusionReasons.Contains("NO_USABLE_EXIT"))),"no exit scored");Log("no_exits");foreach(var p in exits)p.gameObject.SetActive(true);
  var ferry=exits.Single(p=>p.Capture().repairRequired);var free=exits.Where(p=>!p.Capture().repairRequired).ToArray();foreach(var p in free)p.gameObject.SetActive(false);s.hero.Inventory.Clear();ferry.SetRepair(false,"");region.InvalidateRoutes();var poor=Find(wood.id);Need(!poor.eligible&&poor.exits.Single().reason=="MISSING_REPAIR_MATERIALS","missing repair materials ignored");Log("missing_repair_materials");s.hero.Inventory.Add(MaterialKind.Wood,2);s.hero.Inventory.Add(MaterialKind.Stone,1);string unchanged=State();var conditional=Find(wood.id);Need(conditional.exitAvailable&&conditional.exits.Single().availability=="REPAIRABLE"&&conditional.exits.Single().requiresRepair&&State()==unchanged&&!ferry.Capture().repaired,"query repaired/charged ferry");Log("repairable_not_paid");
  ferry.SetRepair(true,"CONTROLLED_NOT_PERSISTED");s.hero.Inventory.Clear();var repaired=Find(wood.id);Need(repaired.exitAvailable&&repaired.exits.Single().availability=="REPAIRED"&&!repaired.exits.Single().requiresRepair&&repaired.extractionConvenience>conditional.extractionConvenience,"repaired exit handling");Log("repaired_ferry");foreach(var p in free)p.gameObject.SetActive(true);
  var lake=Find("poi:6");var avoid=region.NavigationRoute(s.hero.transform.position,"poi:6",true);Need(lake.risk.pathElites>0&&lake.bypassDistance>=0&&avoid.reachable&&!avoid.nearElite,"elite bypass not real");Log("elite_bypass");
  var chest=s.Events.Points.Single(e=>e.entityId=="poi:1:chest");var beforeChest=Find(wood.id);chest.RestoreUsed(true);var afterChest=Find(wood.id);Need(afterChest.unopenedChests==beforeChest.unopenedChests-1&&Mathf.Abs(beforeChest.expectedMaterialUnits-afterChest.expectedMaterialUnits-8)<.00001f&&Mathf.Abs(beforeChest.expectedGear-afterChest.expectedGear-.25f)<.00001f,"opened chest still rewarded");Log("opened_chest");
  foreach(var node in UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).Where(n=>CraftingTargets.At(region.Layout,n.transform.position,wood.id)))node.Restore(0);foreach(var enemy in s.enemies.Where(e=>e.GetComponent<EncounterGuard>()?.poi==wood.id))enemy.RestoreState(new SavedEnemy{entityId=enemy.entityId,position=enemy.transform.position,health=0});var empty=Find(wood.id);Need(!empty.eligible&&empty.expectedMaterialUnits==0&&empty.expectedGear==0&&empty.exclusionReasons.Contains("NO_REMAINING_YIELD"),"depleted site recommended");Log("exhausted");
  s.Flow.Profile.craftingTarget=new CraftingTarget{type=GearType.Sword,quality=GearQuality.White};var warehouse=(int[])s.Camp.Materials.Clone();for(int i=0;i<6;i++)s.Camp.Materials[i]=100;Need(CraftingTargets.Needs(s).All(n=>n.Missing==0),"zero shortage setup");var noNeed=RouteScoring.Evaluate(s);Need(noNeed.All(e=>e.targetMatch==0),"fulfilled goal invented shortage");Finite(noNeed);Log("zero_target_gap");for(int i=0;i<6;i++)s.Camp.Materials[i]=warehouse[i];Mark("ZERO_DISTANCE_NO_PATH_NO_EXIT_FERRY_FEES_BYPASS_OPENED_DEPLETED_ZERO_GAP_PASS controlled setups");
  // Discard every synthetic boundary mutation by restoring the original legitimate departure checkpoint.
  Need(s.Saves.LoadProgress(false),"original valid departure restore");Freeze();region.InvalidateRoutes();Need(s.Saves.SaveProgress(false),"score checkpoint save");File.WriteAllText(Root+"/score-checkpoint.txt",Data(RouteScoring.Evaluate(s)));Mark("ORIGINAL_VALID_CHECKPOINT_SAVED_FOR_NEW_PROCESS");Finish();
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
 static void Finish(){EditorApplication.update-=Tick;EditorApplication.Exit(0);}
}
