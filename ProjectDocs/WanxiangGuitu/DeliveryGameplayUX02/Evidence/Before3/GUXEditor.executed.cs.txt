using System;using System.IO;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.InputSystem;
public static class GUXEditor {
 static string root;static double deadline;static int frame=-1;static bool started;
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+240;EditorSceneManager.playModeStartScene=null;EditorSceneManager.OpenScene(GameEntry.ForestPath);EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("bounded GAMEPLAY-UX02 Editor deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f||Time.frameCount==frame)return;frame=Time.frameCount;if(!started){var s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||s.Saves==null||!s.Saves.IsInitialized||Time.frameCount<6)return;InputSystem.settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;GUXCases.Begin(s,root,"controlled-editor");started=true;}GUXCases.Tick();if(GUXCases.Done){EditorApplication.update-=Tick;EditorApplication.Exit(GUXCases.ExitCode);}}catch(Exception e){File.WriteAllText(root+"/exception.txt",e.ToString());EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
