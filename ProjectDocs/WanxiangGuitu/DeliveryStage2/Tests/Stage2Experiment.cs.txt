using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;
[Serializable] public class Stage2Case {public string id,build,strategy;public int seed,weapon,target;}
[Serializable] public class Stage2RunMetrics {public Stage2Case input;public string scope="Editor normal enemy AI, synthetic camp, actual CC/InputSystem device movement; NOT OS native or human",status="IN_PROGRESS",profile,run,region,exitId,error,departureUtc,settlementUtc,secondRun,secondRegion,growthItem;public int requestedSeed,actualSeed,revision,generationVersion,randomVersion,potionsUsed,pendingItems;public bool fallback,extracted,committed,acknowledged,secondForest;public float elapsedSeconds,meters,damageTaken,attack,defense,moveMultiplier,growthAttack,growthDefense;public int harvestBonus,bagCells;public string[] visited,killed,badges;public int[] obtained=new int[6],consumed=new int[6],runConsumed=new int[6],campConsumed=new int[6],kept=new int[6],lost=new int[6],warehouseBefore,warehouseAfter,assetsStart,assetsEnd;public int keptGear,lostGear;}
public static class Stage2Experiment {
 static string root;static Stage2Case config;static Stage2RunMetrics data;static int frame=-1,lastPhase=-1;static float lastDamage;static Vector3 lastPosition;static bool positionReady;static HashSet<string> visited,killed;static CombatSession session;
 public static Stage2Case Config=>config??(config=JsonUtility.FromJson<Stage2Case>(File.ReadAllText(Environment.GetEnvironmentVariable("WG_STAGE2_CONFIG")??throw new Exception("explicit experiment config required"))));
 public static void Validate(string storage){ValidationStorage.ValidateRoot(storage);var c=Config;if(c.seed<=0||c.weapon<0||c.weapon>3||c.target<4||c.target>8||!new[]{"Safety","Gather","Boss"}.Contains(c.build)||!new[]{"Gather","Treasure","Bypass","Land","Lake"}.Contains(c.strategy)||string.IsNullOrEmpty(c.id))throw new Exception("invalid explicit Stage2 case");}
 public static void Attach(CombatSession s,string storage,bool resume){root=storage;session=s;Validate(storage);data=resume?JsonUtility.FromJson<Stage2RunMetrics>(File.ReadAllText(root+"/stage2-metrics.json")):new Stage2RunMetrics{input=Config,requestedSeed=Config.seed};if(JsonUtility.ToJson(data.input)!=JsonUtility.ToJson(Config))throw new Exception("resume config changed");visited=new HashSet<string>(data.visited??new string[0]);killed=new HashSet<string>(data.killed??new string[0]);lastDamage=s.hero.Health.TotalDamageTaken;frame=lastPhase=-1;positionReady=false;}
 public static void Depart(CombatSession s){var run=s.Flow.ActiveRun;var g=s.hero.GetComponent<AdventureLoadout>();data.departureUtc=DateTime.UtcNow.ToString("o");data.profile=s.Flow.Profile.profileId;data.run=run.runId;data.region=run.regionInstanceId;data.actualSeed=s.GetComponent<ForestRegionRuntime>().Layout.seed;data.revision=s.GetComponent<ForestRegionRuntime>().Layout.revision;data.generationVersion=run.generationVersion;data.randomVersion=run.randomStreams.version;data.fallback=s.GetComponent<ForestRegionRuntime>().Layout.fallback;data.attack=s.hero.AttackPower;data.defense=s.hero.Defense;data.moveMultiplier=s.hero.MoveMultiplier;data.harvestBonus=g.HarvestBonus;data.bagCells=g.Bag.Capacity;data.warehouseBefore=(int[])s.Flow.Profile.materials.Clone();data.assetsStart=Materials().Select((n,i)=>n+s.Flow.Profile.materials[i]).ToArray();Record("depart");}
 static int[] Materials()=>Enum.GetValues(typeof(MaterialKind)).Cast<MaterialKind>().Select(session.hero.Inventory.Count).ToArray();
 public static void Record(string label){if(data==null)return;ObserveDamage();var s=session;var run=s.Flow.ActiveRun;var g=s.hero.GetComponent<AdventureLoadout>();if(run!=null&&run.runId==data.run&&!string.IsNullOrEmpty(run.exitId))data.exitId=run.exitId;var snap=new Snapshot{label=label,utc=DateTime.UtcNow.ToString("o"),profile=s.Flow.Profile.profileId,run=run?.runId,region=run?.regionInstanceId,materials=Materials(),warehouse=(int[])s.Flow.Profile.materials.Clone(),potions=s.hero.Inventory.HealthPotionCount,level=s.hero.Progression.Level,experience=s.hero.Progression.Experience,ranks=s.hero.Progression.CaptureRanks(),pending=s.Flow.Profile.pending.Length,brought=run?.broughtRemaining,earned=run?.earnedRemaining,main=g.mainWeapon,reserve=g.reserveWeapon,pendingRewards=s.Flow.Profile.pending,random=run==null?null:JsonUtility.ToJson(run.randomStreams),items=g.items.Select(i=>new Item{id=i.id,type=(int)i.type,quality=(int)i.quality,definition=i.definitionId,origin=(int)i.origin,originRun=i.originRunId,board=i.board,worn=i.worn,overflow=i.overflow,attack=i.Attack,defense=i.Defense}).ToArray(),settlement=s.Flow.Settlement?.settlementId,settlementStatus=s.Flow.Settlement==null?"none":s.Flow.Settlement.status.ToString(),ack=s.Flow.Settlement?.acknowledged??false};File.AppendAllText(root+"/stage2-transactions.jsonl",JsonUtility.ToJson(snap)+"\n");Save();}
 public static bool Pickup(LootPickup loot,PlayerInventory inventory){var s=session;int[] before=Materials();Record("before-pickup:"+loot.InteractionId);bool ok=loot.PickUp(inventory);if(ok){var after=Materials();for(int i=0;i<6;i++)data.obtained[i]+=Math.Max(0,after[i]-before[i]);}Record("after-pickup:"+loot.InteractionId+":"+ok);return ok;}
 public static void Potion(){data.potionsUsed++;Record("actual-potion-used");}
 static void ObserveDamage(){if(data==null||session==null)return;float now=session.hero.Health.TotalDamageTaken;if(now>=lastDamage)data.damageTaken+=now-lastDamage;lastDamage=now;}
 public static void Observe(CombatSession s,int phase){if(data==null||frame==Time.frameCount)return;frame=Time.frameCount;ObserveDamage();if(s.Flow.ActiveRun?.runId==data.run){data.elapsedSeconds+=Time.deltaTime;if(positionReady)data.meters+=Vector3.ProjectOnPlane(s.hero.transform.position-lastPosition,Vector3.up).magnitude;lastPosition=s.hero.transform.position;positionReady=true;foreach(var p in s.GetComponent<ForestRegionRuntime>().Layout.points)if(Vector3.Distance(s.hero.transform.position,p.position)<9)visited.Add(p.id);foreach(var e in s.enemies.Where(e=>e.Health.Dead))killed.Add(e.entityId);}else positionReady=false;if(phase!=lastPhase){lastPhase=phase;Record("phase:"+phase);}}
 public static bool Cost(string label,Func<bool> action){var before=Materials();Record("before-cost:"+label);bool ok=action();var after=Materials();if(data.run!=null)for(int i=0;i<6;i++){int spent=Math.Max(0,before[i]-after[i]);data.consumed[i]+=spent;if(session.Flow.ActiveRun?.runId==data.run)data.runConsumed[i]+=spent;else data.campConsumed[i]+=spent;}Record("after-cost:"+label+":"+ok);return ok;}
 public static void BeforeCost(string name){Record("before-cost:"+name);}
 public static void AfterCost(string name,int[] before){var after=Materials();for(int i=0;i<6;i++)data.consumed[i]+=Math.Max(0,before[i]-after[i]);Record("after-cost:"+name);}
 public static int[] BeforeMaterials()=>Materials();
 public static void Pending(CombatSession s){var result=s.Flow.Settlement;data.settlementUtc=DateTime.UtcNow.ToString("o");data.extracted=result.outcome==AdventureOutcome.Extracted;data.kept=(int[])result.keptMaterials.Clone();data.lost=(int[])result.lostMaterials.Clone();data.keptGear=result.keptGear.Length;data.lostGear=result.lostGear.Length;Record("settlement-pending");}
 public static void Growth(CombatSession s,float attack,float defense,string itemId){data.growthItem=itemId;data.growthAttack=s.hero.AttackPower-attack;data.growthDefense=s.hero.Defense-defense;if(data.growthAttack<=0&&data.growthDefense<=0)throw new Exception("manufactured gear did not actually improve equipped attributes");data.committed=true;data.acknowledged=true;data.warehouseAfter=(int[])s.Flow.Profile.materials.Clone();data.assetsEnd=Materials().Select((n,i)=>n+s.Flow.Profile.materials[i]).ToArray();data.pendingItems=s.Flow.Profile.pending.Length;data.badges=s.Flow.Profile.badges;Record("committed-withdraw-manufacture-equip");}
 public static void Complete(CombatSession s){data.secondRun=s.Flow.ActiveRun.runId;data.secondRegion=s.Flow.ActiveRun.regionInstanceId;if(!s.hero.GetComponent<AdventureLoadout>().items.Any(i=>i.id==data.growthItem)||!s.Flow.ActiveRun.entryGear.Any(i=>i.id==data.growthItem))throw new Exception("manufactured growth item not carried/protected into next Run");data.secondForest=s.Flow.ActiveRun.mode==AdventureMode.ForestExploration&&s.Flow.ActiveRun.runId!=data.run;if(!data.extracted||!data.committed||!data.acknowledged||!data.secondForest)throw new Exception("incomplete Stage2 economic/reentry result");data.status="PASS";Record("verified-new-process-second-forest");}
 public static void Failure(Exception e){if(data==null)return;data.status="FAIL";data.error=e.ToString();Record("failed");}
 static void Save(){data.visited=visited.OrderBy(x=>x).ToArray();data.killed=killed.OrderBy(x=>x).ToArray();File.WriteAllText(root+"/stage2-metrics.json",JsonUtility.ToJson(data,true));}
 [Serializable] class Snapshot {public string label,utc,profile,run,region,random,settlement,settlementStatus,main,reserve;public PendingReward[] pendingRewards;public int[] materials,warehouse,brought,earned;public int potions,pending,level,experience;public int[] ranks;public bool ack;public Item[] items;}
 [Serializable] class Item {public string id,definition,originRun;public int type,quality,origin;public bool board,worn,overflow;public float attack,defense;}
}
