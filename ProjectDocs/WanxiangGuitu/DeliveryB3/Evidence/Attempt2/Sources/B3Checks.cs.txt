using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class B3Checks {
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")??throw new Exception("isolated root required");
 static CombatSession s;static double deadline;static bool resume;
 static void Need(bool value,string why){if(!value)throw new Exception(why);}
 static void Mark(string text){Debug.Log("B3_"+text);File.AppendAllText(Path.Combine(Root,"b3-results.txt"),text+"\n");}
 public static void Run(){Start(false);}public static void Resume(){Start(true);}
 static void Start(bool load){resume=load;ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+120;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static string Rewards(){return string.Join("|",Enumerable.Range(0,24).Select(_=>{var g=AdventureItems.Roll(12,true);var m=ForestRewards.Chest(ForestSiteKind.Ruins);return (int)g.type+":"+(int)g.quality+":"+string.Join(",",g.affixes??new int[0])+":"+string.Join(",",m.Select(x=>x.amount));}));}
 static string Ground()=>string.Join("|",UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Where(l=>l.gameObject.activeSelf).Select(l=>l.Gear==null?l.Kind+":"+l.Amount:JsonUtility.ToJson(l.Gear)).OrderBy(x=>x));
 static string Streams()=>JsonUtility.ToJson(s.Flow.ActiveRun);
 static void Noise(){for(int i=0;i<500;i++){var v=UnityEngine.Random.value;}ForestLayout.Generate(9182);}
 static void Queries(){CraftingTargets.Needs(s);CraftingTargets.Recommendations(s);CraftingTargets.Compact(s);foreach(var p in s.GetComponent<ForestRegionRuntime>().Layout.points)CraftingTargets.Available(s,p);}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("B3 timeout");if(!EditorApplication.isPlaying)return;
 if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||Time.time<.3f){s=null;return;}s.Saves.InitializeStorage(Path.Combine(Root,"b3-save"),resume);
  if(resume){Need(s.Saves.LoadProgress(false),"new process load");var expected=File.ReadAllLines(Path.Combine(Root,"b3-expected.txt"));Need(s.Flow.ActiveRun.runId==expected[0]&&s.Flow.ActiveRun.regionInstanceId==expected[1]&&s.Flow.ActiveRun.seed.ToString()==expected[2],"new process identity");Need(JsonUtility.ToJson(s.GetComponent<ForestRegionRuntime>().Capture())==expected[3],"layout rerolled");Need(Ground()==expected[5]&&string.Join(",",s.Flow.ActiveRun.earnedRemaining)==expected[6],"new process generated gear/ledger changed");Need(s.Events.Points.Single(p=>p.entityId==expected[7]).Used,"new process chest reopened");Need(Rewards()==expected[4],"new process reward stream changed");Mark("NEW_PROCESS_ID_LAYOUT_NEXT_REWARDS_PASS");Finish();return;}
  Need(CraftingTargets.Select(s.hero,GearType.Sword,GearQuality.White),"target");Need(s.Flow.Depart(AdventureMode.ForestExploration),"depart");foreach(var e in s.enemies)e.enabled=false;
  Need(s.Saves.SaveProgress(false),"checkpoint");var layout=JsonUtility.ToJson(s.GetComponent<ForestRegionRuntime>().Capture());var expectedRewards=Rewards();Need(s.Saves.LoadProgress(false),"restore baseline");foreach(var e in s.enemies)e.enabled=false;Noise();Need(Rewards()==expectedRewards,"GLOBAL_NOISE_COUPLED_REWARDS");Mark("GLOBAL_NOISE_AND_LAYOUT_GENERATION_DO_NOT_CHANGE_REWARDS_PASS");
  Need(s.Saves.LoadProgress(false),"restore query baseline");var before=Streams();var global=JsonUtility.ToJson(UnityEngine.Random.state);Queries();Need(before==Streams()&&global==JsonUtility.ToJson(UnityEngine.Random.state),"query state changed");Need(Rewards()==expectedRewards,"query reward changed");Mark("READ_ONLY_QUERIES_PRESERVE_STREAM_AND_REWARD_SEQUENCE_PASS");
  Need(s.Saves.LoadProgress(false),"restore repeat");Need(Rewards()==expectedRewards,"repeat checkpoint rewards");Need(layout==JsonUtility.ToJson(s.GetComponent<ForestRegionRuntime>().Capture()),"layout changed");Mark("SAME_CHECKPOINT_SEQUENCE_REPRODUCIBLE_PASS");
  Need(s.Saves.LoadProgress(false),"restore before real chest");foreach(var e in s.enemies)e.enabled=false;
  var region=s.GetComponent<ForestRegionRuntime>();var site=region.Layout.points.Single(p=>p.siteKind==ForestSiteKind.Ruins);var chest=s.Events.Points.Single(p=>p.entityId==site.id+":chest");s.hero.RestorePosition(chest.transform.position+Vector3.back*1.6f+Vector3.up*.1f,0);Physics.SyncTransforms();var ledger=string.Join(",",s.Flow.ActiveRun.earnedRemaining);Need(chest.Interact(s.hero),"actual ruins open");Need(!chest.Interact(s.hero),"duplicate open");Need(ledger==string.Join(",",s.Flow.ActiveRun.earnedRemaining),"unpicked auto claimed");Need(s.Saves.SaveProgress(false),"opened chest checkpoint");var ground=string.Join("|",UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Select(l=>l.Gear==null?l.Kind+":"+l.Amount:JsonUtility.ToJson(l.Gear)).OrderBy(x=>x));Noise();Need(s.Saves.LoadProgress(false),"opened chest restore");Need(s.Events.Points.Single(p=>p.entityId==site.id+":chest").Used,"opened chest reset");Need(ground==string.Join("|",UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Where(l=>l.gameObject.activeSelf).Select(l=>l.Gear==null?l.Kind+":"+l.Amount:JsonUtility.ToJson(l.Gear)).OrderBy(x=>x)),"determined ground gear/material rerolled");Mark("ACTUAL_RUINS_USED_AND_GEAR_IDS_RESTORE_WITHOUT_REROLL_PASS");
  var isolated=AdventureData.Clone(s.Flow.ActiveRun.randomStreams);s.Flow.ActiveRun.randomStreams=null;Need(s.Saves.SaveProgress(false),"legacy null fixture checkpoint");var legacySequence=Rewards();Need(s.Saves.LoadProgress(false)&&s.Flow.ActiveRun.randomStreams==null,"legacy null stream auto upgraded");Need(Rewards()==legacySequence,"legacy reward semantics changed");s.Flow.ActiveRun.randomStreams=isolated;Need(s.Saves.SaveProgress(false),"return isolated checkpoint");Mark("OLD_V9_NULL_STREAM_RESTORES_LEGACY_SEQUENCE_PASS");
  var failureExpected=Rewards();s.Saves.WriteValidationHook=point=>{if(point=="before-publish")throw new IOException("B3_SYNTHETIC_STATE_WRITE_FAILURE");};Need(!s.Saves.SaveProgress(false),"write fault hidden");s.Saves.WriteValidationHook=null;Need(s.Saves.LoadProgress(false),"write failure recover");Need(Rewards()==failureExpected,"failed write advanced durable random stream");Mark("FAILED_PUBLISH_RELOAD_PRESERVES_DURABLE_RANDOM_SEQUENCE_PASS");
  Need(s.Saves.SaveProgress(false),"save next sequence");var next=Rewards();Need(s.Saves.LoadProgress(false),"reload next");File.WriteAllLines(Path.Combine(Root,"b3-expected.txt"),new[]{s.Flow.ActiveRun.runId,s.Flow.ActiveRun.regionInstanceId,s.Flow.ActiveRun.seed.ToString(),layout,next,Ground(),string.Join(",",s.Flow.ActiveRun.earnedRemaining),site.id+":chest"});Mark("PASS controlled; enemies disabled; not native UI");Finish();
 }
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
 static void Finish(){EditorApplication.update-=Tick;EditorApplication.Exit(0);}
}
