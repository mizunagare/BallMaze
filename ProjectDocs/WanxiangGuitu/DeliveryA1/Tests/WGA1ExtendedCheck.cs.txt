using System;
using System.IO;
using System.Linq;
using System.Security.Cryptography;
using UnityEditor;
using UnityEngine;
public static class WGA1ExtendedCheck {
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")??throw new Exception("explicit root");
 static void Need(bool b,string m){if(!b)throw new Exception(m);}static void Copy(string a,string b){Directory.CreateDirectory(b);foreach(var f in Directory.GetFiles(a,"*",SearchOption.AllDirectories)){string target=Path.Combine(b,f.Substring(a.Length+1));Directory.CreateDirectory(Path.GetDirectoryName(target));File.Copy(f,target);}}
 static string GameHash(string p){using(var sha=SHA256.Create())return string.Join("|",Directory.GetFiles(p,"*",SearchOption.AllDirectories).Where(f=>!Path.GetFileName(f).StartsWith("wanxiang-")&&!f.EndsWith(".tmp")).OrderBy(f=>f).Select(f=>f.Substring(p.Length)+":"+Convert.ToBase64String(sha.ComputeHash(File.ReadAllBytes(f)))));}
 public static void InterruptedWrite(){try{string old=Path.Combine(Root,"restart-old-"+Guid.NewGuid().ToString("N")),next=old+"-new";Directory.CreateDirectory(old);File.Copy(Path.Combine(Root,"fixtures/v8-camp.json"),Path.Combine(old,"save.json"));Copy(Directory.GetDirectories(Root,"v9-run-*").Last(),next);SaveLocationMigration.ChooseNew(old,next,SaveLocationMigration.Resolve(old,next));File.WriteAllText(Path.Combine(old,"old-version-changed"),"new state");string hash=GameHash(next),receipt=File.ReadAllText(Path.Combine(next,"wanxiang-choice.json"));SaveLocationMigration.ChoiceValidationHook=p=>{if(p=="before-choice-publish")throw new IOException("interrupted confirmation");};bool failed=false;try{SaveLocationMigration.ChooseNew(old,next,SaveLocationMigration.Resolve(old,next));}catch(IOException){failed=true;}finally{SaveLocationMigration.ChoiceValidationHook=null;}Need(failed&&File.ReadAllText(Path.Combine(next,"wanxiang-choice.json"))==receipt,"failed choice overwrote old receipt");File.WriteAllLines(Path.Combine(Root,"choice-process-resume.txt"),new[]{old,next,hash});Debug.Log("A1_CHOICE_INTERRUPTED_PROCESS_WRITTEN_PASS");EditorApplication.Exit(0);}catch(Exception e){SaveLocationMigration.ChoiceValidationHook=null;Debug.LogException(e);EditorApplication.Exit(1);}}
 public static void InterruptedResume(){try{var lines=File.ReadAllLines(Path.Combine(Root,"choice-process-resume.txt"));string old=lines[0],next=lines[1];var conflict=SaveLocationMigration.Resolve(old,next);Need(conflict.status==SaveLocationStatus.Conflict&&GameHash(next)==lines[2],"new process did not retain conflict/save");SaveLocationMigration.ChooseNew(old,next,conflict);Need(SaveLocationMigration.Resolve(old,next).status==SaveLocationStatus.Ready&&GameHash(next)==lines[2],"new process retry altered game data");CombatSaveSystem.InspectReadOnly(Path.Combine(next,"save.json"));Debug.Log("A1_CHOICE_NEW_PROCESS_RETRY_PASS");EditorApplication.Exit(0);}catch(Exception e){Debug.LogException(e);EditorApplication.Exit(1);}}
 public static void Matrix(){try{foreach(string kind in new[]{"camp","run","pending","committed"})foreach(int version in new[]{8,9}){string old=Path.Combine(Root,"a1-old-"+Guid.NewGuid().ToString("N")),next=old+"-new";if(version==9)Copy(Directory.GetDirectories(Root,"v9-"+kind+"-*").Last(),old);else{Directory.CreateDirectory(old);File.Copy(Path.Combine(Root,"fixtures","v8-"+kind+".json"),Path.Combine(old,"save.json"));}Copy(old,next);string oldBytes=GameHash(old),newBytes=GameHash(next);var conflict=SaveLocationMigration.Resolve(old,next);Need(conflict.status==SaveLocationStatus.Conflict,"not initial conflict");SaveLocationMigration.ChooseNew(old,next,conflict);File.WriteAllText(Path.Combine(old,"old-version-marker"),"change");oldBytes=GameHash(old);conflict=SaveLocationMigration.Resolve(old,next);Need(conflict.status==SaveLocationStatus.Conflict,"new change not detected");File.WriteAllText(Path.Combine(old,"old-version-marker"),"later change");bool stopped=false;try{SaveLocationMigration.ChooseNew(old,next,conflict);}catch(IOException){stopped=true;}Need(stopped,"stale UI confirmation accepted");oldBytes=GameHash(old);conflict=SaveLocationMigration.Resolve(old,next);SaveLocationMigration.ChooseNew(old,next,conflict);Need(SaveLocationMigration.Resolve(old,next).status==SaveLocationStatus.Ready&&GameHash(old)==oldBytes&&GameHash(next)==newBytes,"restart altered games/loot/settlement");CombatSaveSystem.InspectReadOnly(Path.Combine(next,"save.json"));
 File.WriteAllText(Path.Combine(old,"old-version-marker"),"durable choice");SaveLocationMigration.ChoiceValidationHook=p=>{if(p=="after-choice-publish")throw new IOException("crash after confirmation publication");};try{SaveLocationMigration.ChooseNew(old,next);}catch(IOException){}finally{SaveLocationMigration.ChoiceValidationHook=null;}Need(SaveLocationMigration.Resolve(old,next).status==SaveLocationStatus.Ready&&GameHash(next)==newBytes,"durable confirmation lost on restart");Debug.Log("A1_MATRIX_PASS v"+version+" "+kind+" paged bytes/identity/settlement unchanged");}
 string link=Path.Combine(Root,"reparse-location"),sentinel=Path.Combine(Root,"reparse-target","sentinel");Need((File.GetAttributes(link)&FileAttributes.ReparsePoint)!=0,"junction fixture missing");string source=Path.Combine(Root,"link-source-"+Guid.NewGuid().ToString("N"));Directory.CreateDirectory(source);File.Copy(Path.Combine(Root,"fixtures/v8-camp.json"),Path.Combine(source,"save.json"));Need(SaveLocationMigration.Resolve(source,link).status==SaveLocationStatus.Blocked&&File.ReadAllText(sentinel)=="untouched","junction traversed/overwritten");bool rejected=false;try{ValidationStorage.ValidateRoot(link);}catch(IOException){rejected=true;}Need(rejected,"isolation root accepted junction");string dangling=Path.Combine(Root,"reparse-dangling");Need((File.GetAttributes(dangling)&FileAttributes.ReparsePoint)!=0,"dangling fixture missing");rejected=false;try{ValidationStorage.ValidateRoot(dangling);}catch(IOException){rejected=true;}Need(rejected&&SaveLocationMigration.Resolve(source,dangling).status==SaveLocationStatus.Blocked,"dangling link accepted as absent");Debug.Log("A1_REPARSE_AND_MATRIX_ALL_PASS");EditorApplication.Exit(0);
 }catch(Exception e){SaveLocationMigration.ChoiceValidationHook=null;Debug.LogException(e);EditorApplication.Exit(1);}}
}
