        IEnumerator Start()
        {
            var stack=new System.Collections.Generic.Stack<IEnumerator>();stack.Push(Run());
            while(stack.Count>0)
            {
                object next=null;bool more;
                try{more=stack.Peek().MoveNext();if(more)next=stack.Peek().Current;}
                catch(Exception e){File.WriteAllText(Path.Combine(root,"failure.txt"),e.ToString());Exit(1);yield break;}
                if(!more)stack.Pop();else if(next is IEnumerator nested)stack.Push(nested);else yield return next;
            }
            File.WriteAllText(Path.Combine(root,"complete.txt"),"PASS");Exit(0);
        }
        IEnumerator Wait(string id,Func<bool> predicate,float seconds=30)
        {float end=Time.realtimeSinceStartup+seconds;while(!predicate()&&Time.realtimeSinceStartup<end)yield return null;Check(id,predicate());}
        IEnumerator Walk(string id,params Vector3[] points)
        {
            foreach(var point in points)
            {
                var path=new NavMeshPath();bool valid=NavMesh.SamplePosition(hero.transform.position,out var a,2,NavMesh.AllAreas)&&NavMesh.SamplePosition(point,out var b,2,NavMesh.AllAreas)&&NavMesh.CalculatePath(a.position,b.position,NavMesh.AllAreas,path)&&path.status==NavMeshPathStatus.PathComplete;
                Check(id+"-nav-"+point,valid);
                File.AppendAllText(Path.Combine(root,"paths.tsv"),id+"\t"+string.Join(";",path.corners.Select(p=>p.ToString("F3")))+"\n");
                foreach(var corner in path.corners.Skip(1))
                {
                    float end=Time.realtimeSinceStartup+40;
                    while(Vector3.ProjectOnPlane(corner-hero.transform.position,Vector3.up).magnitude>.35f&&Time.realtimeSinceStartup<end&&!hero.Health.Dead)
                    {
                        var delta=Vector3.ProjectOnPlane(corner-hero.transform.position,Vector3.up);
                        hero.MoveCharacter(Vector3.ClampMagnitude(delta,hero.moveSpeed*hero.MoveMultiplier*Time.deltaTime));yield return null;
                    }
                    Check(id+"-move-"+corner,Vector3.ProjectOnPlane(corner-hero.transform.position,Vector3.up).magnitude<=.35f,hero.transform.position.ToString());
                }
            }
        }

        {
            // Controlled threshold fixture: only real weapon attacks reduce HP.
            // AI is suspended during setup, then runs normally for the observed cycle.
            boss.enabled=false;
            hero.RestorePosition(enemy.transform.position-Vector3.forward*1.8f,0);
            s.orbit.RestoreView(0,9);yield return null;

            hero.Health.Observed+=h=>File.AppendAllText(Path.Combine(root,"phase-hero-health.tsv"),$"{h.frame}\t{h.kind}\t{h.before}\t{h.after}\n");

                combat.Strike();yield return null;
            }
            Check("threshold-real-weapon-damage",!enemy.Health.Dead&&enemy.Health.Current<=enemy.Health.maxHealth*.5f);
            // Avoid changing health or sequence: enabling resumes the actual state machine.
            boss.enabled=true;
            yield return Wait("phase-transition",()=>boss.Phase2,5);
            Check("transition-harmless-state",boss.State==CrystalCavernBossBrain.CombatState.Transition);
            int released=boss.Releases;
            Time.timeScale=0;float remaining=boss.Remaining;Vector3 position=enemy.transform.position;
            yield return new WaitForSecondsRealtime(.5f);
            Check("transition-pause",boss.Releases==released&&boss.Remaining==remaining&&enemy.transform.position==position);
            Time.timeScale=1;
            var moves=new System.Collections.Generic.HashSet<CrystalCavernBossRules.Attack>();

                moves.Add(e.attack);
                File.AppendAllText(Path.Combine(root,"phase-releases.tsv"),$"{e.frame}\t{e.time:R}\t{e.release}\t{e.phase2}\t{e.attack}\t{e.rawDamage}\t{e.origin}\t{e.direction}\n");

            {
                // Observe without freezing the player. Keep ring-safe close range;
                // lateral rolling is a real dodge request, not an invulnerability edit.

                if(delta.sqrMagnitude>.001f){hero.transform.rotation=Quaternion.LookRotation(delta);s.orbit.RestoreView(hero.transform.eulerAngles.y,9);}
                if(boss.State==CrystalCavernBossBrain.CombatState.Warning&&boss.Remaining<.25f)
                    hero.Dodge.TryDodge(Vector3.Cross(Vector3.up,delta.normalized));
                if(!hero.Dodge.IsRolling&&boss.State==CrystalCavernBossBrain.CombatState.Recovery&&delta.magnitude>1.8f)
                    hero.MoveCharacter(delta.normalized*hero.moveSpeed*hero.MoveMultiplier*Time.deltaTime);
                File.AppendAllText(Path.Combine(root,"phase-frames.tsv"),$"{Time.frameCount}\t{Time.time:R}\t{boss.State}\t{boss.Attack}\t{boss.CrossStep}\t{boss.Remaining:R}\t{hero.transform.position}\t{hero.Health.Current}\t{enemy.Health.Current}\n");

            Check("phase2-save",s.Saves.SaveProgress(false),s.Saves.Status);
            File.WriteAllText(Path.Combine(root,"saved-run.json"),JsonUtility.ToJson(s.Flow.ActiveRun,true));

        IEnumerator WindupPair(CrystalCavernBossBrain boss,CombatEnemy enemy)
        {
            bool cross=mode=="windup-b";
            boss.enabled=false;
            hero.RestorePosition(enemy.transform.position-Vector3.forward*1.8f,0);
            s.orbit.RestoreView(0,9);yield return null;

            enemy.Health.Observed+=h=>{
                if(h.kind==HealthObservationKind.Damage)lastDamage=h.before-h.after;
                File.AppendAllText(Path.Combine(root,"windup-health.tsv"),$"{h.frame}\t{Time.time:R}\t{h.kind}\t{h.before:R}\t{h.after:R}\n");
            };
            float end=Time.realtimeSinceStartup+90;
            if(cross)while((lastDamage<=0||enemy.Health.Current>180+lastDamage)&&Time.realtimeSinceStartup<end)
            {
                if(!PlayerInput.Focused)PlayerInput.Focus(true);
                combat.Strike();yield return null;
            }
            Check("fixture-above-half",enemy.Health.Current>180);
            yield return new WaitForSeconds(2);
            boss.enabled=true;
            yield return Wait("claw-warning",()=>boss.State==CrystalCavernBossBrain.CombatState.Warning&&boss.Attack==CrystalCavernBossRules.Attack.Claw,8);
            float warning=Time.time,deadline=Time.time+boss.Remaining,before=enemy.Health.Current;
            float released=-1,recovery=-1,transition=-1;Vector3 locked=Vector3.zero;bool haveLock=false,requested=false;
            boss.Released+=e=>{
                File.AppendAllText(Path.Combine(root,"windup-release.tsv"),$"{e.frame}\t{e.time:R}\t{e.release}\t{e.phase2}\t{e.attack}\t{e.rawDamage:R}\t{e.origin}\t{e.direction}\n");
                if(released<0){released=e.time;Check("release-original-phase",!e.phase2&&e.attack==CrystalCavernBossRules.Attack.Claw&&e.rawDamage==16);}
            };
            end=Time.realtimeSinceStartup+7;
            while(Time.realtimeSinceStartup<end)
            {
                if(!requested&&Time.time-warning>.2f){if(!PlayerInput.Focused)PlayerInput.Focus(true);requested=combat.Strike();}
                if(boss.State==CrystalCavernBossBrain.CombatState.Warning&&boss.Remaining<=.75f)
                {if(!haveLock){locked=boss.AttackDirection;haveLock=true;}else Check("locked-direction-frame-"+Time.frameCount,Vector3.Angle(locked,boss.AttackDirection)<.001f);}
                if(released>=0&&recovery<0&&boss.State==CrystalCavernBossBrain.CombatState.Recovery&&boss.Remaining==0)recovery=Time.time;
                if(boss.State==CrystalCavernBossBrain.CombatState.Transition){transition=Time.time;recovery=Time.time;File.AppendAllText(Path.Combine(root,"windup-transition.tsv"),$"{Time.frameCount}\t{Time.time:R}\t{boss.State}\t{enemy.Health.Current:R}\n");break;}
                File.AppendAllText(Path.Combine(root,"windup-timeline.tsv"),$"{Time.frameCount}\t{Time.time:R}\t{boss.State}\t{boss.Attack}\t{boss.Remaining:R}\t{boss.AttackDirection}\t{enemy.Health.Current:R}\t{hero.Health.Current:R}\n");
                if(!cross&&released>=0&&Time.time>=released+1.35f)break;
                yield return null;
            }
            Check("real-hit-in-warning",enemy.Health.Current<before);
            Check("threshold-branch",cross?enemy.Health.Current<=180:enemy.Health.Current>180);
            Check("not-early-release",released>=deadline-.002f,$"deadline={deadline:R},release={released:R}");
            Check("full-original-recovery",!cross||transition>=released+1.35f,$"release={released:R},transition={transition:R}");
            Check("phase-branch",boss.Phase2==cross);
            File.WriteAllText(Path.Combine(root,"windup-summary.json"),JsonUtility.ToJson(new WindupSummary{warning=warning,deadline=deadline,release=released,recoveryObserved=recovery,transition=transition},true));
        }

                Check("disk-full-reward-proof",RewardState(CombatSaveSystem.InspectReadOnly(Path.Combine(root,"WanxiangGuitu/ForestRuins/save.json")))==expected);
                Check("continue-existing",GameEntry.Instance.Continue(),GameEntry.Error);
                yield return Wait("restored-game",()=>(s=FindAnyObjectByType<CombatSession>())!=null&&s.Saves.IsInitialized&&!GameEntry.Loading);
                Check("runtime-full-reward-proof",RewardState(s.Flow.Compose((CombatSaveData)typeof(CombatSaveSystem).GetMethod("CaptureLegacy",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic).Invoke(s.Saves,null)))==expected);
                yield break;
            }
            if(mode=="phase2-restore")
            {
                var saved=CombatSaveSystem.InspectReadOnly(Path.Combine(root,"WanxiangGuitu/ForestRuins/save.json"));
                var savedBoss=saved.activeRun.checkpoint.enemies.Single(e=>e.entityId==CrystalCavernLayout.BossId);
                Check("continue-existing",GameEntry.Instance.Continue(),GameEntry.Error);
                yield return Wait("restore-game",()=>(s=FindAnyObjectByType<CombatSession>())!=null&&s.Saves.IsInitialized&&!GameEntry.Loading);
                hero=s.hero;region=s.GetComponent<ForestRegionRuntime>();
                var restored=s.enemies.Single(e=>e.entityId==CrystalCavernLayout.BossId);
                var brain=restored.GetComponent<CrystalCavernBossBrain>();
                Check("same-run",s.Flow.ActiveRun.runId==saved.activeRun.runId);
                Check("same-content",s.Flow.ActiveRun.contentRevision==saved.activeRun.contentRevision&&CrystalCavernLayout.HasBoss(region.Layout));
                Check("phase-from-saved-hp",brain.Phase2&&restored.Health.Current==savedBoss.health);
                Check("challenge-restored",brain.ChallengeCommitted);
                Check("no-replayed-release",brain.Releases==0);
                yield return new WaitForSeconds(.7f);
                Check("restored-safety-window",brain.Releases==0);
                yield break;
            }
            GameEntry.Instance.AskNew();Check("new-confirm",GameEntry.Instance.ConfirmNew(),GameEntry.Error);
            yield return Wait("game",()=>(s=FindAnyObjectByType<CombatSession>())!=null&&s.Saves.IsInitialized&&!GameEntry.Loading);
            hero=s.hero;PlayerInput.Focus(true);yield return null;
            if(mode=="stage1-staff")
            {
                var loadout=hero.GetComponent<AdventureLoadout>();
                Check("synthetic-camp-staff",loadout.Add(AdventureItems.New(GearType.Staff,GearQuality.White,hero)));
                Check("equip-staff",loadout.SelectMain(loadout.items.First(i=>i.type==GearType.Staff)));
            }
            Check("depart",s.Flow.DepartCavern(),s.Flow.Error);
            region=s.GetComponent<ForestRegionRuntime>();
            Check("exact-version",CrystalCavernLayout.HasBoss(region.Layout)&&s.Flow.ActiveRun.contentRevision==CrystalCavernLayout.Content24);
            var enemy=s.enemies.Single(e=>e.entityId==CrystalCavernLayout.BossId);
            var boss=enemy.GetComponent<CrystalCavernBossBrain>();
            var challenge=s.Events.Points.Single(e=>e.entityId==CrystalCavernLayout.Challenge);
            var near=new[]{P(110,410),P(118,410)};
            var bypass=new[]{P(78,410),P(78,432),P(40,432)};
            Check("dormant-no-exposure",!CrystalCavernBossRules.RouteExposure(region.Layout,enemy,near,2));
            Check("dormant-choice",!challenge.Used&&!boss.ChallengeCommitted);
            // Controlled placement is explicitly separate from the later normal-AI expedition.
            hero.RestorePosition(P(89,420),90);s.orbit.RestoreView(90,9);yield return null;
            if(mode=="stage1-bypass")
            {
                Check("optional-challenge-not-started",!boss.ChallengeCommitted);
                yield return Walk("bypass-return",P(78,420),P(78,432),P(40,432),P(40,410));
                yield return Extract();yield break;
            }
            Check("choice-interactable",challenge.CanInteract(hero));
            string before=File.ReadAllText(s.Saves.SavePath);
            s.Saves.WriteValidationHook=stage=>{if(stage=="before-publish")throw new IOException("C24 deliberate pre-publish failure");};
            bool failed=challenge.Interact(hero);s.Saves.WriteValidationHook=null;
            Check("failed-save-no-battle",!failed&&!boss.ChallengeCommitted&&!challenge.Used);
            Check("failed-save-disk-preserved",File.ReadAllText(s.Saves.SavePath)==before);
            Check("retry-challenge",challenge.Interact(hero),s.Saves.Status);
            Check("challenge-durable",CombatSaveSystem.InspectReadOnly(s.Saves.SavePath).activeRun.checkpoint.events.Single(e=>e.entityId==CrystalCavernLayout.Challenge).used);
            Check("challenged-near-exposure",CrystalCavernBossRules.RouteExposure(region.Layout,enemy,near,2));
            Check("challenged-lake-no-exposure",!CrystalCavernBossRules.RouteExposure(region.Layout,enemy,bypass,2));
            var oldLayouts=new[]{CrystalCavernLayout.Generate(17),CrystalCavernLayout.GenerateExpanded(17),CrystalCavernLayout.GenerateLake(17),RuinsLayout.Generate(17)};
            foreach(var map in oldLayouts)Check("old-map-no-c24-risk-"+map.revision,!CrystalCavernBossRules.RouteExposure(map,enemy,near,2));
            if(mode=="stage1-sword") { yield return SwordSample(boss,enemy); yield break; }
            if(mode=="stage1-staff") { yield return StaffSample(boss,enemy); yield break; }
            if(mode=="phase2") { yield return PhaseSample(boss,enemy); yield break; }
            if(mode=="windup-a"||mode=="windup-b") { yield return WindupPair(boss,enemy); yield break; }
            if(mode=="reward-before"||mode=="reward-after") { yield return RewardFault(boss,enemy); yield break; }
            var scored=RouteScoring.Evaluate(s).Single(r=>r.poiId==CrystalCavernLayout.Arena);
            var navigated=region.NavigationRoute(hero.transform.position,CrystalCavernLayout.Arena);
            Check("route-risk-consistent",navigated.nearElite&&scored.risk.pathElites>0);
            if(mode=="stage1-open")
            {
                yield return Walk("reach-bridge",P(78,420),P(76,414.5f));
                var lever=s.Events.Points.Single(e=>e.entityId==CrystalCavernLayout.BridgeLever);
                Check("bridge-interact",lever.Interact(hero));
                yield return Walk("open-bridge-return",P(76,410),P(43,410),P(40,410));
            }
            else
                yield return Walk("closed-bridge-return",P(78,420),P(78,432),P(40,432),P(40,410));
            Check("boss-no-lakeshore-attack",boss.Releases==0);
            Check("save",s.Saves.SaveProgress(false),s.Saves.Status);
            File.WriteAllText(Path.Combine(root,"run.json"),JsonUtility.ToJson(s.Flow.ActiveRun,true));
            if(!Application.isEditor){yield return new WaitForEndOfFrame();var shot=ScreenCapture.CaptureScreenshotAsTexture();File.WriteAllBytes(Path.Combine(root,"shoulder.png"),shot.EncodeToPNG());Destroy(shot);}
        }
    }
}
