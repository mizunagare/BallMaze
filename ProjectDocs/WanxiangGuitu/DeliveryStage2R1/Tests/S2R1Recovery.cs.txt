using System;using System.IO;using System.Linq;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEngine.AI;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;
// Mid-fight process continuation, separate controlled recovery probes, not main paired samples.
public static class S2R1Recovery {
 static CombatSession s;static ForestRegionRuntime region;static Keyboard kb;static Mouse mouse;static EditorWindow view;static CombatEnemy targetEnemy;static S2R1Observation observation;static S2R1Case current;static int phase,finishFrames;static double deadline;static float nextShot;static bool saving;static string kind;static HashSet<string> shots=new HashSet<string>();
 static string Root=>Path.Combine(ValidationStorage.ValidateRoot(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")),"Recovery-"+kind);static string Strategy=>kind=="Lake"?"Lake":"Gather";
 static void Need(bool b,string m){if(!b)throw new Exception(m);}static void Mark(string m){File.AppendAllText(Root+"/recovery-actions.txt",m+"\n");}
 public static void OrdinaryStart(){Start("Ordinary",false);}public static void OrdinaryResume(){Start("Ordinary",true);}public static void LakeStart(){Start("Lake",false);}public static void LakeResume(){Start("Lake",true);}
 static void Start(string k,bool resume){try{StartCore(k,resume);}catch(Exception error){Debug.LogException(error);EditorApplication.Exit(1);}}
 static void StartCore(string k,bool resume){ValidationStorage.Root(Environment.GetCommandLineArgs(),true);kind=k;saving=!resume;Need(resume?File.Exists(Root+"/observation-checkpoint.json"):!Directory.Exists(Root),"fresh start or exact prior checkpoint required");Directory.CreateDirectory(Root);current=new S2R1Case{id="recovery-"+kind,build="Safety",strategy=Strategy,seed=1,weapon=3,target=4,kind=kind};deadline=EditorApplication.timeSinceStartup+420;view=UI01Before.Size(1280,720);ShaderUtil.allowAsyncCompilation=false;EditorSceneManager.OpenScene(GameEntry.ForestPath);EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Setup(){foreach(var d in InputSystem.devices.Where(d=>d is Keyboard||d is Mouse).ToArray())InputSystem.DisableDevice(d);kb=InputSystem.AddDevice<Keyboard>();mouse=InputSystem.AddDevice<Mouse>();InputSystem.settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;InputSystem.settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;
  if(saving){s.Saves.InitializeStorage(Root+"/Player",false);Stage2Builds.Common(s.hero);Stage2Builds.Apply(s.hero,"Safety",GearType.Bow);s.hero.Inventory.Add(MaterialKind.MonsterHide,5);s.hero.Inventory.Add(MaterialKind.CrystalShard,5);Need(s.hero.Inventory.CraftPotions(5),"real potions");AdventureRandom.SetNextSeedForValidation(1);Need(s.Flow.Depart(AdventureMode.ForestExploration),"actual recovery Run");region=s.GetComponent<ForestRegionRuntime>();targetEnemy=kind=="Lake"?s.enemies.Single(e=>e.GetComponent<ForestBossBrain>()?.Kind==ForestBossKind.Tide):s.enemies.Where(e=>e.archetype==EnemyArchetype.Melee).OrderBy(e=>e.entityId).First();bool placed=false;foreach(var v in new[]{Vector3.back,Vector3.left,Vector3.forward,Vector3.right})if(region.TryGround(targetEnemy.transform.position+v*5,3,out var ground)&&NavMesh.SamplePosition(ground,out var hit,2,NavMesh.AllAreas)){s.hero.RestorePosition(hit.position,0);placed=true;break;}Need(placed,"controlled pretrial approach");observation=new S2R1Observation(s,targetEnemy,Root);Mark("DECLARED_BASIC_ONLY_UNTIL_PAUSED_CHECKPOINT; resume uses original fight policy; all AI live");}
  else{var prior=JsonUtility.FromJson<S2R1Measure>(File.ReadAllText(Root+"/observation-checkpoint.json"));s.Saves.InitializeStorage(Root+"/Player",true);Need(s.Flow.ActiveRun!=null&&s.Flow.ActiveRun.runId==prior.runId,"same actual Run");s.hero.Inventory.SetOpen(true);Need(Time.timeScale==0,"pause immediately after synchronous restore, before next AI frame");region=s.GetComponent<ForestRegionRuntime>();targetEnemy=s.enemies.Single(e=>e.entityId==prior.targetId);observation=new S2R1Observation(s,targetEnemy,Root,prior);observation.VerifyResume();Mark("NEW_PROCESS_SAME_RUN_ENTITY_HP_RNG_GEAR_LEDGER_GROUND_BADGES_CHECKPOINT_ALIGNED; paused activity gap zero");s.hero.Inventory.SetOpen(false);}
  phase=kind=="Lake"?80:1;Need(s.enemies.Where(e=>!e.Health.Dead).All(e=>e.enabled&&e.GetComponent<CharacterController>().enabled),"normal AI/colliders in recovery");}
 static void Exit(int code){StopInput();if(kb!=null)InputSystem.RemoveDevice(kb);if(mouse!=null)InputSystem.RemoveDevice(mouse);observation?.Dispose();EditorApplication.update-=Tick;EditorApplication.Exit(code);}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();Application.runInBackground=true;view?.Repaint();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("recovery deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.5f)return;if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||!s.Saves.IsInitialized){s=null;return;}Setup();return;}observation.Tick();Need(!s.hero.Health.Dead,"recovery hero died; keep censored sample, no fabricated kill");if(s.hero.Progression.IsChoosing)s.hero.Progression.SelectChoice(0);if(s.hero.Progression.PendingChoices>0)return;
  if(saving&&observation.Data.hits>0){Need(!targetEnemy.Health.Dead,"first basic unexpectedly killed target; cannot create mid-fight checkpoint");StopInput();if(s.hero.Dodge.IsRolling)return;s.hero.Inventory.SetOpen(true);observation.SavePausedCheckpoint();Mark("ACTUAL_MID_DAMAGE_PAUSED_SAVE; no extra gameplay commands before normal Editor self-exit");Exit(0);return;}
  if(!saving&&targetEnemy.Health.Dead){StopInput();if(++finishFrames<6)return;Need(observation.Data.status=="killed"&&observation.Data.resumedSegments==1&&observation.Data.hits>=2&&observation.Data.ttk>=0,"complete resumed damage-to-death observation");var run=s.Flow.ActiveRun.runId;var before=JsonUtility.ToJson(s.Flow.Profile);if(kind=="Lake")Need(ForestBossRules.Find(s.Flow.Profile,ForestBossKind.Tide)?.kills==1,"Boss reward exactly once");var ids=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Select(l=>l.InteractionId).OrderBy(x=>x).ToArray();Need(ids.Distinct().Count()==ids.Length,"duplicate actual drops");targetEnemy.DropLootOnce();Need(ids.SequenceEqual(UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Select(l=>l.InteractionId).OrderBy(x=>x)),"repeat drop created reward");s.Flow.OpenAbandon();Need(s.Flow.ConfirmAbandon()&&s.Flow.CommitSettlement()&&s.Flow.AcknowledgeSettlement(),"real post-combat terminal");Need(s.Flow.Profile.committedRuns.Count(x=>x==run)==1,"unique settlement");var committed=JsonUtility.ToJson(s.Flow.Profile);Need(s.Flow.CommitSettlement()&&committed==JsonUtility.ToJson(s.Flow.Profile),"repeat settlement changed profile");Need(s.Saves.SaveProgress(false),"final durable save");Mark("ACTUAL_KILL_AFTER_NEW_PROCESS; DROP_AND_SETTLEMENT_ONCE; no automatic pickup");File.WriteAllText(Root+"/recovery-result.json",JsonUtility.ToJson(new Result{status="PASS",run=run,target=observation.Data.targetId,hits=observation.Data.hits,ttk=observation.Data.ttk,sourceRevision=observation.Data.revision,finalRevision=s.Flow.Revision,dropIds=ids},true));Exit(0);return;}
  if(!Fight())Walk(targetEnemy.transform.position,4);
 }catch(Exception e){Debug.LogException(e);try{observation?.Censor("technical-recovery-error:"+e.Message);File.WriteAllText(Root+"/failure.txt",e.ToString());}catch(Exception diagnostic){Debug.LogException(diagnostic);}Exit(1);}}
 [Serializable]class Result {public string status,run,target;public int hits;public double ttk;public long sourceRevision,finalRevision;public string[] dropIds;}
 static void StopInput(){InputSystem.QueueStateEvent(kb,new KeyboardState());}
 static void Aim(Vector3 p){var d=Vector3.ProjectOnPlane(p-s.hero.transform.position,Vector3.up);if(d.sqrMagnitude>.01f)s.hero.transform.rotation=Quaternion.LookRotation(d);}
 static bool Walk(Vector3 goal,float arrival=1.1f){if(Vector3.ProjectOnPlane(goal-s.hero.transform.position,Vector3.up).magnitude<arrival){StopInput();return true;}var path=region.NavigationPath(s.hero.transform.position,goal,"stage2-walk:"+goal.ToString("F3"),Strategy=="Bypass");Need(path.reachable&&path.status==NavMeshPathStatus.PathComplete,"normal walk no path "+goal);var next=path.corners.Skip(1).FirstOrDefault(p=>Vector3.ProjectOnPlane(p-s.hero.transform.position,Vector3.up).magnitude>1);if(next==Vector3.zero)next=goal;var d=next-s.hero.transform.position;s.orbit.RestoreView(Mathf.Atan2(d.x,d.z)*Mathf.Rad2Deg,28);InputSystem.QueueStateEvent(kb,new KeyboardState(Key.W,Key.LeftShift));return false;}
 static readonly FieldInfo strikeClock=typeof(CombatEnemy).GetField("strikeAt",BindingFlags.Instance|BindingFlags.NonPublic);
 static bool EvadeThreat(){
  foreach(var enemy in s.enemies.Where(e=>!e.Health.Dead).OrderBy(e=>e.GetComponent<ForestBossBrain>()!=null?0:1)){
   var boss=enemy.GetComponent<ForestBossBrain>();Vector3 away;
   if(boss!=null){if(boss.State!=ForestBossState.Telegraph||boss.Remaining>.16f)continue;away=s.hero.transform.position-boss.LockedPoint;if(boss.Attack==ForestBossAttack.AntlerCharge||boss.Attack==ForestBossAttack.WaterFan||boss.Attack==ForestBossAttack.SeedVolley)away=Vector3.Cross(Vector3.up,enemy.transform.forward);}
   else{if(!enemy.WindingUp||(float)strikeClock.GetValue(enemy)-Time.time>.16f||Vector3.Distance(enemy.transform.position,s.hero.transform.position)>25)continue;away=enemy.archetype==EnemyArchetype.Ranged?Vector3.Cross(Vector3.up,enemy.transform.forward):s.hero.transform.position-enemy.transform.position;}
   if(away.sqrMagnitude<.01f)away=-s.hero.transform.forward;
   if(s.hero.Dodge.TryDodge(away)){File.AppendAllText(Root+"/dodge-actions.csv",$"{Time.time:0.00},{enemy.entityId},{(boss==null?"ordinary":boss.Attack.ToString())}\n");return true;}
  }return false;
 }
 static bool Observing(CombatEnemy e)=>phase==80&&e.GetComponent<ForestBossBrain>() is ForestBossBrain b&&b.Kind==(Strategy=="Land"?ForestBossKind.Warden:ForestBossKind.Tide)&&b.Releases<3;
 static bool Fight(){
  File.AppendAllText(Root+"/boss-world.csv",string.Join("",s.enemies.Where(e=>e.GetComponent<ForestBossBrain>()!=null).Select(e=>{var b=e.GetComponent<ForestBossBrain>();return $"{Time.time:0.00},{phase},{e.entityId},{b.State},{b.Attack},{e.Health.Current},{b.Releases},{e.transform.position.x:0.00},{e.transform.position.z:0.00}\n";})));
  if(s.hero.Health.Current<65&&s.hero.Inventory.UseHealthPotion())observation.Potion();if(EvadeThreat()){StopInput();return true;}
  var enemy=s.enemies.Where(e=>!e.Health.Dead&&(Observing(e)&&Vector3.Distance(e.transform.position,s.hero.transform.position)<26||Vector3.Distance(e.transform.position,s.hero.transform.position)<10||e.GetComponent<EncounterGuard>()?.State==1&&Vector3.Distance(e.transform.position,s.hero.transform.position)<22)).OrderBy(e=>Observing(e)?0:1).ThenBy(e=>e.GetComponent<ForestBossBrain>()!=null?1:0).ThenBy(e=>Vector3.Distance(e.transform.position,s.hero.transform.position)).FirstOrDefault();
  if(enemy==null)return false;Need(Strategy!="Bypass"||enemy.GetComponent<EncounterGuard>()?.poi!="poi:4","elite bypass accidentally engaged elite outpost");
  if(Observing(enemy)&&enemy.GetComponent<ForestBossBrain>().State!=ForestBossState.Telegraph&&enemy.GetComponent<ForestBossBrain>().State!=ForestBossState.Attacking&&Vector3.Distance(enemy.transform.position,s.hero.transform.position)>6){Walk(enemy.transform.position,3.5f);return true;}if(enemy.GetComponent<ForestBossBrain>()==null&&enemy.archetype==EnemyArchetype.Ranged&&Vector3.Distance(enemy.transform.position,s.hero.transform.position)>3){Walk(enemy.transform.position,2);return true;}StopInput();var combat=s.hero.GetComponent<AdventureCombat>();var chosen=s.hero.GetComponent<AdventureLoadout>().Weapon;Need(chosen==(GearType)current.weapon,"weapon coverage silently changed");float range=chosen==GearType.Sword?2.0f:chosen==GearType.Spear?3.1f:10f;if(Vector3.Distance(enemy.transform.position,s.hero.transform.position)>range){Walk(enemy.transform.position,range);return true;}Aim(enemy.transform.position);var brain=enemy.GetComponent<ForestBossBrain>();if(brain!=null){File.AppendAllText(Root+"/boss-combat.csv",$"{Time.time:0.00},{enemy.entityId},{brain.State},{brain.Attack},{enemy.Health.Current},{s.hero.Health.Current},{brain.Releases}\n");if(brain.State==ForestBossState.Telegraph){string key=enemy.entityId+"-"+brain.Attack;if(shots.Add(key))ScreenCapture.CaptureScreenshot(Root+"/boss-telegraph-"+key.Replace(':','-')+".png");if(brain.Remaining<.16f){var away=Vector3.ProjectOnPlane(s.hero.transform.position-brain.LockedPoint,Vector3.up);if(brain.Attack==ForestBossAttack.AntlerCharge||brain.Attack==ForestBossAttack.WaterFan||brain.Attack==ForestBossAttack.SeedVolley)away=Vector3.Cross(Vector3.up,enemy.transform.forward);s.hero.Dodge.TryDodge(away);}}}if(!saving&&(brain==null||brain.Releases>=3)&&combat.SkillRemaining<=0&&combat.UseSkill()){observation.Attack("skill");File.AppendAllText(Root+"/attack-actions.csv",$"{Time.time:0.00},skill,{enemy.entityId}\n");}if((brain==null||brain.Releases>=3)&&Time.time>nextShot){nextShot=Time.time+.55f;if(combat.Strike()){observation.Attack("basic");File.AppendAllText(Root+"/attack-actions.csv",$"{Time.time:0.00},basic,{enemy.entityId}\n");}}
  if((int)s.hero.GetComponent<AdventureLoadout>().Weapon>=2&&Vector3.Distance(enemy.transform.position,s.hero.transform.position)<3){var away=s.hero.transform.position+Vector3.ProjectOnPlane(s.hero.transform.position-enemy.transform.position,Vector3.up).normalized*4;if(region.TryGround(away,3,out var p)&&region.NavigationPath(s.hero.transform.position,p,"s2r1-optional-kite").reachable)Walk(p,.7f);else StopInput();}return true;
 }
}
