using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class B09Rules {
 static CombatSession s;static string mode;static double deadline;static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT");static string Current=>Root+"/b09-current";static string Legacy=>Root+"/b09-legacy";
 static void Need(bool b,string x){if(!b)throw new Exception(x);}static void Mark(string x){File.AppendAllText(Root+"/b09-rules.txt",x+"\n");}
 public static void Run(){Start("run");}public static void Resume(){Start("resume");}public static void Interrupt(){Start("interrupt");}public static void Recover(){Start("recover");}
 static void Start(string m){mode=m;ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+180;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void End(){EditorApplication.update-=Tick;EditorApplication.Exit(0);}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("B09 rules deadline");if(!EditorApplication.isPlaying||s!=null)return;s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.autoSaveInterval=3600;
 if(mode=="recover"){Need(!Directory.Exists(Current),"interruption should leave archive boundary");Need(NewProfileTransaction.Recover(Current)=="","recovery failed");Need(SaveCatalog.Fingerprint(Current)==File.ReadAllText(Root+"/b09-old-hash.txt"),"old collection not exactly restored");Need(s.Saves.InitializeFromEntry(Current,Legacy,false,SaveCatalog.Fingerprint(Current)),"load recovered");Need(s.Flow.Profile.profileId==File.ReadAllText(Root+"/b09-profile.txt"),"recovery profile changed");Mark("NEW_PROCESS_INTERRUPTION_ROLLBACK_AND_PROFILE_PASS");End();return;}
 if(mode=="resume"||mode=="interrupt"){Need(s.Saves.InitializeFromEntry(Current,Legacy,false,SaveCatalog.Fingerprint(Current)),"continue new profile");Need(s.Flow.Profile.profileId==File.ReadAllText(Root+"/b09-profile.txt")&&s.Flow.ActiveRun==null&&s.Flow.Profile.hasTutorial,"profile restore");if(mode=="resume"){Mark("NEW_PROCESS_EXACT_NEW_PROFILE_AND_TUTORIAL");End();return;}File.WriteAllText(Root+"/b09-old-hash.txt",SaveCatalog.Fingerprint(Current));s.Saves.EntryValidationHook=p=>{if(p=="after-archive")throw new EntryInterruption();};Need(!s.Saves.InitializeFromEntry(Current,Legacy,true,SaveCatalog.Fingerprint(Current)),"interruption fixture");s.Saves.SuspendEntry();Mark("CONTROLLED_PROCESS_ENDED_AFTER_ARCHIVE_JOURNAL");End();return;}
 Need(SaveCatalog.Inspect(Current).status==SaveAvailability.Missing&&!Directory.Exists(Current),"missing probe wrote state");s.Saves.InitializeStorage(Current,false);Need(SaveCatalog.Inspect(Current).canContinue,"valid current");string read=SaveCatalog.Fingerprint(Current),rng=JsonUtility.ToJson(UnityEngine.Random.state),profile=JsonUtility.ToJson(s.Flow.Profile);for(int i=0;i<4;i++)SaveCatalog.Inspect(Current);Need(read==SaveCatalog.Fingerprint(Current)&&rng==JsonUtility.ToJson(UnityEngine.Random.state)&&profile==JsonUtility.ToJson(s.Flow.Profile),"read-only catalog mutated state");
 s.Camp.Restore(new GearItem[0],new[]{7,3,1,0,0,0});Need(s.Saves.SaveProgress(false),"old permanent materials fixture");Directory.CreateDirectory(Current+"/ground-loot");File.WriteAllText(Current+"/ground-loot/synthetic-unreferenced-page.txt","full-collection sentinel, not player data");Need(s.Flow.Depart(AdventureMode.ForestExploration)&&s.Saves.SaveProgress(false),"real forest fixture");string run=s.Flow.ActiveRun.runId,old=s.Flow.Profile.profileId;var info=SaveCatalog.Inspect(Current);Need(info.runId==run&&info.phase=="在途冒险","actual run summary");
 read=SaveCatalog.Fingerprint(Current);s.Saves.WriteValidationHook=p=>{if(p=="before-publish")throw new IOException("B09_EXPECTED_CANDIDATE_WRITE_FAILURE");};Need(!s.Saves.InitializeFromEntry(Current,Legacy,true,read)&&SaveCatalog.Fingerprint(Current)==read,"candidate save failure replaced old collection");s.Saves.WriteValidationHook=null;Need(s.Saves.InitializeFromEntry(Current,Legacy,false,read)&&s.Flow.ActiveRun.runId==run,"failed candidate not recoverable");
 s.Saves.EntryValidationHook=p=>{if(p=="after-archive")throw new IOException("B09_EXPECTED_ARCHIVE_FAULT");};read=SaveCatalog.Fingerprint(Current);Need(!s.Saves.InitializeFromEntry(Current,Legacy,true,read)&&SaveCatalog.Fingerprint(Current)==read,"new profile fault damaged old");Need(s.Saves.InitializeFromEntry(Current,Legacy,false,read)&&s.Flow.ActiveRun.runId==run&&s.Flow.Profile.profileId==old,"old exact run after failure");
 s.Saves.EntryValidationHook=p=>{if(p=="after-publish")throw new IOException("B09_EXPECTED_AFTER_PUBLISH");};Need(s.Saves.InitializeFromEntry(Current,Legacy,true,SaveCatalog.Fingerprint(Current)),"published callback reconciliation");Need(s.Flow.Profile.profileId!=old&&s.Flow.ActiveRun==null&&s.Flow.Profile.materials.Sum()==0&&s.Flow.Profile.warehouse.Length==0&&s.Flow.Profile.badges.Length==0&&s.Flow.Profile.unlockedRecipes.Length==0&&Onboarding.Active(s),"new profile mixed old state");s.Saves.EntryValidationHook=null;File.WriteAllText(Root+"/b09-profile.txt",s.Flow.Profile.profileId);Need(NewProfileTransaction.Recover(Current)=="","repeat journal recovery");
 string cases=Root+"/b09-cases";Directory.CreateDirectory(cases);File.WriteAllText(cases+"/save.json","broken");Need(SaveCatalog.Inspect(cases).status==SaveAvailability.Corrupt,"corrupt classified missing");File.Copy(Current+"/save.json",cases+"/save.json.bak");Need(SaveCatalog.Inspect(cases).status==SaveAvailability.Backup&&SaveCatalog.Inspect(cases).canContinue,"valid backup unavailable");File.Copy(Current+"/save.json",cases+"/save.json",true);var e=JsonUtility.FromJson<CombatSaveEnvelope>(File.ReadAllText(cases+"/save.json"));e.version=99;File.WriteAllText(cases+"/save.json",JsonUtility.ToJson(e));Need(SaveCatalog.Inspect(cases).status==SaveAvailability.Incompatible&&!SaveCatalog.Inspect(cases).canContinue,"unsupported");File.Copy(Current+"/save.json",cases+"/save.json",true);File.WriteAllText(cases+"/save.json.tmp","not published");Need(SaveCatalog.Inspect(cases).status==SaveAvailability.Uncertain,"uncertain temp");
 for(int version=1;version<=6;version++){string path=Root+"/b09-legacy-catalog-"+version;Directory.CreateDirectory(path);File.Copy(Root+"/fixtures/v"+version+".json",path+"/save.json");string stamp=SaveCatalog.Fingerprint(path);var old=SaveCatalog.Inspect(path);Need(old.canContinue&&old.version==version&&old.region=="旧版波次场景"&&old.phase.Contains("迁移")&&SaveCatalog.Fingerprint(path)==stamp,"legacy catalog lies or migrates during read");Mark("LEGACY_CATALOG_READONLY v"+version+" level="+old.level+" phase="+old.phase);}
 Mark("SINGLE_AUTOSAVE_READONLY_STATES_FAILED_RESET_ATOMIC_RESET_NO_MIX_AND_CALLBACK_RECONCILE_PASS");End();
 }catch(Exception e){Debug.LogException(e);EditorApplication.Exit(1);}}
}
