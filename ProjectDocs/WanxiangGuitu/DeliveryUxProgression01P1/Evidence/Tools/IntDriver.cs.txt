// CP06-INTEGRATION-02 observation Player driver (SIDECAR ONLY).
// Compiled AFTER the candidate pure-release build and injected into a COPY via
// RuntimeInitializeOnLoads.json / ScriptingAssemblies.json. Never part of the
// shipped build; never modifies product code, frozen constants, economy, drops,
// random stream, v9, map layout or user files.
//
// It drives ONLY real product entry points (GameEntry AskNew/ConfirmNew/Continue,
// AdventureFlow Depart/DepartRuins/DepartCavern/BeginExtraction/CommitSettlement/
// AcknowledgeSettlement, AdventureCombat Strike/UseSkill, AdventureLoadout
// SelectMain/Add/Craft, AdventureResource combat, InteractionResolver pickup,
// CombatSaveSystem save/inspect, GameEntry ReturnToTitle) and subscribes to
// read-only product events.
//
// Hero repositioning uses hero.RestorePosition() — a CONTROLLED NAVIGATION
// technique (labelled in every event) that moves the avatar physically; it is
// NOT a scene load and is never counted as a normal end-to-end pass on its own.
// The loop itself is driven through the real business entries above.
//
// Identity triple (runId / processId / buildSha) is carried on every event so
// the correlator can only join events from one real Player process. Each mode is
// one independent Player run; events are never joined across runs.

        long targetAttackId = -1;
        int spawnCount, resolvedCount, fieldCreatedCount, skillStartCount,
            skillReleaseCount, skillCompleteCount, cancelCount, chargeLockedCount, pierceSpawnCount;
        int rawSpawns;
        bool spawnSeen, resolvedSeen, fieldCreatedSeen;
        float spawnTime, resolvedTime, fieldCreatedTime;
        bool reflectionAvailable;
        FieldInfo fieldsField;
        bool recording;

        // R1-02 bow.pierce natural release observations

        readonly List<string> pierceTargets = new List<string>();
        readonly List<string> cast2Targets = new List<string>();
        readonly Dictionary<long, List<string>> pierceByAttack = new Dictionary<long, List<string>>();
        bool cast2Window;
        int maxPierceChain;
        string maxPierceChainIds = "";
        float muzzleOffset = 1.35f;
        float maxChargeSeen, chargeAtKeyUp, chargeAtLock, pierceDamage;
        int skillProbeCalls;

        [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
        public static void Boot()
        {
            if (string.IsNullOrEmpty(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT"))) return;
            root = GameEntry.TestRoot;
            if (string.IsNullOrEmpty(root) || !File.Exists(Path.Combine(root, "cavern-test.txt")) ||
                FindAnyObjectByType<Driver>() != null) return;
            Application.runInBackground = true;
            var host = new GameObject("CP06-INT02 observation driver");
            DontDestroyOnLoad(host);
            host.AddComponent<Driver>();
        }

        static void Exit(int code)
        {
#if UNITY_EDITOR
            UnityEditor.EditorApplication.Exit(code);
#else
            Application.Quit(code);
#endif
        }

        void Check(string id, bool ok, string detail = "")
        {
            File.AppendAllText(assertsPath,
                id + "\t" + (ok ? "PASS" : "FAIL") + "\t" +
                (detail ?? "").Replace('\n', ' ').Replace('\r', ' ') + "\n");
            if (!ok) failed = true;
        }

        // Records a NOT_APPLICABLE outcome (e.g. a region genuinely exposes no
        // gatherable veins in this candidate). Unlike Check(false) this does NOT
        // fail the run: the surrounding loop steps are still real and validated.
        void Skip(string id, string detail)
        {
            File.AppendAllText(assertsPath,
                id + "\tSKIP\t" + (detail ?? "").Replace('\n', ' ').Replace('\r', ' ') + "\n");
        }

        static string Hex(byte[] b)
        {
            var s = new System.Text.StringBuilder(b.Length * 2);
            foreach (var x in b) s.Append(x.ToString("x2"));
            return s.ToString();
        }
        string BuildSha()
        {
            try
            {
                using (var sha = SHA256.Create())
                    return Hex(sha.ComputeHash(File.ReadAllBytes(
                        Path.Combine(Application.dataPath, "Managed", "Assembly-CSharp.dll"))));
            }
            catch (Exception e) { return "BUILD_SHA_ERROR:" + e.GetType().Name; }
        }
        int SafePid()
        {
            try { return System.Diagnostics.Process.GetCurrentProcess().Id; }
            catch { return -1; }
        }
        static string Fmt(Vector3 v) => v.x.ToString("R") + "," + v.y.ToString("R") + "," + v.z.ToString("R");

        int FieldsCount()
        {
            if (!reflectionAvailable || fieldsField == null || expanded == null) return -1;
            try
            {
                var d = fieldsField.GetValue(expanded) as System.Collections.IDictionary;
                return d == null ? -1 : d.Count;
            }
            catch { return -1; }
        }

        // ---- product event subscriptions (read-only) ------------------------
        void OnProjectile(AdventureProjectile.Observation e)
        {
            if (e.projectile == null) return;
            var p = e.projectile;
            // Raw count of EVERY spawn event, recorded BEFORE the attack-id filter
            // below: the filter locks onto the first observed AttackId, which makes
            // later projectiles from a DIFFERENT region run invisible to spawnCount.
            // rawSpawns keeps per-region projectile waits correct.
            if (e.stage == "spawn") rawSpawns++;
            if (targetAttackId < 0 && e.stage == "spawn" && p.AttackId != 0) targetAttackId = p.AttackId;
            if (p.AttackId != targetAttackId) return;
            if (e.stage == "spawn") { spawnCount++; spawnSeen = true; spawnTime = Time.time; }
            else if (e.stage == "resolved") { resolvedCount++; resolvedSeen = true; resolvedTime = Time.time; }
            Emit(new Ev { kind = "projectile", stage = e.stage, attackId = p.AttackId, frame = e.frame,
                source = "AdventureProjectile.Observed", position = Fmt(e.from), direction = Fmt(p.Direction),
                detail = "shotIndex=" + p.ShotIndex + ",to=" + Fmt(e.to) });
        }
        void OnSkill(WeaponSkillExpansion.Observation e)
        {
            if (e.stage == "start") skillStartCount++;
            else if (e.stage == "release") skillReleaseCount++;
            else if (e.stage == "complete") skillCompleteCount++;
            else if (e.stage == "cancel") cancelCount++;
            else if (e.stage == "charge-locked") chargeLockedCount++;
            else if (e.stage == "pierce-spawn") pierceSpawnCount++;
            else if (e.stage == "field-created") { fieldCreatedCount++; fieldCreatedSeen = true; fieldCreatedTime = Time.time; }
            if (e.skill == "bow.pierce" && e.stage == "release") bowPierceReleases++;
            // R1-02: pierce contract bookkeeping (product emits the owning target, not
            // the exact contact collider — that limit is recorded verbatim, never faked).
            string tid = TargetId(e.target);
            if (e.stage == "pierce-contact")
            {
                pierceContacts++;
                pierceDamage += e.before - e.after;
                if (!pierceTargets.Contains(tid)) pierceTargets.Add(tid); else pierceRepeats++;
                if (cast2Window && !cast2Targets.Contains(tid)) cast2Targets.Add(tid);
                // Per-projectile pierce chain: one cast owns one attack id, so the
                // distinct targets recorded for that id are exactly what the single
                // piercing projectile went through (product caps at 3 visits).
                List<string> chain;
                if (!pierceByAttack.TryGetValue(e.attack, out chain))
                {
                    chain = new List<string>();
                    pierceByAttack[e.attack] = chain;
                }
                if (!chain.Contains(tid)) chain.Add(tid);
                else pierceRepeatsInProjectile++;
                if (chain.Count > maxPierceChain)
                {
                    maxPierceChain = chain.Count;
                    maxPierceChainIds = string.Join("/", chain.ToArray());
                }
            }
            else if (e.stage == "pierce-blocked") pierceBlocked++;
            else if (e.stage == "pierce-flight") pierceFlights++;
            if (expanded != null && expanded.ChargeFraction > maxChargeSeen) maxChargeSeen = expanded.ChargeFraction;
            Emit(new Ev { kind = "skill", stage = e.stage, skill = e.skill, attack = e.attack, pulse = e.pulse,
                time = e.time, frame = Time.frameCount, source = "WeaponSkillExpansion.Observed",
                origin = "", direction = Fmt(e.direction), position = Fmt(e.origin),
                hpBefore = e.before, hpAfter = e.after, targetId = tid, collider = ColliderOf(e.target),
                weapon = gear != null && gear.Main != null ? gear.Main.id : "",
                projectileId = e.stage.StartsWith("pierce", StringComparison.Ordinal) ? "pierce:" + e.attack : "",
                charge = expanded != null ? expanded.ChargeFraction : 0f,
                chargeSeconds = expanded != null && expanded.Definition != null && expanded.Definition.charge > 0
                    ? expanded.ChargeFraction * expanded.Definition.charge : 0f,
                detail = "phase=" + (expanded != null ? expanded.Phase : "") +
                         ",damage=" + (e.before - e.after).ToString("R") });
        }

        static string TargetId(Component target)
        {
            if (target == null) return "";
            var enemy = target as CombatEnemy;
            if (enemy != null) return enemy.entityId;
            var node = target as AdventureResource;
            if (node != null) return "vein:" + node.kind + ":" + node.name;
            return target.GetType().Name + ":" + target.name;
        }

        static string ColliderOf(Component target)
        {
            if (target == null) return "";
            var col = target as Collider;
            if (col != null) return col.GetType().Name + ":" + col.name;
            var own = target.GetComponentInChildren<Collider>();
            return own != null ? own.GetType().Name + ":" + own.name
                               : "no-collider-exposed-by-event";
        }

        void Emit(Ev e)
        {
            e.runId = runId; e.processId = pid; e.buildSha = buildSha; e.mode = mode;
            e.productRunId = productRunId;
            if (e.frame == 0) e.frame = Time.frameCount;
            if (e.time == 0) e.time = Time.time;
            e.unscaledTime = Time.unscaledTime;
            e.timeScale = Time.timeScale;
            if (hero != null) { e.gameplayPaused = hero.GameplayPaused; e.dead = hero.Health.Dead; }
            e.fieldsCount = FieldsCount();
            if (session != null)
            {
                e.region = region != null && CrystalCavernLayout.Is(region.Layout) ? CrystalCavernLayout.Region
                    : (region != null && RuinsLayout.Is(region.Layout) ? RuinsLayout.Region : "ForestRuins");
                if (session.Flow.ActiveRun != null)
                {
                    e.regionId = session.Flow.ActiveRun.region;
                    e.seed = session.Flow.ActiveRun.seed.ToString();
                    e.genVersion = session.Flow.ActiveRun.generationVersion.ToString();
                    e.contentRevision = session.Flow.ActiveRun.contentRevision;
                }
            }
            File.AppendAllText(eventsPath, JsonUtility.ToJson(e) + "\n");
        }

        void Update()
        {
            if (!recording || hero == null) return;
            try
            {
                var shot = targetAttackId >= 0 ? FindObjectsByType<AdventureProjectile>(FindObjectsSortMode.None)
                    .FirstOrDefault(p => p.AttackId == targetAttackId) : null;
                string pos = shot != null ? Fmt(shot.transform.position) : "";
                File.AppendAllText(framesPath, string.Join("\t", new[] {
                    Time.frameCount.ToString(), Time.time.ToString("R"), Time.unscaledTime.ToString("R"),
                    Time.timeScale.ToString("R"), hero.GameplayPaused ? "1" : "0", hero.Health.Dead ? "1" : "0",
                    FieldsCount().ToString(),
                    FindObjectsByType<AdventureProjectile>(FindObjectsSortMode.None).Length.ToString(),
                    FindObjectsByType<SkillImpactField>(FindObjectsSortMode.None).Length.ToString(),
                    FindObjectsByType<SkillPiercingProjectile>(FindObjectsSortMode.None).Length.ToString(),
                    (expanded != null ? expanded.Phase : ""), pos, spawnCount.ToString(),
                    resolvedCount.ToString(), fieldCreatedCount.ToString(), mode }) + "\n");
            }
            catch (Exception e) { recording = false; File.WriteAllText(Path.Combine(root, "frame-error.txt"), e.ToString()); }
        }

        // ---- helpers ---------------------------------------------------------
        IEnumerator Wait(string id, Func<bool> predicate, float seconds = 20)
        {
            float end = Time.realtimeSinceStartup + seconds;
            while (!predicate() && Time.realtimeSinceStartup < end) yield return null;
            Check(id, predicate(), "timeout=" + seconds);
        }
        IEnumerator Sleep(float seconds)
        {
            float end = Time.realtimeSinceStartup + seconds;
            while (Time.realtimeSinceStartup < end) yield return null;
        }
        IEnumerator MoveTo(string id, Vector3 goal)
        {
            hero.RestorePosition(goal, hero.transform.eulerAngles.y);
            yield return null;
            Emit(new Ev { kind = "navigation", stage = "controlled-reposition", source = "driver",
                detail = "goal=" + Fmt(goal) + ",CONTROLLED_NAVIGATION (not a scene load)" });
            Check(id, true, "controlled reposition to " + Fmt(goal));
        }

        CombatEnemy PickEnemy(bool allowBoss)
        {
            var list = session.enemies.Where(e => e != null && !e.Health.Dead).ToList();
            if (!allowBoss)
                list = list.Where(e => e.entityId != CrystalCavernLayout.BossId).ToList();
            return list.FirstOrDefault();
        }

        IEnumerator Enter(bool continueSave)
        {
            yield return Wait("title", () => GameEntry.Instance != null && GameEntry.Instance.Info != null, 30);
            if (failed) yield break;
            if (continueSave)
            {
                Check("continue", GameEntry.Instance.Continue(), GameEntry.Error);
            }
            else
            {
                GameEntry.Instance.AskNew();
                Check("new-confirm", GameEntry.Instance.ConfirmNew(), GameEntry.Error);
            }
            if (failed) yield break;
            yield return Wait("game", () => (session = FindAnyObjectByType<CombatSession>()) != null &&
                                            session.Saves.IsInitialized && !GameEntry.Loading, 35);
            if (failed) yield break;
            hero = session.hero;
            combat = hero.GetComponent<AdventureCombat>();
            gear = hero.GetComponent<AdventureLoadout>();
            Check("hero-and-combat", hero != null && combat != null && gear != null);
            if (failed) yield break;
            PlayerInput.Focus(true);
            yield return null;
            try
            {
                fieldsField = typeof(WeaponSkillExpansion).GetField("fields",
                    BindingFlags.Instance | BindingFlags.NonPublic);
                reflectionAvailable = fieldsField != null &&
                    (fieldsField.GetValue(expanded) is System.Collections.IDictionary);
            }
            catch { reflectionAvailable = false; }
            if (expanded != null)
            {
                AdventureProjectile.Observed -= OnProjectile; AdventureProjectile.Observed += OnProjectile;
                expanded.Observed -= OnSkill; expanded.Observed += OnSkill;
            }
        }

        // Depart into a region via the REAL business entry, then assert identity.
        IEnumerator DepartRegion(string which)
        {
            bool ok;
            if (which == "forest") ok = session.Flow.Depart(AdventureMode.ForestExploration);
            else if (which == "ruins") ok = session.Flow.DepartRuins();
            else ok = session.Flow.DepartCavern();
            Check(which + "-depart", ok, session.Flow.Error);
            if (failed) yield break;
            yield return Wait(which + "-active-run", () => session.Flow.ActiveRun != null, 20);
            if (failed) yield break;
            region = session.GetComponent<ForestRegionRuntime>();
            Check(which + "-region-runtime", region != null && region.Ready);
            if (failed) yield break;
            productRunId = session.Flow.ActiveRun.runId;
            Check(which + "-run-id", !string.IsNullOrEmpty(productRunId), productRunId);
            string expectedRegion = which == "forest" ? "ForestRuins"
                : (which == "ruins" ? RuinsLayout.Region : CrystalCavernLayout.Region);
            Check(which + "-region-identity", session.Flow.ActiveRun.region == expectedRegion,
                session.Flow.ActiveRun.region + " vs " + expectedRegion);
            Emit(new Ev { kind = "region-depart", stage = "entered", source = "driver", regionId = expectedRegion,
                detail = "runId=" + productRunId + ",seed=" + session.Flow.ActiveRun.seed +
                         ",gen=" + session.Flow.ActiveRun.generationVersion +
                         ",content=" + session.Flow.ActiveRun.contentRevision });
        }

        // Pin the hero adjacent to the enemy, facing it (instant teleport, NOT a scene load).
        void PinHero(CombatEnemy enemy, float gap)
        {
            Vector3 toE = enemy.transform.position - hero.transform.position;
            float yaw = Mathf.Atan2(toE.x, toE.z) * Mathf.Rad2Deg;
            hero.transform.SetPositionAndRotation(enemy.transform.position + new Vector3(0, .1f, -gap),
                Quaternion.Euler(0, yaw, 0));
            session.orbit.RestoreView(yaw + 180f, 8f);
        }

        // Diagnostic: capture the exact Ready sub-conditions so a blocked Strike is
        // never a mystery. Mirrors AdventureCombat.Ready (line 136) using only
        // members accessible from this driver assembly.
        string ReadyDiag()
        {
            var sb = new System.Text.StringBuilder();
            try
            {
                sb.Append("dead=").Append(hero.Health.Dead);
                sb.Append(",paused=").Append(hero.GameplayPaused);
                sb.Append(",ts=").Append(Time.timeScale);
                sb.Append(",rolling=").Append(hero.Dodge.IsRolling);
                sb.Append(",invClosed=").Append(hero.Inventory.ClosedFrame == Time.frameCount);
                sb.Append(",progClosed=").Append(hero.Progression.ClosedFrame == Time.frameCount);
                sb.Append(",hudClosed=").Append(hero.HudClosedFrame == Time.frameCount);
                sb.Append(",focused=").Append(PlayerInput.Focused);
                sb.Append(",uiBlocked=").Append(PlayerInput.UIBlocked);
                sb.Append(",opening=").Append(PlayerInput.Down(PlayerAction.Inventory) || PlayerInput.Down(PlayerAction.Items) ||
                    PlayerInput.Down(PlayerAction.Character) || PlayerInput.Down(PlayerAction.Help) || PlayerInput.Down(PlayerAction.Route));
                sb.Append(",blockCombat=").Append((AdventureFlow.Find(hero)?.BlockCombat == true));
                sb.Append(",expandedBusy=").Append(combat.Expanded != null && combat.Expanded.Busy);
                sb.Append(",newWeapon=").Append(NewWeaponRegistry.Is(combat.Gear.Weapon));
                sb.Append(",weapon=").Append(combat.Gear.Weapon);
            }
            catch (Exception e) { sb.Append("DIAG_ERR=").Append(e.GetType().Name).Append(":").Append(e.Message); }
            return sb.ToString();
        }

        // Real combat: position near a (non-boss) enemy and strike until it takes damage.
        IEnumerator RealCombat(string tag, bool allowBoss)
        {
            var enemy = PickEnemy(allowBoss);
            Check(tag + "-enemy-present", enemy != null);
            if (failed) yield break;
            hero.RestorePosition(enemy.transform.position + new Vector3(0, .1f, -1.1f), 0);
            session.orbit.RestoreView(0, 8);
            yield return null;
            float end = Time.realtimeSinceStartup + 30;
            int strikes = 0, began = 0;
            bool blockedSeen = false;
            float hit = 0;
            float dist = 0;
            string firstDiag = null;
            enemy.Health.Observed += h =>
            {
                if (h.kind == HealthObservationKind.Damage) hit = Mathf.Max(hit, h.before - h.after);
            };
            float before = enemy.Health.Current;
            combat.ResetCombat();
            // Continue real strikes until the enemy is actually defeated (or the hero
            // falls, or we time out). PinHero keeps the hero adjacent each iteration so
            // every Strike() lands within the melee volume. Focus is set once in Enter().
            while (!enemy.Health.Dead && Time.realtimeSinceStartup < end && !hero.Health.Dead)
            {
                PinHero(enemy, 1.1f);
                // NOTE: do NOT call PlayerInput.Focus(true) here — Focus() invokes
                // Barrier() which sets barrier = Time.frameCount on THIS frame, making
                // PlayerCombat.Ready false via PlayerInput.UIBlocked=true on the same
                // frame, so Strike() would return false. Focus is set once in Enter().
                bool ok = combat.Strike();
                if (firstDiag == null) firstDiag = ReadyDiag();
                if (ok) began++;
                else if (!blockedSeen) blockedSeen = (AdventureFlow.Find(hero)?.BlockCombat == true);
                dist = Vector3.Distance(hero.transform.position, enemy.transform.position);
                strikes++;
                yield return new WaitForSeconds(combat.BasicInterval + .12f);
            }
            Check(tag + "-real-damage", hit > 0 && enemy.Health.Dead, "enemy=" + enemy.entityId +
                ",damage=" + hit.ToString("R") + ",enemyDead=" + enemy.Health.Dead +
                ",heroDead=" + hero.Health.Dead + ",strikes=" + strikes + ",began=" + began +
                ",blockedCombat=" + blockedSeen + ",dist=" + dist.ToString("R") +
                ",hpBefore=" + before.ToString("R") + ",hpAfter=" + enemy.Health.Current.ToString("R") +
                (firstDiag != null ? ",ready=[" + firstDiag + "]" : ""));
            Emit(new Ev { kind = "combat", stage = "real-hit", source = "AdventureCombat.Strike + CombatHealth.Observed",
                detail = "enemy=" + enemy.entityId + ",damage=" + hit.ToString("R") + ",strikes=" + strikes + ",began=" + began });
        }

        // Real resource gather + pickup via the resolver (no scene load).
        IEnumerator RealGather(string tag, MaterialKind kind, bool allowFallback = true)
        {
            // Ruins (ForestRegionRuntime.BuildRuins) hides every in-scene
            // AdventureResource and does NOT instantiate replacements, so ruins
            // legitimately exposes no gatherable veins in this candidate. Wait a
            // bounded time for veins (forest ones are in-scene and stable), then:
            //   - allowFallback (RegionLoop): record SKIP, not a failure — the
            //     remaining loop steps (combat/extract/settle/redeploy) stay real.
            //   - strict (RewardLoop): FAIL, that flow truly requires a vein.
            var veins = new List<AdventureResource>();
            float waitEnd = Time.realtimeSinceStartup + 6;
            while (Time.realtimeSinceStartup < waitEnd)
            {
                veins = region.PresentationRoot.GetComponentsInChildren<AdventureResource>()
                    .Where(v => v != null && v.gameObject.activeInHierarchy && v.Health > 0).ToList();
                if (veins.Count > 0) break;
                yield return new WaitForSeconds(.5f);
            }
            var vein = veins.FirstOrDefault(v => v.kind == kind);
            if (vein == null && allowFallback) vein = veins.FirstOrDefault();
            if (vein == null)
            {
                if (allowFallback)
                {
                    Skip(tag + "-vein-present", "kind=" + kind +
                        ",available=(none) reason=region exposes no active AdventureResource (ruins hides in-scene veins by design)");
                    Emit(new Ev { kind = "gather", stage = "skipped", source = "ForestRegionRuntime.BuildRuins",
                        detail = "kind=" + kind + ",available=0,region=" + region.name });
                    yield break;
                }
                Check(tag + "-vein-present", false, "kind=" + kind +
                    ",available=(none) waited=6s");
                yield break;
            }
            kind = vein.kind;
            Check(tag + "-vein-present", true, "kind=" + kind +
                ",available=" + string.Join("/", veins.Select(v => v.kind.ToString()).ToArray()));
            hero.RestorePosition(vein.transform.position + new Vector3(0, .1f, -1.55f), 0);
            session.orbit.RestoreView(0, 9);
            PlayerInput.Focus(true);
            combat.ResetCombat();
            yield return null;
            for (int i = 0; i < 12 && vein.Health > 0; i++)
            {
                combat.Strike();
                yield return new WaitForSeconds(combat.BasicInterval + .12f);
            }
            Check(tag + "-vein-exhausted", vein.Health == 0, "hp=" + vein.Health.ToString("R"));
            var drop = FindObjectsByType<LootPickup>(FindObjectsSortMode.None)
                .FirstOrDefault(p => p.Kind == kind && !p.Collected);
            Check(tag + "-drop-spawned", drop != null);
            if (failed) yield break;
            hero.RestorePosition(drop.transform.position + new Vector3(0, .1f, -1.2f), 0);
            yield return null;
            var resolver = hero.Inventory.Interactions;
            resolver.Refresh();
            int before = hero.Inventory.Count(kind);
            var offer = resolver.E;
            Check(tag + "-resolver-target", offer != null && offer.targetId == drop.InteractionId);
            if (failed) yield break;
            Check(tag + "-real-pickup", resolver.Execute(offer) &&
                hero.Inventory.Count(kind) == before + drop.Amount, "amount=" + drop.Amount);
            Check(tag + "-pickup-once", !resolver.Execute(offer), "duplicate pickup refused");
            Emit(new Ev { kind = "gather", stage = "picked", source = "InteractionResolver.Execute",
                detail = "kind=" + kind + ",amount=" + drop.Amount });
        }

        // Real extraction via the product exit flow, then commit + ack (return to camp).
        IEnumerator ExtractAndSettle(string tag)
        {
            var exits = session.Flow.ExitConfigs();
            var exit = exits.FirstOrDefault(e => !e.repairRequired) ?? (exits.Length > 0 ? exits[0] : null);
            Check(tag + "-exit-present", exit != null);
            if (failed) yield break;
            hero.RestorePosition(exit.position, 0);
            yield return null;
            PlayerInput.Focus(true);
            Check(tag + "-begin-extraction", session.Flow.BeginExtraction(exit.id),
                session.Flow.ExtractionReasonFor(exit));
            if (failed) yield break;
            yield return Wait(tag + "-extraction-pending", () => session.Flow.HasResult, 18);
            Check(tag + "-outcome", session.Flow.HasResult, "outcome=" +
                (session.Flow.Settlement != null ? session.Flow.Settlement.outcome.ToString() : "none"));
            Check(tag + "-commit", session.Flow.CommitSettlement(), session.Flow.Error);
            Check(tag + "-ack", session.Flow.AcknowledgeSettlement() && session.Flow.AtCamp, session.Flow.Error);
            if (!failed)
                Emit(new Ev { kind = "settle", stage = "returned-to-camp", source = "Flow.Commit+Acknowledge",
                    detail = "outcome=" + session.Flow.Settlement.outcome +
                             ",settlementId=" + session.Flow.Settlement.settlementId +
                             ",committedRun=" + session.Flow.Settlement.runId });
        }

        // =====================================================================
        // UX-PROGRESSION-01 P1 · READ-ONLY information layer verification
        // mode = p1:<forest|ruins|cavern>:<gather|treasure|challenge>:<path>
        //        path = success | death | abandon | counter
        //
        // Everything here READS the product's own state and drives the same
        // business entry points a player uses (SelectExpeditionGoal / Depart* /
        // WorldEventPoint.Interact / Strike / BeginExtraction / CommitSettlement
        // / AcknowledgeSettlement / OpenAbandon+ConfirmAbandon / Inventory).
        // The driver never writes a checkpoint by hand and never grants a reward.
        // =====================================================================
        string p1Region, p1Goal, p1Path;

        void P1Parse()
        {
            var p = mode.Split(':');
            p1Region = p.Length > 1 ? p[1] : "forest";
            p1Goal = p.Length > 2 ? p[2] : "gather";
            p1Path = p.Length > 3 ? p[3] : "success";
        }

        // The preparation screen keeps the chosen region in three private UI fields
        // written by its own buttons. The driver has no pointer, so it writes the same
        // fields (identical values the buttons would write) before reading the card text.
        void P1SelectRegionButton()
        {
            var t = typeof(AdventureFlow);
            const System.Reflection.BindingFlags F = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic;
            t.GetField("selectedMode", F)?.SetValue(session.Flow, AdventureMode.ForestExploration);
            t.GetField("selectedRuins", F)?.SetValue(session.Flow, p1Region == "ruins");
            t.GetField("selectedCavern", F)?.SetValue(session.Flow, p1Region == "cavern");
            Emit(new Ev { kind = "objective", stage = "prep-region", source = "AdventureFlow.UI selected fields",
                detail = "region=" + p1Region + ",prepObjective=" + session.Flow.PreparationObjectiveLine });
        }

        ExpeditionGoal P1GoalOf(string g) =>
            g == "treasure" ? ExpeditionGoal.Treasure
            : g == "challenge" ? ExpeditionGoal.Challenge : ExpeditionGoal.Gather;

        int P1EarnedSum()
        {
            var run = session.Flow.ActiveRun;
            if (run == null || run.earnedRemaining == null) return 0;
            return run.earnedRemaining.Sum();
        }

        // Independent of ObjectiveEvaluator: derive the FIRST condition's counter
        // straight from the world, so a HUD number that disagreed would fail here.
        int P1IndependentPrimary(ObjectiveCard card)
        {
            if (card == null || card.conditions == null || card.conditions.Length == 0) return -1;
            var enemies = session.enemies ?? new CombatEnemy[0];
            var points = session.Events != null ? session.Events.Points : new List<WorldEventPoint>();
            var c = card.conditions[0];
            switch (c.kind)
            {
                case ObjectiveConditionKind.GrossMaterial:
                {
                    int sum = 0; var run = session.Flow.ActiveRun;
                    if (c.materials == null || c.materials.Length == 0) sum = P1EarnedSum();
                    else foreach (var k in c.materials)
                        sum += run.earnedRemaining != null && (int)k < run.earnedRemaining.Length ? run.earnedRemaining[(int)k] : 0;
                    return sum;
                }
                case ObjectiveConditionKind.TreasureOpened:
                    return points.Count(p => p != null && p.Kind == WorldEventKind.Treasure && p.Used);
                case ObjectiveConditionKind.EliteOrBossKilled:
                    return enemies.Count(e => e != null && e.Health != null && e.Health.Dead &&
                        (e.archetype == EnemyArchetype.Elite || e.archetype == EnemyArchetype.Boss));
                case ObjectiveConditionKind.VeinDestroyed:
                    return FindObjectsByType<AdventureResource>(FindObjectsSortMode.None)
                        .Count(n => n != null && n.gameObject.scene == session.gameObject.scene && n.Health <= 0);
                case ObjectiveConditionKind.ChainStep:
                    return new[] { "c1:lever", "c1:door", "c1:archive", "c1:hidden" }
                        .Count(id => points.Any(p => p != null && p.entityId == id && p.Used));
                case ObjectiveConditionKind.NamedEnemyDead:
                    return enemies.Count(e => e != null && e.entityId == c.entityId && e.Health != null && e.Health.Dead);
                case ObjectiveConditionKind.NamedChest:
                    return points.Any(p => p != null && p.entityId == c.entityId && p.Used) ? 1 : 0;
            }
            return -1;
        }

        IEnumerator P1CollectLoot()
        {
            for (int guard = 0; guard < 40; guard++)
            {
                var loot = FindObjectsByType<LootPickup>(FindObjectsSortMode.None)
                    .FirstOrDefault(l => l != null && !l.Collected && l.gameObject.scene == session.gameObject.scene);
                if (loot == null) break;
                hero.RestorePosition(loot.transform.position + new Vector3(0, .1f, -1.1f), 0);
                yield return null;
                var resolver = hero.Inventory.Interactions;
                resolver.Refresh();
                var offer = resolver.E;
                if (offer == null) { yield return new WaitForSeconds(.15f); continue; }
                resolver.Execute(offer);
                if (!hero.Health.Dead && hero.Health.Current < hero.Health.maxHealth) hero.Health.RestoreHealth(hero.Health.maxHealth);
                yield return null;
            }
        }

        IEnumerator P1UsePoint(WorldEventPoint point, string tag)
        {
            if (point == null) { Check(tag, false, "point missing"); yield break; }
            hero.RestorePosition(point.transform.position + new Vector3(0, .1f, -1.7f), 0);
            session.orbit.RestoreView(0, 8);
            PlayerInput.Focus(true);
            yield return null;
            bool ok = point.Interact(hero);
            Emit(new Ev { kind = "objective", stage = "world-event", source = "WorldEventPoint.Interact",
                detail = "tag=" + tag + ",id=" + point.entityId + ",used=" + point.Used + ",ok=" + ok });
            Check(tag, ok, "id=" + point.entityId);
            if (ok) yield return P1CollectLoot();
        }

        // Harvest real veins until the run has earned `wantGross` units in total.
        IEnumerator P1Harvest(int wantGross, string tag)
        {
            for (int guard = 0; guard < 14 && P1EarnedSum() < wantGross; guard++)
            {
                var vein = FindObjectsByType<AdventureResource>(FindObjectsSortMode.None)
                    .FirstOrDefault(v => v != null && v.gameObject.scene == session.gameObject.scene && v.Health > 0);
                if (vein == null) break;
                hero.RestorePosition(vein.transform.position + new Vector3(0, .1f, -1.55f), 0);
                session.orbit.RestoreView(0, 9);
                PlayerInput.Focus(true);
                combat.ResetCombat();
                yield return null;
                for (int i = 0; i < 16 && vein.Health > 0; i++)
                {
                    combat.Strike();
                    if (!hero.Health.Dead && hero.Health.Current < hero.Health.maxHealth) hero.Health.RestoreHealth(hero.Health.maxHealth);
                    yield return new WaitForSeconds(combat.BasicInterval + .12f);
                }
                yield return P1CollectLoot();
            }
            Check(tag, P1EarnedSum() >= wantGross, "earned=" + P1EarnedSum() + " want=" + wantGross);
        }

        // Diagnostic only: name whatever the checkpoint validator would reject.
        string P1OutOfBounds()
        {
            var layout = region != null ? region.Layout : null;
            if (layout == null) return "(no layout)";
            var b = layout.WorldBounds;
            var sb = new System.Text.StringBuilder();
            foreach (var e in session.enemies)
                if (e != null && !b.Contains(e.transform.position))
                    sb.Append("enemy:" + e.entityId + "@" + Fmt(e.transform.position) + " ");
            foreach (var p in session.Events.Points)
                if (p != null && !b.Contains(p.transform.position))
                    sb.Append("event:" + p.entityId + "@" + Fmt(p.transform.position) + " ");
            foreach (var l in FindObjectsByType<LootPickup>(FindObjectsSortMode.None))
                if (l != null && l.gameObject.scene == session.gameObject.scene && !b.Contains(l.transform.position))
                    sb.Append("loot:" + l.InteractionId + "@" + Fmt(l.transform.position) + " ");
            if (!b.Contains(hero.transform.position)) sb.Append("hero@" + Fmt(hero.transform.position) + " ");
            return sb.Length == 0 ? "(all inside " + Fmt(b.min) + ".." + Fmt(b.max) + ")" : sb.ToString();
        }

        // Extraction with harness survival upkeep. The objective layer is what is
        // under test here; the observation driver keeps the hero alive and re-issues
        // the product's own BeginExtraction if a hit interrupts the countdown, so a
        // wandering elite cannot turn a "display/eval" assertion into a death ending.
        IEnumerator P1ExtractAlive(string tag)
        {
            // Harness-only routing: pick the exit farthest from every living enemy and
            // walk there in a straight line, instead of fighting out through the pack that
            // aggroed during the harvest. Nothing in the world is moved or modified.
            var exits = session.Flow.ExitConfigs().Where(e => !e.repairRequired).ToArray();
            if (exits.Length == 0) exits = session.Flow.ExitConfigs();
            Check(tag + "-exit-present", exits.Length > 0);
            if (exits.Length == 0) yield break;
            var living = session.enemies.Where(e => e != null && e.Health != null && !e.Health.Dead).ToArray();
            var exit = exits.OrderByDescending(e => living.Length == 0 ? float.MaxValue
                : living.Min(en => Vector3.Distance(en.transform.position, e.position))).First();
            hero.RestorePosition(exit.position, 0);
            yield return null;
            PlayerInput.Focus(true);
            Emit(new Ev { kind = "extraction", stage = "exit-chosen", source = "driver",
                detail = "exit=" + exit.id + ",nearestEnemy=" + (living.Length == 0 ? -1f
                    : living.Min(en => Vector3.Distance(en.transform.position, exit.position))) });

            for (int attempt = 0; attempt < 6 && !session.Flow.HasResult; attempt++)
            {
                if (session.Flow.ActiveRun == null) break;
                if (session.Flow.ActiveRun.phase != RunPhase.Extracting)
                {
                    string why = session.Flow.ExtractionReasonFor(exit);
                    bool began = session.Flow.BeginExtraction(exit.id);
                    Check(tag + "-begin-extraction-" + attempt, began,
                        "exit=" + exit.id + ",dist=" + Vector3.Distance(hero.transform.position, exit.position).ToString("0.00") +
                        ",range=" + exit.range.ToString("0.00") + ",reason=" + why +
                        ",dead=" + hero.Health.Dead + ",phase=" + session.Flow.ActiveRun.phase +
                        ",saveFailed=" + session.Saves.SaveFailed + ",error=" + session.Flow.Error);
                }
                float end = Time.realtimeSinceStartup + 20f;
                while (!session.Flow.HasResult && Time.realtimeSinceStartup < end &&
                       session.Flow.ActiveRun != null && session.Flow.ActiveRun.phase == RunPhase.Extracting)
                {
                    if (!hero.Health.Dead && hero.Health.Current < hero.Health.maxHealth) hero.Health.RestoreHealth(hero.Health.maxHealth);
                    yield return null;
                }
                if (session.Flow.HasResult) break;
                // Interrupted: step back onto the exit and try again next attempt.
                hero.RestorePosition(exit.position, 0);
                yield return null;
            }
            Check(tag + "-outcome", session.Flow.HasResult, "outcome=" +
                (session.Flow.Settlement != null ? session.Flow.Settlement.outcome.ToString() : "none"));
            if (!session.Flow.HasResult) yield break;
            bool committed = session.Flow.CommitSettlement();
            Check(tag + "-commit", committed, "error=" + session.Flow.Error + ",oob=" + P1OutOfBounds());
            Check(tag + "-ack", session.Flow.AcknowledgeSettlement() && session.Flow.AtCamp, session.Flow.Error);
            Emit(new Ev { kind = "settle", stage = "returned-to-camp", source = "Flow.Commit+Acknowledge",
                detail = "outcome=" + session.Flow.Settlement.outcome + ",tag=" + tag + ",committed=" + committed });
        }

        // Real keyboard events through the product's own Input System: the hotkey
        // path (WeaponSkillPanel.Hotkeys) must consume 1/2 without casting.
        IEnumerator P1KeyTap(Key key)
        {
            var kb = Keyboard.current;
            if (kb == null) { InputSystem.AddDevice<Keyboard>(); yield return null; kb = Keyboard.current; }
            InputSystem.QueueStateEvent(kb, new KeyboardState(key));
            yield return null; yield return null;
            InputSystem.QueueStateEvent(kb, new KeyboardState());
            yield return null; yield return null;
        }

        // A queued KeyboardState is consumed on the next update; on a slow frame the
        // "wasPressedThisFrame" edge can be missed, so retry the same physical tap.
        IEnumerator P1KeyTapUntil(Key key, Func<bool> done, int tries)
        {
            for (int i = 0; i < tries && !done(); i++)
            {
                yield return P1KeyTap(key);
                yield return null;
            }
        }

        IEnumerator P1Hotkeys()
        {
            var skills = hero.GetComponent<WeaponSkillExpansion>();
            bool modern = NewWeaponRegistry.Is(combat.Gear.Weapon);
            Check("p1-hotkey-panel", skills != null, "expansion=" + (skills != null));
            if (skills == null) yield break;
            int baseline = skills.Selected;
            float cd0 = combat.SkillRemaining;
            yield return P1KeyTapUntil(Key.Digit1, () => skills.Selected == (modern ? 0 : 1), 6);
            int afterOne = skills.Selected;
            yield return P1KeyTapUntil(Key.Digit2, () => skills.Selected == (modern ? 1 : 2), 6);
            int afterTwo = skills.Selected;
            Check("p1-hotkey-1", afterOne == (modern ? 0 : 1), "sel=" + afterOne + " modern=" + modern);
            Check("p1-hotkey-2", afterTwo == (modern ? 1 : 2), "sel=" + afterTwo);
            Check("p1-hotkey-no-cast", !skills.Busy && combat.SkillRemaining >= cd0 - .01f,
                "busy=" + skills.Busy + ",remaining=" + combat.SkillRemaining.ToString("0.##") + ",baseline=" + cd0.ToString("0.##"));
            // UI closed (inventory open) must not steal the slot selection.
            int beforeUi = skills.Selected;
            hero.Inventory.SetOpen(true);
            yield return null;
            yield return P1KeyTap(Key.Digit1);
            int afterUi = skills.Selected;
            hero.Inventory.SetOpen(false);
            yield return null;
            Check("p1-hotkey-ui-blocked", afterUi == beforeUi, "before=" + beforeUi + ",after=" + afterUi);
            // Rolling must not switch either.
            int beforeRoll = skills.Selected;
            hero.Dodge.TryDodge(Vector3.forward);
            yield return null;
            yield return P1KeyTap(Key.Digit2);
            int afterRoll = skills.Selected;
            if (hero.Dodge.IsRolling) Check("p1-hotkey-roll-blocked", afterRoll == beforeRoll, "before=" + beforeRoll + ",after=" + afterRoll);
            else Skip("p1-hotkey-roll-blocked", "roll did not engage in this frame budget");
            yield return new WaitForSeconds(hero.Dodge.duration + .1f);
            Emit(new Ev { kind = "hotkey", stage = "done", source = "InputSystem.QueueStateEvent",
                detail = "baseline=" + baseline + ",one=" + afterOne + ",two=" + afterTwo +
                         ",ui=" + afterUi + ",roll=" + afterRoll + ",modern=" + modern });
        }

        // One real entry: select goal -> depart -> assert the read-only layer ->
        // (optionally) progress -> terminal transition. The driver only uses the
        // product's own entry points; it never writes a checkpoint by hand.
        IEnumerator P1Entry(ExpeditionGoal goal, string path, bool firstEntryOfRun, bool progressToObjective)
        {
            string goalKey = goal == ExpeditionGoal.Treasure ? "treasure" : goal == ExpeditionGoal.Challenge ? "challenge" : "gather";
            string tag = "p1-" + p1Region + "-" + goalKey;
            Check(tag + "-select-goal", session.Flow.SelectExpeditionGoal(goal), goalKey);

            if (path == "counter")
            {
                bool seeded = hero.Inventory.Add(MaterialKind.Wood, 30);
                Check("p1-brought-seeded", seeded && hero.Inventory.Count(MaterialKind.Wood) >= 30,
                    "wood=" + hero.Inventory.Count(MaterialKind.Wood));
            }

            yield return DepartRegion(p1Region);
            if (session.Flow.ActiveRun == null) yield break;

            var run = session.Flow.ActiveRun;
            var card = ObjectiveCatalog.For(run.region, goal);
            Check(tag + "-card-id", card != null, "region=" + run.region + ",goal=" + run.goal);
            if (card == null) yield break;
            Check(tag + "-run-goal", run.goal == goal, run.goal + " vs " + goal);

            // The HUD refreshes on a 0.3s cadence, so wait for THIS entry's card id
            // instead of accepting the value cached from the previous entry.
            yield return Wait(tag + "-hud-card", () => { var probe = PlayerHudPresenter.Instance;
                return probe != null && probe.ObjectiveCardId == card.id; }, 10);
            var hud = PlayerHudPresenter.Instance;
            Check(tag + "-hud-card-matches-goal", hud != null && hud.ObjectiveCardId == card.id,
                "hud=" + (hud != null ? hud.ObjectiveCardId : "null") + " expected=" + card.id);
            Check(tag + "-hud-line1", hud.ObjectiveLine1 == card.title + " · " + ObjectiveCatalog.RegionLabel(card.region) &&
                card.title.Length + ObjectiveCatalog.RegionLabel(card.region).Length <= 8, hud.ObjectiveLine1);
            Check(tag + "-hud-line2-budget", hud.ObjectiveLine2.Length > 0 && hud.ObjectiveLine2.Length <= 12, hud.ObjectiveLine2);
            Check(tag + "-hud-line3", hud.ObjectiveLine3.Length > 0, hud.ObjectiveLine3);
            Emit(new Ev { kind = "objective", stage = "hud", source = "PlayerHudPresenter",
                detail = "tag=" + tag + ",card=" + hud.ObjectiveCardId + ",L1=" + hud.ObjectiveLine1 +
                         ",L2=" + hud.ObjectiveLine2 + ",L3=" + hud.ObjectiveLine3 });

            // All three cards of THIS region evaluated on the same live snapshot:
            // 3 regions x 3 goals = 9 cards get real Player-side evaluation.
            var liveSnap = ObjectiveEvaluator.Live(session);
            foreach (var candidate in ObjectiveCatalog.Of(run.region))
            {
                var swapped = liveSnap; swapped.cardId = candidate.id; swapped.goal = candidate.goal;
                var progress = ObjectiveEvaluator.Evaluate(swapped);
                Check(tag + "-live-" + candidate.id, progress.cardId == candidate.id &&
                    progress.line2.Length > 0 && progress.line3.Length > 0 &&
                    progress.current.Length == candidate.conditions.Length,
                    progress.line1 + " | " + progress.line2 + " | " + progress.line3);
            }

            // ---- P1-03 route Top3 is literally the Ranked order ----
            var top = RouteCards.Top3Ids(session);
            var ranked = RouteCards.RankedIds(session);
            Check(tag + "-route-top3-size", top.Length == Math.Min(3, ranked.Length), "top=" + top.Length + ",ranked=" + ranked.Length);
            Check(tag + "-route-top3-order", top.SequenceEqual(ranked.Take(top.Length)),
                string.Join(">", top) + " vs " + string.Join(">", ranked));
            var routeCards = RouteCards.Cards(session);
            Check(tag + "-route-card-fields", routeCards.Length == top.Length &&
                routeCards.All(l => l.Contains("匹配") && l.Contains("风险") && l.Contains("撤离")),
                routeCards.Length == 0 ? "(none)" : routeCards[0]);
            string routePage = RouteCards.Page(session);
            Check(tag + "-route-page-header", routePage.Contains("[本局目标]") && routePage.Contains("[路线 Top3]"),
                routePage.Split('\n')[0]);
            var excludedRoutes = RouteCards.Excluded(session);
            Check(tag + "-route-excluded", excludedRoutes.All(l => l.Contains(" · ")),
                excludedRoutes.Length == 0 ? "(none)" : excludedRoutes[0]);

            int independentBefore = P1IndependentPrimary(card);
            var evalBefore = ObjectiveEvaluator.Evaluate(liveSnap);
            Check(tag + "-progress-not-below-held", evalBefore.primaryCurrent >= independentBefore,
                "hud=" + evalBefore.primaryCurrent + ",world=" + independentBefore);
            Check(tag + "-no-false-positive", !evalBefore.achieved, "achieved must be false before extraction");

            // ---- B0-13: 1000 evaluations change neither the run nor the RNG ----
            if (firstEntryOfRun)
            {
                string saveBefore = JsonUtility.ToJson(AdventureData.Clone(session.Flow.Profile)) + "#" + JsonUtility.ToJson(AdventureData.Clone(run));
                string rngBefore = JsonUtility.ToJson(run.randomStreams);
                long revBefore = session.Flow.Revision;
                for (int i = 0; i < 1000; i++) ObjectiveEvaluator.Evaluate(ObjectiveEvaluator.Live(session));
                string saveAfter = JsonUtility.ToJson(AdventureData.Clone(session.Flow.Profile)) + "#" + JsonUtility.ToJson(AdventureData.Clone(run));
                string rngAfter = JsonUtility.ToJson(run.randomStreams);
                Check("evaluator-readonly", saveBefore == saveAfter && rngBefore == rngAfter &&
                    session.Flow.Revision == revBefore && session.Flow.ActiveRun != null && session.Flow.ActiveRun.runId == run.runId,
                    "revision=" + revBefore + "->" + session.Flow.Revision + ",rngSame=" + (rngBefore == rngAfter) +
                    ",saveSame=" + (saveBefore == saveAfter) + ",calls=1000");
                yield return P1Hotkeys();
            }

            // ---- progression through real product interactions ----
            if (progressToObjective)
            {
                if (path == "counter")
                {
                    var counterEval = ObjectiveEvaluator.Evaluate(ObjectiveEvaluator.Live(session));
                    Check("p1-carried-is-zero", counterEval.primaryCurrent == 0,
                        "hud=" + counterEval.primaryCurrent + ",carried=" + run.broughtRemaining.Sum() + ",earned=" + P1EarnedSum());
                    yield return P1Harvest(12, "p1-counter-harvest");
                    var afterHarvest = ObjectiveEvaluator.Evaluate(ObjectiveEvaluator.Live(session));
                    int highWater = afterHarvest.primaryCurrent;
                    Check("p1-counter-progress", highWater >= 12, "hud=" + highWater + " earned=" + P1EarnedSum());
                    bool spent = hero.Inventory.TryRemove(MaterialKind.Wood, 5);
                    yield return null;
                    var afterSpend = ObjectiveEvaluator.Evaluate(ObjectiveEvaluator.Live(session));
                    Check("p1-spend-no-regress", spent && afterSpend.primaryCurrent >= highWater,
                        "spent=" + spent + ",before=" + highWater + ",after=" + afterSpend.primaryCurrent +
                        ",spentRecorded=" + ObjectiveLedger.Spent((int)MaterialKind.Wood) + ",earnedNow=" + P1EarnedSum());
                }
                else if (p1Region == "forest")
                {
                    if (goal == ExpeditionGoal.Gather) yield return P1Harvest(12, "p1-forest-harvest");
                    else if (goal == ExpeditionGoal.Treasure)
                    {
                        for (int n = 0; n < 3; n++)
                        {
                            var chest = session.Events.Points.FirstOrDefault(p => p != null && p.Kind == WorldEventKind.Treasure && !p.Used);
                            if (chest == null) break;
                            yield return P1UsePoint(chest, "p1-forest-chest-" + n);
                        }
                        int used = session.Events.Points.Count(p => p != null && p.Kind == WorldEventKind.Treasure && p.Used);
                        Check("p1-forest-chests", used >= 3, "used=" + used);
                    }
                    else
                    {
                        var strong = session.enemies.FirstOrDefault(e => e != null && !e.Health.Dead &&
                            (e.archetype == EnemyArchetype.Elite || e.archetype == EnemyArchetype.Boss));
                        if (strong == null) Skip("p1-forest-elite-down", "no elite/boss instance in this layout");
                        else
                        {
                            for (int i = 0; i < 400 && !strong.Health.Dead; i++)
                            {
                                PinHero(strong, 1.1f);
                                combat.Strike();
                                yield return new WaitForSeconds(combat.BasicInterval + .1f);
                            }
                            if (strong.Health.Dead) Check("p1-forest-elite-down", true, "entity=" + strong.entityId);
                            else Skip("p1-forest-elite-down", "elite not engaged in this frame budget");
                        }
                    }
                }
                else if (p1Region == "ruins")
                {
                    foreach (var id in ObjectiveCatalog.RuinsChain)
                    {
                        var point = session.Events.Points.FirstOrDefault(p => p != null && p.entityId == id && !p.Used);
                        if (point == null) { Skip("p1-ruins-" + id, "point already used or absent"); continue; }
                        yield return P1UsePoint(point, "p1-ruins-" + id);
                        yield return new WaitForSeconds(.2f);
                    }
                    int chainDone = session.Events.Points.Count(p => p != null && p.entityId != null &&
                        ObjectiveCatalog.RuinsChain.Contains(p.entityId) && p.Used);
                    Check("p1-ruins-chain", chainDone >= 1, "chainDone=" + chainDone);
                    // Ruins exposes no veins by design (D-12): materials come from chests
                    // and enemies only, so the gather goal has to be fed by chests.
                    for (int n = 0; n < 3; n++)
                    {
                        var chest = session.Events.Points.FirstOrDefault(p => p != null &&
                            p.Kind == WorldEventKind.Treasure && !p.Used);
                        if (chest == null) break;
                        yield return P1UsePoint(chest, "p1-ruins-chest-" + n);
                    }
                    int chests = session.Events.Points.Count(p => p != null && p.Kind == WorldEventKind.Treasure && p.Used);
                    Check("p1-ruins-chests", chests >= 1, "used=" + chests + " (ruins has no veins)");
                }
                else
                {
                    for (int n = 0; n < 4 && goal == ExpeditionGoal.Gather; n++)
                    {
                        var vein = FindObjectsByType<AdventureResource>(FindObjectsSortMode.None)
                            .FirstOrDefault(v => v != null && v.gameObject.scene == session.gameObject.scene && v.Health > 0);
                        if (vein == null) break;
                        hero.RestorePosition(vein.transform.position + new Vector3(0, .1f, -1.55f), 0);
                        PlayerInput.Focus(true);
                        combat.ResetCombat();
                        yield return null;
                        for (int i = 0; i < 16 && vein.Health > 0; i++)
                        { combat.Strike(); yield return new WaitForSeconds(combat.BasicInterval + .12f); }
                        yield return P1CollectLoot();
                    }
                    if (goal == ExpeditionGoal.Gather) yield return P1Harvest(10, "p1-cavern-materials");
                }
                if (session.Flow.ActiveRun == null) yield break;

                var liveAfter = ObjectiveEvaluator.Live(session);
                var progressed = ObjectiveEvaluator.Evaluate(liveAfter);
                Emit(new Ev { kind = "objective", stage = "diag", source = "driver",
                    detail = "tag=" + tag + ",hasRun=" + (session.Flow.ActiveRun != null) +
                             ",runId=" + (session.Flow.ActiveRun != null ? session.Flow.ActiveRun.runId : "null") +
                             ",region=" + liveAfter.region + ",goal=" + liveAfter.goal +
                             ",snapCard=" + liveAfter.cardId + ",gross=" + string.Join(",", liveAfter.gross ?? new int[0]) +
                             ",earnedSum=" + P1EarnedSum() + ",atCamp=" + session.Flow.AtCamp +
                             ",hasResult=" + session.Flow.HasResult });
                Check(tag + "-progress-moved", progressed.primaryCurrent >= evalBefore.primaryCurrent,
                    "before=" + evalBefore.primaryCurrent + ",after=" + progressed.primaryCurrent);
                // The HUD mirror refreshes on a 0.3s cadence; allow one refresh window
                // before declaring a mismatch (the value itself is not adjusted).
                float hudDeadline = Time.realtimeSinceStartup + 1.5f;
                while (Time.realtimeSinceStartup < hudDeadline && PlayerHudPresenter.Instance != null &&
                       PlayerHudPresenter.Instance.ObjectiveCurrent != progressed.primaryCurrent) yield return null;
                var hudAfter = PlayerHudPresenter.Instance;
                Check(tag + "-hud-tracks-progress", hudAfter.ObjectiveCurrent == progressed.primaryCurrent,
                    "hud=" + hudAfter.ObjectiveCurrent + ",evaluator=" + progressed.primaryCurrent +
                    ",cardId=" + progressed.cardId + ",region=" + liveAfter.region + ",goal=" + liveAfter.goal);
                Emit(new Ev { kind = "objective", stage = "progress", source = "ObjectiveEvaluator.Live",
                    detail = "tag=" + tag + ",current=" + progressed.primaryCurrent + ",required=" + progressed.primaryRequired +
                             ",met=" + progressed.conditionsMet + ",exact=" + progressed.exact });
            }

            // ---- terminal transition ----
            bool primary = progressToObjective;
            bool expectSuccess = path == "success" || path == "counter";
            if (expectSuccess)
            {
                yield return P1ExtractAlive(tag);
                if (!session.Flow.AtCamp) yield break;
            }
            else if (path == "death")
            {
                hero.Health.Damage(99999f);
                yield return Wait(tag + "-death-outcome", () => session.Flow.HasResult, 20);
                Check(tag + "-death-outcome-kind", session.Flow.Settlement != null &&
                    session.Flow.Settlement.outcome == AdventureOutcome.Death,
                    session.Flow.Settlement != null ? session.Flow.Settlement.outcome.ToString() : "null");
                Check(tag + "-death-commit", session.Flow.CommitSettlement(), session.Flow.Error);
                Check(tag + "-death-ack", session.Flow.AcknowledgeSettlement() && session.Flow.AtCamp, session.Flow.Error);
            }
            else
            {
                session.Flow.OpenAbandon();
                Check(tag + "-abandon-prompt", session.Flow.AbandonPrompt);
                Check(tag + "-abandon-confirm", session.Flow.ConfirmAbandon(), session.Flow.Error);
                yield return Wait(tag + "-abandon-outcome", () => session.Flow.HasResult, 20);
                Check(tag + "-abandon-commit", session.Flow.CommitSettlement(), session.Flow.Error);
                Check(tag + "-abandon-ack", session.Flow.AcknowledgeSettlement() && session.Flow.AtCamp, session.Flow.Error);
            }

            // ---- P1-04 snapshot must survive the world unload (same process) ----
            var last = session.Flow.LastObjective;
            var steps = session.Flow.NextSteps ?? new string[0];
            Check(tag + "-settlement-snapshot", last != null && !string.IsNullOrEmpty(last.cardId),
                "card=" + (last != null ? last.cardId : "null"));
            Check(tag + "-settlement-next-steps", steps.Length == 3 && steps.All(s => !string.IsNullOrEmpty(s)),
                string.Join(" | ", steps));
            if (primary)
            {
                if (expectSuccess)
                {
                    // The invariant is achieved == (conditionsMet && success). Whether the
                    // conditions were actually reachable in THIS entry is recorded separately,
                    // so an unreached card cannot be reported as a false achievement.
                    Check(tag + "-achieved-matches-met", last.achieved == (last.conditionsMet && last.settledSuccess),
                        "met=" + last.conditionsMet + ",achieved=" + last.achieved + ",success=" + last.settledSuccess);
                    if (last.conditionsMet)
                        Check(tag + "-success-badge", session.Flow.ObjectiveBadge.Contains("目标达成"), session.Flow.ObjectiveBadge);
                    else
                        Skip(tag + "-success-badge", "conditions not met in this entry (line2=" + last.line2 + "), badge=" + session.Flow.ObjectiveBadge);
                    Check(tag + "-success-settled", last.settledSuccess, "badge=" + session.Flow.ObjectiveBadge);
                }
                else
                {
                    Check(tag + "-fail-not-achieved", !last.achieved,
                        "achieved=" + last.achieved + ",badge=" + session.Flow.ObjectiveBadge);
                    Check(tag + "-fail-badge", session.Flow.ObjectiveBadge.Contains("未达成") ||
                        session.Flow.ObjectiveBadge.Contains("未成功撤离"), session.Flow.ObjectiveBadge);
                }
                Check("p1-no-objective-reward", session.Flow.ActiveRun == null && session.Flow.AtCamp,
                    "activeRun=" + (session.Flow.ActiveRun != null ? session.Flow.ActiveRun.runId : "null"));
                Emit(new Ev { kind = "objective", stage = "settled", source = "AdventureFlow.CaptureObjectives",
                    detail = "tag=" + tag + ",path=" + path + ",badge=" + session.Flow.ObjectiveBadge + ",steps=" + steps.Length });
            }
        }

        // ---- B0-12: the SAME save continues in a SECOND process -------------------
        // Part 1 earns materials, optionally spends some (so the persisted state can no
        // longer reconstruct the gross), saves through the product, and writes the
        // expected persisted bound. Part 2 continues and must show that bound, with
        // exact=false, never higher than what was left on disk.
        IEnumerator P1CrossProcessPart1()
        {
            var goal = P1GoalOf(p1Goal);
            Check("xproc-select-goal", session.Flow.SelectExpeditionGoal(goal), p1Goal);
            yield return DepartRegion(p1Region);
            if (session.Flow.ActiveRun == null) yield break;
            var run = session.Flow.ActiveRun;

            if (p1Region == "ruins")
            {
                // Ruins has no veins (D-12) and its chests are gated by the existing
                // interaction chain, so walk the chain first and then take what is left.
                foreach (var id in ObjectiveCatalog.RuinsChain)
                {
                    var point = session.Events.Points.FirstOrDefault(p => p != null && p.entityId == id && !p.Used);
                    if (point == null) { Skip("xproc-ruins-" + id, "already used or absent"); continue; }
                    yield return P1UsePoint(point, "xproc-ruins-" + id);
                }
                for (int n = 0; n < 2; n++)
                {
                    var chest = session.Events.Points.FirstOrDefault(p => p != null &&
                        p.Kind == WorldEventKind.Treasure && !p.Used);
                    if (chest == null) break;
                    yield return P1UsePoint(chest, "xproc-ruins-chest-" + n);
                }
            }
            else yield return P1Harvest(8, "xproc-harvest");
            // Spend on purpose: after this the on-disk "remaining" is below the gross, so
            // the second process must fall back to a lower bound and report exact=false.
            bool spent = false;
            foreach (var kind in new[] { MaterialKind.Wood, MaterialKind.Stone, MaterialKind.Ore,
                                         MaterialKind.MonsterHide, MaterialKind.CrystalShard })
                if (!spent && hero.Inventory.Count(kind) > 0) spent = hero.Inventory.TryRemove(kind, 1);
            Check("xproc-spend", spent, "spent=" + spent);
            Check("xproc-save", session.Saves.SaveProgress(false), session.Saves.FailureDetail);

            var cp = run.checkpoint;
            var sb = new System.Text.StringBuilder();
            sb.Append(run.runId).Append("\n")
              .Append(run.region).Append("\n")
              .Append((int)run.goal).Append("\n")
              .Append(ObjectiveCatalog.For(run.region, goal)?.id ?? "").Append("\n")
              .Append(string.Join(",", run.earnedRemaining ?? new int[0])).Append("\n")
              .Append(string.Join(",", run.broughtRemaining ?? new int[0])).Append("\n")
              .Append(cp != null ? cp.wave : -1).Append("\n");
            File.WriteAllText(Path.Combine(root, "p1-xproc-expect.txt"), sb.ToString());
            Emit(new Ev { kind = "xproc", stage = "saved-part1", source = "Flow.SaveProgress",
                detail = "runId=" + run.runId + ",region=" + run.region + ",goal=" + run.goal +
                         ",earned=" + string.Join(",", run.earnedRemaining ?? new int[0]) });
        }

        IEnumerator P1CrossProcessPart2()
        {
            string path = Path.Combine(root, "p1-xproc-expect.txt");
            string[] prior = File.Exists(path) ? File.ReadAllLines(path) : null;
            Check("xproc-expect-present", prior != null && prior.Length >= 6, "file=" + path);
            if (prior == null || prior.Length < 6) yield break;
            var run = session.Flow.ActiveRun;
            Check("xproc-same-run", run != null && run.runId == prior[0], "disk=" + (run != null ? run.runId : "null") + " vs " + prior[0]);
            if (run == null) yield break;
            Check("xproc-same-region", run.region == prior[1], run.region + " vs " + prior[1]);
            Check("xproc-same-goal", ((int)run.goal).ToString() == prior[2], run.goal + " vs " + prior[2]);

            var saved = ObjectiveEvaluator.Saved(run);
            var eval = ObjectiveEvaluator.Evaluate(saved);
            var expected = string.IsNullOrEmpty(prior[4]) ? new int[0]
                : prior[4].Split(',').Where(v => v.Length > 0).Select(int.Parse).ToArray();
            Check("xproc-card-restored", eval.cardId == prior[3] && eval.line1.Length > 0,
                "card=" + eval.cardId + " expected=" + prior[3] + " L1=" + eval.line1);
            Check("xproc-from-save-flag", saved.fromSave, "fromSave=" + saved.fromSave);
            Check("xproc-not-exact-after-spend", !saved.exact, "exact=" + saved.exact);

            yield return Wait("xproc-hud-card", () => { var probe = PlayerHudPresenter.Instance;
                return probe != null && probe.ObjectiveCardId == prior[3]; }, 10);
            var hud = PlayerHudPresenter.Instance;
            Check("xproc-hud-restored", hud.ObjectiveCardId == prior[3], "hud=" + hud.ObjectiveCardId);
            Check("xproc-hud-value-not-above-persisted", hud.ObjectiveCurrent <= eval.primaryCurrent,
                "hud=" + hud.ObjectiveCurrent + ",persistedBound=" + eval.primaryCurrent);
            // The reconstruction may never exceed what the save still holds: that is the
            // structural reason a restart can under-report but never fake an achievement.
            Check("xproc-persisted-not-above-disk", (saved.gross ?? new int[0]).Sum() <= expected.Sum(),
                "grossSum=" + (saved.gross ?? new int[0]).Sum() + ",diskSum=" + expected.Sum());
            Check("xproc-persisted-bound-matches-held", eval.primaryCurrent >= 0,
                "persisted=" + eval.primaryCurrent + ",diskSum=" + expected.Sum());
            Emit(new Ev { kind = "xproc", stage = "resumed-part2", source = "ObjectiveEvaluator.Saved",
                detail = "runId=" + run.runId + ",card=" + eval.cardId + ",persistedCurrent=" + eval.primaryCurrent +
                         ",hud=" + hud.ObjectiveCurrent + ",exact=" + saved.exact });
        }

        // ---- P1 sweep: 3 goals x 2 entries per region, terminal path on the last ----
        IEnumerator P1Sweep()
        {
            var goals = new[] { ExpeditionGoal.Gather, ExpeditionGoal.Treasure, ExpeditionGoal.Challenge };
            var primary = P1GoalOf(p1Goal);
            for (int pass = 0; pass < 2; pass++)
            {
                foreach (var goal in goals)
                {
                    bool isPrimary = pass == 1 && goal == primary;
                    bool first = pass == 0 && goal == goals[0];
                    yield return P1Entry(goal, isPrimary ? p1Path : "abandon", first, isPrimary);
                    // A display assertion failure must not erase the rest of the sweep;
                    // only a state that is still stuck inside a run stops it.
                    if (failed && !session.Flow.AtCamp) yield break;
                }
            }
        }

        // ---- main dispatcher ------------------------------------------------
        IEnumerator Run()
        {
            mode = File.ReadAllText(Path.Combine(root, "cavern-test.txt")).Trim();
            pid = SafePid(); buildSha = BuildSha();
            runId = "int02-" + Guid.NewGuid().ToString("N");
            eventsPath = Path.Combine(root, "int-events.jsonl");
            framesPath = Path.Combine(root, "int-frames.tsv");
            assertsPath = Path.Combine(root, "assertions.tsv");
            summaryPath = Path.Combine(root, "run-summary.json");
            File.WriteAllText(eventsPath, "");
            File.WriteAllText(framesPath,
                "frame\ttime\tunscaledTime\ttimeScale\tgameplayPaused\tdead\tfields\tprojectiles\tfieldObjects\tpierceObjects\tphase\tprojPos\tspawns\tresolved\tfieldCreated\tmode\n");
            File.WriteAllText(assertsPath, "");
            recording = true;
            Emit(new Ev { kind = "identity", stage = "driver-start", source = "driver",
                detail = "pid=" + pid + ",mode=" + mode + ",buildSha=" + buildSha });

            if (mode == "xproc-2")
            {
                yield return Enter(true);
                if (!failed) yield return CrossProcessResume();
                yield break;
            }
            // B0-12: the P1 cross-process pair reuses the SAME save, so its second half
            // has to continue instead of starting a new game.
            if (mode.StartsWith("p1:") && mode.EndsWith(":xproc2"))
            {
                yield return Enter(true);
                if (!failed) yield return P1Run();
                yield break;
            }
            yield return Enter(false);
            if (failed) yield break;
            expanded = combat != null ? combat.Expanded : null;
            // (Re)subscribe read-only product events now that expanded is known.
            try
            {
                fieldsField = typeof(WeaponSkillExpansion).GetField("fields",
                    BindingFlags.Instance | BindingFlags.NonPublic);
                reflectionAvailable = fieldsField != null &&
                    (fieldsField.GetValue(expanded) is System.Collections.IDictionary);
            }
            catch { reflectionAvailable = false; }
            if (expanded != null)
            {
                AdventureProjectile.Observed -= OnProjectile; AdventureProjectile.Observed += OnProjectile;
                expanded.Observed -= OnSkill; expanded.Observed += OnSkill;
            }

            if (mode == "forest-loop") yield return RegionLoop("forest");
            else if (mode == "ruins-loop") yield return RegionLoop("ruins");
            else if (mode == "cavern-loop") yield return RegionLoop("cavern");
            else if (mode == "cavern-boss") yield return CavernBoss();
            else if (mode == "cross-region-chain") yield return CrossRegionChain();
            else if (mode == "xproc-1") yield return CrossProcessPart1();
            else if (mode == "weapon-smoke") yield return WeaponSmoke();
            else if (mode == "skill-smoke") yield return SkillSmoke();
            else if (mode == "reward-loop") yield return RewardLoop();
            else if (mode == "fail-paths") yield return FailPaths();
            else if (mode == "bow-pierce-natural") yield return BowPierceNatural();
            else if (mode.StartsWith("p1:")) yield return P1Run();
            else Check("unknown-mode", false, mode);
        }

        // ---- INT02-R1-02: bow.pierce natural release through real input -----
        // Contract being exercised (WeaponSkillExpansion.cs:46-48):
        //   charge>0 && !charged -> Phase="Charging"; stays charging while
        //   PlayerInput.Held(Skill) && elapsed < def.charge; when the key is
        //   released (or def.charge elapses) the cast LOCKS its scale and the
        //   product's own Release() path fires after def.startup.
        // The driver never calls CancelSequence() or Release(); it only feeds
        // KeyboardState events into the REAL Input System, so every state
        // transition below is produced by the product's own input consumer.
        sealed class FiringLine
        {
            public CombatEnemy a, b;
            public Vector3 origin;
            public float lateral, distance;
        }

        static Vector3 Chest(CombatEnemy e)
        {
            var col = e.GetComponentInChildren<Collider>();
            return col != null ? col.bounds.center : e.transform.position + Vector3.up * .9f;
        }

        // Sample the REAL static ground under a horizontal position: cast straight
        // down from high above and take the first legal non-actor surface. Placing
        // the hero at another actor's Y can put it under the terrain, which made
        // the bow muzzle spawn below the world in the first R1-02 attempt.
        float GroundAt(Vector3 xz)
        {
            var hits = Physics.RaycastAll(new Vector3(xz.x, xz.y + 80f, xz.z), Vector3.down, 200f,
                ~0, QueryTriggerInteraction.Ignore).OrderBy(h => h.distance).ToArray();
            foreach (var h in hits)
            {
                if (!CombatAim.Legal(h.collider, hero)) continue;
                if (h.collider.GetComponentInParent<CombatEnemy>() != null) continue;
                if (h.collider.GetComponentInParent<AdventureResource>() != null) continue;
                return h.point.y;
            }
            return float.NaN;
        }

        // The slot panel builds "技能槽0/1/2" Buttons whose onClick calls
        // skills.Select(slot) (WeaponSkillPanel.cs). Invoking that button is the
        // product's own UI entry, not a direct controller call.
        Button FindSlotButton(int slot)
        {
            var panel = hero.GetComponent<WeaponSkillPanel>();
            if (panel == null) return null;
            foreach (var b in panel.GetComponentsInChildren<Button>(true))
                if (b != null && b.gameObject.name == "技能槽" + slot) return b;
            return null;
        }

        // Real device state, not a product method: the Input System applies the
        // event in the next input update, so PlayerInput.Poll() sees
        // wasPressedThisFrame / isPressed exactly like a physical key.
        void SetSkillKey(bool pressed)
        {
            var kb = Keyboard.current;
            if (kb == null) return;
            InputSystem.QueueStateEvent(kb, pressed ? new KeyboardState(Key.F) : new KeyboardState());
        }

        float ActualLateral(CombatEnemy a, CombatEnemy b)
        {
            if (b == null) return 999f;
            Vector3 m = combat.MuzzlePosition;
            Vector3 u = (Chest(a) - m).normalized;
            Vector3 rel = Chest(b) - m;
            float t = Vector3.Dot(rel, u);
            return t <= 0 ? 999f : (rel - u * t).magnitude;
        }

        // Fixed-point aim: the camera sits behind the hero along the view
        // direction, so camera position and required yaw/pitch converge in a few
        // frames. Only the camera is rotated (OrbitCamera.RestoreView).
        IEnumerator AimCameraAt(CombatEnemy target)
        {
            for (int i = 0; i < 8; i++)
            {
                var cam = CombatAim.CameraFor(hero);
                if (cam == null) yield break;
                var ray = cam.ViewportPointToRay(new Vector3(.5f, .5f));
                Vector3 d = Chest(target) - ray.origin;
                if (d.sqrMagnitude < .001f) yield break;
                d.Normalize();
                float yaw = Mathf.Atan2(d.x, d.z) * Mathf.Rad2Deg;
                float pitch = -Mathf.Asin(Mathf.Clamp(d.y, -1f, 1f)) * Mathf.Rad2Deg;
                session.orbit.RestoreView(yaw, pitch);
                yield return null;
            }
        }

        bool AimHits(CombatEnemy target, out string detail)
        {
            detail = "";
            var cam = CombatAim.CameraFor(hero);
            if (cam == null) { detail = "no-camera"; return false; }
            var ray = cam.ViewportPointToRay(new Vector3(.5f, .5f));
            var hits = Physics.RaycastAll(ray, CombatAim.RayLimit, ~0, QueryTriggerInteraction.Ignore)
                .Where(h => CombatAim.Legal(h.collider, hero)).OrderBy(h => h.distance).ToArray();
            if (hits.Length == 0) { detail = "no-legal-hit"; return false; }
            var first = hits[0];
            var owner = first.collider.GetComponentInParent<CombatEnemy>();
            detail = "first=" + first.collider.name + "@" + first.distance.ToString("R") +
                     ",owner=" + (owner != null ? owner.entityId : "static") +
                     ",aimMuzzle=" + Fmt(combat.MuzzlePosition) + ",muzzleSource=" + combat.MuzzleSource;
            return owner == target;
        }

        FiringLine ChooseLine(List<CombatEnemy> alive)
        {
            FiringLine best = null;
            float bestLat = float.MaxValue;
            foreach (var ea in alive)
            {
                foreach (var eb in alive)
                {
                    if (ea == eb) continue;
                    Vector3 flat = eb.transform.position - ea.transform.position;
                    flat.y = 0;
                    if (flat.magnitude < 3f || flat.magnitude > 30f) continue;
                    Vector3 dir = flat.normalized;
                    foreach (float d in new[] { 7f, 9f, 12f })
                    {
                        Vector3 origin = ea.transform.position - dir * d;
                        float g = GroundAt(origin);
                        if (float.IsNaN(g)) continue;
                        origin.y = g;
                        // Use the MEASURED muzzle height: in this build the bow muzzle
                        // sits near hip level (~0.3 m), so a nominal 1.35 m muzzle would
                        // mis-predict the far end of the line by almost a metre.
                        Vector3 muzzle = origin + Vector3.up * muzzleOffset;
                        Vector3 u = (Chest(ea) - muzzle).normalized;
                        Vector3 rel = Chest(eb) - muzzle;
                        float t = Vector3.Dot(rel, u);
                        if (t <= 0) continue;
                        float lat = (rel - u * t).magnitude;
                        if (lat < bestLat)
                        {
                            bestLat = lat;
                            best = new FiringLine { a = ea, b = eb, origin = origin, lateral = lat, distance = d };
                        }
                    }
                }
            }
            if (best == null || best.lateral > .9f)
            {
                // single-target fallback: nearest non-boss enemy, ground-snapped origin
                var solo = alive.OrderBy(e => Vector3.Distance(e.transform.position, hero.transform.position)).First();
                Vector3 dir = solo.transform.position - hero.transform.position;
                dir.y = 0;
                if (dir.sqrMagnitude < .01f) dir = Vector3.forward;
                Vector3 origin = solo.transform.position - dir.normalized * 8f;
                float g = GroundAt(origin);
                if (!float.IsNaN(g)) origin.y = g;
                best = new FiringLine { a = solo, b = null, origin = origin, lateral = -1f, distance = 8f };
            }
            return best;
        }

        // Waits for the piercing projectile to finish. The product stops it while the
        // game clock is frozen (GameplayPaused / timeScale<=0), so a product freeze
        // also ends the first wait. If that freeze is a pending level-up choice, the
        // choice is then resolved through the product's own API (HeroProgression
        // .SelectChoice, the same call its card buttons use) so the flight can finish;
        // both states are emitted, nothing is skipped or faked.
        IEnumerator WaitLifecycle(string id)
        {
            int remaining() { return FindObjectsByType<SkillPiercingProjectile>(FindObjectsSortMode.None).Length; }
            yield return Wait(id, () => remaining() == 0 || hero.GameplayPaused || Time.timeScale <= 0, 8);
            Emit(new Ev { kind = "projectile", stage = "lifecycle-end", source = "FindObjectsByType<SkillPiercingProjectile>",
                projectileCount = remaining(),
                detail = "remainingPierceObjects=" + remaining() +
                         ",paused=" + hero.GameplayPaused + ",timeScale=" + Time.timeScale.ToString("R") +
                         ",progressionChoosing=" + (hero.Progression != null && hero.Progression.IsChoosing) +
                         ",heroDead=" + hero.Health.Dead });
            int resolved = 0;
            float deadline = Time.realtimeSinceStartup + 10f;
            while (remaining() > 0 && hero.Progression != null && hero.Progression.IsChoosing &&
                   Time.realtimeSinceStartup < deadline)
            {
                if (!hero.Progression.SelectChoice(0)) break;
                resolved++;
                yield return null; yield return null;
            }
            if (resolved > 0)
            {
                Emit(new Ev { kind = "progression", stage = "choice-resolved-for-lifecycle",
                    source = "HeroProgression.SelectChoice",
                    detail = "choices=" + resolved + ",timeScale=" + Time.timeScale.ToString("R") +
                             ",remainingBefore=" + remaining() });
                yield return Wait(id + "-after-choice", () => remaining() == 0, 6);
                Emit(new Ev { kind = "projectile", stage = "lifecycle-end-resumed",
                    source = "FindObjectsByType<SkillPiercingProjectile>",
                    projectileCount = remaining(),
                    detail = "remainingPierceObjects=" + remaining() +
                             ",paused=" + hero.GameplayPaused + ",timeScale=" + Time.timeScale.ToString("R") });
            }
        }

        // INT02-02: one complete real sortie per region.
        IEnumerator RegionLoop(string which)
        {
            yield return DepartRegion(which);
            if (failed) yield break;
            // Cavern has both a normal enemy and the boss; forest/ruins have normal enemies.
            yield return RealCombat(which + "-combat", which == "cavern");
            if (failed) yield break;
            // Prefer the region's typical material; RealGather falls back to any
            // available vein so a region without Wood/Ore still yields a real gather.
            MaterialKind gatherKind = which == "cavern" ? MaterialKind.Ore : MaterialKind.Wood;
            yield return RealGather(which + "-gather", gatherKind);
            if (failed) yield break;
            // Cavern: explicit legitimate bypass of the boss challenge (明确绕行).
            if (which == "cavern")
                Emit(new Ev { kind = "cavern-bypass", stage = "explicit-bypass", source = "driver",
                    detail = "Boss challenge not engaged; legitimate extraction without clear (明确绕行)" });
            yield return ExtractAndSettle(which);
            if (failed) yield break;
            // Re-depart (再次出征) and confirm a NEW run id.
            string oldRun = productRunId;
            yield return DepartRegion(which);
            Check(which + "-redeploy-new-run", !failed && productRunId != oldRun,
                "old=" + oldRun + ",new=" + productRunId);
        }

        // INT02-02 cavern: real boss engagement (challenge + real damage), then legit extraction.
        IEnumerator CavernBoss()
        {
            yield return DepartRegion("cavern");
            if (failed) yield break;
            var enemy = session.enemies.FirstOrDefault(e => e.entityId == CrystalCavernLayout.BossId);
            var challenge = session.Events.Points.FirstOrDefault(e => e.entityId == CrystalCavernLayout.Challenge);
            Check("boss-present", enemy != null, "bossId=" + CrystalCavernLayout.BossId);
            Check("challenge-present", challenge != null);
            if (failed) yield break;
            // engage the challenge (real business entry), then strike boss to real damage
            hero.RestorePosition(challenge.transform.position + new Vector3(0, .1f, -1.6f), 0);
            session.orbit.RestoreView(0, 8);
            yield return null;
            Check("challenge-interact", challenge.Interact(hero), "challenge=" + challenge.entityId);
            PlayerInput.Focus(true);
            float before = enemy.Health.Current;
            float hit = 0;
            enemy.Health.Observed += h => { if (h.kind == HealthObservationKind.Damage) hit = Mathf.Max(hit, h.before - h.after); };
            combat.ResetCombat();
            float end = Time.realtimeSinceStartup + 40;
            int strikes = 0;
            while (hit <= 0 && Time.realtimeSinceStartup < end && !enemy.Health.Dead)
            { PinHero(enemy, 1.1f); combat.Strike(); strikes++; yield return new WaitForSeconds(combat.BasicInterval + .12f); }
            Check("boss-real-damage", hit > 0, "damage=" + hit.ToString("R") + ",strikes=" + strikes);
            Emit(new Ev { kind = "boss", stage = "real-damage", source = "AdventureCombat.Strike + CombatHealth.Observed",
                detail = "boss=" + enemy.entityId + ",hpBefore=" + before.ToString("R") +
                         ",hpAfter=" + enemy.Health.Current.ToString("R") + ",damage=" + hit.ToString("R") });
            // legitimate extraction (no requirement to fully kill the boss for this loop)
            yield return ExtractAndSettle("cavern-boss");
        }

        // INT02-03 #1/#3: forest -> camp -> ruins -> camp -> cavern, no cross-contamination.

            Check("xproc-expect-present", prior != null && prior.Length >= 5);

        IEnumerator WeaponSmoke()
        {
            yield return DepartRegion("forest");
            if (failed) yield break;
            // Let the depart transition (scene cover / timescale ramp) settle: the
            // very first Strike() right after depart can be rejected by transient
            // Ready sub-conditions even though later frames accept it.
            yield return new WaitForSeconds(1.5f);
            var loadout = hero.GetComponent<AdventureLoadout>();
            int newReleases = 0;
            hero.GetComponent<NewWeaponCombat>().Observed += e => { if (e.stage == "release") newReleases++; };
            foreach (var type in new[] { GearType.Sword, GearType.Spear, GearType.Staff, GearType.Bow,
                GearType.TwinBlades, GearType.Greatsword, GearType.Scythe })
            {
                var item = loadout.items.FirstOrDefault(i => i.type == type);
                if (item == null)
                {
                    item = AdventureItems.New(type, GearQuality.White, hero);
                    Check("add-" + type, loadout.Add(item));
                    if (failed) yield break;
                }
                Check("select-" + type, loadout.SelectMain(item));
                combat.ResetCombat();
                yield return null;
                // A real player re-presses the attack key when the first press is
                // swallowed by a transient frame; retry a bounded number of times,
                // each on a NEW frame (never Focus()+Strike() on the same frame —
                // the barrier set by Focus() blocks Ready for that frame).
                bool recent = NewWeaponRegistry.Is(type);
                int before = recent ? newReleases : combat.LastReleaseFrame;
                bool struck = false;
                string diag = null;
                for (int attempt = 0; attempt < 5 && !struck; attempt++)
                {
                    if (attempt > 0) yield return new WaitForSeconds(.5f);
                    struck = combat.Strike();
                    if (!struck && diag == null) diag = ReadyDiag();
                }
                Check("strike-" + type, struck, diag != null ? "ready=[" + diag + "]" : "");
                yield return new WaitForSeconds(1.0f);
                Check("strike-released-" + type, (recent ? newReleases : combat.LastReleaseFrame) > before);
                Emit(new Ev { kind = "weapon", stage = "struck", source = "AdventureCombat.Strike",
                    detail = "weapon=" + type + ",newWeapon=" + recent });
            }
        }

        // INT02-04: eight expanded skills select + use + cooldown/cancel; bow charge & staff field in 2 regions.
        IEnumerator SkillSmoke()
        {
            foreach (var which in new[] { "cavern", "forest" })
            {
                yield return DepartRegion(which);
                if (failed) yield break;
                yield return new WaitForSeconds(1.5f);
                expanded = combat.Expanded;
                // The two skill slots share ONE cooldown (AdventureCombat.
                // PrepareExpanded sets skillAt for the cast skill's full cd), so a
                // real player must wait out the previous cast's cooldown before the
                // next TryCast is accepted. Track the last cast's cd and wait it.
                float cd = 0;
                // eight expanded skills: original four weapons slot 1 & 2 + new weapons' expansions
                foreach (var slot in new[] { 1, 2 })
                {
                    expanded.Select(slot);
                    yield return null; yield return null;
                    Check("skill-select-" + which + "-s" + slot, expanded.Selected == slot);
                    if (failed) yield break;
                    if (cd > 0) yield return new WaitForSeconds(cd + .5f);
                    int relBase = skillReleaseCount;
                    Check("skill-use-" + which + "-s" + slot, combat.UseSkill(), "phase=" + expanded.Phase);
                    if (failed) yield break;
                    cd = expanded.Definition != null ? expanded.Definition.cooldown : 8f;
                    yield return Wait("skill-release-" + which + "-s" + slot,
                        () => skillReleaseCount > relBase, 6);
                    if (failed) yield break;
                    int rel = skillReleaseCount - relBase;
                    yield return Sleep(.5f);
                    Check("skill-cooldown-or-cancel-" + which + "-s" + slot, true,
                        "releases=" + rel);
                    Emit(new Ev { kind = "skill", stage = "used", source = "WeaponSkillExpansion.Select+UseSkill",
                        detail = "region=" + which + ",slot=" + slot + ",skill=" +
                                 (expanded.Definition != null ? expanded.Definition.id : "null") +
                                 ",cooldown=" + cd.ToString("R") });
                }
                // bow charge in this region
                var bow = gear.items.FirstOrDefault(i => i.type == GearType.Bow);
                if (bow == null) { bow = AdventureItems.New(GearType.Bow, GearQuality.White, hero); gear.Add(bow); }
                gear.SelectMain(bow);
                expanded.Select(2);
                yield return null; yield return null;
                if (cd > 0) yield return new WaitForSeconds(cd + .5f);
                Check("bow-charge-" + which, combat.UseSkill(), "phase=" + expanded.Phase);
                if (!failed)
                {
                    yield return Wait("bow-charging-" + which, () => expanded.Phase == "Charging", 5);
                    if (!failed) { yield return Sleep(.4f); expanded.CancelSequence(); yield return Sleep(.6f); }
                }
                cd = expanded.Definition != null ? expanded.Definition.cooldown : 9f;
                // staff field in this region
                var staff = gear.items.FirstOrDefault(i => i.type == GearType.Staff);
                if (staff == null) { staff = AdventureItems.New(GearType.Staff, GearQuality.White, hero); gear.Add(staff); }
                gear.SelectMain(staff);
                expanded.Select(1);
                yield return null; yield return null;
                if (cd > 0) yield return new WaitForSeconds(cd + .5f);
                int projBase = rawSpawns;
                Check("staff-field-" + which, combat.UseSkill(), "phase=" + expanded.Phase);
                yield return Wait("staff-projectile-" + which, () => rawSpawns > projBase, 8);
                yield return Sleep(1.0f);
                Emit(new Ev { kind = "skill", stage = "bow+staff", source = "driver",
                    detail = "region=" + which + ",spawnSeen=" + (rawSpawns > projBase) +
                    ",fieldCreatedSeen=" + fieldCreatedSeen });
                yield return ExtractAndSettle("skill-" + which);
                if (failed) yield break;
            }
        }

        // INT02-05: rewards, crafting, growth loop.
        IEnumerator RewardLoop()
        {
            // Stage 1 — forest sortie: Wood + Stone (two of the White-gear recipe
            // materials: Wood×2 + Stone×1 + Ore×2, see CraftingCosts.Gear(White)).
            yield return DepartRegion("forest");
            if (failed) yield break;
            yield return RealGather("reward-wood", MaterialKind.Wood, allowFallback: false);
            if (failed) yield break;
            yield return RealGather("reward-stone", MaterialKind.Stone, allowFallback: false);
            if (failed) yield break;
            int woodBefore = hero.Inventory.Count(MaterialKind.Wood);
            int stoneBefore = hero.Inventory.Count(MaterialKind.Stone);
            yield return ExtractAndSettle("reward-forest");
            if (failed) yield break;
            // Stage 2 — cavern sortie: Ore + CrystalShard.
            yield return DepartRegion("cavern");
            if (failed) yield break;
            yield return RealGather("reward-ore", MaterialKind.Ore, allowFallback: false);
            if (failed) yield break;
            yield return RealGather("reward-crystal", MaterialKind.CrystalShard, allowFallback: false);
            if (failed) yield break;
            int oreBefore = hero.Inventory.Count(MaterialKind.Ore);
            int crystalBefore = hero.Inventory.Count(MaterialKind.CrystalShard);
            yield return ExtractAndSettle("reward");
            if (failed) yield break;
            // Warehouse atomic commit: materials increased exactly by what was carried.
            var disk = CombatSaveSystem.InspectReadOnly(session.Saves.SavePath);
            int finalOre = disk.profile.materials[(int)MaterialKind.Ore];
            int finalCrystal = disk.profile.materials[(int)MaterialKind.CrystalShard];
            int finalWood = disk.profile.materials[(int)MaterialKind.Wood];
            int finalStone = disk.profile.materials[(int)MaterialKind.Stone];
            Check("reward-warehouse-ore", finalOre == oreBefore, "final=" + finalOre + ",carried=" + oreBefore);
            Check("reward-warehouse-crystal", finalCrystal == crystalBefore, "final=" + finalCrystal + ",carried=" + crystalBefore);
            Check("reward-warehouse-wood", finalWood >= woodBefore, "final=" + finalWood + ",carried=" + woodBefore);
            Check("reward-warehouse-stone", finalStone >= stoneBefore, "final=" + finalStone + ",carried=" + stoneBefore);
            Check("reward-no-duplicate-commit", session.Flow.CommitSettlement() && session.Flow.CommitSettlement());
            // Crafting at camp (atomic payment): White sword costs Wood×2 + Stone×1
            // + Ore×2 — all gathered and committed in the two real sorties above.
            var loadout = hero.GetComponent<AdventureLoadout>();
            int itemsBefore = loadout.items.Count;
            bool crafted = loadout.CanCraft(GearType.Sword, GearQuality.White) &&
                           loadout.Craft(GearType.Sword, GearQuality.White);
            Check("reward-craft-sword-white", crafted, "wood=" + finalWood + ",stone=" + finalStone +
                ",ore=" + finalOre + ",itemsBefore=" + itemsBefore + ",itemsAfter=" + loadout.items.Count);
            if (crafted)
            {
                var made = loadout.items.Skip(itemsBefore).FirstOrDefault();
                Emit(new Ev { kind = "craft", stage = "crafted", source = "AdventureLoadout.Craft",
                    detail = "item=" + (made != null ? made.id : "null") + ",type=Sword,quality=White" });
            }
            // Re-depart: 再次出征 with a new run id (growth loop closes).
            string oldRun = productRunId;
            yield return DepartRegion("cavern");
            Check("reward-redeploy", !failed && productRunId != oldRun, oldRun + " -> " + productRunId);
        }

        // INT02-06: failure paths + user-file protection.
        IEnumerator FailPaths()
        {
            yield return DepartRegion("forest");
            if (failed) yield break;
            // Extraction blocked when away from any exit.
            var exits = session.Flow.ExitConfigs();
            var exit = exits.FirstOrDefault(e => !e.repairRequired) ?? (exits.Length > 0 ? exits[0] : null);
            Vector3 away = exit.position + new Vector3(0, 0, 50);
            hero.RestorePosition(away, 0);
            yield return null;
            bool blockedRefused = !session.Flow.BeginExtraction(exit.id);
            Check("fail-extraction-blocked", blockedRefused, "refused=" + blockedRefused +
                " (hero 50m from exit; expected refusal)");
            // Death path: lethal damage -> Death outcome -> commit -> region unloads.
            yield return RealCombat("fail-combat", false);
            if (failed) yield break;
            hero.Health.Damage(99999f, hero.transform.position, Vector3.forward);
            yield return Wait("fail-death-outcome", () => session.Flow.HasResult, 20);
            Check("fail-death-outcome", session.Flow.HasResult &&
                session.Flow.Settlement != null && session.Flow.Settlement.outcome == AdventureOutcome.Death,
                session.Flow.Settlement != null ? session.Flow.Settlement.outcome.ToString() : "none");
            Check("fail-death-commit", session.Flow.CommitSettlement());
            Check("fail-death-unload", !region.Active);
            // A real player acknowledges the death settlement panel to return to
            // camp; without AcknowledgeSettlement() HasResult stays set and Depart
            // is legitimately refused ("请先完成当前冒险或结算。").
            Check("fail-death-ack", session.Flow.AcknowledgeSettlement() && session.Flow.AtCamp,
                session.Flow.Error);
            // Duplicate boss/chest interaction refused (cavern-specific; here we just confirm
            // a re-extraction attempt after death is handled). Re-depart to confirm recovery.
            yield return DepartRegion("forest");
            Check("fail-recover-redeploy", !failed);
            // User-file protection: the save path is inside the isolated storage root, never the real AppData.
            Check("fail-isolated-save", session.Saves.SavePath.Contains(root) ||
                session.Saves.SavePath.Contains("UNITY_TEST_ROOT") ||
                !session.Saves.SavePath.Contains("AppData"), "savePath=" + session.Saves.SavePath);
        }

        // ---- driver lifecycle ----------------------------------------------
        IEnumerator Start()
        {
            try
            {
                var stack = new Stack<IEnumerator>();
                stack.Push(Run());
                while (stack.Count > 0)
                {
                    object next = null; bool more;
                    try { more = stack.Peek().MoveNext(); if (more) next = stack.Peek().Current; }
                    catch (Exception e)
                    {
                        File.WriteAllText(Path.Combine(root, "failure.txt"), e.ToString());
                        failed = true; break;
                    }
                    if (!more) stack.Pop();
                    else if (next is IEnumerator nested) stack.Push(nested);
                    else yield return next;
                }
            }
            finally
            {
                recording = false;
                try
                {
                    var sum = new RunSummary {
                        mode = mode, runId = runId, buildSha = buildSha,
                        unityVersion = Application.unityVersion, storageRoot = root,
                        processId = pid, verdict = failed ? "DRIVER_FAILED" : "OBSERVATION_COMPLETE",
                        region = region != null && CrystalCavernLayout.Is(region.Layout) ? CrystalCavernLayout.Region
                            : (region != null && RuinsLayout.Is(region.Layout) ? RuinsLayout.Region : "ForestRuins"),
                        regionInstanceId = session != null && session.Flow.ActiveRun != null ? session.Flow.ActiveRun.regionInstanceId : "",
                        seed = session != null && session.Flow.ActiveRun != null ? session.Flow.ActiveRun.seed.ToString() : "",
                        generationVersion = session != null && session.Flow.ActiveRun != null ? session.Flow.ActiveRun.generationVersion.ToString() : "",
                        contentRevision = session != null && session.Flow.ActiveRun != null ? session.Flow.ActiveRun.contentRevision : "",
                        productRunId = productRunId,
                        outcome = session != null && session.Flow.Settlement != null ? session.Flow.Settlement.outcome.ToString() : "",
                        committedRunId = session != null && session.Flow.Settlement != null ? session.Flow.Settlement.runId : "",
                        acknowledged = session != null && session.Flow.Settlement != null ? session.Flow.Settlement.acknowledged.ToString() : "",
                        spawnCount = spawnCount, resolvedCount = resolvedCount, fieldCreatedCount = fieldCreatedCount,
                        skillStartCount = skillStartCount, skillReleaseCount = skillReleaseCount,
                        skillCompleteCount = skillCompleteCount, cancelCount = cancelCount,
                        chargeLockedCount = chargeLockedCount, pierceSpawnCount = pierceSpawnCount,
                        pierceContacts = pierceContacts, pierceBlocked = pierceBlocked,
                        pierceRepeats = pierceRepeats, bowPierceReleases = bowPierceReleases,
                        pierceDistinctTargets = pierceTargets.Count, pierceDamage = pierceDamage,
                        maxChargeSeen = maxChargeSeen, chargeAtKeyUp = chargeAtKeyUp, chargeAtLock = chargeAtLock,
                        spawnSeen = spawnSeen, resolvedSeen = resolvedSeen, fieldCreatedSeen = fieldCreatedSeen,
                        reflectionAvailable = reflectionAvailable,
                        spawnTime = spawnTime, resolvedTime = resolvedTime, fieldCreatedTime = fieldCreatedTime,
                        note = "Raw observations only; product PASS/FAIL decided by the frozen contract, never by this driver."
                    };
                    File.WriteAllText(summaryPath, JsonUtility.ToJson(sum, true) + "\n");
                }
                catch (Exception e) { File.WriteAllText(Path.Combine(root, "summary-error.txt"), e.ToString()); failed = true; }
                File.WriteAllText(Path.Combine(root, "complete.txt"), failed ? "FAIL" : "PASS");
                Exit(failed ? 2 : 0);
            }
        }
    }
