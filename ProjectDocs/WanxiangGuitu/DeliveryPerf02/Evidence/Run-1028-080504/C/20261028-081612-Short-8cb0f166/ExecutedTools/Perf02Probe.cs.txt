using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using System.Threading;using UnityEngine;using UnityEngine.Profiling;using Unity.Profiling;using Unity.Profiling.LowLevel.Unsafe;
// Controlled allocations only in the isolated capability experiment, never in A/B/C workload windows.
public static class Perf02Probe{
 [Serializable]class Counter{public string name,category,unit;public bool available;}
 [Serializable]class CounterList{public string unity,scope;public Counter[] counters;}
 class R:IDisposable{public string name,unit,category,error,options;public ProfilerRecorder p;public long Value=>p.Valid&&p.Count>0?p.LastValue:-1;public void Dispose(){p.Dispose();}}
 [Serializable]class Sample{public string phase,edge,utc;public int frame,thread;public long payload,threadAllocated,monoUsed,osWorking,osPrivate;public CounterValue[] counters;}
 [Serializable]class CounterValue{public string name,unit,category,error,options,dataType;public long value,current;public int count;public bool valid,running;}
 static R Start(ProfilerCategory category,string name,ProfilerRecorderOptions options){var r=new R{name=name,category=category.Name,unit="UNAVAILABLE",options=options.ToString()};try{r.p=ProfilerRecorder.StartNew(category,name,128,options);if(r.p.Valid)r.unit=r.p.UnitType.ToString();}catch(Exception e){r.error=e.ToString();}return r;}
 public static IEnumerator Execute(string root){Perf02Dxgi.WriteActualUnityAdapter(Path.Combine(root,"unity-native-adapter.json"));var available=new List<ProfilerRecorderHandle>();ProfilerRecorderHandle.GetAvailable(available);var descriptions=available.Select(h=>{var d=ProfilerRecorderHandle.GetDescription(h);return new Counter{name=d.Name,category=d.Category.Name,unit=d.UnitType.ToString(),available=h.Valid};}).ToArray();File.WriteAllText(Path.Combine(root,"probe-counters.json"),JsonUtility.ToJson(new CounterList{unity=Application.unityVersion,scope="Actual GetAvailable before explicit probe recorder creation; absent is not assumed supported",counters=descriptions},true));
  var rs=new List<R>();var opts=ProfilerRecorderOptions.Default;foreach(var h in available){var d=ProfilerRecorderHandle.GetDescription(h);if(d.Name.IndexOf("GC",StringComparison.OrdinalIgnoreCase)>=0||d.Name.IndexOf("GPU",StringComparison.OrdinalIgnoreCase)>=0)rs.Add(Start(d.Category,d.Name,opts));}
  rs.Add(Start(ProfilerCategory.Memory,"GC Allocated In Frame",opts));rs.Add(Start(ProfilerCategory.Memory,"GC Alloc",opts));rs.Add(Start(ProfilerCategory.Internal,"GC.Alloc",opts));rs.Add(Start(ProfilerCategory.Internal,"GC.Alloc",opts|ProfilerRecorderOptions.CollectOnlyOnCurrentThread));
  object[] keep=new object[128];var samples=new List<Sample>();int k=0;
  Action<string,string,long,long> observe=(phase,edge,payload,allocated)=>{var os=Perf02OS.Read();samples.Add(new Sample{phase=phase,edge=edge,utc=DateTime.UtcNow.ToString("o"),frame=Time.frameCount,thread=Thread.CurrentThread.ManagedThreadId,payload=payload,threadAllocated=allocated,monoUsed=Profiler.GetMonoUsedSizeLong(),osWorking=os.workingSet,osPrivate=os.privateBytes,counters=rs.Select(r=>new CounterValue{name=r.name,category=r.category,unit=r.unit,error=r.error,options=r.options,value=r.Value,current=r.p.Valid?r.p.CurrentValue:-1,dataType=r.p.Valid?r.p.DataType.ToString():"UNAVAILABLE",count=r.p.Valid?r.p.Count:0,valid=r.p.Valid,running=r.p.Valid&&r.p.IsRunning}).ToArray()});};
  foreach(bool alloc in new[]{false,true}){for(int i=0;i<8;i++){string phase=alloc?"known-main-65536":"zero-action-main";long before=GC.GetAllocatedBytesForCurrentThread();if(alloc)keep[k++]=new byte[65536];long delta=GC.GetAllocatedBytesForCurrentThread()-before;observe(phase,"same-frame",alloc?65536:0,delta);yield return null;observe(phase,"next-frame",alloc?65536:0,delta);yield return new WaitForEndOfFrame();observe(phase,"end-of-frame",alloc?65536:0,delta);yield return null;}}
  var worker=new WorkerResult();var t=new Thread(()=>{worker.thread=Thread.CurrentThread.ManagedThreadId;long before=GC.GetAllocatedBytesForCurrentThread();worker.zeroDelta=GC.GetAllocatedBytesForCurrentThread()-before;before=GC.GetAllocatedBytesForCurrentThread();for(int i=0;i<16;i++)keep[64+i]=new byte[65536];worker.knownDelta=GC.GetAllocatedBytesForCurrentThread()-before;worker.payload=16*65536;worker.done=true;});int startFrame=Time.frameCount;t.Start();float deadline=Time.realtimeSinceStartup+10;while(!worker.done&&Time.realtimeSinceStartup<deadline){observe("worker-allocation","main-frame-observation",0,-1);yield return null;}bool joined=t.Join(100);File.WriteAllText(Path.Combine(root,"probe-worker.json"),JsonUtility.ToJson(new WorkerReceipt{thread=worker.thread,payload=worker.payload,zeroDelta=worker.zeroDelta,knownDelta=worker.knownDelta,done=worker.done,joined=joined,startFrame=startFrame,endFrame=Time.frameCount,scope="Background thread uses managed allocations only; no Unity API there. Delta is per-thread API; process/Mono memory never substituted as allocation bytes"},true));
  for(int i=0;i<6;i++){observe("post-worker","next-frame",0,-1);yield return null;}Perf02Dxgi.Write(Path.Combine(root,"dxgi-start.json"));for(int i=0;i<30;i++)yield return null;Perf02Dxgi.Write(Path.Combine(root,"dxgi-end.json"));using(var w=new StreamWriter(Path.Combine(root,"probe-samples.jsonl")))foreach(var sample in samples)w.WriteLine(JsonUtility.ToJson(sample));foreach(var r in rs)r.Dispose();GC.KeepAlive(keep);
 }
 class WorkerResult{public int thread;public long zeroDelta,knownDelta,payload;public volatile bool done;}
 [Serializable]class WorkerReceipt{public int thread,startFrame,endFrame;public long zeroDelta,knownDelta,payload;public bool done,joined;public string scope;}
}
