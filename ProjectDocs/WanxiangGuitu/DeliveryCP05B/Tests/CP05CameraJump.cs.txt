using System;using System.IO;using System.Reflection;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class CP05CameraJump {
 [Serializable]class Result{public string status;public Vector3 oldCamera,newCamera,target;public float targetDistance;public bool clear;}
 public static void Run(){var root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);var target=new GameObject("Owned jump fixture");var c=new GameObject("Owned camera",typeof(Camera),typeof(OrbitCamera)).GetComponent<OrbitCamera>();c.target=target.transform;c.RestoreView(0,28);var tick=typeof(OrbitCamera).GetMethod("LateUpdate",BindingFlags.NonPublic|BindingFlags.Instance);tick.Invoke(c,null);var before=c.transform.position;var wall=GameObject.CreatePrimitive(PrimitiveType.Cube);wall.transform.position=new Vector3(25,4,-5.3f);wall.transform.localScale=new Vector3(1,10,8);target.transform.position=Vector3.right*50;Physics.SyncTransforms();tick.Invoke(c,null);float d=Vector3.Distance(c.transform.position,target.transform.position);var result=new Result{status=d<9&&c.CameraPositionClear?"PASS":"FAIL",oldCamera=before,newCamera=c.transform.position,target=target.transform.position,targetDistance=d,clear=c.CameraPositionClear};File.WriteAllText(root+"/camera-jump.json",JsonUtility.ToJson(result,true));EditorApplication.Exit(result.status=="PASS"?0:1);}
}
