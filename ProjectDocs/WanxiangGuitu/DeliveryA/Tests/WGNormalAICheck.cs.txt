using System;
using System.IO;
using System.Linq;
using System.Reflection;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.AI;
using UnityEngine.InputSystem;
using UnityEngine.InputSystem.LowLevel;
public static class WGNormalAICheck {
 static CombatSession s;static ForestRegionRuntime region;static Keyboard kb;static EditorWindow view;static CombatEnemy target;static string targetId;static int phase;static float since,nextShot;static double deadline,nextLog,phaseReal;static Vector3 shotPosition;static float healthAfterShot;static bool shot;static AdventureResource resource;static int woodBefore;static string Root=>ValidationStorage.ValidateRoot(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")??throw new Exception("explicit root required"));
 static void Need(bool b,string m){if(!b)throw new Exception(m);}static void Mark(string m){Debug.Log("WG_NORMAL_AI_"+m);File.AppendAllText(Root+"/normal-results.txt",m+"\n");}static void Step(int next){phase=next;since=Time.time;phaseReal=EditorApplication.timeSinceStartup;InputSystem.QueueStateEvent(kb,new KeyboardState());}
 public static string ValidateLaunch(string[] args,string expected){string root=ValidationStorage.Root(args,true);if(root!=ValidationStorage.ValidateRoot(expected))throw new InvalidOperationException("测试输出与启动隔离根必须一致");return root;}
 public static void Run(){try{ValidateLaunch(Environment.GetCommandLineArgs(),Root);if(EditorApplication.isPlayingOrWillChangePlaymode)throw new InvalidOperationException("隔离检查必须先于Play");}catch(Exception e){Debug.LogError("A1_ISOLATION_REJECTED "+e.Message);EditorApplication.Exit(1);return;}Directory.CreateDirectory(Root);ValidationStorage.Record("pre-play",Root);deadline=EditorApplication.timeSinceStartup+240;ShaderUtil.allowAsyncCompilation=false;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static bool Walk(Vector3 goal,float arrival=1.1f){if(Vector3.ProjectOnPlane(goal-s.hero.transform.position,Vector3.up).magnitude<arrival){InputSystem.QueueStateEvent(kb,new KeyboardState());return true;}var path=new NavMeshPath();Need(NavMesh.SamplePosition(s.hero.transform.position,out var a,3,NavMesh.AllAreas)&&NavMesh.SamplePosition(goal,out var b,3,NavMesh.AllAreas)&&NavMesh.CalculatePath(a.position,b.position,NavMesh.AllAreas,path)&&path.status==NavMeshPathStatus.PathComplete,"walk path missing "+goal);Vector3 waypoint=path.corners.Skip(1).FirstOrDefault(p=>Vector3.ProjectOnPlane(p-s.hero.transform.position,Vector3.up).magnitude>1);if(waypoint==Vector3.zero)waypoint=goal;var dir=waypoint-s.hero.transform.position;s.orbit.RestoreView(Mathf.Atan2(dir.x,dir.z)*Mathf.Rad2Deg,28);InputSystem.QueueStateEvent(kb,new KeyboardState(Key.W,Key.LeftShift));return false;}
 static Vector3 Away(float radius){var direction=Vector3.ProjectOnPlane(s.hero.transform.position-target.GetComponent<EncounterGuard>().home,Vector3.up).normalized;return target.GetComponent<EncounterGuard>().home+direction*radius;}
 static void Aim(Vector3 p){var dir=Vector3.ProjectOnPlane(p-s.hero.transform.position,Vector3.up);if(dir.sqrMagnitude>.01f)s.hero.transform.rotation=Quaternion.LookRotation(dir);}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();view?.Repaint();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("normal AI timeout phase="+phase);if(!EditorApplication.isPlaying)return;
 if(phase==0){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||!s.Saves.IsInitialized)return;typeof(B06OriginalPlay).GetMethod("Size",BindingFlags.NonPublic|BindingFlags.Static).Invoke(null,null);view=(EditorWindow)typeof(B06OriginalPlay).GetField("view",BindingFlags.NonPublic|BindingFlags.Static).GetValue(null);kb=InputSystem.AddDevice<Keyboard>();InputSystem.settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;InputSystem.settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;
 Need(s.Saves.IsInitialized&&s.Saves.SavePath==Path.Combine(Root,"WanxiangGuitu","ForestRuins","save.json"),"default first initialization did not use launch root");var g=s.hero.GetComponent<AdventureLoadout>();Need(g.SelectMain(g.items.First(i=>i.type==GearType.Bow)),"bow preparation");Need(g.SelectReserve(g.items.First(i=>i.type==GearType.Sword)),"reserve preparation");s.hero.Inventory.Add(MaterialKind.Stone,1);s.hero.Inventory.Add(MaterialKind.Ore,2);
 region=s.gameObject.GetComponent<ForestRegionRuntime>()??s.gameObject.AddComponent<ForestRegionRuntime>();region.LayoutFactory=seed=>ForestLayout.Generate(418);Need(s.Flow.Depart(AdventureMode.ForestExploration),"AI departure");target=s.enemies.Single(e=>e.entityId=="enemy:2");targetId=target.entityId;
 var candidates=Enumerable.Range(0,72).Select(i=>target.transform.position+Quaternion.Euler(0,i*5,0)*Vector3.forward*17).Where(p=>region.TryGround(p,2,out _)&&s.enemies.All(e=>Vector3.Distance(e.transform.position,p)>12.5f)&&AdventureCombat.Sight(p+Vector3.up*1.25f,target.transform.position+Vector3.up*1.25f,target)).ToArray();Need(candidates.Length>0,"no remote-shot approach");shotPosition=candidates.OrderBy(p=>Vector3.Distance(p,s.hero.transform.position)).First();File.WriteAllText(Root+"/metadata.txt","RequestedSeed="+s.Flow.ActiveRun.seed+"\nActualSeed="+region.Layout.seed+"\nGeneration="+s.Flow.ActiveRun.generationVersion+"\nContent="+s.Flow.ActiveRun.contentRevision+"\nPreparation: starter bow/sword; brought stone=1 ore=2; no health/damage/speed overrides; no AI frozen; no teleports after departure\n");Step(1);return;}
 if(EditorApplication.timeSinceStartup>nextLog){nextLog=EditorApplication.timeSinceStartup+.5;File.AppendAllText(Root+"/normal-ai.csv",$"{Time.time:0.00},{phase},{s.hero.transform.position.x:0.0},{s.hero.transform.position.z:0.0},{s.hero.Health.Current:0.0},{target?.GetComponent<EncounterGuard>()?.State},{target?.Health.Current:0.0},{Vector3.Distance(s.hero.transform.position,target.transform.position):0.0}\n");}
 if(s.hero.Health.Dead)throw new Exception("normal AI hero died phase="+phase);if(s.hero.Progression.IsChoosing)s.hero.Progression.SelectChoice(0);
 if(phase==1){if(Walk(shotPosition)){Need(target.GetComponent<EncounterGuard>().State==0,"undetected target already alerted");Mark("BYPASS_AND_REMOTE_SETUP_PASS");Step(2);}return;}
 if(phase==2){if(Time.time-since<.5f)return;Need(Vector3.Distance(target.transform.position,s.hero.transform.position)>11,"shot inside discovery circle");Aim(target.transform.position);if(!shot){shot=s.hero.GetComponent<AdventureCombat>().Strike();return;}if(target.Health.Current<target.Health.maxHealth){Need(!target.Health.Dead,"single starter shot unexpectedly killed target");healthAfterShot=target.Health.Current;Need(target.GetComponent<EncounterGuard>().State==1&&s.enemies.Where(e=>e.GetComponent<EncounterGuard>().poi==target.GetComponent<EncounterGuard>().poi).All(e=>e.GetComponent<EncounterGuard>().State==1),"outpost alarm missing");shotPosition=target.transform.position;Need(s.Saves.SaveProgress(false)&&s.Saves.LoadProgress(false),"chase restore save");target=s.enemies.Single(e=>e.entityId==targetId);Need(target.GetComponent<EncounterGuard>().State==1&&target.Health.Current==healthAfterShot,"chase state/health restore");Mark("REMOTE_HIT_GROUP_ALERT_AND_CHASE_RESTORE_PASS");Step(3);}return;}
 if(phase==3){if(Time.time-since<1.2f)return;Need(Vector3.Distance(target.transform.position,shotPosition)>1.2f&&target.GetComponent<EncounterGuard>().State==1,"alerted enemy stopped outside discovery circle");Mark("PURSUIT_BEYOND_11M_PASS");Step(4);return;}
 if(phase==4){Walk(Away(37),1.8f);if(target.GetComponent<EncounterGuard>().State==2){Need(s.Saves.SaveProgress(false)&&s.Saves.LoadProgress(false),"return save");target=s.enemies.Single(e=>e.entityId==targetId);Need(target.GetComponent<EncounterGuard>().State==2,"return restore");ScreenCapture.CaptureScreenshot(Root+"/01-returning.png");Step(5);}return;}
 if(phase==5){if(Vector3.Distance(s.hero.transform.position,target.GetComponent<EncounterGuard>().home)<34){Walk(Away(37),1.8f);return;}InputSystem.QueueStateEvent(kb,new KeyboardState());if(target.GetComponent<EncounterGuard>().State==0){Need(Mathf.Abs(target.Health.Current-healthAfterShot)<.01f,"return healed damage");Mark("RETURN_RESTORE_HOME_NO_HEAL_PASS");Step(6);}return;}
 if(phase==6){if(target.GetComponent<EncounterGuard>().State==1){Mark("REDETECTION_PASS");Step(7);}else Walk(target.GetComponent<EncounterGuard>().home,8);return;}
 if(phase==7){Aim(target.transform.position);InputSystem.QueueStateEvent(kb,new KeyboardState());if(Time.time>nextShot&&!target.Health.Dead){nextShot=Time.time+.65f;s.hero.GetComponent<AdventureCombat>().Strike();}if(target.Health.Dead){foreach(var loot in UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Where(l=>Vector3.Distance(l.transform.position,s.hero.transform.position)<2))loot.PickUp(s.hero.Inventory);resource=UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).Where(r=>r.kind==MaterialKind.Wood&&r.Health>0).OrderBy(r=>Vector3.Distance(r.transform.position,s.hero.transform.position)).First();Mark("ACTUAL_BOW_KILL_PASS");Step(8);}return;}
 if(phase==8){if(Walk(resource.transform.position,4)){woodBefore=s.hero.Inventory.Count(MaterialKind.Wood);Step(9);}return;}
 if(phase==9){InputSystem.QueueStateEvent(kb,new KeyboardState());Aim(resource.transform.position);if(Time.time>nextShot&&resource.Health>0){nextShot=Time.time+.65f;s.hero.GetComponent<AdventureCombat>().Strike();}if(resource.Health<=0){Walk(resource.transform.position,.8f);foreach(var loot in UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Where(l=>l.Kind==MaterialKind.Wood))loot.PickUp(s.hero.Inventory);if(s.hero.Inventory.Count(MaterialKind.Wood)>woodBefore){Mark("ACTUAL_HARVEST_PICKUP_PASS wood="+s.hero.Inventory.Count(MaterialKind.Wood));Step(10);}}return;}
 if(phase==10){Need(region.TryInteractionApproach(region.Layout.exits.First(e=>e.id=="exit:entry"),out var approach),"exit approach");if(Walk(approach,1)){Need(s.Flow.BeginExtraction(),"normal AI extraction "+s.Flow.ExtractionReason);ScreenCapture.CaptureScreenshot(Root+"/02-extracting.png");Step(11);}return;}
 if(phase==11){InputSystem.QueueStateEvent(kb,new KeyboardState());if(s.Flow.HasResult){Need(s.Flow.Settlement.outcome==AdventureOutcome.Extracted&&s.Flow.Settlement.keptMaterials[(int)MaterialKind.Wood]>0,"nonzero harvest not settled");ScreenCapture.CaptureScreenshot(Root+"/03-earned-settlement.png");Step(12);}else if(s.Flow.ActiveRun.phase==RunPhase.Running){Need(s.Flow.BeginExtraction(),"interrupted exit retry");}return;}
 if(phase==12){if(EditorApplication.timeSinceStartup-phaseReal<.5)return;Need(s.Flow.CommitSettlement()&&s.Flow.AcknowledgeSettlement(),"normal camp settlement");var b=s.hero.Inventory;Need(s.Camp.Materials[(int)MaterialKind.Wood]>0,"harvest not stored");Need(s.Camp.TransferMaterial(MaterialKind.Wood,2,false),"withdraw harvested wood");Need(s.hero.GetComponent<AdventureLoadout>().Craft(GearType.Sword,GearQuality.White),"manufacture from collected wood");Need(s.Flow.Depart(AdventureMode.EndlessWaves)&&s.Saves.SaveProgress(false)&&s.Saves.LoadProgress(false),"normal next expedition recovery");target=s.enemies[0];Mark("NONZERO_SETTLEMENT_WITHDRAW_MANUFACTURE_REDEPART_PASS");ScreenCapture.CaptureScreenshot(Root+"/04-second-expedition.png");Step(13);return;}
 if(phase==13&&Time.time-since>1){InputSystem.RemoveDevice(kb);EditorApplication.update-=Tick;bool rejected=false;try{s.Saves.InitializeStorage(Path.Combine(Path.GetDirectoryName(Root),"Protected","DoNotTouch"),false);}catch(IOException){rejected=true;}Need(rejected,"root boundary accepted outside storage");Mark("A1_ROOT_BOUNDARY_GUARD_PASS");
 File.WriteAllLines(Path.Combine(Root,"expected-normal.txt"),new[]{s.Flow.Profile.profileId,s.Flow.ActiveRun.runId,string.Join(",",s.Camp.Materials),string.Join(",",s.hero.GetComponent<AdventureLoadout>().items.Select(i=>i.id).OrderBy(i=>i))});ValidationStorage.Record("normal-ai-finish",s.Saves.SavePath);Mark("ALL_PASS");EditorApplication.Exit(0);}
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
