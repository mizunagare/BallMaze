using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class RunloopCycle {
 static string root,mode;static CombatSession s;static ThirdPersonHero h;static IEnumerator cases;static int frame=-1;static double deadline;static bool initialized;static List<Row> rows=new List<Row>();
 [Serializable]class Row{public string check,status,detail;}
 [Serializable]class Report{public string mode,scope="Controlled synthetic Editor domain operations, not native/human/normal AI";public Row[] checks;}
 [Serializable]class Expected{public string profile,run,region,layout,streams,gear,warehouse,ledger,ground,settlement;public int seed,collected;public long revision;}
 public static void Run()=>Start("run");public static void Running()=>Start("running");public static void Pending()=>Start("pending");public static void Committed()=>Start("committed");public static void Final()=>Start("final");public static void Death()=>Start("death");
 static void Start(string m){mode=m;root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+180;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Need(bool ok,string n,string detail=""){rows.Add(new Row{check=n,status=ok?"PASS":"FAIL",detail=detail});if(!ok)throw new Exception(n+" "+detail);}
 static void FreezeEnemies(){foreach(var e in s.enemies){e.enabled=false;var b=e.GetComponent<ForestBossBrain>();if(b!=null)b.enabled=false;var g=e.GetComponent<EncounterGuard>();if(g!=null)g.enabled=false;}}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("bounded cycle deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f||frame==Time.frameCount)return;frame=Time.frameCount;if(!initialized){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||!Application.isBatchMode&&!s.Saves.IsInitialized)return;bool fresh=mode=="run"||mode=="death";s.Saves.InitializeStorage(root+"/Store",!fresh);s.Saves.autoSaveInterval=3600;s.Saves.enabled=false;h=s.hero;h.enabled=false;s.orbit.enabled=false;FreezeEnemies();initialized=true;cases=Cases();}if(!cases.MoveNext())Finish();}catch(Exception e){rows.Add(new Row{check="exception",status="FAIL",detail=e.ToString()});Finish();}}
 static void Finish(){EditorApplication.update-=Tick;if(s!=null){s.Saves.WriteValidationHook=null;s.Saves.SuspendEntry();}File.WriteAllText(root+"/cycle-"+mode+".json",JsonUtility.ToJson(new Report{mode=mode,checks=rows.ToArray()},true));EditorApplication.Exit(rows.Any(r=>r.status=="FAIL")?1:0);}
 static string Gear()=>string.Join("|",h.GetComponent<AdventureLoadout>().items.OrderBy(i=>i.id).Select(i=>i.id+":"+i.origin+":"+i.originRunId));
 static string Counts()=>string.Join(",",Enumerable.Range(0,6).Select(k=>h.Inventory.Count((MaterialKind)k)))+"/"+string.Join(",",s.Camp.Materials);
 static Expected Snapshot(int collected){var r=s.Flow.ActiveRun;return new Expected{profile=s.Flow.Profile.profileId,run=r?.runId??"",region=r?.regionInstanceId??"",seed=r?.seed??0,revision=s.Flow.Revision,layout=JsonUtility.ToJson(r?.checkpoint.forest),streams=JsonUtility.ToJson(r?.randomStreams),gear=Gear(),warehouse=string.Join(",",s.Camp.Materials),ledger=r==null?"":string.Join(",",r.broughtRemaining)+"/"+string.Join(",",r.earnedRemaining),ground=string.Join("|",UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).OrderBy(l=>l.InteractionId).Select(l=>l.InteractionId+":"+l.Amount)),settlement=s.Flow.Settlement==null?"":s.Flow.Settlement.settlementId+":"+s.Flow.Settlement.status,collected=collected};}
 static Expected Read()=>JsonUtility.FromJson<Expected>(File.ReadAllText(root+"/expected.json"));
 static void SaveExpected(int collected){Need(s.Saves.SaveProgress(false),"explicit checkpoint save");File.WriteAllText(root+"/expected.json",JsonUtility.ToJson(Snapshot(collected),true));File.AppendAllText(root+"/transitions.jsonl",JsonUtility.ToJson(Snapshot(collected))+"\n");}
 static void Match(Expected e){var now=Snapshot(e.collected);Need(e.profile==now.profile&&e.run==now.run&&e.region==now.region&&e.seed==now.seed,"same Profile Run region seed");Need(e.layout==now.layout&&e.streams==now.streams,"same layout and three random streams");Need(e.gear==now.gear&&e.warehouse==now.warehouse&&e.ledger==now.ledger&&e.ground==now.ground&&e.settlement==now.settlement,"same actual gear warehouse ledger ground receipt");}
 static IEnumerator Cases(){
 int collected=0;
 if(mode=="run"||mode=="death"){
  s.Camp.Restore(s.Camp.Items.ToArray(),Enumerable.Repeat(100,6).ToArray());s.Flow.StartFresh();h.Inventory.Add(MaterialKind.Wood,2);Need(s.Saves.SaveProgress(false),"synthetic camp input saved");h.Inventory.SetOpen(false);h.SaveMenuOpen=false;AdventureRandom.SetNextSeedForValidation(418);Need(s.Flow.Depart(AdventureMode.ForestExploration),"real forest departure");FreezeEnemies();
  var nodes=UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).Where(n=>n.kind==MaterialKind.Wood).OrderBy(n=>n.StableId).Take(2).ToArray();Need(nodes.Length==2,"existing wood nodes");int before=h.Inventory.Count(MaterialKind.Wood);nodes[0].Hit(9999,h);yield return null;var drop=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).First(l=>l.Kind==MaterialKind.Wood);collected=drop.Amount;Need(h.Inventory.Count(MaterialKind.Wood)==before&&s.Flow.ActiveRun.earnedRemaining[(int)MaterialKind.Wood]==0,"ground does not become owned");h.RestorePosition(drop.SavePosition-Vector3.up*.5f,0);Physics.SyncTransforms();Need(drop.PickUp(h.Inventory),"actual pickup");yield return null;Need(h.Inventory.Count(MaterialKind.Wood)==before+collected&&s.Flow.ActiveRun.earnedRemaining[(int)MaterialKind.Wood]==collected,"actual material and source ledger");
  nodes[1].Hit(9999,h);yield return null;Need(h.Inventory.Count(MaterialKind.Wood)==before+collected,"second unpicked ground excluded");
  string rng=JsonUtility.ToJson(UnityEngine.Random.state),streams=JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams),counts=Counts();string advice=CraftingTargets.Recommendations(s);CraftingTargets.Recommendations(s);Need(rng==JsonUtility.ToJson(UnityEngine.Random.state)&&streams==JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams)&&counts==Counts(),"risk route queries read only");File.WriteAllText(root+"/route-advice.txt",advice);
  if(mode=="death"){h.Health.Damage(999999);s.Flow.ResolveTerminal();Need(s.Flow.Settlement.outcome==AdventureOutcome.Death&&s.Flow.Settlement.lostMaterials[(int)MaterialKind.Wood]==collected,"death loses acquired run materials");Need(s.Flow.CommitSettlement()&&s.Camp.Materials[(int)MaterialKind.Wood]==100,"death commit does not deposit ground or run loot");yield break;}
  SaveExpected(collected);yield break;
 }
 var e=Read();collected=e.collected;Match(e);
 if(mode=="running"){
  h.Inventory.SetOpen(false);h.SaveMenuOpen=h.HudPanelOpen=false;var ferry=s.Flow.ExitConfigs().First(x=>x.repairRequired);h.RestorePosition(ferry.position,0);yield return null;var resolver=h.GetComponent<InteractionResolver>()??h.gameObject.AddComponent<InteractionResolver>();resolver.Refresh();resolver.Present();string reason=s.Flow.RepairReason(ferry.id),counts=Counts();Need(reason!=""&&resolver.DisplayedH!=null&&resolver.DisplayedH.reason==reason&&!resolver.DisplayedH.executable,"unmet ferry prompt matches actual reason");Need(!resolver.ExecuteDisplayed(true)&&counts==Counts(),"rejected paid action has zero consumption");
  var exit=s.Flow.ExitConfigs().First(x=>!x.repairRequired);h.RestorePosition(exit.position+Vector3.right*(exit.range+5),0);reason=s.Flow.ExtractionReasonFor(exit);Need(reason!=""&&!s.Flow.BeginExtraction(exit.id),"out of range extraction rejected");h.RestorePosition(exit.position,0);
  s.Saves.WriteValidationHook=p=>{if(p=="before-publish")throw new IOException("RUN01_EXPECTED_EXTRACT_SAVE_FAIL");};Need(!s.Flow.BeginExtraction(exit.id)&&s.Flow.ActiveRun.phase==RunPhase.Running,"extraction save failure no false channel");s.Saves.WriteValidationHook=null;Need(s.Saves.SaveProgress(false),"clear checkpoint failure");Need(s.Flow.BeginExtraction(exit.id),"free exit begins");s.Flow.CancelExtraction("RUN01 controlled cancel");Need(s.Flow.ActiveRun.phase==RunPhase.Running,"cancel returns Running");Need(s.Flow.BeginExtraction(exit.id),"channel again");h.Health.Damage(1);Need(s.Flow.ActiveRun.phase==RunPhase.Running,"actual damage interrupts channel");Need(s.Flow.BeginExtraction(exit.id),"legal extraction after interruption");while(!s.Flow.HasResult)yield return null;
  Need(s.Flow.Settlement.status==SettlementStatus.Pending&&s.Flow.Settlement.outcome==AdventureOutcome.Extracted,"real countdown enters Pending");Need(s.Camp.Materials[(int)MaterialKind.Wood]==100&&s.Flow.Settlement.keptMaterials[(int)MaterialKind.Wood]==collected,"Pending preview is not permanent warehouse");SaveExpected(collected);yield break;
 }
 if(mode=="pending"){
  long rev=s.Flow.Revision;s.Saves.WriteValidationHook=p=>{if(p=="before-publish")throw new IOException("RUN01_EXPECTED_COMMIT_PREFLIGHT_FAIL");};Need(!s.Flow.CommitSettlement()&&!s.Flow.SettlementNeedsRead&&s.Flow.Settlement.status==SettlementStatus.Pending&&s.Flow.Revision==rev,"unpublished commit retains retryable Pending");s.Saves.WriteValidationHook=p=>{if(p=="after-publish")throw new IOException("RUN01_EXPECTED_COMMIT_POST_CALLBACK");};Need(s.Flow.CommitSettlement(),"published commit reconciles disk");s.Saves.WriteValidationHook=null;rev=s.Flow.Revision;Need(s.Flow.CommitSettlement()&&rev==s.Flow.Revision,"duplicate commit no publication");Need(s.Camp.Materials[(int)MaterialKind.Wood]==100+collected&&s.Flow.Profile.committedRuns.Count(x=>x==s.Flow.Settlement.runId)==1,"picked-only warehouse and one receipt");SaveExpected(collected);yield break;
 }
 if(mode=="committed"){
  Need(s.Flow.AcknowledgeSettlement(),"real acknowledgement enters camp");var g=h.GetComponent<AdventureLoadout>();var costs=CraftingCosts.Gear(GearQuality.White);int[] before=Enumerable.Range(0,6).Select(k=>h.Inventory.Count((MaterialKind)k)+s.Camp.Materials[k]).ToArray();string exact=Counts()+"/"+Gear();s.Saves.WriteValidationHook=p=>{if(p=="before-publish")throw new IOException("RUN01_EXPECTED_CRAFT_ROLLBACK");};Need(!g.Craft(GearType.Sword,GearQuality.White)&&exact==Counts()+"/"+Gear(),"camp craft failure rolls back bag plus warehouse");s.Saves.WriteValidationHook=null;int count=g.items.Count;Need(g.Craft(GearType.Sword,GearQuality.White)&&g.items.Count==count+1,"real camp manufacture");Need(Enumerable.Range(0,6).All(k=>before[k]-h.Inventory.Count((MaterialKind)k)-s.Camp.Materials[k]==costs.Where(c=>(int)c.material==k).Sum(c=>c.amount)),"atomic exact combined material costs");var item=g.items.Last(x=>x.Weapon&&x.id!=g.mainWeapon&&x.id!=g.reserveWeapon);Need(s.Camp.DepositEquipment(item.id)&&s.Camp.WithdrawEquipment(item.id),"real warehouse equipment transaction");string previous=s.Flow.Settlement.runId,profile=s.Flow.Profile.profileId;h.Inventory.SetOpen(false);h.SaveMenuOpen=false;AdventureRandom.SetNextSeedForValidation(419);Need(s.Flow.Depart(AdventureMode.ForestExploration)&&s.Flow.ActiveRun.runId!=previous&&s.Flow.Profile.profileId==profile,"successor is new forest Run same Profile");FreezeEnemies();SaveExpected(collected);yield break;
 }
 Need(mode=="final"&&s.Flow.ActiveRun.mode==AdventureMode.ForestExploration,"new process successor remains forest exploration");
 }
}
