using System;
using System.IO;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
public static class WGLegacyForestCheck {
 static double deadline;public static void Run(){deadline=EditorApplication.timeSinceStartup+90;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{if(EditorApplication.timeSinceStartup>deadline)throw new Exception("legacy forest timeout");if(!EditorApplication.isPlaying||Time.time<.5f)return;var s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;string root=Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")??throw new Exception("explicit root required");s.Saves.InitializeStorage(Path.Combine(root,"legacy-forest-v9"),true);if(s.Flow.ActiveRun==null||!s.Flow.ActiveRun.checkpoint.forestWorld||!string.IsNullOrEmpty(s.Flow.Error)||!s.Saves.SaveProgress(false))throw new Exception("legacy forest recovery: "+s.Flow.Error);Debug.Log("WG_LEGACY_FOREST_V9_PASS");EditorApplication.update-=Tick;EditorApplication.Exit(0);}catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
