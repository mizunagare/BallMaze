using System;using System.IO;using System.Linq;using System.Reflection;using UnityEngine;using UnityEditor;using UnityEditor.Build.Profile;using UnityEditor.Build.Reporting;
public static class PrivacyBuild{
 const string Asset="Assets/PrivacyResearch/DiagnosticsDisabled.asset";
 static BindingFlags F=BindingFlags.Instance|BindingFlags.Static|BindingFlags.Public|BindingFlags.NonPublic;
 static string Root=>ValidationStorage.Root(Environment.GetCommandLineArgs(),true);
 static string Hash(string p)=>BitConverter.ToString(System.Security.Cryptography.SHA256.Create().ComputeHash(File.ReadAllBytes(p))).Replace("-","").ToLowerInvariant();
 [Serializable]class State{public string stage,utc,unity,profilePath,profileState,engineDiagnostics,crashCanUpload,connectEnabled,connectSHA;public string[] packages;}
 [Serializable]class Result{public string result,options;public int errors,warnings;public double seconds;public string[] messages;}
 static string Read(string type,string prop){var t=AppDomain.CurrentDomain.GetAssemblies().Select(a=>a.GetType(type)).FirstOrDefault(x=>x!=null);return t?.GetProperty(prop,F)?.GetValue(null)?.ToString()??"UNAVAILABLE";}
 static void Snapshot(string stage,BuildProfile profile){var platform=typeof(BuildProfile).GetProperty("platformBuildProfile",F).GetValue(profile);var container=platform.GetType().GetProperty("insightsSettingsContainer",F).GetValue(platform);var state=container.GetType().GetProperty("buildProfileEngineDiagnosticsState",F).GetValue(container);File.WriteAllText(Root+"/"+stage+".json",JsonUtility.ToJson(new State{stage=stage,utc=DateTime.UtcNow.ToString("o"),unity=Application.unityVersion,profilePath=Asset,profileState=state.ToString(),engineDiagnostics=Read("UnityEditor.EngineDiagnostics.EngineDiagnosticsSettings","enabled"),crashCanUpload=Read("UnityEditor.CrashReporting.CrashReportingSettings","canUploadReports"),connectEnabled=Read("UnityEngine.Connect.UnityConnectSettings","enabled"),connectSHA=Hash("ProjectSettings/UnityConnectSettings.asset"),packages=UnityEditor.PackageManager.PackageInfo.GetAllRegisteredPackages().Where(x=>x.name.Contains("render-pipelines")||x.name=="com.unity.inputsystem"||x.name=="com.unity.ugui").Select(x=>x.name+"="+x.version).ToArray()},true));}
 public static void Run(){try{
 if(Application.unityVersion!="6000.6.5f1")throw new Exception("Pinned Unity only");string original=Hash("ProjectSettings/UnityConnectSettings.asset");
 var create=typeof(BuildProfile).GetMethod("CreateInstance",F,null,new[]{typeof(BuildTarget),typeof(StandaloneBuildSubtarget)},null);var p=(BuildProfile)create.Invoke(null,new object[]{BuildTarget.StandaloneWindows64,StandaloneBuildSubtarget.Player});
 var platform=typeof(BuildProfile).GetProperty("platformBuildProfile",F).GetValue(p);var property=platform.GetType().GetProperty("insightsSettingsContainer",F);object container=property.GetValue(platform);var diagnostic=container.GetType().GetProperty("buildProfileEngineDiagnosticsState",F);diagnostic.SetValue(container,Enum.Parse(diagnostic.PropertyType,"Disabled"));property.SetValue(platform,container);
 typeof(BuildProfile).GetProperty("overrideGlobalScenes",F).SetValue(p,true);typeof(BuildProfile).GetProperty("scenes",F).SetValue(p,new[]{new EditorBuildSettingsScene(GameEntry.BootPath,true),new EditorBuildSettingsScene(GameEntry.TitlePath,true),new EditorBuildSettingsScene(GameEntry.ForestPath,true)});
 Directory.CreateDirectory("Assets/PrivacyResearch");AssetDatabase.CreateAsset(p,Asset);AssetDatabase.SaveAssets();BuildProfile.SetActiveBuildProfile(p);Snapshot("before-build",p);
 if(Read("UnityEditor.EngineDiagnostics.EngineDiagnosticsSettings","enabled")!="False")throw new Exception("Disabled profile not effective; refuse build/run claim");
 var report=BuildPipeline.BuildPlayer(new BuildPlayerWithProfileOptions{buildProfile=p,locationPathName=Root+"/Release/WanxiangGuitu.exe",options=BuildOptions.None});Snapshot("after-build",p);File.Copy(Asset,Root+"/DiagnosticsDisabled.asset.txt");
 File.WriteAllText(Root+"/build.json",JsonUtility.ToJson(new Result{result=report.summary.result.ToString(),options=report.summary.options.ToString(),errors=report.summary.totalErrors,warnings=report.summary.totalWarnings,seconds=report.summary.totalTime.TotalSeconds,messages=report.steps.SelectMany(s=>s.messages.Select(m=>m.type+" "+m.content)).ToArray()},true));
 bool unchanged=Hash("ProjectSettings/UnityConnectSettings.asset")==original;File.WriteAllText(Root+"/connect-unchanged.json","{\"unchanged\":"+unchanged.ToString().ToLowerInvariant()+",\"originalSHA\":\""+original+"\"}");EditorApplication.Exit(report.summary.result==BuildResult.Succeeded&&unchanged?0:1);
 }catch(Exception e){File.WriteAllText(Root+"/exception.txt",e.ToString());EditorApplication.Exit(1);}}
}
