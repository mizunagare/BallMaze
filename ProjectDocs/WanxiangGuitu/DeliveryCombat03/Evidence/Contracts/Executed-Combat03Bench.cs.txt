 static IEnumerator Timing(int w,bool skill){Setup(w,normal,1.3f,0);yield return null;float start=Time.time,hit=-1;Action<HealthObservation> observe=x=>{if(x.kind==HealthObservationKind.Damage&&hit<0)hit=(float)x.activeTime;};normal.Health.Observed+=observe;bool first=skill?combat.UseSkill():combat.Strike();float cooldown=skill?combat.SkillRemaining:combat.AttackRemaining;string motion=h.GetComponent<FantasyActorVisual>().Motion;while(combat.LastAttackPhase!="Idle")yield return null;float recoveryEnd=Time.time-start;bool second=false;while(!second){yield return null;second=skill?combat.UseSkill():combat.Strike();}float cycle=Time.time-start;normal.Health.Observed-=observe;File.AppendAllText(root+"/timing.csv",$"{w},{(skill?"skill":"basic")},{F(cooldown)},{F(combat.LastReleaseTime-start)},{F(recoveryEnd)},{F(cycle)},{F(hit<0?-1:hit-start)},{motion},{first}\n");combat.ResetCombat();yield return null;}
 static int TotalExperience(){int result=h.Progression.Experience;for(int level=1;level<h.Progression.Level;level++)result+=h.Progression.firstLevelExperience+(level-1)*h.Progression.experienceGrowth;return result;}

 int accepted=0,damageEvents=0;float actualDamage=0;Action<HealthObservation> observer=x=>{if(x.kind==HealthObservationKind.Damage){actualDamage+=x.applied;damageEvents++;}};e.Health.Observed+=observer;float start=Time.time,startHero=h.Health.Current;int xp=TotalExperience();int ground=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Length;
 while(!e.Health.Dead&&!h.Health.Dead&&Time.time-start<15){var direction=Vector3.ProjectOnPlane(e.transform.position-h.transform.position,Vector3.up);if(direction.sqrMagnitude>.001f)h.transform.rotation=Quaternion.LookRotation(direction);if(combat.Strike())accepted++;yield return null;}
 float elapsed=Time.time-start;e.Health.Observed-=observer;string terminal=e.Health.Dead?"ENEMY_DEAD":h.Health.Dead?"HERO_DEAD":"TIME_LIMIT";e.enabled=false;if(e.Health.Dead){e.DropLootOnce();int count=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Length,experience=TotalExperience();e.DropLootOnce();if(count!=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Length||experience!=TotalExperience())throw new Exception("Duplicate death reward");}
 File.AppendAllText(root+"/duels.csv",$"{w},{body},{repeat},{terminal},{F(elapsed)},{F(actualDamage)},{F(actualDamage/Mathf.Max(.001f,elapsed))},{F(startHero-h.Health.Current)},{accepted},{damageEvents},{TotalExperience()-xp},{UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Length-ground}\n");combat.ResetCombat();var wait=Wait(.1f);while(wait.MoveNext())yield return null;}
 static IEnumerator Cases(){
 var region=s.GetComponent<ForestRegionRuntime>()??s.gameObject.AddComponent<ForestRegionRuntime>();region.LayoutFactory=_=>ForestLayout.Generate(418);if(!s.Flow.Depart(AdventureMode.ForestExploration))throw new Exception("fixture depart");s.Saves.SuspendEntry();s.Flow.enabled=false;h.enabled=false;h.Progression.enabled=false;s.orbit.enabled=false;foreach(var e in s.enemies)e.enabled=false;
 var floor=GameObject.CreatePrimitive(PrimitiveType.Cube);floor.name="C02 controlled flat floor";floor.transform.position=new Vector3(300,19.5f,300);floor.transform.localScale=new Vector3(80,1,80);
 normal=s.Waves.CreateEncounter(s.transform,"c02:normal",Origin+Vector3.forward*2,EnemyArchetype.Melee);elite=s.Waves.CreateEncounter(s.transform,"c02:elite",Origin+Vector3.forward*3,EnemyArchetype.Elite);boss=s.Waves.CreateEncounter(s.transform,"c02:boss",Origin+Vector3.forward*4,EnemyArchetype.Boss);foreach(var e in new[]{normal,elite,boss}){e.enabled=false;var b=e.GetComponent<BossBrain>();if(b!=null)b.enabled=false;}
 weapons=Enumerable.Range(0,4).Select(w=>{Setup(w,normal,2,0);return new WeaponRow{weapon=((GearType)w).ToString(),gear=gear.Main.definitionId,quality="White",affixes="none",level=1,power=h.AttackPower,interval=h.AttackInterval,reach=h.AttackRange};}).ToArray();File.WriteAllText(root+"/weapons.json",JsonUtility.ToJson(new Summary{weapons=weapons},true));
 File.WriteAllText(root+"/matrix.csv","weapon,action,body,distance,angle,height,duplicate,wall,accepted,hit,applied,cooldown,releaseLatency,hitLatency,hitFrame,damageEvents,numberAtHit,colliderRadius,colliderHeight,colliderCenterY\n");File.WriteAllText(root+"/timing.csv","weapon,action,cooldown,releaseLatency,actionEnd,nextAcceptance,hitLatency,inputMotion,accepted\n");File.WriteAllText(root+"/duels.csv","weapon,body,repeat,terminal,seconds,appliedDamage,killWindowDPS,receivedDamage,acceptedAttacks,damageEvents,experienceGain,groundDropDelta\n");
 if(slope){
 for(int sign=-1;sign<=1;sign+=2){floor.transform.rotation=Quaternion.Euler(-sign*10,0,0);Physics.SyncTransforms();for(int w=0;w<4;w++)for(int skill=0;skill<2;skill++){
 var test=Sample(w,skill==1,normal,"slope"+(sign*10),1.5f,0,false,false,sign*1.5f*Mathf.Tan(10*Mathf.Deg2Rad));while(test.MoveNext())yield return null;}}
 yield break;}
 var targets=new[]{normal,elite,boss};string[] names={"normal","elite","boss"};

 if(boundary){for(int t=0;t<targets.Length;t++)for(int w=0;w<2;w++){int count=w==0?11:16;for(int i=0;i<count;i++){var test=Sample(w,false,targets[t],names[t],(w==0?1.5f:3f)+i*.1f,0);while(test.MoveNext())yield return null;}}yield break;}
 for(int t=0;t<targets.Length;t++)for(int w=0;w<4;w++)for(int skill=0;skill<2;skill++){float[] distances=skill==0?new[]{.75f,3f,6f,12f}:new[]{.75f,3f,6f,12f};float[] angles=skill==0?new[]{-30f,0f,30f}:new[]{-30f,0f,30f};foreach(float d in distances)foreach(float angle in angles){var test=Sample(w,skill==1,targets[t],names[t],d,angle);while(test.MoveNext())yield return null;}}
 for(int w=0;w<4;w++)for(int skill=0;skill<2;skill++){if(w>=2)foreach(float d in new[]{2f,6f,12f})foreach(float angle in new[]{-15f,0f,15f}){var test=Sample(w,skill==1,normal,"normal",d,angle);while(test.MoveNext())yield return null;}var timing=Timing(w,skill==1);while(timing.MoveNext())yield return null;}
 for(int w=0;w<4;w++){foreach(var variant in new[]{"duplicate","wall","high"}){var test=Sample(w,false,normal,"normal",1.5f,0,variant=="duplicate",variant=="wall",variant=="high"?2:0);while(test.MoveNext())yield return null;}}

 static IEnumerator Extra(){
 File.WriteAllText(root+"/contracts.csv","weapon,quality,primaryLoss,secondaryLoss,expectedPrimary,status\n");
 if(visual){for(int w=0;w<4;w++){Setup(w,normal,2,0);var cam=Camera.main;cam.transform.position=Origin+new Vector3(5,4,-5);cam.transform.LookAt(Origin+Vector3.up+Vector3.forward*1.5f);yield return null;combat.Strike();while(combat.LastAttackPhase=="Startup")yield return null;ScreenCapture.CaptureScreenshot(root+"/raw-"+w+".png");Capture("release-"+w);var wait=Wait(.7f);while(wait.MoveNext())yield return null;}yield break;}

 File.AppendAllText(root+"/contracts.csv",$"{w},{q},{F(loss)},{F(other)},{F(expected)},{(ok?"PASS":"FAIL")}\n");samples++;gear.Main.quality=GearQuality.White;combat.ResetCombat();if(!ok)throw new Exception("quality damage regression "+w+"/"+q+" "+loss+"/"+other);yield return null;
 }
