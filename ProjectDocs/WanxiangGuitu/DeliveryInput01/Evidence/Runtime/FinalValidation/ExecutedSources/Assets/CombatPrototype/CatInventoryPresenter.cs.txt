public sealed class CatGridSlot:MonoBehaviour,IPointerClickHandler,IPointerEnterHandler,IPointerExitHandler,IBeginDragHandler,IDragHandler,IEndDragHandler,IDropHandler {
 public CatInventoryPresenter owner;public bool board;public int x,y;public string id;public Vector2Int grab;
 public void OnPointerEnter(PointerEventData e){owner.Hover(id);}
 public void OnPointerExit(PointerEventData e){owner.Hover(null);}
 public void OnPointerClick(PointerEventData e){if(e.button==PointerEventData.InputButton.Left&&!string.IsNullOrEmpty(id))owner.Select(id);}
 public void OnBeginDrag(PointerEventData e){if(e.button==PointerEventData.InputButton.Left&&!string.IsNullOrEmpty(id))owner.StartDrag(id,grab);}
 public void OnDrag(PointerEventData e){}
 public void OnEndDrag(PointerEventData e){if(ContainerInventoryView.Dragging)owner.CancelDrag();}
 public void OnDrop(PointerEventData e){owner.Drop(board,x,y);}
}
