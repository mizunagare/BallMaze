using System;using System.IO;using System.Linq;using System.Collections;using UnityEngine;using UnityEngine.AI;using UnityEditor;using UnityEditor.SceneManagement;
// Real checkpoint capture/restore of an accepted but unreleased attack. Controlled positioning, not AI/native input.
public static class Combat01Save {
 [Serializable]class Identity{public string run,region,node,streams,ledger;public int seed;public float health;}
 static CombatSession s;static IEnumerator task;static EditorWindow view;static string root;static bool resume;static int frame=-1;static double deadline;
 public static void Run(){Begin(false);}public static void Resume(){Begin(true);}
 static void Begin(bool continued){resume=continued;root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);view=UI01Before.Size(1280,720);deadline=EditorApplication.timeSinceStartup+180;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Need(bool ok,string why){if(!ok)throw new Exception(why);File.AppendAllText(root+"/combat01-save-"+(resume?"Resume":"Run")+".txt","PASS "+why+"\n");}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();view?.Repaint();Application.runInBackground=true;if(EditorApplication.timeSinceStartup>deadline)throw new Exception("save test deadline");if(!EditorApplication.isPlaying||frame==Time.frameCount)return;frame=Time.frameCount;if(task==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||!s.Saves.IsInitialized)return;task=Cases();}if(!task.MoveNext()){EditorApplication.update-=Tick;EditorApplication.Exit(0);}}catch(Exception e){File.WriteAllText(root+"/combat01-save-failure-"+(resume?"Resume":"Run")+".txt",e.ToString());Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
 static IEnumerator Cases(){
 if(!resume){(s.GetComponent<ForestRegionRuntime>()??s.gameObject.AddComponent<ForestRegionRuntime>()).LayoutFactory=_=>ForestLayout.Generate(418);Need(s.Flow.Depart(AdventureMode.ForestExploration),"forest departure");}
 s.hero.enabled=false;s.orbit.enabled=false;foreach(var enemy in s.enemies){enemy.enabled=false;var boss=enemy.GetComponent<ForestBossBrain>();if(boss!=null)boss.enabled=false;}
 yield return null;
 if(!resume){var node=UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).First(n=>n.kind==MaterialKind.Wood&&n.Health>0);Need(NavMesh.SamplePosition(node.transform.position+Vector3.back*1.5f,out var hit,3,NavMesh.AllAreas),"real navmesh approach");s.hero.RestorePosition(hit.position,0);s.hero.transform.forward=Vector3.ProjectOnPlane(node.transform.position-s.hero.transform.position,Vector3.up).normalized;var gear=s.hero.GetComponent<AdventureLoadout>();Need(gear.SelectMain(gear.items.First(i=>i.InBag&&i.type==GearType.Sword)),"real equipped starter sword");var combat=s.hero.GetComponent<AdventureCombat>();combat.ResetCombat();Physics.SyncTransforms();yield return null;float before=node.Health;Need(combat.Strike()&&combat.PendingAttackCount==1,"accepted pending attack");Need(node.Health==before,"no input-frame resource damage");Need(s.Saves.SaveProgress(false),"actual atomic checkpoint saves pending acceptance cooldown");var expected=new Identity{run=s.Flow.ActiveRun.runId,region=s.Flow.ActiveRun.regionInstanceId,node=node.StableId,seed=s.Flow.ActiveRun.seed,health=before,streams=JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams),ledger=string.Join(",",s.Flow.ActiveRun.broughtRemaining)+"/"+string.Join(",",s.Flow.ActiveRun.earnedRemaining)};File.WriteAllText(root+"/combat01-save-identity.json",JsonUtility.ToJson(expected,true));Need(s.Saves.LoadProgress(false),"actual LoadProgress");Need(combat.PendingAttackCount==0,"restore cancels transient attack instead of replay");}
 var id=JsonUtility.FromJson<Identity>(File.ReadAllText(root+"/combat01-save-identity.json"));Need(s.Flow.ActiveRun.runId==id.run&&s.Flow.ActiveRun.regionInstanceId==id.region&&s.Flow.ActiveRun.seed==id.seed,"same run region seed");Need(JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams)==id.streams,"same gameplay random streams");Need(string.Join(",",s.Flow.ActiveRun.broughtRemaining)+"/"+string.Join(",",s.Flow.ActiveRun.earnedRemaining)==id.ledger,"same material ledger");float end=Time.time+.8f;while(Time.time<end)yield return null;var restored=UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).Single(n=>n.StableId==id.node);Need(restored.Health==id.health,"saved uncommitted attack never replays resource damage");Need(s.hero.GetComponent<AdventureCombat>().PendingAttackCount==0,"no restored pending action");Need(s.Saves.SaveProgress(false),"checkpoint remains saveable");
 }
}
