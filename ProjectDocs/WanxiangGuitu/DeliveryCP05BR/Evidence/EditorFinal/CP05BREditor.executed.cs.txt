using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEditor;using UnityEditor.SceneManagement;using UnityEditorInternal;using UnityEditor.Profiling;using UnityEngine;using UnityEngine.Profiling;
public static class CP05BREditor {
 [Serializable]class Raw{public string marker,thread;public int frame;public double cpuMs;public long[] allocations;}
 [Serializable]class Report{public string scope="Unity Profiler raw CPU scope/GC.Alloc metadata; query scopes only, not coroutine intervals";public Raw[] rows;public string[] errors;}
 static EditorWindow view;static bool started,oldEnabled,oldEditor,oldEngine;static int tail;static double deadline;static string root;static HashSet<int> seen=new HashSet<int>();static List<Raw> rows=new List<Raw>();static List<string> errors=new List<string>();
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+240;view=UI01Before.Size(1280,720);EditorSceneManager.playModeStartScene=null;EditorSceneManager.OpenScene(GameEntry.TitlePath);EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Capture(){int last=ProfilerDriver.lastFrameIndex;for(int f=Math.Max(ProfilerDriver.firstFrameIndex,last-5);f<last;f++){if(seen.Contains(f))continue;using(var d=ProfilerDriver.GetRawFrameDataView(f,0)){if(!d.valid)continue;seen.Add(f);if(d.sampleCount>200000){errors.Add("Raw frame capacity exceeded: "+f+" samples="+d.sampleCount);continue;}for(int i=0;i<d.sampleCount;i++){string n=d.GetSampleName(i);if(!n.StartsWith("CP05BR."))continue;int end=i+d.GetSampleChildrenCountRecursive(i);var values=new List<long>();for(int j=i+1;j<=end&&j<d.sampleCount;j++)if(d.GetSampleName(j)=="GC.Alloc"){try{values.Add(d.GetSampleMetadataAsLong(j,0));}catch(Exception e){errors.Add(e.ToString());}}rows.Add(new Raw{marker=n,thread=d.threadName,frame=f,cpuMs=d.GetSampleTimeMs(i),allocations=values.ToArray()});}}}}
 static void Restore(){ProfilerDriver.enabled=oldEnabled;ProfilerDriver.profileEditor=oldEditor;Profiler.enabled=oldEngine;}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();view?.Repaint();if(started)Capture();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("CP05BR Editor bounded deadline");if(!EditorApplication.isPlaying)return;if(!started){if(UnityEngine.SceneManagement.SceneManager.GetActiveScene().path!=GameEntry.TitlePath)return;oldEnabled=ProfilerDriver.enabled;oldEditor=ProfilerDriver.profileEditor;oldEngine=Profiler.enabled;ProfilerDriver.profileEditor=false;ProfilerDriver.enabled=true;ProfilerDriver.SetAreaEnabled(ProfilerArea.CPU,true);Profiler.enableAllocationCallstacks=false;Profiler.enabled=true;CP05Training.StartIsolated();var g=new GameObject("Editor isolated performance probe");UnityEngine.Object.DontDestroyOnLoad(g);g.AddComponent<CP05BRProbe>();started=true;}if(CP05BRProbe.Done&&++tail>=5){Capture();File.WriteAllText(root+"/unity-profiler.json",JsonUtility.ToJson(new Report{rows=rows.ToArray(),errors=errors.ToArray()},true));EditorApplication.update-=Tick;Restore();EditorApplication.Exit(CP05BRProbe.ExitCode);}}catch(Exception e){File.WriteAllText(root+"/exception.txt",e.ToString());EditorApplication.update-=Tick;if(started)Restore();EditorApplication.Exit(1);}}
}
