using System;using System.IO;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class B09Cases {
 static string mode;static CombatSession s;static double deadline;static string Root=>ValidationStorage.ValidateRoot(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT"));
 static void Need(bool b,string x){if(!b)throw new Exception(x);}public static void Run(){Start("create");}public static void Pending(){Start("Pending");}public static void Committed(){Start("Committed");}
 static void Start(string m){mode=m;if(ValidationStorage.Root(Environment.GetCommandLineArgs(),true)!=Root)throw new Exception("B09 pre-play root mismatch");deadline=EditorApplication.timeSinceStartup+150;EditorSceneManager.OpenScene(GameEntry.ForestPath);EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Copy(string from,string to){Directory.CreateDirectory(to);foreach(var f in Directory.GetFiles(from,"*",SearchOption.AllDirectories)){ValidationStorage.RejectLinks(f);string p=Path.Combine(to,f.Substring(from.Length+1));Directory.CreateDirectory(Path.GetDirectoryName(p));File.Copy(f,p,false);}}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("B09 catalog cases deadline");if(!EditorApplication.isPlaying||s!=null)return;s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;
  if(mode=="create"){string path=Root+"/b09-menu-source";s.Saves.InitializeStorage(path,false);Need(s.Flow.Depart(AdventureMode.ForestExploration),"actual synthetic forest");s.Flow.OpenAbandon();Need(s.Flow.ConfirmAbandon()&&s.Flow.HasResult&&s.Saves.SaveProgress(false),"pending fixture through real terminal");Need(SaveCatalog.Inspect(path).phase=="结算待确认","pending catalog");Copy(path,Root+"/b09-menu-Pending");Need(s.Flow.CommitSettlement()&&s.Saves.SaveProgress(false),"commit real pending");Need(SaveCatalog.Inspect(path).phase=="结算已保存，待回营","committed catalog");Copy(path,Root+"/b09-menu-Committed");File.WriteAllText(Root+"/b09-case-ids.txt",s.Flow.Profile.profileId+"/"+s.Flow.Settlement.runId+"/"+s.Flow.Settlement.settlementId);}
  else{string path=Root+"/b09-menu-"+mode;var before=CombatSaveSystem.InspectReadOnly(path+"/save.json");Need(s.Saves.InitializeFromEntry(path,Root+"/b09-unused-legacy",false,SaveCatalog.Fingerprint(path)),"entry continue terminal");Need(s.Flow.Profile.profileId+"/"+s.Flow.Settlement.runId+"/"+s.Flow.Settlement.settlementId==File.ReadAllText(Root+"/b09-case-ids.txt"),"terminal identities changed");Need(JsonUtility.ToJson(before.profile)==JsonUtility.ToJson(s.Flow.Profile)&&JsonUtility.ToJson(before.settlement)==JsonUtility.ToJson(s.Flow.Settlement),"terminal entry repaid or modified profile");Need(s.Flow.Settlement.status==(mode=="Pending"?SettlementStatus.Pending:SettlementStatus.Committed),"wrong terminal phase");}
  File.AppendAllText(Root+"/b09-cases.txt","PASS "+mode+" actual terminal entry, IDs/profile/settlement preserved\n");EditorApplication.update-=Tick;EditorApplication.Exit(0);
 }catch(Exception e){Debug.LogException(e);EditorApplication.Exit(1);}}
}
