using System.Collections.Generic;using System.Linq;using UnityEngine;
// Renderer-only landscape extension. No collision, navigation source, loot or RNG calls.
public sealed class ScenePresentation:MonoBehaviour {
 CombatSession session;GameObject scenery;bool? wasCamp;GameObject forestRoot;readonly Dictionary<Renderer,bool> hidden=new Dictionary<Renderer,bool>();readonly List<Material> materials=new List<Material>();readonly List<Mesh> meshes=new List<Mesh>();
 public GameObject DecorationRoot=>scenery;
 void Awake(){session=GetComponent<CombatSession>();}
 static GameObject Shape(Transform parent,string name,PrimitiveType kind,Vector3 p,Vector3 scale,Material mat){var g=GameObject.CreatePrimitive(kind);g.name=name;g.transform.SetParent(parent,false);g.transform.localPosition=p;g.transform.localScale=scale;g.layer=2;var c=g.GetComponent<Collider>();c.enabled=false;Object.Destroy(c);g.GetComponent<Renderer>().sharedMaterial=mat;return g;}
 Material Material(Color c){var m=new Material(Shader.Find("Universal Render Pipeline/Lit")){color=c};m.SetFloat("_Smoothness",.05f);materials.Add(m);return m;}
 public static void TitleBackdrop(Transform parent){var root=new GameObject("入口 · 森林远景");root.transform.SetParent(parent,false);var own=root.AddComponent<ScenePresentationArt>();Material earth=own.Color(new Color(.13f,.22f,.16f)),leaf=own.Color(new Color(.22f,.38f,.25f)),wood=own.Color(new Color(.19f,.12f,.09f));Shape(root.transform,"林间地面",PrimitiveType.Cube,new Vector3(0,-.3f,8),new Vector3(100,.4f,100),earth);for(int i=0;i<42;i++){float x=(i%7-3)*5+Mathf.Sin(i*2.31f)*2,z=6+i/7*7;Shape(root.transform,"林木",PrimitiveType.Cylinder,new Vector3(x,2,z),new Vector3(.5f,2,.5f),wood);Shape(root.transform,"树冠",PrimitiveType.Sphere,new Vector3(x,5,z),new Vector3(3,5,3),leaf);}var camera=new GameObject("标题镜头",typeof(Camera),typeof(AudioListener));camera.transform.SetParent(root.transform);camera.transform.position=new Vector3(14,7,-10);camera.transform.LookAt(new Vector3(0,2,18));var cam=camera.GetComponent<Camera>();cam.backgroundColor=new Color(.32f,.48f,.42f);cam.clearFlags=CameraClearFlags.SolidColor;cam.farClipPlane=180;var light=new GameObject("林间日光",typeof(Light));light.transform.SetParent(root.transform);light.transform.rotation=Quaternion.Euler(38,-35,0);light.GetComponent<Light>().type=LightType.Directional;light.GetComponent<Light>().intensity=1.15f;RenderSettings.ambientLight=new Color(.4f,.48f,.39f);RenderSettings.fog=true;RenderSettings.fogMode=FogMode.ExponentialSquared;RenderSettings.fogColor=cam.backgroundColor;RenderSettings.fogDensity=.012f;}
 void LateUpdate(){if(session==null||!session.Saves.IsInitialized)return;bool camp=session.Flow.AtCamp;var region=session.GetComponent<ForestRegionRuntime>();var root=region?.PresentationRoot;if(wasCamp==camp&&forestRoot==root)return;Restore();wasCamp=camp;forestRoot=root;Build(camp,region);}
 void Restore(){foreach(var item in hidden)if(item.Key!=null)item.Key.enabled=item.Value;hidden.Clear();if(scenery!=null){scenery.SetActive(false);Destroy(scenery);}foreach(var m in materials)if(m!=null)Destroy(m);materials.Clear();foreach(var m in meshes)if(m!=null)Destroy(m);meshes.Clear();}
 void Build(bool camp,ForestRegionRuntime region){if(!camp&&(region==null||!region.Active||!region.Ready))return;scenery=new GameObject(camp?"营地连续远景（仅表现）":"森林边界延景（仅表现）");scenery.transform.SetParent(transform);var mat=Material(new Color(.24f,.33f,.23f));var leaf=Material(new Color(.23f,.38f,.26f));var wood=Material(new Color(.24f,.16f,.1f));
  var campRoot=session.Camp.PresentationRoot;foreach(var r in FindObjectsByType<Renderer>(FindObjectsSortMode.None)){if(r.gameObject.scene!=gameObject.scene||r.transform.IsChildOf(session.hero.transform)||r.transform.IsChildOf(scenery.transform))continue;bool keep=camp?campRoot!=null&&r.transform.IsChildOf(campRoot.transform):region.PresentationRoot!=null&&r.transform.IsChildOf(region.PresentationRoot.transform)||r.GetComponentInParent<CombatEnemy>()!=null||r.GetComponentInParent<LootPickup>()!=null||r.GetComponentInParent<AdventureResource>()!=null||r.GetComponentInParent<WorldEventPoint>()!=null;if(!keep){hidden[r]=r.enabled;r.enabled=false;}}
  Vector3 center=camp?new Vector3(80,0,-80):region.Layout.boundsCenter;center.y=0;var mesh=new Mesh{name="连续地表表现"};var vertices=new List<Vector3>();var tris=new List<int>();const int steps=64;
  if(camp){vertices.Add(center+Vector3.down*.015f);for(int i=0;i<=steps;i++){float a=i*Mathf.PI*2/steps;vertices.Add(center+new Vector3(Mathf.Cos(a)*130,-.015f,Mathf.Sin(a)*130));if(i>0){tris.Add(0);tris.Add(i+1);tris.Add(i);}}}
  else {var l=region.Layout;for(int i=0;i<=steps;i++){float a=i*Mathf.PI*2/steps;var d=new Vector3(Mathf.Cos(a),0,Mathf.Sin(a));float t=Mathf.Min(l.boundsSize.x*.5f/Mathf.Max(.001f,Mathf.Abs(d.x)),l.boundsSize.z*.5f/Mathf.Max(.001f,Mathf.Abs(d.z)));var inner=center+d*t;inner.y=l.revision>=5?l.terrain.Height(l,inner):0;vertices.Add(inner-Vector3.up*.04f);var outer=center+d*(t+110);outer.y=-3+Mathf.Sin(a*5)*2;vertices.Add(outer);if(i>0){int b=i*2;tris.AddRange(new[]{b-2,b,b-1,b,b+1,b-1});}}}
  mesh.SetVertices(vertices);mesh.SetTriangles(tris,0);mesh.RecalculateNormals();meshes.Add(mesh);var ground=new GameObject("无碰撞地表延伸",typeof(MeshFilter),typeof(MeshRenderer));ground.transform.SetParent(scenery.transform,false);ground.GetComponent<MeshFilter>().sharedMesh=mesh;ground.GetComponent<MeshRenderer>().sharedMaterial=mat;
  for(int i=0;i<48;i++){float a=i*2.39996f;float radius=camp?20+i%7*8:130+i%7*9;var p=center+new Vector3(Mathf.Cos(a)*radius,0,Mathf.Sin(a)*radius);Shape(scenery.transform,"远景树干",PrimitiveType.Cylinder,p+Vector3.up*2,new Vector3(.7f,2,.7f),wood);Shape(scenery.transform,"远景树冠",PrimitiveType.Sphere,p+Vector3.up*5,new Vector3(3.8f,5.8f,3.8f),leaf);}
  if(!camp){foreach(var r in region.PresentationRoot.GetComponentsInChildren<Renderer>()){if(!r.name.EndsWith("边界"))continue;hidden[r]=r.enabled;r.enabled=false;}for(int i=0;i<72;i++){float a=i*Mathf.PI*2/72;var d=new Vector3(Mathf.Cos(a),0,Mathf.Sin(a));float t=Mathf.Min(region.Layout.boundsSize.x*.5f/Mathf.Max(.001f,Mathf.Abs(d.x)),region.Layout.boundsSize.z*.5f/Mathf.Max(.001f,Mathf.Abs(d.z)))+1.5f;var p=center+d*t;p.y=region.Layout.revision>=5?region.Layout.terrain.Height(region.Layout,p):0;Shape(scenery.transform,"边林轮廓（无碰撞）",PrimitiveType.Sphere,p+Vector3.up*2,new Vector3(8,6+i%3,6),leaf);}}
  RenderSettings.fog=true;RenderSettings.fogMode=FogMode.ExponentialSquared;RenderSettings.fogColor=new Color(.42f,.56f,.51f);RenderSettings.fogDensity=.0048f;session.orbit.GetComponent<Camera>().backgroundColor=RenderSettings.fogColor;
 }
 void OnDestroy(){Restore();}
}
