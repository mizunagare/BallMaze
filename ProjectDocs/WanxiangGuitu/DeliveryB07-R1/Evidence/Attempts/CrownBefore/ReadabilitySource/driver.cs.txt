using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
// Deliberate camera/actor placement, AI disabled. Never a normal victory/native input claim.
public static class B07ActorOcclusion {
 static CombatSession s;static CameraActorFade fade;static CombatEnemy enemy;static Renderer[] renderers;static Material[][] originals;static EditorWindow view;static int phase,frame;static double deadline;static string file,random,streams,profile;static float hp,heroHp;static CharacterController controller;
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT");
 static void Need(bool b,string message){if(!b)throw new Exception(message);}
 public static void Run(){ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+240;view=UI01Before.Size(1280,720);EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();Application.runInBackground=true;view?.Repaint();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("occlusion deadline "+phase);if(!EditorApplication.isPlaying||Time.unscaledTime<.5f)return;
  if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||!s.Saves.IsInitialized){s=null;return;}s.Saves.InitializeStorage(Root+"/occlusion",false);AdventureRandom.SetNextSeedForValidation(1132820888);Need(s.Flow.Depart(AdventureMode.ForestExploration),"depart");s.hero.enabled=false;foreach(var e in s.enemies){e.enabled=false;var b=e.GetComponent<ForestBossBrain>();if(b!=null)b.enabled=false;}
   enemy=s.enemies.Single(e=>e.GetComponent<ForestBossBrain>()?.Kind==ForestBossKind.Warden);controller=enemy.GetComponent<CharacterController>();enemy.GetComponent<EncounterGuard>().Restore(1);s.hero.RestorePosition(enemy.transform.position+Vector3.forward*4,180);s.orbit.RestoreView(0,28);fade=s.orbit.GetComponent<CameraActorFade>();Need(fade!=null,"missing camera fade");fade.enabled=false;renderers=enemy.GetComponentsInChildren<Renderer>().Where(r=>r.GetComponentInParent<ForestBossTelegraph>()==null).ToArray();originals=renderers.Select(r=>r.sharedMaterials).ToArray();hp=enemy.Health.Current;heroHp=s.hero.Health.Current;frame=Time.frameCount;return;}
  if(phase==0){if(Time.frameCount<=frame+30||ShaderUtil.anythingCompiling)return;enemy.GetComponent<ForestBossBrain>().ResetBrain();Need(enemy.GetComponent<ForestBossBrain>().Begin(ForestBossAttack.RootSlam),"off warning");file=Root+"/b07r1-occlusion-controlled-off.png";ScreenCapture.CaptureScreenshot(file);frame=Time.frameCount;phase++;return;}
  if(phase==1){if(Time.frameCount<=frame+2||!File.Exists(file))return;random=JsonUtility.ToJson(UnityEngine.Random.state);streams=JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams);profile=JsonUtility.ToJson(s.Flow.Profile);float scale=Time.timeScale;fade.enabled=true;fade.Refresh();Need(fade.OccludedActors>0&&fade.FadedRenderers>0,"visual occluder not detected");Need(renderers.All(r=>r.sharedMaterials.All(m=>m.shader.name=="Combat/ActorFade")),"Boss body or crown still opaque");Need(controller.enabled&&controller.radius==.6f&&controller.height==2.5f&&enemy.Health.Current==hp&&s.hero.Health.Current==heroHp,"presentation changed combat");Need(JsonUtility.ToJson(UnityEngine.Random.state)==random&&JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams)==streams&&JsonUtility.ToJson(s.Flow.Profile)==profile&&Time.timeScale==scale,"fade changed RNG/profile/time scale");frame=Time.frameCount;phase++;return;}
  if(phase==2){if(Time.frameCount<=frame+30||ShaderUtil.anythingCompiling)return;enemy.GetComponent<ForestBossBrain>().ResetBrain();Need(enemy.GetComponent<ForestBossBrain>().Begin(ForestBossAttack.RootSlam),"on warning");Need(enemy.GetComponent<ForestBossBrain>().Warning.GetComponentsInChildren<Renderer>().All(r=>r.sharedMaterial.shader.name=="Combat/BossWarning"),"warning was faded with model");file=Root+"/b07r1-occlusion-controlled-on.png";ScreenCapture.CaptureScreenshot(file);frame=Time.frameCount;phase++;return;}
  if(phase==3){if(Time.frameCount<=frame+2||!File.Exists(file))return;s.orbit.RestoreView(180,28);frame=Time.frameCount;phase++;return;}
  if(phase==4){if(Time.frameCount<=frame+4)return;for(int i=0;i<renderers.Length;i++)Need(renderers[i].sharedMaterials.SequenceEqual(originals[i]),"material not restored after camera orbit");fade.enabled=false;Need(fade.FadedRenderers==0&&fade.OwnedMaterialCount==0,"fade material cache not cleared");Need(enemy.Health.Current==hp&&s.hero.Health.Current==heroHp&&controller.enabled,"restore changed combat");File.WriteAllText(Root+"/b07r1-occlusion.txt","PASS actual renderer obstruction detected, transparent materials, camera-away restoration, disable cleanup; collision/HP/RNG/profile/time scale preserved; CONTROLLED not normal/native\n");EditorApplication.update-=Tick;EditorApplication.Exit(0);return;}
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
