using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;
// Read-only real scene/navigation audit. Not physical player movement or human play.
public static class WorldAudit {
 [Serializable]public class Row{public string id,kind,status;public Vector3 position;public float meters;public Vector3[] path;}
 [Serializable]public class Report{public string profile,run,region,content,layoutJson,streams;public int requestedSeed,actualSeed,revision,generationVersion,attempt;public bool fallback,unique,readOnly;public string[] failures;public Row[] rows;public string advice;}
 static float Length(Vector3[] points){float n=0;for(int i=1;i<points.Length;i++)n+=Vector3.Distance(points[i-1],points[i]);return n;}
 public static void Capture(CombatSession s,string root){var region=s.GetComponent<ForestRegionRuntime>();var map=region.Capture();var rows=new List<Row>();var ids=new List<string>();
 Action<string,string,Vector3> add=(id,kind,to)=>{ids.Add(id);var route=region.NavigationPath(map.entry,to,kind+":"+id);if(route.reachable&&Mathf.Abs(route.distance-Length(route.corners))>.01f)throw new Exception("length differs "+id);rows.Add(new Row{id=id,kind=kind,position=to,status=route.reachable?"PASS":"FAIL",meters=route.reachable?route.distance:-1,path=route.corners});};
 foreach(var p in map.points)add(p.id,"poi",p.position);
 foreach(var n in map.nodes){if(!region.TryGround(n.position,4,out var approach))throw new Exception("resource approach "+n.id);add(n.id,"resource",approach);}
 foreach(var e in map.encounters)add(e.id,e.boss==ForestBossKind.None?"ordinary":"boss",e.home);
 foreach(var e in map.exits){if(!region.TryInteractionApproach(e,out var approach))throw new Exception("exit interaction "+e.id);add(e.id,e.repairRequired?"ferry":"free-exit",approach);}
 foreach(var e in s.Events.Points){bool found=false;Vector3 approach=default;for(int i=0;i<16&&!found;i++){float a=i*Mathf.PI/8;var p=e.transform.position+new Vector3(Mathf.Cos(a),0,Mathf.Sin(a))*1.8f;found=region.TryGround(p,1,out approach)&&region.PathComplete(map.entry,approach);}if(!found)throw new Exception("event approach "+e.entityId);add(e.entityId,e.Kind.ToString(),approach);}
 foreach(var edge in map.routes){var from=map.points.Single(p=>p.id==edge.from).position;var route=region.NavigationRoute(from,edge.to);rows.Add(new Row{id=edge.from+"->"+edge.to,kind="edge",status=route.reachable?"PASS":"FAIL",meters=route.reachable?route.distance:-1,path=route.corners});}
 var safe=region.NavigationRoute(map.entry,"poi:5",true);rows.Add(new Row{id="poi:5",kind="elite-bypass",status=safe.reachable&&!safe.nearElite?"PASS":"FAIL",meters=safe.reachable?safe.distance:-1,path=safe.corners});
 string before=JsonUtility.ToJson(s.Flow.ActiveRun)+JsonUtility.ToJson(s.Flow.Profile)+JsonUtility.ToJson(UnityEngine.Random.state);string advice=CraftingTargets.Details(s);var first=JsonUtility.ToJson(new Scores{values=RouteScoring.Evaluate(s)});for(int i=0;i<5;i++){CraftingTargets.Details(s);if(first!=JsonUtility.ToJson(new Scores{values=RouteScoring.Evaluate(s)}))throw new Exception("same state score differs");}bool readOnly=before==JsonUtility.ToJson(s.Flow.ActiveRun)+JsonUtility.ToJson(s.Flow.Profile)+JsonUtility.ToJson(UnityEngine.Random.state);
 var report=new Report{profile=s.Flow.Profile.profileId,run=s.Flow.ActiveRun.runId,region=s.Flow.ActiveRun.regionInstanceId,content=s.Flow.ActiveRun.contentRevision,requestedSeed=map.requestedSeed,actualSeed=map.seed,revision=map.revision,generationVersion=s.Flow.ActiveRun.generationVersion,attempt=map.generationAttempt,fallback=map.fallback,failures=map.generationFailures,unique=ids.Distinct().Count()==ids.Count,readOnly=readOnly,rows=rows.ToArray(),layoutJson=JsonUtility.ToJson(map),streams=JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams),advice=advice};File.WriteAllText(root+"/content-nav.json",JsonUtility.ToJson(report,true));File.WriteAllText(root+"/route-evaluations.json",first);if(!report.unique||!readOnly||rows.Any(x=>x.status!="PASS"))throw new Exception("world nav/identity/query failure; raw matrix preserved");}
 [Serializable]class Scores{public RouteEvaluation[] values;}
 public static bool Restored(CombatSession s){var cp=s.Flow.ActiveRun?.checkpoint;if(cp==null)return s.Flow.AtCamp;var nodes=UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).Where(n=>n.gameObject.scene==s.gameObject.scene).ToArray();return cp.enemies.All(e=>s.enemies.Any(a=>a.entityId==e.entityId&&Mathf.Abs(a.Health.Current-e.health)<.001f))&&cp.resources.All(e=>nodes.Any(n=>n.StableId==e.entityId&&Mathf.Abs(n.Health-e.health)<.001f))&&cp.events.All(e=>s.Events.Points.Any(p=>p.entityId==e.entityId&&p.Used==e.used));}
}
