using System;using System.IO;using System.Collections;using System.Collections.Generic;using System.Linq;using System.Reflection;using UnityEngine;
// Isolated controlled diagnostics only. Uses Strike/UseSkill and observes production projectiles/Damage.
public static class CP04Cases {
 [Serializable]public class Hit {public string name,owner,type;public int layer;public bool legal,trigger;public Vector3 point,center,size;public float distance;}
 [Serializable]public class Trace {public string stage,caseId,collider,owner;public long attack;public int shot,frame;public float time;public Vector3 from,to;}
 [Serializable]public class Shot {public string caseId,reason,muzzleSource,muzzlePath,clip;public long attack;public int shot,frame;public bool valid,barrelBlocked,hasContact,obstructed,orthographic;public float nearClip,farClip,fov,aspect,speed,life,poseTime,surfaceProjection;public Vector3 cameraPosition,cameraForward,rayOrigin,rayDirection,muzzle,origin,direction,intent,contact,target,hero;public Matrix4x4 projection;public Hit[] rayHits;public Hit predictedContact;}
 [Serializable]public class Damage {public string caseId;public int frame;public float before,after,applied;}
 [Serializable]public class Case {public string id,camera,occlusion;public float distance,before,after,hpLoss;public bool skill,accepted;public int spawns,remainingProjectiles;public Hit[] colliders;}
 [Serializable]public class Report {public string scope="Controlled Editor real Strike/UseSkill, native aim/projectiles/collision/Damage; not native input, normal AI or human";public Case[] cases;public Shot[] shots;public Trace[] trace;public Damage[] damage;}
 static CombatSession s;static ThirdPersonHero h;static AdventureCombat combat;static AdventureLoadout gear;static FantasyActorVisual actor;static string root,current;static IEnumerator run;static Case active;static CombatEnemy target;static List<Case> cases=new List<Case>();static List<Shot> shots=new List<Shot>();static List<Trace> trace=new List<Trace>();static List<Damage> damage=new List<Damage>();
 public static bool Done;public static int ExitCode;
 static object F(object x,string n)=>x.GetType().GetField(n,BindingFlags.Instance|BindingFlags.NonPublic|BindingFlags.Public).GetValue(x);
 static Hit Info(Collider c,Vector3 point,float distance=0)=>c==null?null:new Hit{name=c.name,owner=c.GetComponentInParent<CombatEnemy>()?.entityId,type=c.GetType().Name,layer=c.gameObject.layer,legal=CombatAim.Legal(c,h),trigger=c.isTrigger,point=point,center=c.bounds.center,size=c.bounds.size,distance=distance};
 public static void Begin(CombatSession session,string output){s=session;h=s.hero;root=output;Application.runInBackground=true;Time.captureFramerate=120;AdventureProjectile.Observed+=Observed;run=Run();}
 public static void Tick(){if(Done)return;try{if(run.MoveNext())return;Finish();}catch(Exception e){File.WriteAllText(root+"/exception.txt",e.ToString());ExitCode=1;Finish();}}
 static void Finish(){AdventureProjectile.Observed-=Observed;Time.captureFramerate=0;File.WriteAllText(root+"/diagnosis.json",JsonUtility.ToJson(new Report{cases=cases.ToArray(),shots=shots.ToArray(),trace=trace.ToArray(),damage=damage.ToArray()},true));s.Saves.SuspendEntry();Done=true;}
 static void Observed(AdventureProjectile.Observation x){var p=x.projectile;var a=p.Aim;trace.Add(new Trace{stage=x.stage,caseId=current,collider=x.collider!=null?x.collider.name:"",owner=x.collider!=null?x.collider.GetComponentInParent<CombatEnemy>()?.entityId:"",attack=p.AttackId,shot=p.ShotIndex,frame=x.frame,time=Time.time,from=x.from,to=x.to});if(x.stage!="spawn")return;active.spawns++;var hits=Physics.RaycastAll(a.cameraOrigin,a.cameraDirection,CombatAim.RayLimit,~0,QueryTriggerInteraction.Ignore).OrderBy(v=>v.distance).ToArray();var first=hits.FirstOrDefault(v=>CombatAim.Legal(v.collider,h));var muzzle=h.GetComponentsInChildren<Transform>(true).FirstOrDefault(t=>t.name=="ProjectileMuzzle");string path="";for(var t=muzzle;t!=null;t=t.parent)path=t.name+"/"+path;
 shots.Add(new Shot{caseId=current,attack=p.AttackId,shot=p.ShotIndex,frame=x.frame,reason=a.reason,muzzleSource=a.muzzleSource,muzzlePath=path,clip=(string)F(actor,"actionClip"),poseTime=actor.PoseTime,valid=a.valid,barrelBlocked=a.barrelBlocked,hasContact=a.hasContact,obstructed=a.obstructed,orthographic=a.camera.orthographic,nearClip=a.camera.nearClipPlane,farClip=a.camera.farClipPlane,fov=a.camera.fieldOfView,aspect=a.camera.aspect,projection=a.camera.projectionMatrix,speed=(float)F(p,"speed"),life=(float)F(p,"life"),cameraPosition=a.camera.transform.position,cameraForward=a.camera.transform.forward,rayOrigin=a.cameraOrigin,rayDirection=a.cameraDirection,muzzle=a.rawMuzzle,origin=p.SpawnOrigin,direction=p.Direction,intent=a.intentPoint,contact=a.contactPoint,target=target.transform.position,hero=h.transform.position,surfaceProjection=first.collider!=null?Vector3.Dot(first.point-a.origin,a.cameraDirection):0,rayHits=hits.Select(v=>Info(v.collider,v.point,v.distance)).ToArray(),predictedContact=Info(a.contactCollider,a.contactPoint)});
 }
 static IEnumerator Run(){s.Saves.InitializeStorage(root+"/Game/ForestRuins",false);s.Saves.enabled=false;AdventureRandom.SetNextSeedForValidation(418);if(!s.Flow.Depart(AdventureMode.ForestExploration))throw new Exception("synthetic forest departure rejected");s.Flow.BeginLegacyRestore();h.Inventory.SetOpen(false);h.HudPanelOpen=h.SaveMenuOpen=false;Time.timeScale=1;h.enabled=false;s.orbit.enabled=false;foreach(var e in s.enemies){e.enabled=false;var b=e.GetComponent<ForestBossBrain>();if(b)b.enabled=false;}combat=h.GetComponent<AdventureCombat>();gear=h.GetComponent<AdventureLoadout>();actor=h.GetComponent<FantasyActorVisual>();for(int i=0;i<4;i++)yield return null;
 if(Array.IndexOf(Environment.GetCommandLineArgs(),"CP04Editor.CameraWall")>=0){for(int skill=0;skill<2;skill++)yield return CaseRun(6,"historical-right",skill==1,"camera-wall-verified");yield break;} foreach(var camera in new[]{"historical-right","center","left","orbit"})foreach(float distance in new[]{.35f,1.5f,6f,20f})for(int skill=0;skill<2;skill++){yield return CaseRun(distance,camera,skill==1,"none");}
 foreach(string block in new[]{"muzzle-wall","camera-wall","ground"})for(int skill=0;skill<2;skill++)yield return CaseRun(6,"historical-right",skill==1,block);
 }
 // Nested coroutine manually advanced so the Editor driver has one bounded update source.
 static IEnumerator CaseRun(float distance,string camera,bool skill,string occlusion){
 current=$"{camera}-d{distance:R}-s{(skill?1:0)}-{occlusion}";active=new Case{id=current,distance=distance,camera=camera,skill=skill,occlusion=occlusion};cases.Add(active);Time.timeScale=1;h.Inventory.SetOpen(false);h.HudPanelOpen=h.SaveMenuOpen=false;h.Dodge.ResetDodge();h.RestorePosition(new Vector3(300,20,300),0);gear.mainWeapon=gear.items.First(i=>i.Weapon&&i.InBag&&(int)i.type==2).id;gear.Refresh();combat.ResetCombat();s.GetComponent<CombatFeedback>().ResetFeedback();s.orbit.RestoreView(0,28);PlayerInput.Focus(true);
 target=s.Waves.CreateEncounter(s.transform,current,h.transform.position+Vector3.forward*distance,EnemyArchetype.Melee);target.enabled=false;target.Health.Observed+=x=>{if(x.kind==HealthObservationKind.Damage)damage.Add(new Damage{caseId=current,frame=x.frame,before=x.before,after=x.after,applied=x.applied});};
 GameObject block=null;if(occlusion!="none"){block=GameObject.CreatePrimitive(PrimitiveType.Cube);block.name="CP04-"+occlusion;if(occlusion=="muzzle-wall"){block.transform.position=h.transform.position+new Vector3(0,1.2f,2);block.transform.localScale=new Vector3(3,3,.3f);}else if(occlusion=="camera-wall"||occlusion=="camera-wall-verified"){block.transform.position=h.transform.position+new Vector3(2.5f,2.5f,occlusion=="camera-wall-verified"?.5f:-2.5f);block.transform.rotation=Quaternion.Euler(0,45,0);block.transform.localScale=new Vector3(3,3,.3f);}else{block.transform.position=h.transform.position+Vector3.down*.11f;block.transform.localScale=new Vector3(60,.2f,60);}}
 if(camera=="orbit")typeof(OrbitCamera).GetMethod("LateUpdate",BindingFlags.Instance|BindingFlags.NonPublic).Invoke(s.orbit,null);else{s.orbit.transform.position=h.transform.position+(camera=="center"?new Vector3(0,3,-6):new Vector3(camera=="left"?-5:5,4,-5));s.orbit.transform.LookAt(target.transform.position+Vector3.up);}
 Physics.SyncTransforms();for(int i=0;i<4;i++)yield return null;active.before=target.Health.Current;active.colliders=target.GetComponentsInChildren<Collider>().Select(c=>Info(c,c.ClosestPoint(combat.MuzzlePosition))).ToArray();active.accepted=skill?combat.UseSkill():combat.Strike();if(!active.accepted)throw new Exception("real action rejected "+current);float until=Time.time+3.2f;double deadline=Time.realtimeSinceStartupAsDouble+25;while(Time.time<until){if(Time.realtimeSinceStartupAsDouble>deadline)throw new Exception("simulation stalled "+current);yield return null;}active.after=target.Health.Current;active.hpLoss=active.before-active.after;active.remainingProjectiles=UnityEngine.Object.FindObjectsByType<AdventureProjectile>(FindObjectsSortMode.None).Count(p=>(ThirdPersonHero)F(p,"hero")==h);File.AppendAllText(root+"/case-progress.jsonl",JsonUtility.ToJson(active)+"\n");target.gameObject.SetActive(false);if(block!=null)UnityEngine.Object.Destroy(block);yield return null;
 }
 // IEnumerator nesting is expanded by this explicit stack, rather than Unity coroutine timing.
 public static void AdvanceNestedBegin(){run=Flatten(run);}
 static IEnumerator Flatten(IEnumerator iterator){while(iterator.MoveNext()){if(iterator.Current is IEnumerator nested){var f=Flatten(nested);while(f.MoveNext())yield return f.Current;}else yield return iterator.Current;}}
