using System;
using System.Collections;
using System.IO;
using System.Linq;
using UnityEngine;
using UnityEngine.AI;

namespace CavernValidation
{
    public sealed class Driver : MonoBehaviour
    {
        static string root;
        CombatSession s; ThirdPersonHero hero; ForestRegionRuntime region; string mode;
        [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
        public static void Boot()
        {
            if(string.IsNullOrEmpty(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")))return;
            root=GameEntry.TestRoot;
            if(string.IsNullOrEmpty(root)||!File.Exists(Path.Combine(root,"cavern-test.txt"))||FindAnyObjectByType<Driver>()!=null)return;
            Application.runInBackground=true;
            var host=new GameObject("C24 isolated controlled validation");DontDestroyOnLoad(host);host.AddComponent<Driver>();
        }
        void Check(string id,bool ok,string detail="")
        {
            File.AppendAllText(Path.Combine(root,"assertions.tsv"),id+"\t"+(ok?"PASS":"FAIL")+"\t"+detail.Replace('\n',' ')+"\n");
            if(!ok)throw new InvalidOperationException(id+": "+detail);
        }
        static void Exit(int code)
        {
#if UNITY_EDITOR
            UnityEditor.EditorApplication.Exit(code);
#else
            Application.Quit(code);
#endif
        }
        IEnumerator Start()
        {
            var stack=new System.Collections.Generic.Stack<IEnumerator>();stack.Push(Run());
            while(stack.Count>0)
            {
                object next=null;bool more;
                try{more=stack.Peek().MoveNext();if(more)next=stack.Peek().Current;}
                catch(Exception e){File.WriteAllText(Path.Combine(root,"failure.txt"),e.ToString());Exit(1);yield break;}
                if(!more)stack.Pop();else if(next is IEnumerator nested)stack.Push(nested);else yield return next;
            }
            File.WriteAllText(Path.Combine(root,"complete.txt"),"PASS");Exit(0);
        }
        IEnumerator Wait(string id,Func<bool> predicate,float seconds=30)
        {float end=Time.realtimeSinceStartup+seconds;while(!predicate()&&Time.realtimeSinceStartup<end)yield return null;Check(id,predicate());}
        IEnumerator Walk(string id,params Vector3[] points)
        {
            foreach(var point in points)
            {
                var path=new NavMeshPath();bool valid=NavMesh.SamplePosition(hero.transform.position,out var a,2,NavMesh.AllAreas)&&NavMesh.SamplePosition(point,out var b,2,NavMesh.AllAreas)&&NavMesh.CalculatePath(a.position,b.position,NavMesh.AllAreas,path)&&path.status==NavMeshPathStatus.PathComplete;
                Check(id+"-nav-"+point,valid);
                File.AppendAllText(Path.Combine(root,"paths.tsv"),id+"\t"+string.Join(";",path.corners.Select(p=>p.ToString("F3")))+"\n");
                foreach(var corner in path.corners.Skip(1))
                {
                    float end=Time.realtimeSinceStartup+40;
                    while(Vector3.ProjectOnPlane(corner-hero.transform.position,Vector3.up).magnitude>.35f&&Time.realtimeSinceStartup<end&&!hero.Health.Dead)
                    {
                        var delta=Vector3.ProjectOnPlane(corner-hero.transform.position,Vector3.up);
                        hero.MoveCharacter(Vector3.ClampMagnitude(delta,hero.moveSpeed*hero.MoveMultiplier*Time.deltaTime));yield return null;
                    }
                    Check(id+"-move-"+corner,Vector3.ProjectOnPlane(corner-hero.transform.position,Vector3.up).magnitude<=.35f,hero.transform.position.ToString());
                }
            }
        }
        Vector3 P(float x,float z)=>new Vector3(x,.1f,z);
        IEnumerator Extract()
        {
            yield return Walk("free-exit",P(33,407.05f),P(17,407.05f),P(15,410),P(0,410),P(0,385));
            var resolver=hero.Inventory.Interactions;resolver.Refresh();resolver.Present();
            Check("free-exit-real-interaction",resolver.Interact(),s.Flow.ExtractionReason);
            yield return Wait("extraction-terminal",()=>s.Flow.HasResult,15);
            Check("settlement-commit",s.Flow.CommitSettlement());Check("settlement-ack",s.Flow.AcknowledgeSettlement());
            Check("back-at-camp",s.Flow.AtCamp);
        }
        IEnumerator Run()
        {
            mode=File.ReadAllText(Path.Combine(root,"cavern-test.txt")).Trim();
            yield return Wait("title",()=>GameEntry.Instance?.Info!=null);
            GameEntry.Instance.AskNew();Check("new-confirm",GameEntry.Instance.ConfirmNew(),GameEntry.Error);
            yield return Wait("game",()=>(s=FindAnyObjectByType<CombatSession>())!=null&&s.Saves.IsInitialized&&!GameEntry.Loading);
            hero=s.hero;PlayerInput.Focus(true);yield return null;
            Check("depart",s.Flow.DepartCavern(),s.Flow.Error);
            region=s.GetComponent<ForestRegionRuntime>();
            Check("exact-version",CrystalCavernLayout.HasBoss(region.Layout)&&s.Flow.ActiveRun.contentRevision==CrystalCavernLayout.Content24);
            var enemy=s.enemies.Single(e=>e.entityId==CrystalCavernLayout.BossId);
            var boss=enemy.GetComponent<CrystalCavernBossBrain>();
            var challenge=s.Events.Points.Single(e=>e.entityId==CrystalCavernLayout.Challenge);
            var near=new[]{P(110,410),P(118,410)};
            var bypass=new[]{P(78,410),P(78,432),P(40,432)};
            Check("dormant-no-exposure",!CrystalCavernBossRules.RouteExposure(region.Layout,enemy,near,2));
            Check("dormant-choice",!challenge.Used&&!boss.ChallengeCommitted);
            // Controlled placement is explicitly separate from the later normal-AI expedition.
            hero.RestorePosition(P(89,420),90);s.orbit.RestoreView(90,9);yield return null;
            if(mode=="stage1-bypass")
            {
                Check("optional-challenge-not-started",!boss.ChallengeCommitted);
                yield return Walk("bypass-return",P(78,420),P(78,432),P(40,432),P(40,410));
                yield return Extract();yield break;
            }
            Check("choice-interactable",challenge.CanInteract(hero));
            string before=File.ReadAllText(s.Saves.SavePath);
            s.Saves.WriteValidationHook=stage=>{if(stage=="before-publish")throw new IOException("C24 deliberate pre-publish failure");};
            bool failed=challenge.Interact(hero);s.Saves.WriteValidationHook=null;
            Check("failed-save-no-battle",!failed&&!boss.ChallengeCommitted&&!challenge.Used);
            Check("failed-save-disk-preserved",File.ReadAllText(s.Saves.SavePath)==before);
            Check("retry-challenge",challenge.Interact(hero),s.Saves.Status);
            Check("challenge-durable",CombatSaveSystem.InspectReadOnly(s.Saves.SavePath).activeRun.checkpoint.events.Single(e=>e.entityId==CrystalCavernLayout.Challenge).used);
            Check("challenged-near-exposure",CrystalCavernBossRules.RouteExposure(region.Layout,enemy,near,2));
            Check("challenged-lake-no-exposure",!CrystalCavernBossRules.RouteExposure(region.Layout,enemy,bypass,2));
            var oldLayouts=new[]{CrystalCavernLayout.Generate(17),CrystalCavernLayout.GenerateExpanded(17),CrystalCavernLayout.GenerateLake(17),RuinsLayout.Generate(17)};
            foreach(var map in oldLayouts)Check("old-map-no-c24-risk-"+map.revision,!CrystalCavernBossRules.RouteExposure(map,enemy,near,2));
            var scored=RouteScoring.Evaluate(s).Single(r=>r.poiId==CrystalCavernLayout.Arena);
            var navigated=region.NavigationRoute(hero.transform.position,CrystalCavernLayout.Arena);
            Check("route-risk-consistent",navigated.nearElite&&scored.risk.pathElites>0);
            if(mode=="stage1-open")
            {
                yield return Walk("reach-bridge",P(78,420),P(76,414.5f));
                var lever=s.Events.Points.Single(e=>e.entityId==CrystalCavernLayout.BridgeLever);
                Check("bridge-interact",lever.Interact(hero));
                yield return Walk("open-bridge-return",P(76,410),P(43,410),P(40,410));
            }
            else
                yield return Walk("closed-bridge-return",P(78,420),P(78,432),P(40,432),P(40,410));
            Check("boss-no-lakeshore-attack",boss.Releases==0);
            Check("save",s.Saves.SaveProgress(false),s.Saves.Status);
            File.WriteAllText(Path.Combine(root,"run.json"),JsonUtility.ToJson(s.Flow.ActiveRun,true));
            if(!Application.isEditor){yield return new WaitForEndOfFrame();var shot=ScreenCapture.CaptureScreenshotAsTexture();File.WriteAllBytes(Path.Combine(root,"shoulder.png"),shot.EncodeToPNG());Destroy(shot);}
        }
    }
}
