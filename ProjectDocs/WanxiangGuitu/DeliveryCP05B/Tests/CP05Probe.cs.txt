using System;using System.IO;using System.Collections;using System.Collections.Generic;using System.Diagnostics;using System.Reflection;using UnityEngine;
public sealed class CP05Probe:MonoBehaviour {
 [Serializable]public class Row{public string name,status,detail;}
 [Serializable]public class Result{public string stage,layer;public Row[] rows;}
 public static bool Done;public static int Code;
 readonly List<Row> rows=new List<Row>();string root;ThirdPersonHero hero;OrbitCamera orbit;IEnumerator routine;float deadline;bool finished;
 [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]static void Entry(){if(Array.IndexOf(Environment.GetCommandLineArgs(),"-cp05-probe")>=0)new GameObject("Isolated CP05 API Probe").AddComponent<CP05Probe>();}
 void Start(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=Time.realtimeSinceStartup+180;routine=Run();StartCoroutine(Drive());}
 IEnumerator Drive(){while(true){bool more=false;object current=null;try{more=routine.MoveNext();if(more)current=routine.Current;}catch(Exception e){Check("probe-exception",false,e.ToString());Finish();yield break;}if(!more){Finish();yield break;}yield return current;}}
 void Update(){if(!finished&&Time.realtimeSinceStartup>deadline){Check("bounded-deadline",false,"no kill");Finish();}}
 void Check(string name,bool pass,string detail){rows.Add(new Row{name=name,status=pass?"PASS":"FAIL",detail=detail});File.AppendAllText(root+"/probe-progress.jsonl",JsonUtility.ToJson(rows[rows.Count-1])+"\n");}
 IEnumerator Run(){while(CP05Training.Instance?.Hero==null)yield return null;hero=CP05Training.Instance.Hero;orbit=hero.cameraTransform.GetComponent<OrbitCamera>();yield return new WaitForSecondsRealtime(1);Check("training-clock",Time.timeScale>0,"timeScale="+Time.timeScale);Check("moving-enemies-created",CP05Training.Instance.Enemies.Length==2,"real CombatEnemy AI, original fields");
  foreach(var e in CP05Training.Instance.Enemies)e.target=null; // controlled fixtures only, no product AI modification
  PlayerInput.Focus(true);hero.GetComponent<CharacterController>().enabled=false;hero.transform.position=new Vector3(0,.1f,0);hero.GetComponent<CharacterController>().enabled=true;orbit.RestoreView(0,28);orbit.shoulderMode=false;yield return null;yield return null;var original=orbit.transform.position;float yaw=orbit.ViewYaw;orbit.shoulderMode=true;yield return null;yield return null;Check("shoulder-offset",orbit.transform.position.x-original.x>.3f,"original="+original+" shoulder="+orbit.transform.position);Check("camera-yaw-player-owned",orbit.ViewYaw==yaw,"mode change does not rotate view");
  var wall=GameObject.CreatePrimitive(PrimitiveType.Cube);wall.name="CP05 camera collision fixture";wall.transform.position=new Vector3(0,2.5f,-3);wall.transform.localScale=new Vector3(4,5,1);Physics.SyncTransforms();yield return null;yield return null;Check("camera-wall-clear",orbit.CameraPositionClear&&orbit.AppliedDistance<6,"distance="+orbit.AppliedDistance);wall.SetActive(false);Destroy(wall);yield return null;
  var fade=orbit.GetComponent<CameraActorFade>();fade.Refresh();fade.enabled=false;Check("fade-restore",fade.FadedRenderers==0&&fade.OwnedMaterialCount==0,"disabled restores owned renderer/material state");fade.enabled=true;
  var before=UnityEngine.Random.state;for(int i=0;i<20;i++)fade.Refresh();Check("camera-fade-rng-readonly",JsonUtility.ToJson(before)==JsonUtility.ToJson(UnityEngine.Random.state),"20 explicit renderer queries");
  PlayerInput.Barrier();Check("input-barrier",PlayerInput.UIBlocked&&!PlayerInput.Down(PlayerAction.Attack),"synthetic context barrier, NOT native input");yield return null;yield return null;
  var late=typeof(OrbitCamera).GetMethod("LateUpdate",BindingFlags.NonPublic|BindingFlags.Instance);long positiveBefore=GC.GetAllocatedBytesForCurrentThread();var calibration=new byte[65536];GC.KeepAlive(calibration);long positive=GC.GetAllocatedBytesForCurrentThread()-positiveBefore;long zeroBefore=GC.GetAllocatedBytesForCurrentThread();for(int i=0;i<100;i++){}long zero=GC.GetAllocatedBytesForCurrentThread()-zeroBefore;var sw=Stopwatch.StartNew();long b=GC.GetAllocatedBytesForCurrentThread();for(int i=0;i<100;i++)late.Invoke(orbit,null);long alloc=GC.GetAllocatedBytesForCurrentThread()-b;sw.Stop();rows.Add(new Row{name="camera-cpu-gc",status=positive>=65536&&zero==0?"PASS":"NOT_RUN",detail="measurement only, no performance budget; reflection-inclusive 100 calls ms="+sw.Elapsed.TotalMilliseconds+" bytes="+alloc+" positive="+positive+" zero="+zero});
  yield return new WaitForEndOfFrame();try{var tex=ScreenCapture.CaptureScreenshotAsTexture();File.WriteAllBytes(root+"/training.png",tex.EncodeToPNG());Destroy(tex);}catch(Exception e){rows.Add(new Row{name="screenshot",status="NOT_RUN",detail=e.Message});}
 }
 void Finish(){if(finished)return;finished=true;Code=rows.Exists(x=>x.status=="FAIL")?1:0;File.WriteAllText(root+"/probe.json",JsonUtility.ToJson(new Result{stage=CombatAssistOptions.Stage,layer=Application.isEditor?"controlled Editor API":"isolated nondevelopment Player API",rows=rows.ToArray()},true));Done=true;if(!Application.isEditor)Application.Quit(Code);}
}
