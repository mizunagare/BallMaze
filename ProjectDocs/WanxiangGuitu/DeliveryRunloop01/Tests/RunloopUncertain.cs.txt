using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
// Explicit isolated storage fault. Restore exactly published bytes, never invent rewards.
public static class RunloopUncertain {
 static string root;static bool corrupt,done;static double deadline;static int frame=-1;static CombatSession s;
 [Serializable]class Result{public bool corrupt,waiting,commitBlocked,saveBlocked,notClaimedCommitted,readRecovered,duplicateSafe;public string error,settlement;public long revision;}
 public static void Missing(){Start(false);}public static void Corrupt(){Start(true);}
 static void Start(bool c){corrupt=c;root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+180;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("uncertain deadline");if(done||!EditorApplication.isPlaying||Time.unscaledTime<.6f||frame==Time.frameCount)return;frame=Time.frameCount;s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;done=true;s.Saves.InitializeStorage(root+"/Synthetic",false);s.Saves.autoSaveInterval=3600;AdventureRandom.SetNextSeedForValidation(418);if(!s.Flow.Depart(AdventureMode.ForestExploration))throw new Exception("depart");foreach(var e in s.enemies){e.enabled=false;var b=e.GetComponent<ForestBossBrain>();if(b!=null)b.enabled=false;}s.Flow.OpenAbandon();if(!s.Flow.ConfirmAbandon())throw new Exception("abandon");string held=s.Saves.SavePath+".published-held";
 s.Saves.WriteValidationHook=p=>{if(p!="after-publish")return;File.Move(s.Saves.SavePath,held);if(corrupt)File.WriteAllText(s.Saves.SavePath,"RUN01_SYNTHETIC_CORRUPTION");throw new IOException("RUN01_EXPECTED_UNREADABLE_AFTER_PUBLISH");};bool first=s.Flow.CommitSettlement();s.Saves.WriteValidationHook=null;var r=new Result{corrupt=corrupt,waiting=s.Flow.SettlementNeedsRead,error=s.Flow.Error,notClaimedCommitted=!first&&s.Flow.Settlement.status==SettlementStatus.Pending,settlement=s.Flow.Settlement.settlementId};r.commitBlocked=!s.Flow.CommitSettlement();r.saveBlocked=!s.Saves.SaveProgress(false);
 byte[] exact=File.ReadAllBytes(held);File.WriteAllBytes(s.Saves.SavePath,exact);r.readRecovered=s.Saves.LoadProgress(false)&&!s.Flow.SettlementNeedsRead&&s.Flow.Settlement.status==SettlementStatus.Committed&&s.Flow.Settlement.settlementId==r.settlement;long rev=s.Flow.Revision;string profile=JsonUtility.ToJson(s.Flow.Profile);r.duplicateSafe=s.Flow.CommitSettlement()&&s.Flow.Revision==rev&&profile==JsonUtility.ToJson(s.Flow.Profile);r.revision=rev;File.WriteAllText(root+"/uncertain.json",JsonUtility.ToJson(r,true));s.Saves.SuspendEntry();EditorApplication.update-=Tick;EditorApplication.Exit(r.waiting&&r.commitBlocked&&r.saveBlocked&&r.notClaimedCommitted&&r.readRecovered&&r.duplicateSafe?0:1);
 }catch(Exception e){File.WriteAllText(root+"/uncertain-exception.txt",e.ToString());if(s!=null){s.Saves.WriteValidationHook=null;s.Saves.SuspendEntry();}EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
