 static bool EvadeThreat(){
  foreach(var enemy in s.enemies.Where(e=>!e.Health.Dead).OrderBy(e=>e.GetComponent<ForestBossBrain>()!=null?0:1)){
   var boss=enemy.GetComponent<ForestBossBrain>();Vector3 away;
   if(boss!=null){if(boss.State!=ForestBossState.Telegraph||boss.Remaining>.16f)continue;away=s.hero.transform.position-boss.LockedPoint;if(boss.Attack==ForestBossAttack.AntlerCharge||boss.Attack==ForestBossAttack.WaterFan||boss.Attack==ForestBossAttack.SeedVolley)away=Vector3.Cross(Vector3.up,enemy.transform.forward);}
   else{if(!enemy.WindingUp||(float)strikeClock.GetValue(enemy)-Time.time>.16f||Vector3.Distance(enemy.transform.position,s.hero.transform.position)>25)continue;away=enemy.archetype==EnemyArchetype.Ranged?Vector3.Cross(Vector3.up,enemy.transform.forward):s.hero.transform.position-enemy.transform.position;}
   if(away.sqrMagnitude<.01f)away=-s.hero.transform.forward;
   if(s.hero.Dodge.TryDodge(away)){File.AppendAllText(Root+"/dodge-actions.csv",$"{Time.time:0.00},{enemy.entityId},{(boss==null?"ordinary":boss.Attack.ToString())}\n");return true;}
  }return false;
 }
