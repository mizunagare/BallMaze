using System;
using System.IO;
using System.Linq;
using UnityEngine;
public class WGPlayerFlow:MonoBehaviour {
 [Serializable] class Expected {public string run,node,usedEvent;public int wood,seed;public string[] gear;}
 string root,mode;CombatSession s;int phase;float elapsed;Expected expected;
 [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]static void Boot(){if(Application.isEditor)return;var args=Environment.GetCommandLineArgs();int index=Array.IndexOf(args,"-wg-validation-root");if(index<0||index+2>=args.Length)return;var obj=new GameObject("explicit isolated validation driver");var d=obj.AddComponent<WGPlayerFlow>();d.root=Path.GetFullPath(args[index+1]);d.mode=args[index+2];var save=FindFirstObjectByType<CombatSaveSystem>();if(save!=null)save.enabled=false;}
 void Need(bool b,string m){if(!b)throw new Exception(m);}string ExpectedPath=>Path.Combine(root,"expected.json");
 void Mark(string m){Debug.Log("WG_PLAYER_"+m);File.AppendAllText(Path.Combine(root,"player-results.txt"),mode+" "+m+"\n");}
 void Update(){try{elapsed+=Time.unscaledDeltaTime;if(elapsed>60)throw new Exception("player timeout "+mode+" phase="+phase);if(elapsed<.6f)return;
 if(phase==0){s=FindFirstObjectByType<CombatSession>();if(s==null)return;Directory.CreateDirectory(root);Need(Application.productName=="万象归途"&&!s.Saves.QuickLoadAllowed,"brand/F9 release boundary");s.Saves.InitializeLocatedStorage(Path.Combine(root,"Legacy","ForestRuins"),Path.Combine(root,"WanxiangGuitu","ForestRuins"));Need(s.Saves.IsInitialized,"location not ready");
 if(mode=="zero-save"){
 s.hero.Inventory.Add(MaterialKind.Stone,1);s.hero.Inventory.Add(MaterialKind.Ore,2);var region=s.gameObject.GetComponent<ForestRegionRuntime>()??s.gameObject.AddComponent<ForestRegionRuntime>();region.LayoutFactory=seed=>{var l=ForestLayout.Generate(seed);l.encounters=new ForestEncounter[0];return l;};Need(s.Flow.Depart(AdventureMode.ForestExploration)&&s.enemies.Length==0&&s.WorldReady,"zero encounter deployment");var node=FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).First(n=>n.kind==MaterialKind.Wood);s.hero.RestorePosition(node.transform.position,0);node.Hit(100000,s.hero);var loot=FindObjectsByType<LootPickup>(FindObjectsSortMode.None).First();Need(loot.PickUp(s.hero.Inventory),"actual zero-world harvest");s.Events.Points[0].RestoreUsed(true);LootPickup.Spawn(MaterialKind.Ore,2,node.transform.position+Vector3.right*4+Vector3.up*.5f);Need(s.Saves.SaveProgress(false),"zero save");expected=new Expected{run=s.Flow.ActiveRun.runId,node=node.StableId,usedEvent=s.Events.Points[0].entityId,wood=s.hero.Inventory.Count(MaterialKind.Wood),seed=region.Layout.seed};File.WriteAllText(ExpectedPath,JsonUtility.ToJson(expected));ScreenCapture.CaptureScreenshot(Path.Combine(root,"zero-world.png"));Mark("ZERO_SAVE_PASS");phase=90;
 }else if(mode=="zero-resume"){
 expected=JsonUtility.FromJson<Expected>(File.ReadAllText(ExpectedPath));Need(s.Flow.ActiveRun.runId==expected.run&&s.enemies.Length==0&&s.WorldReady,"new process empty run");Need(FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).Single(n=>n.StableId==expected.node).Health==0&&s.Events.Points.Single(p=>p.entityId==expected.usedEvent).Used,"completed world reset");Need(s.hero.Inventory.Count(MaterialKind.Wood)==expected.wood&&FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Count(l=>!l.Collected)==1,"materials/page loss");s.hero.RestorePosition(s.Flow.ExitPosition,0);Need(s.Flow.BeginExtraction(),"empty-world extraction");Mark("ZERO_NEW_PROCESS_RESUME_PASS");phase=1;
 }else if(mode=="terminal-resume"){
 expected=JsonUtility.FromJson<Expected>(File.ReadAllText(ExpectedPath));Need(s.Flow.ActiveRun.runId==expected.run&&s.Flow.Settlement.outcome==AdventureOutcome.Death&&s.Flow.Settlement.status==SettlementStatus.Pending,"death pending not recovered");Need(s.Flow.CommitSettlement()&&s.Flow.CommitSettlement()&&s.Flow.AcknowledgeSettlement(),"death commit");Need(expected.gear.OrderBy(x=>x).SequenceEqual(s.hero.GetComponent<AdventureLoadout>().items.Select(i=>i.id).OrderBy(x=>x)),"entry gear lost");Need(s.Flow.Profile.committedRuns.Length==2,"death duplicate history");Need(s.Flow.Depart(AdventureMode.EndlessWaves),"abandon departure");s.Flow.OpenAbandon();Need(s.Flow.ConfirmAbandon()&&s.Flow.CommitSettlement()&&s.Flow.CommitSettlement()&&s.Flow.AcknowledgeSettlement()&&s.Flow.Profile.committedRuns.Length==3,"abandon unique settlement");Need(s.Flow.Depart(AdventureMode.EndlessWaves)&&s.Saves.SaveProgress(false)&&s.Saves.LoadProgress(false),"last redepart/reload");Mark("DEATH_ABANDON_UNIQUE_SETTLEMENT_REDEPART_PASS");phase=90;
 }else throw new Exception("unknown mode");return;}
 if(phase==1&&s.Flow.HasResult){Need(s.Flow.Settlement.outcome==AdventureOutcome.Extracted&&s.Flow.CommitSettlement()&&s.Flow.CommitSettlement()&&s.Flow.AcknowledgeSettlement(),"empty extraction settlement");var gear=s.hero.GetComponent<AdventureLoadout>();Need(s.Camp.Materials[(int)MaterialKind.Wood]==expected.wood,"zero harvest lost/duplicated");Need(s.Camp.TransferMaterial(MaterialKind.Wood,2,false)&&gear.Craft(GearType.Sword,GearQuality.White),"camp withdrawal/manufacture");var sword=gear.items.Last(i=>i.type==GearType.Sword&&i.id!=gear.mainWeapon&&i.id!=gear.reserveWeapon);string itemId=sword.id;Need(s.Camp.DepositEquipment(itemId)&&s.Camp.WithdrawEquipment(itemId)&&gear.items.Count(i=>i.id==itemId)==1,"equipment storage roundtrip");Need(s.Flow.Depart(AdventureMode.ForestExploration),"second forest deployment");expected.run=s.Flow.ActiveRun.runId;expected.gear=gear.items.Select(i=>i.id).ToArray();File.WriteAllText(ExpectedPath,JsonUtility.ToJson(expected));s.hero.Health.Damage(1000000);s.Flow.ResolveTerminal();Need(s.Flow.HasResult&&s.Saves.SaveProgress(false),"durable death terminal");Mark("ZERO_EXTRACT_STORE_CRAFT_REDEPART_PENDING_PASS");ScreenCapture.CaptureScreenshot(Path.Combine(root,"pending-death.png"));phase=90;}
 if(phase==90&&elapsed>3){Application.Quit(0);phase=99;}
 }catch(Exception e){Debug.LogException(e);Mark("FAIL "+e.Message);Application.Quit(1);phase=99;}}
}
