using System;using System.IO;using System.Linq;using System.Collections.Generic;using System.Reflection;using System.Security.Cryptography;using System.Text;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
// Controlled Editor business-path fault injection. Never native input or player evidence.
public static class R1Matrix {
 static string root;static CombatSession s;static bool done;static double end;
 [Serializable]class Row{public string id,status,detail;public bool validV9,needsRead;public long revision;}
 [Serializable]class Results{public string layer="controlled Editor, synthetic v9 only";public Row[] rows;}
 static List<Row> rows=new List<Row>();
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);end=EditorApplication.timeSinceStartup+300;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>end)throw new Exception("R1 deadline");if(done||!EditorApplication.isPlaying||Time.unscaledTime<.6f)return;s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;done=true;

 foreach(string kind in new[]{"committed","pending","parse"})Recover(kind);
 Finish();}catch(Exception e){File.WriteAllText(root+"/exception.txt",e.ToString());Finish(2);}}
 static void Record(string id,bool pass,string detail,bool valid=true){rows.Add(new Row{id=id,status=pass?"PASS":"FAIL",detail=detail,validV9=valid,needsRead=s.Flow.SettlementNeedsRead,revision=s.Flow.Revision});}
 static string Canon<T>(T value)=>JsonUtility.ToJson(AdventureData.Clone(value));
 static void Setup(string name){s.Saves.WriteValidationHook=null;s.Flow.ValidationHook=null;ClearRegionHook();s.Saves.SuspendEntry();string dir=root+"/"+name;if(!s.Saves.InitializeFromEntry(dir,dir+"-legacy",true,SaveCatalog.Fingerprint(dir)))throw new Exception("fresh isolated profile: "+s.Saves.Status);s.Saves.autoSaveInterval=3600;s.Saves.enabled=false;AdventureRandom.SetNextSeedForValidation(418);if(!s.Flow.Depart(AdventureMode.ForestExploration))throw new Exception("depart "+s.Flow.DepartureReason);foreach(var e in s.enemies){e.enabled=false;var b=e.GetComponent<ForestBossBrain>();if(b!=null)b.enabled=false;}s.hero.enabled=false;s.orbit.enabled=false;
 var resource=UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).First(r=>r.gameObject.scene==s.gameObject.scene&&r.Health>0);resource.Hit(10000,s.hero);
 s.Flow.OpenAbandon();if(!s.Flow.ConfirmAbandon())throw new Exception("terminal");if(!Directory.GetFiles(dir+"/ground-loot","*.json").Any())throw new Exception("real paged ground loot required");}
 static void Envelope(CombatSaveData data,string path){AdventureValidation.Validate(data);GroundLootStore.Pack(data,Path.GetDirectoryName(path));string payload=JsonUtility.ToJson(data);string sum;using(var sha=SHA256.Create())sum=BitConverter.ToString(sha.ComputeHash(Encoding.UTF8.GetBytes(payload))).Replace("-","");File.WriteAllText(path,JsonUtility.ToJson(new CombatSaveEnvelope{version=9,revision=data.revision,savedAtUtc=DateTime.UtcNow.ToString("o"),payload=payload,checksum=sum}));CombatSaveSystem.InspectReadOnly(path);}
 static string Fingerprint(){string dir=Path.GetDirectoryName(s.Saves.SavePath);return string.Join("\n",Directory.GetFiles(dir,"*",SearchOption.AllDirectories).Where(p=>p==s.Saves.SavePath||p==s.Saves.BackupPath||p.Contains("ground-loot")&&!p.EndsWith(".tmp")).OrderBy(p=>p).Select(p=>{using(var sha=SHA256.Create())return p+":"+BitConverter.ToString(sha.ComputeHash(File.ReadAllBytes(p)));}));}
 static void Invoke(string name,params object[] args){typeof(CombatSaveSystem).GetMethod(name,BindingFlags.Instance|BindingFlags.NonPublic).Invoke(s.Saves,args);}
 static void Guards(string id){string before=Fingerprint(),profile=Canon(s.Flow.Profile),run=Canon(s.Flow.ActiveRun);bool commit=!s.Flow.CommitSettlement(),save=!s.Saves.SaveProgress(false);typeof(CombatSaveSystem).GetField("nextSave",BindingFlags.Instance|BindingFlags.NonPublic).SetValue(s.Saves,0f);Invoke("LateUpdate");var keyboard=UnityEngine.InputSystem.InputSystem.AddDevice<UnityEngine.InputSystem.Keyboard>();try{UnityEngine.InputSystem.InputSystem.QueueStateEvent(keyboard,new UnityEngine.InputSystem.LowLevel.KeyboardState(UnityEngine.InputSystem.Key.F5));UnityEngine.InputSystem.InputSystem.Update();Invoke("Update");}finally{UnityEngine.InputSystem.InputSystem.RemoveDevice(keyboard);}Invoke("OnApplicationFocus",false);Invoke("OnApplicationPause",true);Invoke("OnApplicationQuit");File.WriteAllText(root+"/"+id+"-before.sha256.txt",before);File.WriteAllText(root+"/"+id+"-after.sha256.txt",Fingerprint());Record(id,commit&&save&&before==Fingerprint()&&profile==Canon(s.Flow.Profile)&&run==Canon(s.Flow.ActiveRun),"save/auto/controlled InputSystem F5/focus/pause/quit callbacks fingerprint main+backup+pages; callbacks are not real process exit");}
 static void Case(string kind){Setup("case-"+kind);var before=CombatSaveSystem.InspectReadOnly(s.Saves.SavePath);byte[] old=File.ReadAllBytes(s.Saves.SavePath);File.WriteAllBytes(s.Saves.SavePath+".prior-pending",old);string oldProfile=Canon(before.profile);bool valid=true;
 var conflict=AdventureData.Clone(before);switch(kind){case "plan":conflict.settlement.plan.materials[0]++;break;case "outcome":conflict.settlement.outcome=AdventureOutcome.Death;break;case "quantities":conflict.settlement.returnedMaterials[0]++;break;case "checkpoint":conflict.activeRun.checkpoint.resources.First(r=>r.health>1).health-=1;break;case "source":conflict.activeRun.broughtRemaining[0]++;conflict.activeRun.checkpoint.materials[0]++;break;case "revision":conflict.revision++;break;case "profile":conflict.profile.profileId=Guid.NewGuid().ToString("N");break;case "identity":string oldId=conflict.activeRun.runId,newId=Guid.NewGuid().ToString("N");string json=JsonUtility.ToJson(conflict).Replace(oldId,newId);conflict=JsonUtility.FromJson<CombatSaveData>(json);break;}
 string candidate=Path.GetDirectoryName(s.Saves.SavePath)+"/candidate.json";Envelope(conflict,candidate);byte[] candidateBytes=File.ReadAllBytes(candidate);
 s.Saves.WriteValidationHook=p=>{if(kind=="published"){if(p=="after-publish")throw new IOException("R1_EXPECTED_AFTER_PUBLISH");return;}if(p!="before-publish")return;if(kind=="missing")File.Move(s.Saves.SavePath,s.Saves.SavePath+".held");else if(kind=="corrupt")File.WriteAllText(s.Saves.SavePath,"R1_NEGATIVE_CORRUPT");else File.WriteAllBytes(s.Saves.SavePath,kind=="exact"?old:candidateBytes);throw new IOException("R1_EXPECTED_PRE_PUBLISH_"+kind);};
 bool ok=s.Flow.CommitSettlement();s.Saves.WriteValidationHook=null;bool expected=kind=="published"?ok&&!s.Flow.SettlementNeedsRead:kind=="exact"?!ok&&!s.Flow.SettlementNeedsRead:!ok&&s.Flow.SettlementNeedsRead;Record("T-"+kind,expected,"valid v9 artificial negative (or exact real prior/publication); candidate preserved",valid);
 if(kind=="exact"){bool retry=s.Flow.CommitSettlement();long rev=s.Flow.Revision;string profile=Canon(s.Flow.Profile);Record("T02-retry",retry&&s.Flow.CommitSettlement()&&s.Flow.Revision==rev&&profile==Canon(s.Flow.Profile),"exact prior with pages retries only once");}else if(kind!="published")Guards("T11-"+kind);
 }
 static string Label(){var p=typeof(AdventureFlow).GetProperty("SettlementPrimaryLabel");return p==null?(s.Flow.SettlementNeedsRead?"读取完整存档核对":s.Flow.Settlement.status==SettlementStatus.Pending?"确认结算并保存":"返回营地 / 重试回营"):(string)p.GetValue(s.Flow);}
 static bool Primary(){var method=typeof(AdventureFlow).GetMethod("ExecuteSettlementPrimaryAction");if(method!=null)return (bool)method.Invoke(s.Flow,null);if(s.Flow.SettlementNeedsRead)return s.Saves.LoadProgress(false);return s.Flow.Settlement.status==SettlementStatus.Pending?s.Flow.CommitSettlement():s.Flow.AcknowledgeSettlement();}
 static void Recover(string kind){Setup("recover-"+kind);string path=s.Saves.SavePath,held=path+".held";s.Saves.WriteValidationHook=p=>{if(p!=(kind=="pending"?"before-publish":"after-publish"))return;File.Move(path,held);throw new IOException("R1_EXPECTED_UNCERTAIN");};s.Flow.CommitSettlement();s.Saves.WriteValidationHook=null;byte[] exact=File.ReadAllBytes(held);File.WriteAllBytes(path,exact);var data=CombatSaveSystem.InspectReadOnly(path);File.WriteAllText(Path.GetDirectoryName(path)+"/expected-expanded.json",JsonUtility.ToJson(data,true));string identity=Canon(data);long rev=data.revision;string fp=Fingerprint();
 if(kind=="parse"){File.WriteAllText(path,"R1_PARSE_FAILURE");if(File.Exists(s.Saves.BackupPath))File.Move(s.Saves.BackupPath,s.Saves.BackupPath+".held");}else if(kind=="committed"){s.Flow.ValidationHook=p=>{if(p=="before-camp-load"){s.Flow.ValidationHook=null;throw new IOException("R1_ONCE_CAMP_LOAD");}};}else{s.GetComponent<ForestRegionRuntime>().ValidationHook=p=>{if(p=="before-build"){ClearRegionHook();throw new IOException("R1_ONCE_WORLD_LOAD");}};}
 bool first=Primary();Record("T08/10-first-"+kind,!first&&s.Flow.SettlementNeedsRead&&!s.Saves.CanCommitWarehouse,"read must remain primary after runtime/parse failure; label="+Label()+"");Guards("T11-recovery-"+kind);s.Flow.ValidationHook=null;ClearRegionHook();if(kind=="parse"){File.WriteAllBytes(path,exact);if(File.Exists(s.Saves.BackupPath+".held"))File.Move(s.Saves.BackupPath+".held",s.Saves.BackupPath);}
 bool second=Primary();Record("T09-second-"+kind,second&&!s.Flow.SettlementNeedsRead&&s.Saves.CanCommitWarehouse&&s.Flow.Revision==rev&&identity==Canon(CombatSaveSystem.InspectReadOnly(path)),"same presenter command; exact original disk; no extra revision/reward");if(second&&kind=="pending")Record("T12-pending",s.Flow.CommitSettlement()&&s.Flow.CommitSettlement(),"normal settlement after recovery");}
 static void ClearRegionHook(){if(s!=null&&s.GetComponent<ForestRegionRuntime>()!=null)s.GetComponent<ForestRegionRuntime>().ValidationHook=null;}
 static void Finish(int forced=0){if(s!=null){s.Saves.WriteValidationHook=null;s.Flow.ValidationHook=null;s.Saves.SuspendEntry();}ClearRegionHook();File.WriteAllText(root+"/matrix.json",JsonUtility.ToJson(new Results{rows=rows.ToArray()},true));EditorApplication.update-=Tick;EditorApplication.Exit(forced!=0?forced:rows.Any(r=>r.status=="FAIL")?1:0);}
}
