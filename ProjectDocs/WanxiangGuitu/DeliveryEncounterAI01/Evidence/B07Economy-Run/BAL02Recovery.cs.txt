using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class BAL02Recovery{
 static string root;static bool loading;static CombatSession s;static ThirdPersonHero h;static IEnumerator run;static int frame=-1;static double deadline;static List<Row> rows=new List<Row>();[Serializable]class Row{public string name,status,detail;}[Serializable]class Report{public Row[] rows;}[Serializable]class Expected{public string profile,run,gear;public float defense;}
 public static void Save(){loading=false;Start();}public static void Load(){loading=true;Start();}static void C(bool b,string n,string d=""){rows.Add(new Row{name=n,status=b?"PASS":"FAIL",detail=d});}
 static void Start(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+180;EditorSceneManager.playModeStartScene=null;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("recovery deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f||frame==Time.frameCount)return;frame=Time.frameCount;if(run==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;h=s.hero;s.Saves.InitializeStorage(root+"/Store",loading);run=Cases();}if(!run.MoveNext())End();}catch(Exception e){C(false,"exception",e.ToString());End();}}
 static void End(){EditorApplication.update-=Tick;Time.captureDeltaTime=0;if(s!=null)s.Saves.SuspendEntry();File.WriteAllText(root+"/recovery-"+(loading?"load":"save")+".json",JsonUtility.ToJson(new Report{rows=rows.ToArray()},true));EditorApplication.Exit(rows.Any(r=>r.status=="FAIL")?1:0);}
 static IEnumerator Cases(){Time.captureDeltaTime=.04f;if(!loading){AdventureRandom.SetNextSeedForValidation(418);C(s.Flow.Depart(AdventureMode.ForestExploration),"real original Run");BAL02Fixture.Init(h);BAL02Fixture.Defense(h,3);}h.enabled=false;foreach(var e in s.enemies){e.enabled=false;e.target=null;var boss=e.GetComponent<ForestBossBrain>();if(boss!=null)boss.enabled=false;}yield return null;
 if(!loading){C(s.Saves.SaveProgress(false),"old product high-defense v9 normal save");File.WriteAllText(root+"/expected.json",JsonUtility.ToJson(new Expected{profile=s.Flow.Profile.profileId,run=s.Flow.ActiveRun.runId,gear=JsonUtility.ToJson(s.Flow.Profile),defense=h.Defense}));yield break;}
 var old=JsonUtility.FromJson<Expected>(File.ReadAllText(root+"/expected.json"));C(s.Flow.Profile.profileId==old.profile&&s.Flow.ActiveRun.runId==old.run&&Mathf.Abs(h.Defense-old.defense)<.001f,"same old v9 Run Profile and 19.2 armor on new code");float hp=h.Health.Current;int count=0;Action<HealthObservation> observer=x=>{if(x.kind==HealthObservationKind.Damage)count++;};h.Health.Observed+=observer;h.Health.Damage(12);h.Health.Observed-=observer;C(Mathf.Abs(hp-h.Health.Current-BAL02Fixture.Expected(12,h.Defense))<.001f&&count==1,"new formula immediate on old armor, exactly one health event");
 while(h.Health.Current>25)h.Health.Damage(12);C(!h.Health.Dead&&h.Health.Current<25,"low HP comes from real Damage, never assigned");var exit=s.Flow.ExitConfigs().First(x=>!x.repairRequired);h.RestorePosition(exit.position,0);yield return null;C(s.Flow.BeginExtraction(exit.id),"controlled low-HP free extraction begins");h.Health.Damage(9);C(s.Flow.ActiveRun.phase==RunPhase.Running,"actual surviving hit interrupts channel once");C(s.Flow.BeginExtraction(exit.id),"retry low-HP extraction");float end=Time.time+exit.duration+3;while(!s.Flow.HasResult&&Time.time<end)yield return null;C(s.Flow.HasResult&&s.Flow.Settlement.status==SettlementStatus.Pending,"actual timed low-HP extraction reaches Pending");C(s.Flow.CommitSettlement(),"normal settlement commit");var profile=JsonUtility.ToJson(s.Flow.Profile);C(s.Saves.LoadProgress(false)&&profile==JsonUtility.ToJson(s.Flow.Profile),"Committed restore without duplicate reward");
 }
}
