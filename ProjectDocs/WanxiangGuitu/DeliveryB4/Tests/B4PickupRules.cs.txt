using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class B4PickupRules {
 static double deadline;static CombatSession s;static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")??throw new Exception("isolated root required");
 static void Need(bool b,string text){if(!b)throw new Exception(text);}
 public static void Run(){ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+120;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("pickup deadline");if(!EditorApplication.isPlaying||Time.time<.4f)return;s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(Root+"/b4-pickup-save",false);Need(s.Flow.Depart(AdventureMode.ForestExploration),"depart");foreach(var e in s.enemies)e.enabled=false;var region=s.GetComponent<ForestRegionRuntime>();var chest=s.Events.Points.Single(e=>e.entityId=="poi:3:chest");var position=chest.transform.position+Vector3.back*1.8f;Need(region.TryGround(position,2,out var approach),"chest approach");s.hero.RestorePosition(approach+Vector3.up*.1f,0);Physics.SyncTransforms();Need(chest.Interact(s.hero),"open");
  var embedded=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Where(l=>chest.GetComponentsInChildren<Collider>().Any(c=>c.enabled&&c.bounds.Contains(l.SavePosition))).ToArray();Need(embedded.Length>0,"legacy embedded fixture absent");
  foreach(var loot in embedded)Need(loot.CanPickUp(s.hero.Inventory),"OPENED_CHEST_BLOCKS_OWN_EMBEDDED_LOOT "+loot.Kind);
  var target=embedded[0];var wall=GameObject.CreatePrimitive(PrimitiveType.Cube);wall.transform.position=(s.hero.transform.position+Vector3.up*.5f+target.transform.position)/2;wall.transform.localScale=new Vector3(3,3,.15f);Physics.SyncTransforms();Need(!target.CanPickUp(s.hero.Inventory),"wall pickup bypass");wall.SetActive(false);Physics.SyncTransforms();Need(target.CanPickUp(s.hero.Inventory)&&target.PickUp(s.hero.Inventory),"embedded pickup failed");Need(!target.PickUp(s.hero.Inventory),"duplicate pickup");
  File.WriteAllText(Root+"/b4-pickup.txt","PASS actual opened chest embedded materials accessible; extra wall still blocks; pickup exactly once; controlled scene setup, not normal AI");EditorApplication.update-=Tick;EditorApplication.Exit(0);
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
