using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class B07Economy {
 static CombatSession s;static double deadline;static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT");static void Need(bool ok,string why){if(!ok)throw new Exception(why);}static void Mark(string m){File.AppendAllText(Root+"/b07-economy.txt",m+"\n");Debug.Log("B07_ECONOMY_"+m);}
 public static void Run(){ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+120;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Freeze(){foreach(var e in s.enemies){e.enabled=false;if(e.GetComponent<ForestBossBrain>()!=null)e.GetComponent<ForestBossBrain>().enabled=false;}}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();Application.runInBackground=true;if(EditorApplication.timeSinceStartup>deadline)throw new Exception("economy deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.5f)return;s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(Root+"/b07-economy-save",false);Need(s.Flow.Depart(AdventureMode.ForestExploration),"departure");Freeze();var gear=s.hero.GetComponent<AdventureLoadout>();var boss=s.enemies.Single(e=>e.entityId=="boss:warden");
  int added=0;GearItem spare=null;while(added<100){var item=AdventureItems.New(GearType.Armor,GearQuality.White,s);if(!gear.Add(item))break;spare=item;added++;}Need(added>0&&added<100,"bag full fixture");int count=gear.items.Count;
  boss.Health.Damage(boss.Health.maxHealth);boss.DropLootOnce();Need(gear.items.Count==count&&s.hero.Inventory.Count(MaterialKind.BossCore)==0,"boss auto credited");var drop=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Single(l=>l.Gear!=null&&l.Gear.quality==GearQuality.Purple);s.hero.RestorePosition(drop.SavePosition-Vector3.up*.4f,0);Physics.SyncTransforms();Need(!drop.PickUp(s.hero.Inventory)&&!drop.Collected&&drop.gameObject.activeSelf,"full bag lost gear");
  // Explicit controlled fixture cleanup of newly inserted filler, never entry gear.
  foreach(var item in gear.items.Where(i=>i.type==GearType.Armor&&i.quality==GearQuality.White&&i.origin==ItemOrigin.RunLoot).ToArray())gear.items.Remove(item);Need(drop.PickUp(s.hero.Inventory)&&!drop.PickUp(s.hero.Inventory),"pickup once after freeing real cells");var itemId=drop.Gear.id;Need(gear.SelectMain(drop.Gear)&&!s.Flow.Protects(itemId),"equipping run loot grants protection");Need(s.Flow.RetryBossCheckpoint(),"first kill checkpoint");
  var world=CombatSaveSystem.InspectReadOnly(s.Saves.SavePath).activeRun.checkpoint;foreach(var outcome in new[]{AdventureOutcome.Death,AdventureOutcome.Abandoned,AdventureOutcome.Extracted}){var result=AdventureRules.Plan(s.Flow.Profile,s.Flow.ActiveRun,world,outcome);Need(result.plan.badges.Contains("forest:warden"),"badge rolled back "+outcome);Need(AdventureRules.Success(outcome)?result.keptGear.Any(i=>i.id==itemId):result.lostGear.Any(i=>i.id==itemId),"equipped loot outcome "+outcome);}Mark("FULL_BAG_GROUND_RETRY_PICKUP_ONCE_EQUIPPED_LOOT_NOT_PROTECTED_THREE_OUTCOMES");
  s.hero.Health.Damage(99999);s.Flow.ResolveTerminal();Need(s.Flow.CommitSettlement()&&s.Flow.AcknowledgeSettlement(),"actual death commit ack");Need(s.Flow.Depart(AdventureMode.ForestExploration),"new forest repeat");Freeze();boss=s.enemies.Single(e=>e.entityId=="boss:warden");boss.Health.Damage(boss.Health.maxHealth);boss.DropLootOnce();Need(ForestBossRules.Find(s.Flow.Profile,ForestBossKind.Warden).kills==2,"actual repeat record");var repeats=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None);Need(repeats.Count(l=>l.Gear!=null&&l.Gear.quality==GearQuality.Green)==1&&repeats.Where(l=>l.Gear==null&&l.Kind==MaterialKind.BossCore).Sum(l=>l.Amount)==1,"repeat reward differs from first rule");Need(s.Flow.RetryBossCheckpoint(),"repeat save");Mark("ACTUAL_SECOND_RUN_REPEAT_GREEN_CORE1_PERMANENT_BADGE_AFTER_DEATH");EditorApplication.update-=Tick;EditorApplication.Exit(0);
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
