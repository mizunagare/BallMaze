using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class B4StateRules {
 static double deadline;static CombatSession s;
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")??throw new Exception("isolated root required");
 static void Need(bool ok,string why){if(!ok)throw new Exception(why);}static void Mark(string text){File.AppendAllText(Root+"/b4-state-results.txt",text+"\n");}
 public static void Run(){ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+180;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("B4 state rules timeout");if(!EditorApplication.isPlaying||Time.time<.4f)return;
  s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(Root+"/b4-state-save",false);Need(s.Flow.Depart(AdventureMode.ForestExploration),"depart");var region=s.GetComponent<ForestRegionRuntime>();var original=region.Capture();var originalSeed=s.Flow.ActiveRun.seed;var rng=AdventureData.Clone(s.Flow.ActiveRun.randomStreams);
  foreach(int revision in new[]{1,2,3}){
   var old=AdventureData.Clone(original);old.revision=revision;old.seed=-100-revision;old.requestedSeed=old.seed;old.routes=new[]{new ForestRoute{from="poi:0",to="poi:1"},new ForestRoute{from="poi:0",to="poi:2"},new ForestRoute{from="poi:1",to="poi:3"},new ForestRoute{from="poi:2",to="poi:4"},new ForestRoute{from="poi:3",to="poi:4"},new ForestRoute{from="poi:3",to="poi:5"},new ForestRoute{from="poi:4",to="poi:6"},new ForestRoute{from="poi:5",to="poi:6"}};
   region.Build(old);foreach(var e in s.enemies)e.enabled=false;var run=s.Flow.ActiveRun;run.seed=old.seed;run.contentRevision=revision==3?"B06-B2":revision==2?"B06-B1":"B06-A1";run.generationVersion=revision+2;run.hasRandomStreams=revision==3;run.randomStreams=revision==3?AdventureData.Clone(rng):null;
   Need(s.Saves.SaveProgress(false),"old revision save "+revision);string snapshot=JsonUtility.ToJson(region.Capture());Need(s.Saves.LoadProgress(false),"old revision restore "+revision);Need(snapshot==JsonUtility.ToJson(region.Capture())&&s.Flow.ActiveRun.seed==old.seed&&s.Flow.ActiveRun.hasRandomStreams==(revision==3),"old layout/seed/rng was regenerated "+revision);Mark("SAVED_REVISION_"+revision+"_EXACT_LAYOUT_SEED_RNG_SEMANTICS_PASS");
  }
  region.Build(original);foreach(var e in s.enemies)e.enabled=false;s.Flow.ActiveRun.seed=originalSeed;s.Flow.ActiveRun.contentRevision=original.revision>=5?"B06-B4-R1":"B06-B4";s.Flow.ActiveRun.generationVersion=original.revision>=5?7:6;s.Flow.ActiveRun.hasRandomStreams=true;s.Flow.ActiveRun.randomStreams=rng;
  var poi=region.Layout.points.Single(p=>p.id=="poi:1");Deplete(region,poi.id);Need(CraftingTargets.Select(s.hero,GearType.Sword,GearQuality.White),"target");var before=JsonUtility.ToJson(s.Flow.ActiveRun);var global=JsonUtility.ToJson(UnityEngine.Random.state);var recommendations=CraftingTargets.Recommendations(s);Need(!recommendations.Split('\n').Any(line=>line.StartsWith(poi.label+"（")),"depleted woodland recommended as yield");Need(before==JsonUtility.ToJson(s.Flow.ActiveRun)&&global==JsonUtility.ToJson(UnityEngine.Random.state),"query RNG/state mutation");Mark("DEPLETED_SOURCES_AND_QUERY_STATE_PASS");
  var ruins=region.Layout.points.Single(p=>p.id=="poi:3");Deplete(region,ruins.id);s.Flow.Profile.craftingTarget=null;s.Flow.ActiveRun.goal=ExpeditionGoal.Treasure;var drop=LootPickup.SpawnGear(AdventureItems.New(GearType.Sword,GearQuality.White),ruins.position+Vector3.up);Need(CraftingTargets.Recommendations(s).Split('\n').Any(line=>line.StartsWith(ruins.label+"（")&&line.Contains("地面装备")),"unpicked gear source hidden");drop.gameObject.SetActive(false);Need(!CraftingTargets.Recommendations(s).Split('\n').Any(line=>line.StartsWith(ruins.label+"（")),"removed ground gear still recommended");Mark("USED_CHEST_CLEARED_ENCOUNTER_AND_UNPICKED_GEAR_QUERY_PASS controlled setup");
  EditorApplication.update-=Tick;EditorApplication.Exit(0);
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
 static void Deplete(ForestRegionRuntime region,string poi){foreach(var n in UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).Where(n=>region.Layout.points.OrderBy(p=>(p.position-n.transform.position).sqrMagnitude).First().id==poi))n.Restore(0);foreach(var e in s.Events.Points.Where(e=>e.entityId.StartsWith(poi+":")))e.RestoreUsed(true);foreach(var e in s.enemies.Where(e=>e.GetComponent<EncounterGuard>()?.poi==poi))e.RestoreState(new SavedEnemy{position=e.transform.position,yaw=e.transform.eulerAngles.y,health=0,entityId=e.entityId});foreach(var l in UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None))l.gameObject.SetActive(false);}
}
