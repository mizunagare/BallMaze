using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class RunloopRuntimeAudit {
 static string root;static double end;static bool done;static CombatSession s;
 static void Need(bool ok,string n){if(!ok)throw new Exception(n);File.AppendAllText(root+"/runtime-audit.txt","PASS "+n+"\n");}
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);end=EditorApplication.timeSinceStartup+180;EditorSceneManager.OpenScene(GameEntry.ForestPath);EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>end)throw new Exception("runtime audit deadline");if(done||!EditorApplication.isPlaying||Time.unscaledTime<.6f)return;s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||!s.Saves.IsInitialized)return;done=true;s.Saves.InitializeStorage(root+"/Store",true);s.Saves.autoSaveInterval=3600;s.hero.enabled=false;foreach(var a in s.enemies){a.enabled=false;var b=a.GetComponent<ForestBossBrain>();if(b!=null)b.enabled=false;var guard=a.GetComponent<EncounterGuard>();if(guard!=null)guard.enabled=false;}
 var cp=s.Flow.ActiveRun.checkpoint;var region=s.GetComponent<ForestRegionRuntime>();Need(region.Active&&region.Ready&&region.MatchesActors(s.enemies)&&JsonUtility.ToJson(region.Capture())==JsonUtility.ToJson(cp.forest),"actual RuntimeForest layout and all actor IDs");Need(cp.enemies.All(e=>s.enemies.Any(a=>a.entityId==e.entityId&&Mathf.Abs(a.Health.Current-e.health)<.001f)),"actual enemy health restored by stable ID");var nodes=UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None);Need(cp.resources.All(e=>nodes.Any(n=>n.StableId==e.entityId&&Mathf.Abs(n.Health-e.health)<.001f)),"actual resource health restored by stable ID");Need(cp.events.All(e=>s.Events.Points.Any(p=>p.entityId==e.entityId&&p.Used==e.used)),"actual event used states restored by stable ID");s.Saves.SuspendEntry();EditorApplication.update-=Tick;EditorApplication.Exit(0);
 }catch(Exception e){File.WriteAllText(root+"/runtime-audit-failure.txt",e.ToString());if(s!=null)s.Saves.SuspendEntry();EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
