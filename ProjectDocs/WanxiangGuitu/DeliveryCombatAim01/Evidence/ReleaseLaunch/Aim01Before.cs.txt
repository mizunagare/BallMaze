using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
// Frozen controlled actual-content cases. Initial hero/camera placement and AI freezing are fixtures,
// not normal AI/native input. No resource collider is raised, resized or replaced.
public static class Aim01Before {
 static string root;static CombatSession s;static ThirdPersonHero h;static AdventureCombat combat;static AdventureLoadout gear;static IEnumerator cases;static int frame=-1;static double deadline;static List<Row> rows=new List<Row>();static List<Content> contents=new List<Content>();static string sample;
 [Serializable]class Col{public string type,name;public Bounds bounds;public int layer;public bool trigger,enabled;}
 [Serializable]class Content{public string id,name,source,scene;public Vector3 position;public Bounds[] renderers;public Col[] colliders;}
 [Serializable]class Row{public string name,status,id;public float before,after;public Vector3 hero,camera;public Quaternion cameraRotation;}
 [Serializable]class Report{public string scope="controlled actual forest resource/enemy geometry, disabled AI and initial camera/hero setup; not normal AI/native/human";public Content[] contents;public Row[] rows;}
 [Serializable]class Trace{public string sample,phase,aim;public int frame,pending,releaseFrame;public float time,hp;public long action;public Vector3 hero,camera;public Quaternion cameraRotation;public Vector3[] projectiles;}
 static void C(bool ok,string name,Component target=null,float before=0,float after=0){rows.Add(new Row{name=name,status=ok?"PASS":"FAIL",id=target is AdventureResource n?n.StableId:target is CombatEnemy e?e.entityId:"",before=before,after=after,hero=h.transform.position,camera=h.cameraTransform.position,cameraRotation=h.cameraTransform.rotation});}
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+240;EditorSceneManager.playModeStartScene=null;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("AIM baseline bounded deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f||frame==Time.frameCount)return;frame=Time.frameCount;if(cases==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(root+"/Store",false);s.Saves.enabled=false;h=s.hero;combat=h.GetComponent<AdventureCombat>();gear=h.GetComponent<AdventureLoadout>();cases=Cases();}if(!cases.MoveNext())End();}catch(Exception e){File.WriteAllText(root+"/exception.txt",e.ToString());rows.Add(new Row{name="fixture exception",status="FAIL"});End();}}
 static void End(){EditorApplication.update-=Tick;if(s!=null)s.Saves.SuspendEntry();Time.captureDeltaTime=0;File.WriteAllText(root+"/before.json",JsonUtility.ToJson(new Report{contents=contents.ToArray(),rows=rows.ToArray()},true));EditorApplication.Exit((rows.Count==0||contents.Count<3||rows.Any(r=>r.status=="FAIL"))?1:0);}
 static IEnumerator Wait(float seconds,Component target){float end=Time.time+seconds;while(Time.time<end){File.AppendAllText(root+"/timeline.jsonl",JsonUtility.ToJson(new Trace{sample=sample,aim=combat.GetType().GetProperty("LastReleaseAim")==null?null:JsonUtility.ToJson(combat.GetType().GetProperty("LastReleaseAim").GetValue(combat)),frame=Time.frameCount,time=Time.time,phase=combat.LastAttackPhase,pending=combat.PendingAttackCount,action=combat.LastAttackId,releaseFrame=combat.LastReleaseFrame,hp=HP(target),hero=h.transform.position,camera=h.cameraTransform.position,cameraRotation=h.cameraTransform.rotation,projectiles=UnityEngine.Object.FindObjectsByType<AdventureProjectile>(FindObjectsSortMode.None).Select(p=>p.transform.position).ToArray()})+"\n");yield return null;}}
 static float HP(Component c)=>c is AdventureResource n?n.Health:c is CombatEnemy e?e.Health.Current:0;
 static Bounds Body(Component c){var b=c.GetComponentsInChildren<Collider>().First(x=>x.enabled&&!x.isTrigger).bounds;foreach(var col in c.GetComponentsInChildren<Collider>().Where(x=>x.enabled&&!x.isTrigger))b.Encapsulate(col.bounds);return b;}
 static void Capture(Component c){contents.Add(new Content{id=c is AdventureResource n?n.StableId:((CombatEnemy)c).entityId,name=c.name,scene=c.gameObject.scene.name,source=c is AdventureResource?"ForestRegionRuntime.Build: cloned existing scene template per MaterialKind":"ForestRegionRuntime/Waves.CreateEncounter existing actor",position=c.transform.position,renderers=c.GetComponentsInChildren<Renderer>().Select(r=>r.bounds).ToArray(),colliders=c.GetComponentsInChildren<Collider>().Select(x=>new Col{type=x.GetType().Name,name=x.name,bounds=x.bounds,layer=x.gameObject.layer,trigger=x.isTrigger,enabled=x.enabled}).ToArray()});}
 static void Place(Component target,GearType weapon,float distance,float lateral=0,float yaw=0){combat.ResetCombat();if(target is AdventureResource n)n.Restore(n.maxHealth);if(target is CombatEnemy e)e.ResetEnemy();h.Progression.Clear();h.Health.ResetHealth();h.Dodge.ResetDodge();h.Inventory.SetOpen(false);h.HudPanelOpen=h.SaveMenuOpen=false;Time.timeScale=1;PlayerInput.Focus(true);gear.mainWeapon=gear.items.First(x=>x.type==weapon).id;gear.Refresh();var p=target.transform.position-Vector3.forward*distance-Vector3.right*lateral;h.RestorePosition(p,yaw);Physics.SyncTransforms();h.cameraTransform.position=p+new Vector3(0,2.8f,-3);h.cameraTransform.LookAt(Body(target).center);}
 static IEnumerator Shoot(Component target,GearType weapon,float distance,float lateral=0,float yaw=0){Place(target,weapon,distance,lateral,yaw);yield return null;yield return null;float hp=HP(target);var allNodes=UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).Where(n=>n.gameObject.scene==s.gameObject.scene&&n.Health>0).ToDictionary(n=>n,n=>n.Health);sample=weapon+"/"+target.name+"/"+distance+"/"+lateral+"/"+yaw;bool accepted=combat.Strike();C(accepted,"actual attack accepted "+sample);var w=Wait(1.1f,target);while(w.MoveNext())yield return null;bool foreground=target is AdventureResource ore&&ore.StableId=="node:41"&&distance==5;C(foreground?HP(target)==hp&&allNodes.Any(p=>p.Key!=target&&p.Key.Health<p.Value):HP(target)<hp,(foreground?"foreground physical resource intercepts visible ore ":"camera-target actual damage ")+sample,target,hp,HP(target));}
 static IEnumerator Cases(){Time.captureDeltaTime=.02f;AdventureRandom.SetNextSeedForValidation(418);C(s.Flow.Depart(AdventureMode.ForestExploration),"real forest departure");h.enabled=false;s.orbit.enabled=false;foreach(var e in s.enemies){e.enabled=false;var b=e.GetComponent<ForestBossBrain>();if(b!=null)b.enabled=false;}var nodes=UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).Where(n=>n.gameObject.activeInHierarchy&&n.entityId!=null&&n.entityId.StartsWith("node:")).ToArray();var stone=nodes.Single(n=>n.StableId=="node:22");var ore=nodes.Single(n=>n.StableId=="node:41");var enemy=s.enemies.Single(e=>e.entityId=="enemy:1");Capture(stone);Capture(ore);Capture(enemy);
 foreach(var target in new Component[]{stone,ore,enemy})foreach(var weapon in new[]{GearType.Bow,GearType.Staff}){var w=Shoot(target,weapon,5);while(w.MoveNext())yield return null;w=Shoot(target,weapon,2.6f,0,90);while(w.MoveNext())yield return null;}
 var near=Shoot(enemy,GearType.Bow,1);while(near.MoveNext())yield return null;
 // Retain actual resource collider shape/position; only the initial hero offset changes.
 var spear=Shoot(stone,GearType.Spear,2.2f,.65f);while(spear.MoveNext())yield return null;
 Place(enemy,GearType.Bow,5);yield return null;yield return null;PlayerInput.Focus(false);C(!combat.Strike(),"focus loss rejects new domain attack");combat.ResetCombat();PlayerInput.Focus(true);
 var t=new CraftingTarget{type=GearType.Sword,quality=GearQuality.White};C(CraftingTargets.Name(t)==new GearItem{type=t.type,quality=t.quality}.Name,"target equals actual item quality name");
 }
}
