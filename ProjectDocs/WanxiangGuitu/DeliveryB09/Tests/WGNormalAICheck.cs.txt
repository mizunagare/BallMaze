  Aim(enemy.transform.position);if(Time.time>shot){shot=Time.time+.53f;s.hero.GetComponent<AdventureCombat>().Strike();}if(Vector3.Distance(enemy.transform.position,s.hero.transform.position)<5){var away=s.hero.transform.position+Vector3.ProjectOnPlane(s.hero.transform.position-enemy.transform.position,Vector3.up).normalized*4;if(region.TryGround(away,3,out var p)){s.hero.Dodge.TryDodge(p-s.hero.transform.position);Walk(p,.6f);}}return true;
 }

 static void Finish(int stage){Need(s.Saves.SaveProgress(false),"checkpoint");expected=Snapshot(stage);File.WriteAllText(Root+"/b08-expected.json",JsonUtility.ToJson(expected,true));File.WriteAllText(Root+"/b08-checkpoint-"+stage+".json",JsonUtility.ToJson(expected,true));ValidationStorage.Record("b08-normal-finish",s.Saves.SavePath);Mark("CHECKPOINT "+stage);EditorApplication.update-=Tick;EditorApplication.Exit(0);}
 static void Photo(string name){ScreenCapture.CaptureScreenshot(Root+"/b08-normal-"+name+".png");}

  if(s==null){var entry=GameEntry.Instance;if(entry!=null&&!GameEntry.Loading){var menu=entry.GetComponent<TitleMenu>();if(menu!=null&&menu.Canvas!=null){string action=continuing?"继续游戏":(menu.Page=="新游戏确认"?"确认建立新档案":"新游戏");var button=menu.Canvas.GetComponentsInChildren<UnityEngine.UI.Button>().FirstOrDefault(b=>b.name==action);Need(button!=null&&button.interactable,"actual menu button "+action);button.onClick.Invoke();Mark("CONTROLLED_UGUI_ENTRY "+action);return;}}s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||!s.Saves.IsInitialized){s=null;return;}var type=typeof(B06OriginalPlay);type.GetMethod("Size",System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Static).Invoke(null,null);view=(EditorWindow)type.GetField("view",System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Static).GetValue(null);region=s.GetComponent<ForestRegionRuntime>()??s.gameObject.AddComponent<ForestRegionRuntime>();if(!File.Exists(Root+"/b08-normal-ai.csv"))File.WriteAllText(Root+"/b08-normal-ai.csv","time,phase,tutorial,moved,x,z,hp,wood,stone,ore,focus,pathDistance,reachable\n");
   if(continuing){expected=JsonUtility.FromJson<B08ExpectedNormal>(File.ReadAllText(Root+"/b08-expected.json"));firstRun=expected.firstRun;File.WriteAllText(Root+"/b08-actual-resume-"+expected.stage+".json",JsonUtility.ToJson(Snapshot(expected.stage),true));Need(JsonUtility.ToJson(Snapshot(expected.stage))==JsonUtility.ToJson(expected),"new process tutorial/run/route/streams/ledger/gear/profile changed");Mark("EXACT_NEW_PROCESS stage="+expected.stage);if(expected.stage==2){Need(!Onboarding.Active(s)&&Onboarding.State(s).stage==TutorialStage.Complete&&s.Flow.ActiveRun.mode==AdventureMode.ForestExploration&&s.Flow.ActiveRun.runId!=firstRun,"successor tutorial restore");Mark("ALL_PASS B08 normal NPC AI, genuine gathering, settlement, workbench, second forest continuation");Finish(3);return;}Need(Onboarding.State(s).stage==TutorialStage.FreeExit,"continue free extraction");exitId=s.Flow.ExitConfigs().First(e=>!e.repairRequired).id;phase=20;return;}
   Need(Onboarding.Active(s)&&s.Flow.AtCamp,"fresh teaching camp");Need(CraftingTargets.Select(s.hero,GearType.Sword,GearQuality.White),"real crafting target");s.Camp.Explore();phase=1;return;
  }
  if(EditorApplication.timeSinceStartup>=nextLog){nextLog=EditorApplication.timeSinceStartup+.5;File.AppendAllText(Root+"/b08-normal-ai.csv",$"{Time.time:0.00},{phase},{Onboarding.State(s)?.stage},{Onboarding.State(s)?.moved:0.00},{s.hero.transform.position.x:0.00},{s.hero.transform.position.z:0.00},{s.hero.Health.Current:0.0},{s.hero.Inventory.Count(MaterialKind.Wood)},{s.hero.Inventory.Count(MaterialKind.Stone)},{s.hero.Inventory.Count(MaterialKind.Ore)},{s.GetComponent<OnboardingFocus>().TargetId},{s.GetComponent<OnboardingFocus>().Distance:0.0},{s.GetComponent<OnboardingFocus>().Reachable}\n");}
  Need(!s.hero.Health.Dead,"normal hero died");if(s.hero.Progression.IsChoosing)s.hero.Progression.SelectChoice(0);if(s.hero.Progression.PendingChoices>0)return;
  if(phase==1){if(s.hero.Inventory.IsOpen){if(facility==1){var g=s.hero.GetComponent<AdventureLoadout>();Need(g.SelectMain(g.items.First(i=>i.type==GearType.Bow))&&g.SelectReserve(g.items.First(i=>i.type==GearType.Sword)),"equipment selection");}s.Camp.Explore();return;}if(facility<3){string id=CampRuntime.FacilityIds[facility];if(Walk(s.Camp.FacilityPosition(id),.8f)&&Interact(id)){Mark("ACTUAL_FACILITY "+id);facility++;}return;}string departure="camp:departure";if(Walk(s.Camp.FacilityPosition(departure),.8f)&&Interact(departure)){Photo("camp-departure");Need(Onboarding.State(s).stage==TutorialStage.Departure&&s.Flow.Preparing,"preparation facility");AdventureRandom.SetNextSeedForValidation(745672572);Need(s.Flow.Depart(AdventureMode.ForestExploration),"real forest deployment");firstRun=s.Flow.ActiveRun.runId;Mark("FIRST_FOREST "+firstRun);phase=2;}return;}

  if(phase==2){var tutorial=Onboarding.State(s);if(tutorial.stage==TutorialStage.FreeExit){Photo("earned-materials");exitId=s.Flow.ExitConfigs().First(e=>!e.repairRequired).id;phase=10;return;}var focus=s.GetComponent<OnboardingFocus>();focus.Refresh();
   if(tutorial.stage==TutorialStage.Move){var point=region.Layout.entry+Vector3.forward*5;Walk(point,.5f);return;}Need(focus.Reachable&&focus.TargetId!=null,"no safe resource hint "+focus.Hint);var loot=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).FirstOrDefault(l=>l.InteractionId==focus.TargetId);if(loot!=null){if(Walk(focus.Position,.8f)&&loot.CanPickUp(s.hero.Inventory)){s.hero.Inventory.Interactions.Refresh();s.hero.Inventory.Interactions.Present();var offer=s.hero.Inventory.Interactions.DisplayedE;if(offer!=null&&offer.action==InteractionAction.Pickup){Need(s.hero.Inventory.Interactions.ExecuteDisplayed(false)&&s.hero.Inventory.Interactions.LastExecutedId==offer.targetId,"displayed pickup execution");Mark("ACTUAL_PICKUP "+offer.title);}}return;}
   node=UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).FirstOrDefault(n=>n.StableId==focus.TargetId);Need(node!=null,"hint resource missing");if(s.hero.GetComponent<AdventureLoadout>().Weapon!=GearType.Sword){s.hero.GetComponent<AdventureCombat>().SwitchWeapon();return;}if(Walk(node.transform.position,1.65f)){Aim(node.transform.position);if(Time.time>shot){shot=Time.time+.53f;float before=node.Health;bool hit=s.hero.GetComponent<AdventureCombat>().Strike();File.AppendAllText(Root+"/b08-attacks.csv",Time.time+","+node.StableId+","+node.kind+",Sword,"+Vector3.Distance(s.hero.transform.position,node.transform.position)+","+before+","+node.Health+","+hit+"\n");}}return;
