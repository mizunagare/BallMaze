using System;
using System.IO;
using System.Linq;
using System.Reflection;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
public static class B05ScenarioCheck
{
 static string Root=>System.Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")??throw new System.Exception("explicit root required");static CombatSession s;static int phase;static double deadline;static float countdown;static double wait;static string reward,run;static string[] entry;static int[] remaining;
 static AdventureFlow F=>s.Flow;static AdventureLoadout G=>s.hero.GetComponent<AdventureLoadout>();static PlayerInventory B=>s.hero.Inventory;
 static void Need(bool b,string m){if(!b)throw new Exception(m);}
 static void Mark(string id,string text){Debug.Log("B05_"+id+"_PASS "+text);File.AppendAllText(Root+"/scenario-results.txt",id+" PASS "+text+"\n");}
 public static void Run(){deadline=EditorApplication.timeSinceStartup+180;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void FreezeEnemies(){foreach(var e in s.enemies){e.enabled=false;var boss=e.GetComponent<BossBrain>();if(boss!=null)boss.enabled=false;}}
 static void End(){typeof(CombatSaveSystem).GetField("initialized",BindingFlags.Instance|BindingFlags.NonPublic).SetValue(s.Saves,false);EditorApplication.update-=Tick;EditorApplication.Exit(0);}
 static void Tick(){try{
 if(EditorApplication.timeSinceStartup>deadline)throw new Exception("B05 scenario timeout phase="+phase);if(!EditorApplication.isPlaying)return;if(s==null){if(Time.time<.4f)return;s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;}
 if(phase==0){s.Saves.InitializeStorage(Root+"/scenario",false);s.hero.enabled=false;B.RestoreInventory(new[]{3,6,0,10,10,10},2,true,0);Need(F.Depart(AdventureMode.TestChallenge),"New v8 departure: "+F.DepartureReason);entry=G.items.Select(i=>i.id).OrderBy(i=>i).ToArray();s.RestoreWave(5);FreezeEnemies();Need(s.enemies.Any(e=>e.archetype==EnemyArchetype.Boss&&!e.Health.Dead),"Boss fixture missing");s.hero.RestorePosition(new Vector3(25,.1f,25),0);Need(!F.BeginExtraction()&&!s.Camp.InCamp,"Distant extraction bypass");Need(!s.Camp.Enter()&&!s.Camp.InCamp&&F.ActiveRun!=null,"Old H camp bypass");Mark("V04","Remote exit/old camp path cannot transfer goods");s.hero.RestorePosition(F.ExitPosition,0);Need(F.BeginExtraction()&&F.ActiveRun.extractionRemaining==5&&s.enemies.Any(e=>!e.Health.Dead),"Live enemies blocked 5 sec exit");Mark("V02","5-second ordinary extraction starts with wave-five boss alive");B.SetOpen(true);countdown=F.ActiveRun.extractionRemaining;wait=EditorApplication.timeSinceStartup+.25;phase=1;return;}
 if(phase==1){if(EditorApplication.timeSinceStartup<wait)return;Need(F.ActiveRun.extractionRemaining==countdown,"Pause advanced extraction");B.SetOpen(false);s.hero.Health.Damage(2);Need(F.ActiveRun.phase==RunPhase.Running&&F.Settlement==null,"Damage did not cancel");Need(F.BeginExtraction(),"Restart after hit");s.hero.RestorePosition(F.ExitPosition+Vector3.right*5,0);F.SendMessage("Update");Need(F.ActiveRun.phase==RunPhase.Running,"Range did not cancel");s.hero.RestorePosition(F.ExitPosition,0);Need(F.BeginExtraction()&&!F.BeginExtraction()&&F.ActiveRun.phase==RunPhase.Running,"Manual cancel failed");Mark("V03","Pause freezes; damage/range/manual cancellation preserves same Run and no reward");
 var p=LootPickup.SpawnGear(AdventureItems.New(GearType.Accessory,GearQuality.Red),s.hero.transform.position+Vector3.forward);reward=p.Gear.id;Need(p.PickUp(B),"Gear pickup");var wood=LootPickup.Spawn(MaterialKind.Wood,9,s.hero.transform.position+Vector3.right);Need(wood.PickUp(B),"Material pickup");Need(F.BeginExtraction(),"Extraction with boss alive");phase=2;return;}
 if(phase==2){if(F.Settlement==null)return;Need(F.Settlement.outcome==AdventureOutcome.Extracted&&F.CommitSettlement()&&F.CommitSettlement()&&F.AcknowledgeSettlement()&&s.Saves.LoadProgress(false),"Success commit/reload");Need(F.Profile.warehouse.Count(i=>i.id==reward)==1&&F.Profile.materials[3]==9,"Duplicate/lost rewards");Mark("V01","Actual timed exit with living boss; reward one copy after reload");Mark("V10","Repeated commit/reopen no duplicated gear/material");
 Need(s.Camp.WithdrawEquipment(reward)&&s.Camp.DepositEquipment(reward),"Camp gear store/withdraw");Need(s.Camp.TransferMaterial(MaterialKind.Wood,9,false),"Camp materials withdraw");Need(G.Craft(GearType.Sword,GearQuality.White),"Camp manufacture after exit");Need(F.Depart(AdventureMode.TestChallenge),"Redeployment");FreezeEnemies();run=F.ActiveRun.runId;
 var armor=AdventureItems.New(GearType.Armor,GearQuality.Red);Need(G.Add(armor)&&G.AutoMove(armor)&&armor.board&&!F.Protects(armor.id),"New loot wear setup");var bag=AdventureItems.NewContainer("bag_relic");Need(G.Add(bag)&&G.EquipContainer(bag)&&!F.Protects(bag.id),"New loot container wear setup");Need(G.items.Single(i=>i.id==bag.id).origin==ItemOrigin.RunLoot,"Container replacement changed origin");Need(!s.Camp.Knows("gear_armor_4")&&!s.Camp.Knows("bag_relic"),"Loot unlocked permanent recipe");Mark("V09","New red gear and container do not unlock recipes before settlement");
 B.Add(MaterialKind.Wood,5);B.TryRemove(MaterialKind.Wood,7);s.hero.Health.Damage(30);Need(B.UseHealthPotion()&&F.ActiveRun.broughtPotions==1,"Brought potion was not consumed");remaining=(int[])F.ActiveRun.broughtRemaining.Clone();s.hero.Health.Damage(1000000);F.ResolveTerminal();Need(F.Settlement.lostGear.Length==2&&F.CommitSettlement()&&F.AcknowledgeSettlement(),"Worn loot death settlement");Need(!G.items.Any(i=>i.id==armor.id||i.id==bag.id)&&entry.All(id=>G.items.Any(i=>i.id==id))&&B.HealthPotionCount==1&&Enumerable.Range(0,6).All(k=>B.Count((MaterialKind)k)==remaining[k]),"Worn loot protected or consumed stock restored");Mark("V05","New worn armor and backpack lost; all entry identities returned");Mark("V06","Earned-first material spend and used brought potion not restored");Need(F.Profile.warehouse.Count(i=>i.id==reward)==1&&s.Saves.SaveProgress(false),"Death rolled back earlier warehouse");
 Need(F.Depart(AdventureMode.EndlessWaves),"Capacity depart");FreezeEnemies();B.TryRemove(MaterialKind.MonsterHide,B.Count(MaterialKind.MonsterHide));B.TryRemove(MaterialKind.CrystalShard,B.Count(MaterialKind.CrystalShard));B.TryRemove(MaterialKind.Wood,B.Count(MaterialKind.Wood));B.TryRemove(MaterialKind.Stone,B.Count(MaterialKind.Stone));B.TryRemove(MaterialKind.Ore,B.Count(MaterialKind.Ore));Need(B.Add(MaterialKind.Wood,495),"Fill supply slots");Need(F.SupplyUsed()==6,"Supply stack count");var full=LootPickup.Spawn(MaterialKind.Ore,1,s.hero.transform.position+Vector3.forward);Need(!full.PickUp(B)&&!full.Collected&&B.Count(MaterialKind.Ore)==0,"Full supplies swallowed material");Mark("V15_SUPPLIES","6 stacks full: failed pickup leaves ground source untouched");F.OpenAbandon();Need(F.ConfirmAbandon()&&F.Settlement.outcome==AdventureOutcome.Abandoned&&F.CommitSettlement()&&F.AcknowledgeSettlement(),"Abandon outcome");Mark("V16","Abandon uses failure result; exits/checkpoints tested separately");End();}
 }catch(Exception e){Debug.LogException(e);if(s!=null)typeof(CombatSaveSystem).GetField("initialized",BindingFlags.Instance|BindingFlags.NonPublic).SetValue(s.Saves,false);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
