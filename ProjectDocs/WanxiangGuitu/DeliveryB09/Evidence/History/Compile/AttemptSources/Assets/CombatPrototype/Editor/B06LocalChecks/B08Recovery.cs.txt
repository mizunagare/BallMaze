using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class B08Recovery {
 static CombatSession s;static int phase;static double deadline;static bool resume;static string Root=>ValidationStorage.ValidateRoot(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT"));
 static void Need(bool b,string m){if(!b)throw new Exception(m);}static void Mark(string m){File.AppendAllText(Root+"/b08-recovery.txt",m+"\n");}
 public static void Run(){Start(false);}public static void Resume(){Start(true);}static void Start(bool load){resume=load;ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+240;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Freeze(){foreach(var e in s.enemies){e.enabled=false;if(e.GetComponent<ForestBossBrain>()!=null)e.GetComponent<ForestBossBrain>().enabled=false;}}
 static void Quit(){EditorApplication.update-=Tick;EditorApplication.Exit(0);}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("B08 recovery deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.5f)return;
  if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.autoSaveInterval=3600;s.Saves.InitializeStorage(Root+"/b08-recovery-save",resume);Freeze();if(resume){Need(Onboarding.State(s).skipped&&!Onboarding.Active(s)&&s.Flow.ActiveRun!=null&&s.Flow.ActiveRun.mode==AdventureMode.ForestExploration,"skip resume state");Need(s.Flow.ActiveRun.runId==File.ReadAllText(Root+"/b08-skipped-run.txt"),"skipped successor identity");Mark("PASS NEW_PROCESS_SKIPPED_SUCCESSOR_NO_REPLAY");Quit();return;}
   Need(s.Flow.Depart(AdventureMode.ForestExploration),"death fixture departure");Freeze();for(int i=0;i<8;i++)s.hero.MoveCharacter(Vector3.forward*.5f);Need(Onboarding.State(s).stage==TutorialStage.Harvest,"actual move before death");s.hero.Health.Damage(99999);s.Flow.ResolveTerminal();Need(s.Flow.Settlement?.outcome==AdventureOutcome.Death&&Onboarding.State(s).stage==TutorialStage.Departure&&Onboarding.State(s).firstRunId==null,"death tutorial recovery");Need(s.Flow.CommitSettlement()&&s.Flow.AcknowledgeSettlement(),"death return");Mark("PASS REAL_DEATH_RETURNS_TO_PREPARATION");
   Need(s.Flow.Depart(AdventureMode.ForestExploration),"abandon fixture departure");Freeze();s.Flow.OpenAbandon();Need(s.Flow.ConfirmAbandon()&&s.Flow.Settlement.outcome==AdventureOutcome.Abandoned,"actual abandon");Need(s.Flow.CommitSettlement()&&s.Flow.AcknowledgeSettlement()&&Onboarding.State(s).stage==TutorialStage.Departure,"abandon tutorial recovery");Mark("PASS ABANDON_NO_TUTORIAL_REWARD");
   Need(Onboarding.Skip(s),"skip configured flow");Need(s.Flow.Depart(AdventureMode.ForestExploration),"skipped departure");Freeze();foreach(var kind in new[]{MaterialKind.Wood,MaterialKind.Stone,MaterialKind.Ore}){var node=UnityEngine.Object.FindObjectsByType<AdventureResource>(FindObjectsSortMode.None).Where(n=>n.kind==kind&&n.Health>0).First();s.hero.RestorePosition(node.transform.position+Vector3.back*1.3f,0);node.Hit(node.maxHealth,s.hero);var loot=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Where(l=>l.Kind==kind&&l.Gear==null).OrderBy(l=>Vector3.Distance(l.transform.position,node.transform.position)).First();Need(loot.PickUp(s.hero.Inventory),"skipped true pickup");}
   var region=s.GetComponent<ForestRegionRuntime>();var exit=s.Flow.ExitConfigs().First(e=>!e.repairRequired);Need(region.TryInteractionApproach(exit,out var p),"skip exit");s.hero.RestorePosition(p,0);Need(s.Flow.BeginExtraction(exit.id),"skipped true extraction");phase=1;return;
  }
  if(phase==1&&s.Flow.HasResult){Need(s.Flow.Settlement.outcome==AdventureOutcome.Extracted&&s.Flow.CommitSettlement()&&s.Flow.AcknowledgeSettlement(),"skip settlement");foreach(var cost in CraftingCosts.Gear(GearQuality.White))Need(s.Camp.TransferMaterial(cost.material,cost.amount,false),"skip real withdrawal");Need(s.Camp.Manufacture(GearType.Sword,GearQuality.White)&&Onboarding.State(s).skipped&&!Onboarding.Active(s),"skip real manufacture");Need(s.Flow.Depart(AdventureMode.ForestExploration),"skip second departure");Freeze();Need(s.Saves.SaveProgress(false),"skip final save");File.WriteAllText(Root+"/b08-skipped-run.txt",s.Flow.ActiveRun.runId);Mark("PASS SKIPPED_REAL_EXTRACTION_STORE_CRAFT_SECOND_FOREST");Quit();}
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
