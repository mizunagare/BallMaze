using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEngine.AI;using UnityEditor;using UnityEditor.SceneManagement;
public static class BAL01Matrix {
 static string root;static CombatSession s;static ThirdPersonHero h;static AdventureLoadout g;static IEnumerator cases;static int frame=-1;static double deadline;static List<Row> rows=new List<Row>();static List<RunRecord> runs=new List<RunRecord>();
 [Serializable]public class RunRecord{public int seed,rep;public string run,profile,layout,streams;}
 [Serializable]public class Row{public string sample,run,profile,type,policy,outcome,gear,initialPositions,initialStreams,finalStreams,initialRanks,finalRanks;public int seed,rep,initialLevel,finalLevel,kills,enemyCount,heals,dodges,playerAttacks,windups,releases,hitActions,damageEvents,dodgeBlocks,experience,projectilesAtCutoff;public float initialHP,initialDefense,initialPower,elapsed,ttk=-1,damage,hitRate;public bool xpMatchesWhenSurviving;}
 [Serializable]class Report{public string scope="controlled enabled actual AI, real combat/drop/XP; setup teleports and fixture resets are NOT normal AI/native/human. Same frozen conditional policy, actual frame events archived; six active-second window, uncompleted kills censored, not zero TTK";public RunRecord[] runs;public Row[] rows;}
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+1100;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();Application.runInBackground=true;if(EditorApplication.timeSinceStartup>deadline)throw new Exception("bounded matrix deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f||frame==Time.frameCount)return;frame=Time.frameCount;if(cases==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;h=s.hero;g=h.GetComponent<AdventureLoadout>();h.enabled=false;s.orbit.enabled=false;cases=Cases();}if(!cases.MoveNext())End(0);}catch(Exception e){File.WriteAllText(root+"/matrix-exception.txt",e.ToString());End(1);}}
 static void End(int code){EditorApplication.update-=Tick;Time.captureDeltaTime=0;if(s!=null)s.Saves.SuspendEntry();File.WriteAllText(root+"/matrix.json",JsonUtility.ToJson(new Report{runs=runs.ToArray(),rows=rows.ToArray()},true));EditorApplication.Exit(code);}
 static IEnumerator Cases(){Time.captureDeltaTime=.04f;PlayerInput.Focus(true);
 foreach(int seed in new[]{418,419,420})for(int rep=0;rep<3;rep++){
  s.Saves.InitializeStorage(root+"/Runs/"+seed+"-"+rep,false);s.Saves.enabled=false;AdventureRandom.SetNextSeedForValidation(seed);if(!s.Flow.Depart(AdventureMode.ForestExploration))throw new Exception("real forest departure "+s.Flow.Error);var region=s.GetComponent<ForestRegionRuntime>();var id=s.Flow.ActiveRun.runId;var profile=s.Flow.Profile.profileId;var runRecord=new RunRecord{seed=seed,rep=rep,run=id,profile=profile,layout=JsonUtility.ToJson(region.Capture()),streams=JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams)};runs.Add(runRecord);
  // The valid initial v9 Run is saved. Arena resets below are deliberately not persisted.
  s.Flow.BeginLegacyRestore();var all=s.enemies.ToArray();var homes=all.ToDictionary(e=>e,e=>e.transform.position);var initialGear=g.items.Select(i=>i.Clone()).ToArray();string main=g.mainWeapon,reserve=g.reserveWeapon;
  foreach(string kind in new[]{"Melee","Ranged","Elite","Warden","Tide","Mixed"})foreach(string policy in rep==0?new[]{"Stationary","CorrectDodge"}:rep==1?new[]{"Walk","WrongDodge"}:new[]{"Stationary","Surrounded"}){
   foreach(var e in all)e.gameObject.SetActive(false);EnemyProjectile.Clear(s.gameObject.scene);foreach(var p in UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None))UnityEngine.Object.Destroy(p.gameObject);yield return null;
   g.Restore(initialGear,main);g.RestoreReserve(reserve);int level=rep==0?1:5;int[] ranks=rep==0?new int[6]:new[]{2,1,1,0,0,0};h.Progression.RestoreProgress(level,0,ranks,new string[0],new int[0]);h.Health.ResetHealth();h.Dodge.ResetDodge();h.Inventory.RestoreInventory(new int[6],2,true,0);h.Inventory.SetOpen(false);h.SaveMenuOpen=h.HudPanelOpen=false;Time.timeScale=1;
   if(rep==2){var armor=AdventureItems.New(GearType.Armor,GearQuality.Purple);if(!g.Add(armor)||!g.AutoMove(armor)||!armor.board)throw new Exception("real armor fixture cannot fit");}
   CombatEnemy primary=kind=="Warden"?all.Single(e=>e.GetComponent<ForestBossBrain>()?.Kind==ForestBossKind.Warden):kind=="Tide"?all.Single(e=>e.GetComponent<ForestBossBrain>()?.Kind==ForestBossKind.Tide):all.First(e=>e.GetComponent<ForestBossBrain>()==null&&e.archetype==(kind=="Ranged"?EnemyArchetype.Ranged:kind=="Elite"?EnemyArchetype.Elite:EnemyArchetype.Melee));
   var cohort=new List<CombatEnemy>{primary};if(kind=="Mixed"){cohort.Add(all.First(e=>e.GetComponent<ForestBossBrain>()==null&&e.archetype==EnemyArchetype.Ranged));cohort.Add(all.First(e=>e.GetComponent<ForestBossBrain>()==null&&e.archetype==EnemyArchetype.Elite));}if(policy=="Surrounded")cohort.AddRange(all.Where(e=>!cohort.Contains(e)&&e.GetComponent<ForestBossBrain>()==null).Take(2));
   Vector3 center=homes[primary];if(!NavMesh.SamplePosition(center+Vector3.back*2.1f,out var nav,4,NavMesh.AllAreas))throw new Exception("hero sample nav");h.RestorePosition(nav.position+Vector3.up*.1f,0);
   for(int i=0;i<cohort.Count;i++){var e=cohort[i];e.gameObject.SetActive(true);e.enabled=true;var where=i==0?homes[e]:h.transform.position+Quaternion.Euler(0,(i-1)*130+65,0)*Vector3.forward*2.8f;if(NavMesh.SamplePosition(where,out var p,3,NavMesh.AllAreas))where=p.position+Vector3.up*.1f;e.SetSpawn(where);e.ResetEnemy();e.transform.position=where;var guard=e.GetComponent<EncounterGuard>();if(guard!=null){guard.Configure(null,new ForestEncounter{id=e.entityId,home=where,poi="controlled",type=(int)e.archetype});guard.enabled=false;}var old=e.GetComponent<BossBrain>();if(old!=null)old.enabled=false;}
   Physics.SyncTransforms();g.Refresh();h.GetComponent<AdventureCombat>().ResetCombat();yield return null;
   string sample=seed+"-"+rep+"-"+kind+"-"+policy;BAL01Trace.Reset(root+"/Samples/"+sample);var row=new Row{sample=sample,seed=seed,rep=rep,run=id,profile=profile,type=kind,policy=policy,enemyCount=cohort.Count,initialLevel=h.Progression.Level,initialHP=h.Health.Current,initialDefense=h.Defense,initialPower=h.GetComponent<AdventureCombat>().BasicPower,initialRanks=string.Join(",",h.Progression.CaptureRanks()),initialPositions=string.Join("|",cohort.Select(e=>e.entityId+":"+e.transform.position))+"/hero:"+h.transform.position,gear=string.Join("|",g.items.Select(i=>i.id+":"+i.type+":"+i.quality+":"+i.board)),initialStreams=JsonUtility.ToJson(UnityEngine.Random.state)};
   float started=Time.time,stop=started+6;float taken=h.Health.TotalDamageTaken;var combat=h.GetComponent<AdventureCombat>();
   while(Time.time<stop&&!h.Health.Dead&&cohort.Any(e=>!e.Health.Dead)){
    if(h.Progression.PendingChoices>0){if(!h.Progression.IsChoosing)h.Progression.BeginChoice();if(h.Progression.IsChoosing)h.Progression.SelectChoice(0);}
    if(h.Health.Current<h.Health.maxHealth*.4f&&h.Inventory.UseHealthPotion())row.heals++;
    var closest=cohort.Where(e=>!e.Health.Dead).OrderBy(e=>Vector3.Distance(e.transform.position,h.transform.position)).First();var toward=Vector3.ProjectOnPlane(closest.transform.position-h.transform.position,Vector3.up);if(toward.sqrMagnitude>.01f&&!h.Dodge.IsRolling)h.transform.rotation=Quaternion.LookRotation(toward);
    bool correct=policy=="CorrectDodge"&&cohort.Any(e=>{var boss=e.GetComponent<ForestBossBrain>();return boss!=null?boss.State==ForestBossState.Telegraph&&boss.Remaining<=.18f:e.WindingUp&&(float)typeof(CombatEnemy).GetField("strikeAt",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(e)-Time.time<=.18f;});
    bool wrong=policy=="WrongDodge"&&cohort.Any(e=>e.WindingUp);
    if((correct||wrong)&&h.Dodge.TryDodge(correct?h.transform.right:toward))row.dodges++;
    if(h.Dodge.IsRolling)h.MoveCharacter(h.Dodge.FrameDisplacement(Time.deltaTime));else if(policy=="Walk"){var direction=toward.magnitude>1.4f?toward.normalized:Vector3.Cross(Vector3.up,toward.normalized);h.MoveCharacter(direction*h.moveSpeed*h.MoveMultiplier*Time.deltaTime);}
    if(combat.Strike())row.playerAttacks++;yield return null;
   }
   // Let actual enemy Update perform the death/Xp payout; no direct reward submission.
   yield return null;row.elapsed=Time.time-started;row.kills=cohort.Count(e=>e.Health.Dead);if(row.kills==cohort.Count)row.ttk=row.elapsed;row.outcome=h.Health.Dead?"PLAYER_DEAD":row.kills==cohort.Count?"ENEMIES_DEAD":"WINDOW_END_CENSORED";row.finalLevel=h.Progression.Level;row.damage=h.Health.TotalDamageTaken-taken;row.windups=BAL01Trace.Starts;row.releases=BAL01Trace.Releases;row.hitActions=BAL01Trace.HitActions;row.hitRate=row.releases==0?-1:(float)row.hitActions/row.releases;row.damageEvents=BAL01Trace.DamageEvents;row.dodgeBlocks=BAL01Trace.Dodges;row.experience=BAL01Trace.XP;row.xpMatchesWhenSurviving=h.Health.Dead||row.experience==cohort.Where(e=>e.Health.Dead).Sum(e=>e.experienceReward);row.finalRanks=string.Join(",",h.Progression.CaptureRanks());row.finalStreams=JsonUtility.ToJson(UnityEngine.Random.state);row.projectilesAtCutoff=UnityEngine.Object.FindObjectsByType<EnemyProjectile>(FindObjectsSortMode.None).Length;rows.Add(row);File.WriteAllText(root+"/matrix-progress.json",JsonUtility.ToJson(new Report{runs=runs.ToArray(),rows=rows.ToArray()},true));if(!row.xpMatchesWhenSurviving)throw new Exception("XP source ledger mismatch "+sample);BAL01Trace.Folder=null;
  }
 }
 }
}
