// Derived from B4Physics; only target sample count and enemy collision policy change. All geometry assertions retained.
using System;using System.IO;using System.Linq;using UnityEngine;using UnityEngine.AI;using UnityEditor;using UnityEditor.SceneManagement;
public static class Stage2Physics {
 [Serializable] class PathProof {public int requested,actual,steps;public string label;public Vector3[] corners;public Vector3 end;public float meters;}
 [Serializable] class Shape {public float radius,height,step,slope,skin,minimum;public Vector3 center;}
 static double deadline;static CombatSession s;static int seed=1;static System.Collections.Generic.HashSet<int> legalSeeds=new System.Collections.Generic.HashSet<int>();static CharacterController cc;
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT")??throw new Exception("isolated root required");
 static void Need(bool b,string why){if(!b)throw new Exception(why);}
 public static void Run(){ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+540;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Walk(Vector3 from,Vector3 to,string label){
  var path=new NavMeshPath();NavMeshHit a=default,b=default;Need(NavMesh.SamplePosition(from,out a,4,NavMesh.AllAreas)&&NavMesh.SamplePosition(to,out b,4,NavMesh.AllAreas)&&NavMesh.CalculatePath(a.position,b.position,NavMesh.AllAreas,path)&&path.status==NavMeshPathStatus.PathComplete,"navigation seed="+seed+" "+label);
  cc.enabled=false;cc.transform.position=a.position+Vector3.up*.1f;cc.enabled=true;Physics.SyncTransforms();
  int actualSteps=0;float moved=0;foreach(var corner in path.corners){int steps=0;while(Vector3.ProjectOnPlane(corner-cc.transform.position,Vector3.up).magnitude>.65f){var dir=Vector3.ProjectOnPlane(corner-cc.transform.position,Vector3.up).normalized;var before=cc.transform.position;cc.Move(dir*.2f+Vector3.down*.05f);moved+=Vector3.Distance(before,cc.transform.position);actualSteps++;Physics.SyncTransforms();Need(++steps<4000,"collision seed="+seed+" "+label+" at="+cc.transform.position);}}
  Need(Vector3.ProjectOnPlane(b.position-cc.transform.position,Vector3.up).magnitude<1,"endpoint seed="+seed+" "+label);File.AppendAllText(Root+"/stage2-physical-paths.jsonl",JsonUtility.ToJson(new PathProof{requested=seed,actual=s.GetComponent<ForestRegionRuntime>().Layout.seed,label=label,corners=path.corners,end=cc.transform.position,steps=actualSteps,meters=moved})+"\n");
 }
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("B4 physical deadline");if(!EditorApplication.isPlaying||Time.time<.4f)return;
  if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(Path.Combine(Root,"stage2-physics-save"),false);Need(s.Flow.Depart(AdventureMode.ForestExploration),"depart");cc=new GameObject("Stage2 controlled collision probe; live solid enemy bodies; NOT normal AI").AddComponent<CharacterController>();cc.gameObject.layer=s.hero.gameObject.layer;cc.transform.localScale=s.hero.transform.lossyScale;var heroController=s.hero.GetComponent<CharacterController>();cc.radius=heroController.radius;cc.height=heroController.height;cc.center=heroController.center;cc.stepOffset=heroController.stepOffset;cc.slopeLimit=heroController.slopeLimit;cc.skinWidth=heroController.skinWidth;cc.minMoveDistance=heroController.minMoveDistance;File.WriteAllText(Root+"/stage2-controller-shape.json",JsonUtility.ToJson(new Shape{radius=cc.radius,height=cc.height,center=cc.center,step=cc.stepOffset,slope=cc.slopeLimit,skin=cc.skinWidth,minimum=cc.minMoveDistance},true));Physics.IgnoreCollision(cc,s.hero.GetComponent<CharacterController>());}
  var region=s.GetComponent<ForestRegionRuntime>();region.GeneratePlayable(seed,true);var map=region.Capture();map.Validate();Need(map.bossVersion==1&&map.revision==5&&map.requestedSeed==seed,"requested seed not recorded "+seed);Need(map.generationAttempt==0||map.generationFailures.Length>0,"retry/fallback failure reasons absent");
  foreach(var enemy in s.enemies){Need(enemy.GetComponent<CharacterController>().enabled&&!Physics.GetIgnoreCollision(cc,enemy.GetComponent<CharacterController>()),"enemy collider must remain active and not ignored");enemy.enabled=false;if(enemy.GetComponent<ForestBossBrain>() is ForestBossBrain boss)boss.enabled=false;if(enemy.GetComponent<EncounterGuard>() is EncounterGuard guard)guard.enabled=false;}
  foreach(var edge in map.routes)Walk(map.points.Single(p=>p.id==edge.from).position,map.points.Single(p=>p.id==edge.to).position,"edge "+edge.from+">"+edge.to);
  foreach(var poi in map.points)Walk(map.entry,poi.position,"POI "+poi.id);
  foreach(var exit in map.exits){Need(region.TryInteractionApproach(exit,out var target),"exit interaction "+exit.id);Walk(map.entry,target,"exit "+exit.id);}
  foreach(var node in map.nodes)Need(region.TryGround(node.position,4,out var p)&&region.PathComplete(map.entry,p),"resource "+node.id);
  foreach(var enemy in map.encounters)Need(region.PathComplete(map.entry,enemy.home),"encounter "+enemy.id);
  File.AppendAllText(Path.Combine(Root,"stage2-physical.csv"),seed+","+map.seed+","+map.revision+","+map.bossVersion+","+map.routes.Length+",7,3,42,"+map.encounters.Length+","+map.generationAttempt+","+map.fallback+","+string.Join(";",map.generationFailures??new string[0])+",PASS\n");
  legalSeeds.Add(map.seed);seed++;Need(seed<=61||legalSeeds.Count>=20,"could not find 20 unique physical legal seeds within 60 requests");if(legalSeeds.Count>=20){File.WriteAllText(Path.Combine(Root,"stage2-physical.txt"),"PASS 20 unique actual legal seeds; requested and actual seeds/retries/fallback recorded separately; all POIs/edges/exits actual CharacterController traversal; nodes/encounters full NavMesh paths; AI disabled ONLY in this controlled test; enemy colliders retained, never ignored; no native/human claim");EditorApplication.update-=Tick;EditorApplication.Exit(0);}
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
