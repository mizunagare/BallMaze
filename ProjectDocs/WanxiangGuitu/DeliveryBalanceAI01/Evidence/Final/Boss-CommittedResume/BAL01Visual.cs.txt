using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEngine.Rendering.Universal;using UnityEditor;using UnityEditor.SceneManagement;
// Controlled actual runtime camera captures; no native input or human readability verdict.
public static class BAL01Visual {
 static string root;static CombatSession s;static IEnumerator run;static int frame=-1;static double deadline;static List<Row> rows=new List<Row>();
 [Serializable]class Row{public string file,type,utc,scope="controlled enabled real AI, frozen target; runtime Camera render request, NOT native/human";public int width,height;public bool telegraph;public Vector3 hero,enemy,camera;}
 [Serializable]class Report{public Row[] rows;}
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+150;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("visual deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f||frame==Time.frameCount)return;frame=Time.frameCount;if(run==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(root+"/Store",false);s.Saves.enabled=false;run=Cases();}if(!run.MoveNext())End(0);}catch(Exception e){File.WriteAllText(root+"/exception.txt",e.ToString());End(1);}}
 static void End(int code){EditorApplication.update-=Tick;if(s!=null)s.Saves.SuspendEntry();File.WriteAllText(root+"/visual.json",JsonUtility.ToJson(new Report{rows=rows.ToArray()},true));EditorApplication.Exit(code);}
 static IEnumerator Cases(){AdventureRandom.SetNextSeedForValidation(418);if(!s.Flow.Depart(AdventureMode.ForestExploration))throw new Exception(s.Flow.Error);var h=s.hero;h.enabled=false;s.orbit.enabled=false;h.Inventory.SetOpen(false);h.SaveMenuOpen=h.HudPanelOpen=false;Time.timeScale=1;foreach(var e in s.enemies)e.gameObject.SetActive(false);var camera=Camera.main;
 foreach(var type in new[]{EnemyArchetype.Melee,EnemyArchetype.Ranged,EnemyArchetype.Elite}){
  var enemy=s.enemies.First(e=>e.archetype==type);enemy.gameObject.SetActive(true);enemy.target=h;enemy.enabled=true;var guard=enemy.GetComponent<EncounterGuard>();if(guard!=null)guard.enabled=false;enemy.ResetEnemy();var at=enemy.transform.position;h.RestorePosition(at-Vector3.forward*(type==EnemyArchetype.Ranged?5:1.3f),0);h.Health.ResetHealth();Physics.SyncTransforms();float until=Time.time+4;while(!enemy.WindingUp&&Time.time<until)yield return null;if(!enemy.WindingUp)throw new Exception("no actual telegraph "+type);Time.timeScale=0;
  camera.transform.position=h.transform.position+new Vector3(4.8f,5.2f,-6.2f);camera.transform.LookAt((h.transform.position+enemy.transform.position)*.5f+Vector3.up*.55f);
  foreach(int width in new[]{1280,1920}){int height=width==1280?720:1080;var rt=new RenderTexture(width,height,24,RenderTextureFormat.ARGB32);if(!rt.Create())throw new Exception("render target create");var previous=RenderTexture.active;var tex=new Texture2D(width,height,TextureFormat.RGB24,false);try{camera.SubmitRenderRequest(new UniversalRenderPipeline.SingleCameraRequest{destination=rt});RenderTexture.active=rt;tex.ReadPixels(new Rect(0,0,width,height),0,0);tex.Apply();string file=type+"-"+width+".png";File.WriteAllBytes(root+"/"+file,tex.EncodeToPNG());rows.Add(new Row{file=file,type=type.ToString(),utc=DateTime.UtcNow.ToString("o"),width=width,height=height,telegraph=enemy.WindingUp,hero=h.transform.position,enemy=enemy.transform.position,camera=camera.transform.position});}finally{RenderTexture.active=previous;UnityEngine.Object.Destroy(tex);rt.Release();UnityEngine.Object.Destroy(rt);}}
  enemy.gameObject.SetActive(false);Time.timeScale=1;yield return null;
 }
 }
}
