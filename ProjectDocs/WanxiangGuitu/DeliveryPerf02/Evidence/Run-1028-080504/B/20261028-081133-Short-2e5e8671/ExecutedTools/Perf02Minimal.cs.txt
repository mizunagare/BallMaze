using System;using System.IO;using System.Globalization;using UnityEngine;
// B only: fixed numeric storage; no discovery, GC recorder, per-frame strings or OS polling.
[DefaultExecutionOrder(32000)]public sealed class Perf02Minimal:MonoBehaviour{
 struct Row{public int frame;public double time,interval,cpu,main,render,gpu,selfUs;public ulong stamp;public bool fresh;}
 Row[] rows=new Row[200000];int n;bool active,overflow;double last,start,seconds;ulong prior;FrameTiming[] ft=new FrameTiming[1];
 public void Begin(double length){n=0;start=last=Time.realtimeSinceStartupAsDouble;seconds=length;active=true;}
 void LateUpdate(){if(!active)return;long begin=System.Diagnostics.Stopwatch.GetTimestamp();double now=Time.realtimeSinceStartupAsDouble;if(now>start+seconds)return;FrameTimingManager.CaptureFrameTimings();uint count=FrameTimingManager.GetLatestTimings(1,ft);var f=ft[0];bool fresh=count>0&&f.frameStartTimestamp!=0&&f.frameStartTimestamp!=prior;if(fresh)prior=f.frameStartTimestamp;if(n>=rows.Length){overflow=true;return;}rows[n++]=new Row{frame=Time.frameCount,time=now,interval=(now-last)*1000,stamp=count>0?f.frameStartTimestamp:0,fresh=fresh,cpu=fresh?f.cpuFrameTime:-1,main=fresh?f.cpuMainThreadFrameTime:-1,render=fresh?f.cpuRenderThreadFrameTime:-1,gpu=fresh?f.gpuFrameTime:-1,selfUs=(System.Diagnostics.Stopwatch.GetTimestamp()-begin)*1000000.0/System.Diagnostics.Stopwatch.Frequency};last=now;}
 public void End(){active=false;string root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);using(var w=new StreamWriter(Path.Combine(root,"minimal-frames.csv"))){w.WriteLine("frame,time,interval_ms,ftm_timestamp,ftm_new,cpu_ms,main_ms,render_ms,gpu_ms,sampler_us");for(int i=0;i<n;i++){var r=rows[i];w.WriteLine(string.Join(",",r.frame,r.time.ToString("R",CultureInfo.InvariantCulture),r.interval.ToString("R",CultureInfo.InvariantCulture),r.stamp,r.fresh?1:0,r.cpu,r.main,r.render,r.gpu,r.selfUs));}}File.WriteAllText(Path.Combine(root,"minimal-receipt.json"),"{\"overflow\":"+(overflow?"true":"false")+",\"rows\":"+n+",\"preallocatedRows\":200000}");}
}
