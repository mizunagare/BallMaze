using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class EncaiRepro{
 static string root,sample;static CombatSession s;static ThirdPersonHero h;static IEnumerator run;static double deadline;static int frame=-1;static List<Row>rows=new List<Row>();static List<CombatEnemy>actors=new List<CombatEnemy>();static List<GameObject>owned=new List<GameObject>();static HashSet<EnemyProjectile>seen=new HashSet<EnemyProjectile>();static List<float>shots=new List<float>(),hits=new List<float>();static int illegal;static CombatEnemy warden;static readonly Vector3 O=new Vector3(300,20.1f,300);
 [Serializable]class Row{public string name,status,detail;}[Serializable]class Report{public string scope="controlled real AI Update and physics; explicit arena, positions and one initial cooldown fixture; not normal AI/native";public Row[] rows;}[Serializable]class Event{public string sample,kind,entity,state;public int frame;public float time,hp,remaining;public Vector3 position,origin,aim;}
 static void C(bool b,string n,string d=""){rows.Add(new Row{name=n,status=b?"PASS":"FAIL",detail=d});}
 static void Log(Event e){e.sample=sample;e.frame=Time.frameCount;e.time=Time.time;e.hp=h.Health.Current;File.AppendAllText(root+"/events.jsonl",JsonUtility.ToJson(e)+"\n");}
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+260;EditorSceneManager.playModeStartScene=null;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("repro deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f||frame==Time.frameCount)return;frame=Time.frameCount;if(run==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(root+"/Store",false);h=s.hero;s.Saves.enabled=false;run=Cases();}if(!run.MoveNext())End();}catch(Exception e){C(false,"exception",e.ToString());End();}}
 static void End(){EditorApplication.update-=Tick;Time.captureDeltaTime=0;Time.timeScale=1;if(h!=null)h.HudPanelOpen=false;if(s!=null)s.Saves.SuspendEntry();File.WriteAllText(root+"/repro.json",JsonUtility.ToJson(new Report{rows=rows.ToArray()},true));EditorApplication.Exit(rows.Count==0||rows.Any(r=>r.status=="FAIL")?1:0);}
 static GameObject Box(string name,Vector3 pos,Vector3 size){var o=GameObject.CreatePrimitive(PrimitiveType.Cube);o.name=name;o.transform.position=pos;o.transform.localScale=size;owned.Add(o);return o;}
 static CombatEnemy New(string id,EnemyArchetype type,Vector3 pos){var e=s.Waves.CreateEncounter(s.transform,id,pos,type);e.target=h;var g=e.GetComponent<EncounterGuard>();if(g!=null)g.enabled=false;e.enabled=true;actors.Add(e);return e;}
 static void Setup(string name){foreach(var e in actors)if(e!=null)e.gameObject.SetActive(false);actors.Clear();foreach(var o in owned)if(o!=null)UnityEngine.Object.Destroy(o);owned.Clear();EnemyProjectile.Clear(s.gameObject.scene);sample=name;seen.Clear();shots.Clear();hits.Clear();illegal=0;h.Health.ResetHealth();h.Dodge.ResetDodge();h.Progression.Clear();h.Inventory.SetOpen(false);h.HudPanelOpen=h.SaveMenuOpen=false;h.RestorePosition(O,0);Time.timeScale=1;PlayerInput.Focus(true);}
 static IEnumerator Wait(float time){float end=Time.time+time;while(Time.time<end){Sample();yield return null;}}
 static bool World(Collider c)=>c.GetComponentInParent<CombatEnemy>()==null&&c.GetComponentInParent<ThirdPersonHero>()==null;
 static bool BarrelBlocked(CombatEnemy e){var a=(Vector3)typeof(CombatEnemy).GetField("attackOrigin",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(e)+Vector3.up*1.25f;var d=(Vector3)typeof(CombatEnemy).GetField("lockedDirection",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(e);return Physics.SphereCastAll(a,.16f,d,.7f,~0,QueryTriggerInteraction.Ignore).Any(x=>World(x.collider));}
 static void Sample(){foreach(var e in actors.Where(e=>e!=null&&e.gameObject.activeInHierarchy))Log(new Event{kind="actor",entity=e.entityId,state=e.GetComponent<ForestBossBrain>()?.State.ToString()??e.State.ToString(),position=e.transform.position,remaining=e.TelegraphRemaining});foreach(var shot in UnityEngine.Object.FindObjectsByType<EnemyProjectile>(FindObjectsSortMode.None)){if(!seen.Add(shot))continue;var e=(CombatEnemy)typeof(EnemyProjectile).GetField("owner",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(shot);shots.Add(Time.time);if(e.archetype==EnemyArchetype.Ranged&&BarrelBlocked(e))illegal++;Log(new Event{kind="shot",entity=e.entityId,position=shot.transform.position,origin=e.transform.position,aim=shot.Velocity});}}
 static float Gap(List<float> a){return a.Count<2?-1:a.Zip(a.Skip(1),(x,y)=>y-x).Min();}
 static IEnumerator Cases(){Time.captureDeltaTime=.02f;AdventureRandom.SetNextSeedForValidation(418);C(s.Flow.Depart(AdventureMode.ForestExploration),"real layout seed418");warden=s.enemies.Single(e=>e.entityId=="boss:warden");s.Flow.BeginLegacyRestore();h.enabled=false;s.orbit.enabled=false;foreach(var e in s.enemies)e.gameObject.SetActive(false);foreach(var item in h.GetComponent<AdventureLoadout>().items)item.board=false;h.GetComponent<AdventureLoadout>().Refresh();var floor=GameObject.CreatePrimitive(PrimitiveType.Cube);floor.name="ENCAI01 controlled floor";floor.transform.position=O+Vector3.down*.6f;floor.transform.localScale=new Vector3(60,1,60);h.Health.Observed+=x=>{if(x.kind==HealthObservationKind.Damage){hits.Add(Time.time);Log(new Event{kind="damage",remaining=x.applied});}};
 Setup("ranged-physical-barrel");var ranged=New("probe:ranged",EnemyArchetype.Ranged,O+Vector3.back*6);var start=ranged.transform.position;Box("controlled high narrow cover",start+new Vector3(0,1.4f,.35f),new Vector3(.8f,.5f,.12f));Physics.SyncTransforms();yield return null;C(ranged.HasSight(),"eye can see target under narrow cover");var wait=Wait(8);while(wait.MoveNext())yield return null;C(shots.Count>0,"actual AI eventually fires");C(illegal==0,"no actual AI shot beyond blocked barrel","illegal="+illegal+" hp="+h.Health.Current);C(Vector3.Distance(start,ranged.transform.position)>.5f,"blocked firing point triggers actual reposition");
 Setup("opposite-ranged-pair");New("probe:left",EnemyArchetype.Ranged,O+Vector3.back*7);New("probe:right",EnemyArchetype.Ranged,O+Vector3.forward*7);Physics.SyncTransforms();wait=Wait(6);while(wait.MoveNext())yield return null;C(shots.Count>=2&&hits.Count>=2,"two real opposing shooters threaten");C(Gap(shots)>=.20f,"opposite shooters release stagger","gap="+Gap(shots));C(Gap(hits)>=.20f,"opposite shooters actual contact stagger","gap="+Gap(hits));
 foreach(bool legacy in new[]{false,true}){Setup(legacy?"legacy-boss-melee":"forest-boss-melee");CombatEnemy boss;if(legacy)boss=New("probe:legacy",EnemyArchetype.Boss,O+Vector3.forward*4);else{boss=warden;boss.gameObject.SetActive(true);boss.SetSpawn(O+Vector3.forward*4);boss.ResetEnemy();boss.target=h;boss.GetComponent<EncounterGuard>().Configure(null,new ForestEncounter{id=boss.entityId,poi="controlled",home=boss.transform.position,type=3});boss.GetComponent<ForestBossBrain>().enabled=true;actors.Add(boss);}var melee=New("probe:melee",EnemyArchetype.Melee,O+Vector3.right*1.3f);typeof(CombatEnemy).GetField("nextAttack",BindingFlags.Instance|BindingFlags.NonPublic).SetValue(melee,Time.time+(legacy?2.75f:2.85f));Physics.SyncTransforms();wait=Wait(5);while(wait.MoveNext())yield return null;C(hits.Count>=2,"actual "+sample+" attacks");C(Gap(hits)>=.20f,"actual "+sample+" contact separation","gap="+Gap(hits));}
 Setup("forest-disable-windup");warden.gameObject.SetActive(true);warden.SetSpawn(O+Vector3.forward*4);warden.ResetEnemy();warden.target=h;warden.GetComponent<EncounterGuard>().Configure(null,new ForestEncounter{id=warden.entityId,poi="controlled",home=warden.transform.position,type=3});var brain=warden.GetComponent<ForestBossBrain>();brain.enabled=true;actors.Add(warden);while(brain.State!=ForestBossState.Telegraph)yield return null;brain.enabled=false;wait=Wait(2);while(wait.MoveNext())yield return null;brain.enabled=true;float hp=h.Health.Current;wait=Wait(.15f);while(wait.MoveNext())yield return null;C(h.Health.Current==hp,"reenabling cancelled Boss cannot release old hidden attack");
 Setup("forest-independent-pause");warden.gameObject.SetActive(true);warden.SetSpawn(O+Vector3.forward*4);warden.ResetEnemy();warden.target=h;brain.enabled=true;actors.Add(warden);while(brain.State!=ForestBossState.Telegraph)yield return null;float pending=brain.Remaining;h.HudPanelOpen=true;wait=Wait(2);while(wait.MoveNext())yield return null;h.HudPanelOpen=false;C(brain.Remaining>=pending-.05f,"GameplayPaused freezes Boss remaining telegraph clock even with active world clock","before="+pending+" after="+brain.Remaining);hp=h.Health.Current;wait=Wait(.15f);while(wait.MoveNext())yield return null;C(h.Health.Current==hp,"resume preserves warning reaction window");
 }
}
