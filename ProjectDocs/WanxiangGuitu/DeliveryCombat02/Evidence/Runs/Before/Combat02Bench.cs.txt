using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using System.Globalization;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
// Explicit controlled experiment: fixture positions/HP resets between samples, never a human victory.
public static class Combat02Bench {
 static CombatSession s;static ThirdPersonHero h;static AdventureCombat combat;static AdventureLoadout gear;static CombatEnemy normal,elite,boss;static IEnumerator run;static string root;static int frame=-1;static double deadline;static EditorWindow view;static int samples;static bool failed;
 static readonly Vector3 Origin=new Vector3(300,20.1f,300);
 [Serializable]class WeaponRow{public string weapon,gear,quality,affixes;public float power,interval,reach;public int level;}
 [Serializable]class Summary{public string scope="Controlled geometry and enabled-enemy-AI fixed-arena duels; not native or human",status;public int samples;public WeaponRow[] weapons;}
 static WeaponRow[] weapons;
 static string F(float n)=>n.ToString("R",CultureInfo.InvariantCulture);
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);Directory.CreateDirectory(root);deadline=EditorApplication.timeSinceStartup+650;view=UI01Before.Size(1280,720);EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();Application.runInBackground=true;view?.Repaint();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("COMBAT02 bounded experiment deadline samples="+samples);if(!EditorApplication.isPlaying||Time.unscaledTime<.5f||frame==Time.frameCount)return;frame=Time.frameCount;if(run==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(root+"/Synthetic",false);s.Saves.enabled=false;h=s.hero;gear=h.GetComponent<AdventureLoadout>();combat=h.GetComponent<AdventureCombat>();run=Cases();}if(!run.MoveNext())Finish();}catch(Exception e){failed=true;File.AppendAllText(root+"/failure.txt",e.ToString());Debug.LogException(e);Finish();}}
 static void Finish(){File.WriteAllText(root+"/measurement.json",JsonUtility.ToJson(new Summary{status=failed?"FAIL":"PASS",samples=samples,weapons=weapons},true));EditorApplication.update-=Tick;EditorApplication.Exit(failed?1:0);}
 static IEnumerator Wait(float seconds){float end=Time.time+seconds;while(Time.time<end)yield return null;}
 static void Place(CombatEnemy e,Vector3 p){var cc=e.GetComponent<CharacterController>();cc.enabled=false;e.transform.SetPositionAndRotation(p,Quaternion.Euler(0,180,0));cc.enabled=true;Physics.SyncTransforms();}
 static void Setup(int w,CombatEnemy e,float distance,float angle,float height=0){Time.timeScale=1;h.Inventory.SetOpen(false);h.SaveMenuOpen=h.HudPanelOpen=false;h.Progression.Clear();h.Dodge.ResetDodge();h.Health.ResetHealth();h.RestorePosition(Origin,0);gear.mainWeapon=gear.items.First(i=>i.Weapon&&i.InBag&&(int)i.type==w).id;gear.Main.affixes=new int[0];gear.Refresh();combat.ResetCombat();int n=0;foreach(var other in new[]{normal,elite,boss}){if(other==null)continue;other.enabled=false;var brain=other.GetComponent<BossBrain>();if(brain!=null)brain.enabled=false;if(other!=e)Place(other,Origin+Vector3.right*(30+n++*4));}e.Health.maxHealth=10000;e.Health.ResetHealth();e.enabled=false;Place(e,Origin+Quaternion.Euler(0,angle,0)*Vector3.forward*distance+Vector3.up*height);s.Feedback.ResetFeedback();}
 static IEnumerator Sample(int w,bool skill,CombatEnemy e,string body,float d,float angle,bool duplicate=false,bool wall=false,float height=0){Setup(w,e,d,angle,height);GameObject extra=null,block=null;if(duplicate){extra=new GameObject("C02 contained duplicate collider");extra.transform.SetParent(e.transform,false);var sphere=extra.AddComponent<SphereCollider>();sphere.center=e.GetComponent<CharacterController>().center;sphere.radius=e.GetComponent<CharacterController>().radius;}
 if(wall){block=GameObject.CreatePrimitive(PrimitiveType.Cube);block.name="C02 occluder";block.transform.position=Origin+Vector3.up+Vector3.forward*.8f;block.transform.localScale=new Vector3(5,4,.15f);}Physics.SyncTransforms();yield return null;
 float before=e.Health.Current,start=Time.time,hitAt=-1,releaseBefore=combat.LastReleaseTime;int firstHitFrame=-1,hitEvents=0,numberAtHit=0;Action<HealthObservation> observer=x=>{if(x.kind==HealthObservationKind.Damage){if(hitAt<0){hitAt=(float)x.activeTime;firstHitFrame=x.frame;}hitEvents++;numberAtHit=s.Feedback.DamageNumberCount;}};e.Health.Observed+=observer;bool accepted=skill?combat.UseSkill():combat.Strike();float cooldown=skill?combat.SkillRemaining:combat.AttackRemaining;float deadline=Time.time+(w<2?.34f:1.35f);while(Time.time<deadline)yield return null;e.Health.Observed-=observer;
 var cc=e.GetComponent<CharacterController>();bool hit=e.Health.Current<before;string row=string.Join(",",new[]{w.ToString(),skill?"skill":"basic",body,F(d),F(angle),F(height),duplicate?"1":"0",wall?"1":"0",accepted?"1":"0",hit?"1":"0",F(before-e.Health.Current),F(cooldown),F(combat.LastReleaseTime>=start?combat.LastReleaseTime-start:-1),F(hitAt<0?-1:hitAt-start),firstHitFrame.ToString(),hitEvents.ToString(),numberAtHit.ToString(),F(cc.radius),F(cc.height),F(cc.center.y)});File.AppendAllText(root+"/matrix.csv",row+"\n");samples++;
 if(!accepted||hit&&numberAtHit==0)throw new Exception("Rejected/feedback sample "+row);if(extra!=null)UnityEngine.Object.Destroy(extra);if(block!=null)UnityEngine.Object.Destroy(block);combat.ResetCombat();yield return null;
 }
 static IEnumerator Timing(int w,bool skill){Setup(w,normal,1.3f,0);yield return null;float start=Time.time,hit=-1;Action<HealthObservation> observe=x=>{if(x.kind==HealthObservationKind.Damage&&hit<0)hit=(float)x.activeTime;};normal.Health.Observed+=observe;bool first=skill?combat.UseSkill():combat.Strike();float cooldown=skill?combat.SkillRemaining:combat.AttackRemaining;string motion=h.GetComponent<FantasyActorVisual>().Motion;while(combat.LastAttackPhase!="Idle")yield return null;float recoveryEnd=Time.time-start;bool second=false;while(!second){yield return null;second=skill?combat.UseSkill():combat.Strike();}float cycle=Time.time-start;normal.Health.Observed-=observe;File.AppendAllText(root+"/timing.csv",$"{w},{(skill?"skill":"basic")},{F(cooldown)},{F(combat.LastReleaseTime-start)},{F(recoveryEnd)},{F(cycle)},{F(hit<0?-1:hit-start)},{motion},{first}\n");combat.ResetCombat();yield return null;}
 static int TotalExperience(){int result=h.Progression.Experience;for(int level=1;level<h.Progression.Level;level++)result+=h.Progression.firstLevelExperience+(level-1)*h.Progression.experienceGrowth;return result;}
 static IEnumerator Duel(int w,CombatEnemy e,string body,int repeat){Setup(w,e,2,0);e.Health.maxHealth=body=="normal"?60:body=="elite"?140:350;e.Health.ResetHealth();e.ResetEnemy();Place(e,Origin+Vector3.forward*2);e.enabled=true;var brain=e.GetComponent<BossBrain>();if(brain!=null){brain.enabled=e.archetype==EnemyArchetype.Boss;brain.ResetBrain();}yield return null;
 int accepted=0,damageEvents=0;float actualDamage=0;Action<HealthObservation> observer=x=>{if(x.kind==HealthObservationKind.Damage){actualDamage+=x.applied;damageEvents++;}};e.Health.Observed+=observer;float start=Time.time,startHero=h.Health.Current;int xp=TotalExperience();int ground=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Length;
 while(!e.Health.Dead&&!h.Health.Dead&&Time.time-start<15){var direction=Vector3.ProjectOnPlane(e.transform.position-h.transform.position,Vector3.up);if(direction.sqrMagnitude>.001f)h.transform.rotation=Quaternion.LookRotation(direction);if(combat.Strike())accepted++;yield return null;}
 float elapsed=Time.time-start;e.Health.Observed-=observer;string terminal=e.Health.Dead?"ENEMY_DEAD":h.Health.Dead?"HERO_DEAD":"TIME_LIMIT";e.enabled=false;if(e.Health.Dead){e.DropLootOnce();int count=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Length,experience=TotalExperience();e.DropLootOnce();if(count!=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Length||experience!=TotalExperience())throw new Exception("Duplicate death reward");}
 File.AppendAllText(root+"/duels.csv",$"{w},{body},{repeat},{terminal},{F(elapsed)},{F(actualDamage)},{F(actualDamage/Mathf.Max(.001f,elapsed))},{F(startHero-h.Health.Current)},{accepted},{damageEvents},{TotalExperience()-xp},{UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Length-ground}\n");combat.ResetCombat();var wait=Wait(.1f);while(wait.MoveNext())yield return null;}
 static IEnumerator Cases(){
 var region=s.GetComponent<ForestRegionRuntime>()??s.gameObject.AddComponent<ForestRegionRuntime>();region.LayoutFactory=_=>ForestLayout.Generate(418);if(!s.Flow.Depart(AdventureMode.ForestExploration))throw new Exception("fixture depart");s.Flow.enabled=false;h.enabled=false;h.Progression.enabled=false;s.orbit.enabled=false;foreach(var e in s.enemies)e.enabled=false;
 var floor=GameObject.CreatePrimitive(PrimitiveType.Cube);floor.name="C02 controlled flat floor";floor.transform.position=new Vector3(300,19.5f,300);floor.transform.localScale=new Vector3(80,1,80);
 normal=s.Waves.CreateEncounter(s.transform,"c02:normal",Origin+Vector3.forward*2,EnemyArchetype.Melee);elite=s.Waves.CreateEncounter(s.transform,"c02:elite",Origin+Vector3.forward*3,EnemyArchetype.Elite);boss=s.Waves.CreateEncounter(s.transform,"c02:boss",Origin+Vector3.forward*4,EnemyArchetype.Boss);foreach(var e in new[]{normal,elite,boss}){e.enabled=false;var b=e.GetComponent<BossBrain>();if(b!=null)b.enabled=false;}
 weapons=Enumerable.Range(0,4).Select(w=>{Setup(w,normal,2,0);return new WeaponRow{weapon=((GearType)w).ToString(),gear=gear.Main.definitionId,quality="White",affixes="none",level=1,power=h.AttackPower,interval=h.AttackInterval,reach=h.AttackRange};}).ToArray();File.WriteAllText(root+"/weapons.json",JsonUtility.ToJson(new Summary{weapons=weapons},true));
 File.WriteAllText(root+"/matrix.csv","weapon,action,body,distance,angle,height,duplicate,wall,accepted,hit,applied,cooldown,releaseLatency,hitLatency,hitFrame,damageEvents,numberAtHit,colliderRadius,colliderHeight,colliderCenterY\n");File.WriteAllText(root+"/timing.csv","weapon,action,cooldown,releaseLatency,actionEnd,nextAcceptance,hitLatency,inputMotion,accepted\n");File.WriteAllText(root+"/duels.csv","weapon,body,repeat,terminal,seconds,appliedDamage,killWindowDPS,receivedDamage,acceptedAttacks,damageEvents,experienceGain,groundDropDelta\n");
 var targets=new[]{normal,elite,boss};string[] names={"normal","elite","boss"};
 for(int t=0;t<targets.Length;t++)for(int w=0;w<2;w++)for(int skill=0;skill<2;skill++){float[] distances=skill==0?new[]{.75f,1.5f,2f,2.5f,3f,3.5f,4f,4.5f,5f}:new[]{1.5f,3f,5f,7f,9f};float[] angles=skill==0?new[]{-90f,-70f,-45f,-20f,0f,20f,45f,70f,90f}:new[]{-60f,-30f,0f,30f,60f};foreach(float d in distances)foreach(float angle in angles){var test=Sample(w,skill==1,targets[t],names[t],d,angle);while(test.MoveNext())yield return null;}}
 for(int w=0;w<4;w++)for(int skill=0;skill<2;skill++){if(w>=2)foreach(float d in new[]{2f,6f,12f})foreach(float angle in new[]{-15f,0f,15f}){var test=Sample(w,skill==1,normal,"normal",d,angle);while(test.MoveNext())yield return null;}var timing=Timing(w,skill==1);while(timing.MoveNext())yield return null;}
 for(int w=0;w<4;w++){foreach(var variant in new[]{"duplicate","wall","high"}){var test=Sample(w,false,normal,"normal",1.5f,0,variant=="duplicate",variant=="wall",variant=="high"?2:0);while(test.MoveNext())yield return null;}}
 for(int t=0;t<targets.Length;t++)for(int w=0;w<4;w++)for(int repeat=0;repeat<2;repeat++){var duel=Duel(w,targets[t],names[t],repeat);while(duel.MoveNext())yield return null;}
 }
}
