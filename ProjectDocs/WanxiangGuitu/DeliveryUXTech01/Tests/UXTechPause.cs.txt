using System;using System.IO;using System.Collections;using System.Collections.Generic;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;
public static class UXTechPause {
 static string root;static double end;static IEnumerator work;static int frame=-1;static List<UXTechCases.Row> rows=new List<UXTechCases.Row>();
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);end=EditorApplication.timeSinceStartup+150;EditorSceneManager.OpenScene(GameEntry.ForestPath);EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Check(bool ok,string name){rows.Add(new UXTechCases.Row{name=name,status=ok?"PASS":"FAIL",frame=Time.frameCount});}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>end)throw new Exception("bounded pause probe deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<1||frame==Time.frameCount)return;frame=Time.frameCount;if(work==null)work=Cases();if(work.MoveNext())return;File.WriteAllText(root+"/pause.json",JsonUtility.ToJson(new UXTechCases.Report{layer="controlled-editor-clock-fixture",rows=rows.ToArray()},true));EditorApplication.update-=Tick;EditorApplication.Exit(rows.Any(x=>x.status=="FAIL")?1:0);}catch(Exception e){File.WriteAllText(root+"/exception.txt",e.ToString());EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
 static IEnumerator Cases(){var s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();s.Saves.InitializeStorage(root+"/Game/ForestRuins",false);s.Saves.enabled=false;AdventureRandom.SetNextSeedForValidation(418);Check(s.Flow.Depart(AdventureMode.ForestExploration),"synthetic forest real departure");s.Flow.BeginLegacyRestore();s.orbit.enabled=false;foreach(var e in s.enemies){e.enabled=false;var b=e.GetComponent<ForestBossBrain>();if(b)b.enabled=false;}var h=s.hero;var owner=s.enemies.First(x=>!x.Health.Dead);var origin=h.transform.position+Vector3.up*30;var shot=EnemyProjectile.Fire(owner,origin,origin+Vector3.right*100,1,1,null);var pos=shot.transform.position;var combat=h.GetComponent<AdventureCombat>();combat.RestoreCooldown(20);h.Inventory.SetOpen(true);yield return null;float clock=Time.time,cd=combat.SkillRemaining,until=Time.realtimeSinceStartup+6.5f;while(Time.realtimeSinceStartup<until)yield return null;Check(Time.time==clock&&Mathf.Abs(cd-combat.SkillRemaining)<.001f,"inventory freezes scaled clock and cooldown");Check(shot!=null&&shot.transform.position==pos,"inventory freezes enemy projectile movement and lifetime beyond six seconds");h.Inventory.SetOpen(false);for(int i=0;i<4;i++)yield return null;Check(shot!=null&&shot.transform.position!=pos&&combat.SkillRemaining<cd,"close inventory resumes projectile and cooldown");string run=JsonUtility.ToJson(s.Flow.ActiveRun),rng=JsonUtility.ToJson(UnityEngine.Random.state);for(int i=0;i<10;i++)RouteScoring.Evaluate(s);Check(run==JsonUtility.ToJson(s.Flow.ActiveRun)&&rng==JsonUtility.ToJson(UnityEngine.Random.state),"route UI query leaves run and global RNG unchanged");InputSettingsPanel.Open();yield return null;clock=Time.time;cd=combat.SkillRemaining;until=Time.realtimeSinceStartup+.4f;while(Time.realtimeSinceStartup<until)yield return null;Check(Time.time==clock&&combat.SkillRemaining==cd,"settings freezes clock and cooldown");InputSettingsPanel.Instance.Close();s.Saves.SuspendEntry();}
}
