using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEngine.UI;using UnityEditor;using UnityEditor.SceneManagement;
public static class UX04Before {
 static string root;static CombatSession s;static IEnumerator cases;static double deadline;static int frame=-1;static List<Row> rows=new List<Row>();
 [Serializable]class Row{public string name,status,detail;}
 [Serializable]class Report{public string scope="controlled Editor domain/UI callbacks; NOT native or human";public Row[] rows;}
 static void Check(bool ok,string name,string detail=""){rows.Add(new Row{name=name,status=ok?"PASS":"FAIL",detail=detail});}
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+180;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("UX04 deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f||frame==Time.frameCount)return;frame=Time.frameCount;if(cases==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(root+"/Store",false);s.Saves.autoSaveInterval=3600;s.Saves.enabled=false;cases=Cases();}if(!cases.MoveNext())Finish();}catch(Exception e){Check(false,"exception",e.ToString());Finish();}}
 static IEnumerator Wait(float seconds){float end=Time.realtimeSinceStartup+seconds;while(Time.realtimeSinceStartup<end)yield return null;}
 static IEnumerator Cases(){var h=s.hero;var g=h.GetComponent<AdventureLoadout>();s.Camp.Restore(s.Camp.Items.ToArray(),Enumerable.Repeat(100,6).ToArray());s.Flow.StartFresh();h.Inventory.OpenPage(PlayerInventory.Page.Crafting);yield return null;var ui=s.GetComponent<CatInventoryPresenter>();var wait=Wait(.4f);while(wait.MoveNext())yield return null;
 Check(CraftingTargets.Select(h,GearType.Sword,GearQuality.White),"real target selected");long revision=s.Flow.Revision;Check(CraftingTargets.Select(h,GearType.Sword,GearQuality.White)&&revision==s.Flow.Revision,"same target select is revision no-op","before="+revision+" after="+s.Flow.Revision);
 // Explicit valid synthetic tutorial state after returned loot; completion is actual UI manufacture.
 s.Flow.Profile.tutorial.stage=TutorialStage.Withdraw;s.Saves.SaveProgress(false);ui.RequestRefresh();wait=Wait(.3f);while(wait.MoveNext())yield return null;
 var recipe=ui.GetComponentsInChildren<ScrollRect>(true).Single(x=>x.name=="RecipeList");recipe.content.anchoredPosition=new Vector2(0,300);Canvas.ForceUpdateCanvases();float before=recipe.content.anchoredPosition.y;long rv=s.Flow.Revision;var buttons=recipe.GetComponentsInChildren<Button>();buttons[8].onClick.Invoke();wait=Wait(.3f);while(wait.MoveNext())yield return null;var current=ui.GetComponentsInChildren<ScrollRect>(true).Single(x=>x.name=="RecipeList");Check(Mathf.Abs(current.content.anchoredPosition.y-before)<1&&rv==s.Flow.Revision,"recipe selection retains scroll without save","before="+before+" after="+current.content.anchoredPosition.y);
 // Select actual white sword recipe again, then invoke the existing real UI button.
 typeof(CatInventoryPresenter).GetField("craftChoice",BindingFlags.Instance|BindingFlags.NonPublic).SetValue(ui,"G:0:0");ui.RequestRefresh();wait=Wait(.3f);while(wait.MoveNext())yield return null;var create=ui.GetComponentsInChildren<Button>().Single(b=>b.name=="制作1件 → 随身");var ids=g.items.Select(i=>i.id).ToArray();create.onClick.Invoke();wait=Wait(.3f);while(wait.MoveNext())yield return null;Check(g.items.Count(i=>!ids.Contains(i.id)&&i.type==GearType.Sword&&i.quality==GearQuality.White)==1,"real UI product created once");Check(!CraftingTargets.Valid(s.Flow.Profile.craftingTarget),"matching manufactured target finishes atomically",CraftingTargets.Compact(s));Check(s.Flow.Profile.tutorial.stage==TutorialStage.DepartAgain,"warehouse unified craft advances equipment tutorial",s.Flow.Profile.tutorial.stage.ToString());File.WriteAllText(root+"/craft-disk.json",JsonUtility.ToJson(CombatSaveSystem.InspectReadOnly(s.Saves.SavePath),true));
 var item=g.items.First(i=>i.type==GearType.Spear&&i.InBag);var size=item.Size;var grab=new Vector2Int(0,Math.Min(1,size.y-1));ContainerInventoryView.BeginDrag(item,grab);ContainerInventoryView.RotateSelection(g);var expected=new Vector2Int(size.y-1-grab.y,grab.x);Check(ContainerInventoryView.Grab==expected,"rotated grab maps original grabbed cell","expected="+expected+" actual="+ContainerInventoryView.Grab);ContainerInventoryView.CancelInteraction();
 h.Inventory.SetOpen(false);h.SaveMenuOpen=h.HudPanelOpen=false;AdventureRandom.SetNextSeedForValidation(418);Check(s.Flow.Depart(AdventureMode.ForestExploration),"real run for combat probe");s.Saves.SuspendEntry();h.enabled=false;s.orbit.enabled=false;foreach(var e in s.enemies){e.enabled=false;var b=e.GetComponent<ForestBossBrain>();if(b!=null)b.enabled=false;}g.SelectMain(g.items.First(i=>i.type==GearType.Sword&&i.InBag));h.Progression.Clear();h.RestorePosition(new Vector3(300,20,300),0);var combat=h.GetComponent<AdventureCombat>();combat.ResetCombat();float radius=h.AttackRange;var nodeObject=GameObject.CreatePrimitive(PrimitiveType.Cube);nodeObject.name="UX04 low resource fixture";nodeObject.transform.position=h.transform.position+new Vector3(0,.04f,radius-.02f);nodeObject.transform.localScale=new Vector3(.08f,.08f,.08f);var node=nodeObject.AddComponent<AdventureResource>();node.entityId="ux04:low-stone";node.kind=MaterialKind.Stone;node.maxHealth=70;node.Restore(70);Physics.SyncTransforms();yield return null;
 var center=h.transform.position+Vector3.up;var collider=node.GetComponent<Collider>();bool candidate=Physics.OverlapSphere(center,radius).Contains(collider);File.WriteAllText(root+"/candidate.txt","center="+center+" anchor="+(node.transform.position+Vector3.up)+" bounds="+collider.bounds+" candidate="+candidate+" closest="+collider.ClosestPoint(center));Check(combat.Strike(),"real melee attack accepted");float releaseDeadline=Time.realtimeSinceStartup+2;while(combat.LastReleaseFrame<0&&combat.PendingAttackCount>0&&Time.realtimeSinceStartup<releaseDeadline)yield return null;File.WriteAllText(root+"/release.txt","phase="+combat.LastAttackPhase+" release="+combat.LastReleaseFrame+" paused="+h.GameplayPaused+" suspended="+PlayerInput.Suspended+" hp="+node.Health);var pulse=UnityEngine.Object.FindObjectsByType<AdventurePulse>(FindObjectsSortMode.None).FirstOrDefault(p=>p.GetComponent<LineRenderer>().positionCount==51);if(pulse==null){Check(false,"release footprint available");yield break;}var line=pulse.GetComponent<LineRenderer>();var vertices=new Vector3[line.positionCount];line.GetPositions(vertices);File.WriteAllText(root+"/vertices.json",JsonUtility.ToJson(new Vertices{values=vertices},true));float extent=vertices.Max(v=>Vector3.ProjectOnPlane(v-h.transform.position,Vector3.up).magnitude);float expectedRadius=Mathf.Sqrt(Mathf.Max(0,radius*radius-.92f*.92f));Check(Mathf.Abs(extent-expectedRadius)<.01f&&!candidate&&node.Health==70,"ground footprint uses actual elevated hit sphere cross-section","radius="+radius+" displayed="+extent+" expected="+expectedRadius+" HP="+node.Health+" release="+combat.LastReleaseFrame+" attack="+combat.LastAttackId);
 }
 [Serializable]class Vertices{public Vector3[] values;}
 static void Finish(){if(s!=null)s.Saves.SuspendEntry();EditorApplication.update-=Tick;File.WriteAllText(root+"/result.json",JsonUtility.ToJson(new Report{rows=rows.ToArray()},true));EditorApplication.Exit(rows.Any(x=>x.status=="FAIL")?1:0);}
}
