using System;using System.IO;using System.Linq;using UnityEngine;using UnityEngine.AI;using UnityEditor;using UnityEditor.SceneManagement;
public static class R1Audit {
 static double until;static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT");
 public static void Run(){ValidationStorage.Root(Environment.GetCommandLineArgs(),true);until=EditorApplication.timeSinceStartup+120;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>until)throw new Exception("audit deadline");if(!EditorApplication.isPlaying||Time.time<.5f)return;var s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(Root+"/audit-save",false);if(!s.Flow.Depart(AdventureMode.ForestExploration))throw new Exception("depart");var region=s.GetComponent<ForestRegionRuntime>();region.GeneratePlayable(1);var map=region.Capture();foreach(var e in s.enemies)e.enabled=false;
 File.WriteAllText(Root+"/r1-audit-paths.csv","id,status,length,corners\n");
 foreach(var item in map.points.Select(p=>(p.id,p.position)).Concat(map.nodes.Select(n=>(n.id,n.position))).Concat(map.exits.Select(e=>(e.id,e.position)))){var path=new NavMeshPath();NavMeshHit a=default,b=default;bool ok=NavMesh.SamplePosition(map.entry,out a,4,NavMesh.AllAreas)&&NavMesh.SamplePosition(item.position,out b,4,NavMesh.AllAreas)&&NavMesh.CalculatePath(a.position,b.position,NavMesh.AllAreas,path);float length=0;for(int i=1;i<path.corners.Length;i++)length+=Vector3.Distance(path.corners[i-1],path.corners[i]);File.AppendAllText(Root+"/r1-audit-paths.csv",item.id+","+(ok?path.status.ToString():"Invalid")+","+length+",\""+string.Join(";",path.corners.Select(p=>p.ToString("F2")))+"\"\n");}
 File.WriteAllText(Root+"/r1-audit-geometry.txt",string.Join("\n",UnityEngine.Object.FindObjectsByType<Collider>(FindObjectsSortMode.None).Where(c=>c.gameObject.scene==s.gameObject.scene).Select(c=>c.name+" "+c.GetType().Name+" enabled="+c.enabled+" trigger="+c.isTrigger)));
 var field=typeof(ForestLayout).GetField("terrain");if(field==null||field.GetValue(map)==null)throw new Exception("R1_FLAT_DECORATIVE_WORLD_NO_SAVED_TERRAIN");EditorApplication.update-=Tick;EditorApplication.Exit(0);
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
