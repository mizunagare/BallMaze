using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
// Synthetic controlled checks; invokes real attack code, never native UI or human play.
public static class Combat03Readability {
 static CombatSession s;static ThirdPersonHero h;static AdventureCombat c;static AdventureLoadout g;static IEnumerator cases;static int frame=-1;static double deadline;static string root;static List<Row> rows=new List<Row>();
 [Serializable]class Row{public string name,status,detail;}
 [Serializable]class Report{public Row[] checks;public string scope="Controlled Editor product attacks";}
 static void Check(bool ok,string name,string detail=""){rows.Add(new Row{name=name,status=ok?"PASS":"FAIL",detail=detail});Debug.Log("C03 "+name+" "+ok+" "+detail);}
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);Directory.CreateDirectory(root);deadline=EditorApplication.timeSinceStartup+180;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("deadline");if(!EditorApplication.isPlaying||frame==Time.frameCount||Time.unscaledTime<.6f)return;frame=Time.frameCount;if(cases==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;if(!s.Saves.IsInitialized)s.Saves.InitializeStorage(root+"/Synthetic",false);h=s.hero;c=h.GetComponent<AdventureCombat>();g=h.GetComponent<AdventureLoadout>();cases=Cases();}if(!cases.MoveNext())Finish();}catch(Exception e){Check(false,"exception",e.ToString());Finish();}}
 static void Finish(){EditorApplication.update-=Tick;File.WriteAllText(root+"/readability.json",JsonUtility.ToJson(new Report{checks=rows.ToArray()},true));EditorApplication.Exit(rows.Any(x=>x.status=="FAIL")?1:0);}
 static IEnumerator Wait(float t){float end=Time.time+t;while(Time.time<end)yield return null;}
 static IEnumerator Cases(){s.Flow.Depart(AdventureMode.ForestExploration);s.Saves.SuspendEntry();s.Flow.enabled=false;s.Saves.enabled=false;h.enabled=false;s.orbit.enabled=false;foreach(var e in s.enemies)e.enabled=false;h.Inventory.SetOpen(false);h.SaveMenuOpen=h.HudPanelOpen=false;h.RestorePosition(new Vector3(300,20,300),0);var field=typeof(AdventureProjectile).GetField("direction",BindingFlags.Instance|BindingFlags.NonPublic);
 for(int w=2;w<4;w++)for(int q=0;q<5;q++){
  g.mainWeapon=g.items.First(x=>x.Weapon&&x.InBag&&(int)x.type==w).id;g.Main.quality=(GearQuality)q;g.Main.affixes=new int[0];g.Refresh();c.ResetCombat();yield return null;
  string rng=JsonUtility.ToJson(UnityEngine.Random.state), streams=JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams);string text=c.SkillDescription;Check(text==c.SkillDescription&&rng==JsonUtility.ToJson(UnityEngine.Random.state)&&streams==JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams),"description query pure "+w+"/"+q);int expected=w==3?(q>=4?9:q>=2?7:5):(q>=3?3:1);
  Check(w==2?text.Contains((q>=2?5:4).ToString()):text.Contains(expected+" 支"),"quality description "+w+"/"+q,text);
  if(w==2&&q>=3)Check(text.Contains("侧"),"staff sideballs disclosed "+q,text);
  c.UseSkill();while(c.PendingAttackCount>0)yield return null;
  var shots=UnityEngine.Object.FindObjectsByType<AdventureProjectile>(FindObjectsSortMode.None);Check(shots.Length==expected,"independent shot count "+w+"/"+q,shots.Length.ToString());
  var attackId=typeof(AdventureProjectile).GetProperty("AttackId");var ordinal=typeof(AdventureProjectile).GetProperty("ShotIndex");Check(attackId!=null&&ordinal!=null&&shots.All(x=>(long)attackId.GetValue(x)==c.LastAttackId)&&shots.Select(x=>(int)ordinal.GetValue(x)).Distinct().Count()==expected,"runtime attack and shot identity "+w+"/"+q);var dirs=shots.Select(x=>(Vector3)field.GetValue(x)).ToArray();Check(dirs.Distinct().Count()==expected,"distinct shot directions "+w+"/"+q);File.AppendAllText(root+"/directions.csv",w+","+q+","+string.Join(";",dirs.Select(d=>Mathf.Atan2(d.x,d.z)*Mathf.Rad2Deg))+"\n");
  c.ResetCombat();yield return null;
 }
 // No numeric rebalance: existing direct-only staff basic and skill AoE contracts.
 g.mainWeapon=g.items.First(x=>x.Weapon&&x.InBag&&x.type==GearType.Staff).id;g.Main.quality=GearQuality.White;g.Refresh();c.ResetCombat();yield return null;
 var enemy=s.Waves.CreateEncounter(s.transform,"c03:direct",h.transform.position+Vector3.forward*3,EnemyArchetype.Melee);enemy.enabled=false;enemy.Health.maxHealth=10000;enemy.Health.ResetHealth();
 var extra=new GameObject("duplicate collider");extra.transform.SetParent(enemy.transform,false);extra.AddComponent<SphereCollider>().center=Vector3.up;Physics.SyncTransforms();yield return null;
 int hits=0;Action<HealthObservation> observe=x=>{if(x.kind==HealthObservationKind.Damage)hits++;};enemy.Health.Observed+=observe;c.Strike();var wait=Wait(.8f);while(wait.MoveNext())yield return null;Check(hits==1,"staff basic duplicate collider once",hits.ToString());c.ResetCombat();enemy.Health.ResetHealth();hits=0;c.UseSkill();wait=Wait(.9f);while(wait.MoveNext())yield return null;Check(hits==1,"staff direct plus AoE overlap once",hits.ToString());enemy.Health.Observed-=observe;var show=typeof(AdventurePulse).GetMethod("SpawnImpact");Check(show!=null,"explicit actual impact boundary");
if(show!=null){var center=h.transform.position+Vector3.up*5;show.Invoke(null,new object[]{center,4f,Color.yellow,h.gameObject.scene,h});var pulse=UnityEngine.Object.FindObjectsByType<AdventurePulse>(FindObjectsSortMode.None).First(x=>x.transform.position==center);var line=pulse.GetComponent<LineRenderer>();Check(Enumerable.Range(0,line.positionCount).All(i=>Mathf.Abs(Vector3.Distance(line.GetPosition(i),center)-4)<.001f),"fixed AoE boundary equals4");h.HudPanelOpen=true;float end=Time.realtimeSinceStartup+.4f;while(Time.realtimeSinceStartup<end)yield return null;Check(pulse!=null,"impact feedback UI-only pause lifetime");h.HudPanelOpen=false;var age=Wait(.3f);while(age.MoveNext())yield return null;Check(pulse==null,"impact feedback expires after resume");}
 }
}
