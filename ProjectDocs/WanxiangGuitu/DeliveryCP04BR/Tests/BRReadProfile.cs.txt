using System;using System.IO;using System.Collections.Generic;using UnityEngine;using UnityEngine.Profiling;using UnityEditor;using UnityEditorInternal;using System.Reflection;
public static class BRReadProfile {
 [Serializable]class Alloc{public int index;public long bytes;}
 [Serializable]class Scope{public string marker,thread;public int frame;public double cpuMs;public Alloc[] allocations;}
 [Serializable]class Report{public string source;public int first,last;public Scope[] scopes;public string[] errors;}
 static string root,source;static int first,last,next;static List<Scope> scopes=new List<Scope>();static List<string> errors=new List<string>();
 public static void Run(){try{root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);var args=Environment.GetCommandLineArgs();int at=Array.IndexOf(args,"-br-profile-file");source=at>=0?args[at+1]:"I:/WGCombat/CP04BR-1102-A/FixedEditorProbe/profiler.raw";if(!Path.GetFullPath(source).StartsWith("I:\\WGCombat\\CP04BR-"))throw new Exception("owned profile only");Profiler.enabled=false;Profiler.enableBinaryLog=false;var method=typeof(ProfilerDriver).GetMethod("LoadProfile",BindingFlags.Public|BindingFlags.NonPublic|BindingFlags.Static,null,new[]{typeof(string),typeof(bool)},null);if(method==null)throw new Exception("LoadProfile missing");var loaded=method.Invoke(null,new object[]{source,false});File.WriteAllText(root+"/load-result.txt",loaded?.ToString()??"void");EditorApplication.delayCall+=StartRead;}catch(Exception e){File.WriteAllText(root+"/exception.txt",e.ToString());EditorApplication.Exit(1);}}
 static void StartRead(){first=ProfilerDriver.firstFrameIndex;last=ProfilerDriver.lastFrameIndex;next=first;EditorApplication.update+=Tick;}
 static void Tick(){try{for(int n=0;n<16&&next<=last;n++,next++){for(int thread=0;thread<64;thread++){using(var d=ProfilerDriver.GetRawFrameDataView(next,thread)){if(!d.valid)break;if(!d.threadName.Contains("Main Thread"))continue;for(int i=0;i<d.sampleCount;i++){string name=d.GetSampleName(i);if(!name.StartsWith("CP04BR.Scope."))continue;int end=i+d.GetSampleChildrenCountRecursive(i);var a=new List<Alloc>();for(int j=i+1;j<=end&&j<d.sampleCount;j++)if(d.GetSampleName(j)=="GC.Alloc"){try{a.Add(new Alloc{index=j,bytes=d.GetSampleMetadataAsLong(j,0)});}catch(Exception e){errors.Add(next+"/"+j+" "+e.GetType().Name);}}scopes.Add(new Scope{marker=name,thread=d.threadName,frame=next,cpuMs=d.GetSampleTimeMs(i),allocations=a.ToArray()});}}}}
 if(next<=last)return;EditorApplication.update-=Tick;File.WriteAllText(root+"/allocation-events.json",JsonUtility.ToJson(new Report{source=source,first=first,last=last,scopes=scopes.ToArray(),errors=errors.ToArray()},true));EditorApplication.Exit(scopes.Count>0&&errors.Count==0?0:1);}catch(Exception e){File.WriteAllText(root+"/exception.txt",e.ToString());EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
