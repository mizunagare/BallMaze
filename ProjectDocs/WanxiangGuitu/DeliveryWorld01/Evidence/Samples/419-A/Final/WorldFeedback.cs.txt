using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class WorldFeedback {
 static CombatSession s;static string root;static bool done;static double end;
 [Serializable]class Result{public string layer="controlled Editor read-only feedback, not native",text,profile,run;public int liveBosses;public float predictedCore;public bool queryReadOnly,consistent;}
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);end=EditorApplication.timeSinceStartup+150;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>end)throw new Exception("feedback deadline");if(done||!EditorApplication.isPlaying||Time.unscaledTime<.6f)return;s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;done=true;s.Saves.InitializeStorage(root+"/Synthetic",false);s.Saves.autoSaveInterval=3600;bool selected=CraftingTargets.Select(s.hero,GearType.Sword,GearQuality.Gold);if(!selected)throw new Exception("real crafting target selection");AdventureRandom.SetNextSeedForValidation(418);if(!s.Flow.Depart(AdventureMode.ForestExploration))throw new Exception("depart");foreach(var e in s.enemies){e.enabled=false;var b=e.GetComponent<ForestBossBrain>();if(b!=null)b.enabled=false;}
 string before=JsonUtility.ToJson(s.Flow.Profile)+JsonUtility.ToJson(s.Flow.ActiveRun)+JsonUtility.ToJson(UnityEngine.Random.state);var scores=RouteScoring.Evaluate(s);string text=CraftingTargets.Recommendations(s);string again=CraftingTargets.Recommendations(s);var r=new Result{profile=s.Flow.Profile.profileId,run=s.Flow.ActiveRun.runId,text=text,liveBosses=s.enemies.Count(e=>e.GetComponent<ForestBossBrain>()!=null&&!e.Health.Dead),predictedCore=scores.Sum(e=>e.expectedMaterials[(int)MaterialKind.BossCore]),queryReadOnly=before==JsonUtility.ToJson(s.Flow.Profile)+JsonUtility.ToJson(s.Flow.ActiveRun)+JsonUtility.ToJson(UnityEngine.Random.state)&&text==again};r.consistent=r.liveBosses==2&&r.predictedCore>0&&!text.Contains("此森林无核心来源")&&text.Contains("可选");File.WriteAllText(root+"/feedback.json",JsonUtility.ToJson(r,true));Finish(r.consistent&&r.queryReadOnly?0:1);
 }catch(Exception e){File.WriteAllText(root+"/feedback-exception.txt",e.ToString());Finish(2);}}
 static void Finish(int code){if(s!=null)s.Saves.SuspendEntry();EditorApplication.update-=Tick;EditorApplication.Exit(code);}
}
