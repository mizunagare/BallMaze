using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
// Controlled clock/movement fixture. No material, reward or tutorial completion is granted.
public static class B08CampPause {
 static CombatSession s;static double start,deadline;static Vector3 position;static float clock,skill,attack,swap,dodge,potion;static bool ready;
 static string Root=>ValidationStorage.ValidateRoot(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT"));
 static void Need(bool value,string reason){if(!value)throw new Exception(reason);}
 public static void Run(){ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+120;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("B08 camp clock deadline");if(!EditorApplication.isPlaying)return;
  if(!ready){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(Root+"/b08-clock",false);s.Saves.autoSaveInterval=3600;s.Flow.StartFresh();Need(s.Saves.SaveProgress(false),"fixture save");var combat=s.hero.GetComponent<AdventureCombat>();combat.RestoreCooldown(6,2,1);s.hero.Dodge.RestoreCooldown(3);s.hero.Inventory.RestoreInventory(Enumerable.Range(0,6).Select(i=>s.hero.Inventory.Count((MaterialKind)i)).ToArray(),s.hero.Inventory.HealthPotionCount,s.hero.Inventory.PotionOnQuickbar,4);s.hero.RestorePosition(s.Camp.FacilityPosition("camp:equipment")+Vector3.back,0);combat.RestoreCooldown(6,2,1);s.hero.Dodge.RestoreCooldown(3);Need(s.Camp.Explore(),"walk camp");start=EditorApplication.timeSinceStartup;position=s.hero.transform.position;clock=Time.time;skill=combat.SkillRemaining;attack=combat.AttackRemaining;swap=combat.SwitchRemaining;dodge=s.hero.Dodge.CooldownRemaining;potion=s.hero.Inventory.PotionCooldownRemaining;ready=true;return;}
  s.hero.MoveCharacter(Vector3.forward*.005f);if(EditorApplication.timeSinceStartup-start<1)return;var c=s.hero.GetComponent<AdventureCombat>();float elapsed=Time.time-clock;File.WriteAllText(Root+"/b08-camp-clock.json",JsonUtility.ToJson(new ClockResult{elapsed=elapsed,moved=Vector3.ProjectOnPlane(s.hero.transform.position-position,Vector3.up).magnitude,skillBefore=skill,skillAfter=c.SkillRemaining,attackBefore=attack,attackAfter=c.AttackRemaining,swapBefore=swap,swapAfter=c.SwitchRemaining,dodgeBefore=dodge,dodgeAfter=s.hero.Dodge.CooldownRemaining,potionBefore=potion,potionAfter=s.hero.Inventory.PotionCooldownRemaining},true));Need(Mathf.Abs(c.SkillRemaining-skill)<.02f&&Mathf.Abs(c.AttackRemaining-attack)<.02f&&Mathf.Abs(c.SwitchRemaining-swap)<.02f&&Mathf.Abs(s.hero.Dodge.CooldownRemaining-dodge)<.02f&&Mathf.Abs(s.hero.Inventory.PotionCooldownRemaining-potion)<.02f,"B08_CAMP_COOLDOWNS_ADVANCED_WHILE_WALKING");Need(Vector3.ProjectOnPlane(s.hero.transform.position-position,Vector3.up).magnitude>.015f,"camp real controller movement blocked");Need(!c.Strike()&&!s.hero.Dodge.TryDodge(Vector3.forward),"camp combat input leaked");s.hero.Inventory.Interactions.Refresh();s.hero.Inventory.Interactions.Present();Need(s.hero.Inventory.Interactions.DisplayedE?.targetId=="camp:equipment"&&s.hero.Inventory.Interactions.ExecuteDisplayed(false)&&Onboarding.State(s).stage==TutorialStage.Warehouse,"paused camp real E interaction");Need(s.Saves.SaveProgress(false),"end save");EditorApplication.update-=Tick;EditorApplication.Exit(0);
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
 [Serializable]public class ClockResult {public float elapsed,moved,skillBefore,skillAfter,attackBefore,attackAfter,swapBefore,swapAfter,dodgeBefore,dodgeAfter,potionBefore,potionAfter;}
}
