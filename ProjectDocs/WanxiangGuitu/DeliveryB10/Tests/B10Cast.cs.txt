using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
// Shared b474/current camera fixture. Real spawned actors; AI disabled, no combat victory claim.
public static class B10Cast {
 static CombatSession s;static EditorWindow view;static CombatEnemy[] cast;static int index,phase,frame;static bool allVisible=true;static string rng,streams;static double deadline;
 static string Root=>ValidationStorage.ValidateRoot(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT"));
 public static void Run(){ValidationStorage.Root(Environment.GetCommandLineArgs(),true);view=UI01Before.Size(1280,720);ShaderUtil.allowAsyncCompilation=false;deadline=EditorApplication.timeSinceStartup+240;EditorSceneManager.OpenScene(GameEntry.ForestPath);EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Need(bool b,string m){if(!b)throw new Exception(m);}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();view?.Repaint();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("cast deadline");if(!EditorApplication.isPlaying)return;
  if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||!s.Saves.IsInitialized){s=null;return;}s.Saves.InitializeStorage(Root+"/cast-store",false);AdventureRandom.SetNextSeedForValidation(31415);Need(s.Flow.Depart(AdventureMode.ForestExploration),"forest cast departure");s.hero.enabled=false;foreach(var e in s.enemies){e.enabled=false;if(e.GetComponent<ForestBossBrain>()!=null)e.GetComponent<ForestBossBrain>().enabled=false;}cast=new[]{s.enemies.First(e=>e.archetype==EnemyArchetype.Melee),s.enemies.First(e=>e.archetype==EnemyArchetype.Ranged),s.enemies.First(e=>e.archetype==EnemyArchetype.Elite),s.enemies.Single(e=>e.GetComponent<ForestBossBrain>()?.Kind==ForestBossKind.Warden),s.enemies.Single(e=>e.GetComponent<ForestBossBrain>()?.Kind==ForestBossKind.Tide)};rng=JsonUtility.ToJson(UnityEngine.Random.state);streams=JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams);}
  if(phase==0){if(index==cast.Length){Need(rng==JsonUtility.ToJson(UnityEngine.Random.state)&&streams==JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams),"cast presentation advanced random");Need(allVisible,"one or more actual spawned actor renderers absent; captures retained");File.WriteAllText(Root+"/b10-cast.txt","PASS actual spawned melee/ranged/elite/land/lake actors, seed31415, identical orbit/size; controlled AI disabled, NOT normal combat/native/human\n");EditorApplication.update-=Tick;EditorApplication.Exit(0);return;}var e=cast[index];var region=s.GetComponent<ForestRegionRuntime>();var wanted=e.transform.position+Vector3.forward*5;if(!region.TryGround(wanted,2,out var at))at=e.transform.position+Vector3.forward*3;s.hero.RestorePosition(at,180);s.orbit.RestoreView(180,28);Physics.SyncTransforms();frame=Time.frameCount;phase=1;return;}
  if(phase==1&&Time.frameCount>frame+45&&!ShaderUtil.anythingCompiling){var e=cast[index];bool visible=e.Health.Current>0&&e.GetComponentsInChildren<Renderer>().Any(r=>r.enabled&&r.gameObject.activeInHierarchy);allVisible&=visible;File.AppendAllText(Root+"/b10-cast-visibility.txt",new[]{"melee","ranged","elite","land","lake"}[index]+" visible="+visible+" hp="+e.Health.Current+"\n");ScreenCapture.CaptureScreenshot(Root+"/b10-cast-"+new[]{"melee","ranged","elite","land","lake"}[index]+".png");frame=Time.frameCount;phase=2;return;}
  if(phase==2&&Time.frameCount>frame+4){index++;phase=0;}
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
