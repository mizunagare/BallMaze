using System;using System.Linq;using UnityEngine;using UnityEngine.UI;using UnityEngine.InputSystem;using UnityEngine.InputSystem.Controls;using UnityEngine.EventSystems;

// Shared title/help editor. Publish the preference file before changing effective input.
[DefaultExecutionOrder(-29000)]public sealed class InputSettingsPanel:MonoBehaviour {
 public static InputSettingsPanel Instance {get;private set;}public static bool IsOpen=>Instance!=null;
 public PlayerBinding[] Draft {get;private set;}public string Status {get;private set;}="";public bool Capturing=>capture>=0;
 GamePreferences preferences;string path;Action closed;int capture=-1,slot,armedFrame;bool waitRelease;GameObject canvas;Text status;Button[] primary,secondary;float previousTime;CursorLockMode previousLock;bool previousCursor;ThirdPersonHero hero;bool previousHud;bool disposed;CanvasGroup rows;bool waitUiRelease;int uiReleaseFrame;
 public static void Open(Action onClose=null){if(IsOpen)return;var g=new GameObject("操作键位设置");var view=g.AddComponent<InputSettingsPanel>();Instance=view;view.closed=onClose;view.Initialize();}
 void Initialize(){previousTime=Time.timeScale;previousLock=Cursor.lockState;previousCursor=Cursor.visible;hero=UnityEngine.Object.FindFirstObjectByType<ThirdPersonHero>();previousHud=hero!=null&&hero.HudPanelOpen;if(hero!=null)hero.HudPanelOpen=true;Time.timeScale=0;Cursor.lockState=CursorLockMode.None;Cursor.visible=true;PlayerInput.Barrier();try{path=GamePreferences.CurrentPath;preferences=GamePreferences.Load(path);Draft=PlayerBindings.Clone(preferences.inputBindings??PlayerBindings.Defaults());Status=preferences.Warning;}catch(Exception e){Status=e.Message;Draft=PlayerInput.Snapshot();}Build();}
 void Build(){canvas=new GameObject("InputSettingsCanvas",typeof(RectTransform),typeof(Canvas),typeof(CanvasScaler),typeof(GraphicRaycaster));canvas.transform.SetParent(transform,false);canvas.GetComponent<Canvas>().renderMode=RenderMode.ScreenSpaceOverlay;canvas.GetComponent<Canvas>().sortingOrder=500;var scale=canvas.GetComponent<CanvasScaler>();scale.uiScaleMode=CanvasScaler.ScaleMode.ScaleWithScreenSize;scale.referenceResolution=new Vector2(1280,720);scale.matchWidthOrHeight=1;
  var shade=CatUI.Panel(canvas.transform,"InputShade",0,0,1280,720,new Color(0,0,0,.65f),true);shade.anchorMin=Vector2.zero;shade.anchorMax=Vector2.one;shade.offsetMin=shade.offsetMax=Vector2.zero;
  var block=CatUI.Panel(canvas.transform,"InputModal",0,0,1100,650,CatUI.Surface,true);block.anchorMin=block.anchorMax=block.pivot=new Vector2(.5f,.5f);block.anchoredPosition=Vector2.zero;CatUI.Text(block,"Title","操作 / 键位",26,15,620,40,27);CatUI.Text(block,"Instruction","点击主键或别名后按新键；Esc 取消捕获 / 返回。冲突不会替换其他动作。",26,59,1048,40,18);
  var list=CatUI.Scroll(block,"BindingList",26,104,1048,414,PlayerBindings.Definitions.Length*46+8);rows=list.gameObject.AddComponent<CanvasGroup>();primary=new Button[Draft.Length];secondary=new Button[Draft.Length];for(int i=0;i<Draft.Length;i++){int row=i;var d=PlayerBindings.Definitions.First(x=>x.id.ToString()==Draft[i].action);float y=i*46+4;CatUI.Text(list,"Action:"+d.id,d.label,12,y+5,410,34,19);primary[i]=CatUI.Button(list,"",430,y,210,38,()=>BeginCapture(row,0));secondary[i]=CatUI.Button(list,"",651,y,210,38,()=>BeginCapture(row,1));CatUI.Button(list,"清除别名",872,y,134,38,()=>{Draft[row].secondary="";Status="草稿未应用";UpdateLabels();});}
  status=CatUI.Text(block,"BindingStatus","",26,525,1048,58,17,CatUI.Gold);CatUI.Button(block,"应用并保存",26,590,205,42,()=>ApplyDraft());CatUI.Button(block,"取消返回",242,590,205,42,Close);CatUI.Button(block,"恢复默认（草稿）",458,590,290,42,RestoreDefaults);CatUI.Button(block,"取消本次捕获",760,590,314,42,CancelCapture);UpdateLabels();
 }
 public void BeginCapture(int row,int which){if(row<0||row>=Draft.Length)return;capture=row;slot=which;armedFrame=Time.frameCount+1;waitRelease=true;rows.interactable=false;Status="请先松开按键，再按新的单键；Esc 取消。";PlayerInput.Barrier();EventSystem.current?.SetSelectedGameObject(null);UpdateLabels();}
 public void CancelCapture(){capture=-1;waitUiRelease=true;uiReleaseFrame=Time.frameCount+1;Status="已取消捕获；有效键位未改变。";PlayerInput.Barrier();UpdateLabels();}
 public void RestoreDefaults(){capture=-1;waitUiRelease=true;uiReleaseFrame=Time.frameCount+1;Draft=PlayerBindings.Defaults();Status="默认键位仅进入草稿；应用前仍可取消。";PlayerInput.Barrier();UpdateLabels();}
 public bool Propose(string key){if(capture<0)return false;var trial=PlayerBindings.Clone(Draft);if(slot==0)trial[capture].primary=key;else trial[capture].secondary=key;string why=PlayerBindings.Validate(trial);capture=-1;waitUiRelease=true;uiReleaseFrame=Time.frameCount+1;if(why!=""){Status=why+" 未修改草稿。";UpdateLabels();return false;}Draft=trial;Status="草稿未应用；点击应用并保存。";UpdateLabels();return true;}
 public bool ApplyDraft(){if(Capturing){Status="请先完成或取消捕获。";UpdateLabels();return false;}string why=PlayerBindings.Validate(Draft);if(why!=""){Status=why;UpdateLabels();return false;}try{if(preferences==null)throw new Exception("设置路径不可用，请返回后重试。");preferences.inputBindings=PlayerBindings.Clone(Draft);preferences.Save(path);Close();return true;}catch(Exception e){Status="保存失败，当前有效键位保持不变，可重试或取消："+e.Message;UpdateLabels();return false;}}
 void UpdateLabels(){if(primary==null)return;for(int i=0;i<Draft.Length;i++){primary[i].GetComponentInChildren<Text>().text=PlayerBindings.Name(Draft[i].primary)+(capture==i&&slot==0?" …":"");secondary[i].GetComponentInChildren<Text>().text=string.IsNullOrEmpty(Draft[i].secondary)?"添加别名":PlayerBindings.Name(Draft[i].secondary);}status.text=Status;}
 static bool AnyHeld()=>Keyboard.current!=null&&Keyboard.current.allKeys.Any(k=>k.isPressed)||Mouse.current!=null&&new ButtonControl[]{Mouse.current.leftButton,Mouse.current.rightButton,Mouse.current.middleButton,Mouse.current.backButton,Mouse.current.forwardButton}.Any(b=>b.isPressed);
 void Update(){if(waitUiRelease&&!AnyHeld()&&Time.frameCount>uiReleaseFrame){waitUiRelease=false;rows.interactable=true;}var k=Keyboard.current;if(k?.escapeKey.wasPressedThisFrame==true){if(Capturing)CancelCapture();else Close();return;}if(!Capturing||Time.frameCount<=armedFrame)return;if(waitRelease){if(!AnyHeld())waitRelease=false;return;}if(k!=null)foreach(var b in k.allKeys)if(b.wasPressedThisFrame){Propose(b.keyCode.ToString());return;}var m=Mouse.current;if(m==null)return;foreach(string name in new[]{"MouseLeft","MouseMiddle","MouseBack","MouseForward"}){ButtonControl b=name=="MouseLeft"?m.leftButton:name=="MouseMiddle"?m.middleButton:name=="MouseBack"?m.backButton:m.forwardButton;if(b.wasPressedThisFrame){Propose(name);return;}}}
 public void Close(){if(disposed)return;disposed=true;capture=-1;PlayerInput.Barrier();if(hero!=null){hero.HudPanelOpen=previousHud;hero.HudClosedFrame=Time.frameCount;}Time.timeScale=previousTime;Cursor.lockState=previousLock;Cursor.visible=previousCursor;Instance=null;canvas?.SetActive(false);closed?.Invoke();Destroy(gameObject);}
 void OnDestroy(){if(!disposed){disposed=true;PlayerInput.Barrier();if(hero!=null){hero.HudPanelOpen=previousHud;hero.HudClosedFrame=Time.frameCount;}Time.timeScale=previousTime;Cursor.lockState=previousLock;Cursor.visible=previousCursor;}if(Instance==this)Instance=null;}
}
