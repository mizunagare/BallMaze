using System;using System.IO;using System.Collections.Generic;using UnityEngine;
// Isolated measurement assembly only. The injector and exact source hashes are archived.
public static class BAL01Trace {
 public static int Starts,Releases,DamageEvents,Dodges,XP;public static float Loss,PlayerDistance,EnemyDistance,BlockedSeconds;public static int RepositionEpisodes,SlotDenials;static Vector3 lastHero;static Dictionary<CombatEnemy,Vector3> poses=new Dictionary<CombatEnemy,Vector3>();static HashSet<CombatEnemy> movingRanged=new HashSet<CombatEnemy>();public static string Folder;
 static long sequence;static Dictionary<CombatEnemy,long> keys=new Dictionary<CombatEnemy,long>();static Dictionary<EnemyProjectile,long> shots=new Dictionary<EnemyProjectile,long>();static HashSet<long> released=new HashSet<long>(),hit=new HashSet<long>();
 [Serializable]class Event{public string kind,entity;public long attack;public int frame,xp,levelBefore,levelAfter,remainder;public float time,before,after,raw,defense,windup;public string state;public Vector3 position,heroPosition;public bool invulnerable;}
 static void Write(Event e){if(Folder==null)return;e.frame=Time.frameCount;e.time=Time.time;File.AppendAllText(Folder+"/events.jsonl",JsonUtility.ToJson(e)+"\n");}
 public static void Reset(string folder){Folder=folder;Directory.CreateDirectory(folder);Starts=Releases=DamageEvents=Dodges=XP=0;Loss=PlayerDistance=EnemyDistance=BlockedSeconds=0;RepositionEpisodes=SlotDenials=0;poses.Clear();movingRanged.Clear();lastHero=Vector3.zero;keys.Clear();shots.Clear();released.Clear();hit.Clear();sequence=0;}
 public static void Begin(CombatEnemy e){keys[e]=++sequence;Starts++;Write(new Event{kind="telegraph",entity=e.entityId,attack=sequence,windup=e.GetComponent<ForestBossBrain>()!=null?ForestBossRules.Definition(e.GetComponent<ForestBossBrain>().Attack).windup:e.windup});}
 static long Key(CombatEnemy e){if(!keys.ContainsKey(e))Begin(e);return keys[e];}
 public static void Release(CombatEnemy e){long k=Key(e);released.Add(k);Releases++;Write(new Event{kind="release",entity=e.entityId,attack=k});}
 public static void Shot(EnemyProjectile p,CombatEnemy e){shots[p]=Key(e);Write(new Event{kind="projectile",entity=e.entityId,attack=Key(e)});}
 public static void Direct(CombatEnemy e,ThirdPersonHero h,float damage,Vector3 point,Vector3 direction){Apply(Key(e),e.entityId,h,damage,point,direction);}
 public static void Projectile(EnemyProjectile p,ThirdPersonHero h,float damage,Vector3 point,Vector3 direction){Apply(shots.ContainsKey(p)?shots[p]:0,"projectile",h,damage,point,direction);}
 static void Apply(long k,string id,ThirdPersonHero h,float damage,Vector3 point,Vector3 direction){float before=h.Health.Current;bool inv=h.Dodge.IsInvulnerable;h.Health.Damage(damage,point,direction);float applied=before-h.Health.Current;if(applied>0){hit.Add(k);DamageEvents++;Loss+=applied;}else if(inv)Dodges++;Write(new Event{kind="contact",entity=id,attack=k,before=before,after=h.Health.Current,raw=damage,defense=h.Defense,invulnerable=inv});}
 public static void Award(CombatEnemy e,int amount){bool accepted=!e.target.Health.Dead&&amount>0;int level=e.target.Progression.Level;e.target.Progression.GainExperience(amount);int credited=accepted?amount:0;XP+=credited;Write(new Event{kind="xp",entity=e.entityId,xp=credited,levelBefore=level,levelAfter=e.target.Progression.Level,remainder=e.target.Progression.Experience});}
 public static void Deny(CombatEnemy e){SlotDenials++;Write(new Event{kind="slot-denied",entity=e.entityId});}
 public static void Frame(CombatEnemy[] cohort,ThirdPersonHero h){if(lastHero!=Vector3.zero)PlayerDistance+=Vector3.Distance(lastHero,h.transform.position);lastHero=h.transform.position;foreach(var e in cohort){if(e==null||e.Health.Dead)continue;float moved=poses.TryGetValue(e,out var p)?Vector3.Distance(p,e.transform.position):0;poses[e]=e.transform.position;EnemyDistance+=moved;bool reposition=e.archetype==EnemyArchetype.Ranged&&e.State==EnemyCombatState.Reposition&&moved>.005f;if(reposition){if(movingRanged.Add(e))RepositionEpisodes++;}else movingRanged.Remove(e);if(e.archetype==EnemyArchetype.Ranged&&!e.HasSight())BlockedSeconds+=Time.deltaTime;Write(new Event{kind="state",entity=e.entityId,state=e.GetComponent<ForestBossBrain>()?.State.ToString()??e.State.ToString(),position=e.transform.position,heroPosition=h.transform.position});}}
 public static int HitActions{get{int n=0;foreach(var k in hit)if(released.Contains(k))n++;return n;}}
}
