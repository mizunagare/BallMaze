using System;using System.IO;using System.Linq;using UnityEngine;using UnityEngine.UI;using UnityEditor;using UnityEditor.SceneManagement;
// Real open/close and legacy pre-profile state. Observes logs; never suppresses them.
public static class B10InventoryReadiness {
 static CombatSession s;static int phase,frame,errors;static double deadline;static string rng,bag;static string Root=>ValidationStorage.ValidateRoot(Environment.GetEnvironmentVariable("UNITY_TEST_ROOT"));
 static void Need(bool ok,string message){if(!ok)throw new Exception(message);}
 static void Log(string message,string trace,LogType type){if(type==LogType.Exception&&trace.Contains("CatInventoryPresenter"))errors++;}
 public static void Run(){ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+100;Application.logMessageReceived+=Log;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("readiness deadline phase="+phase);if(!EditorApplication.isPlaying)return;
 if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null||CatInventoryPresenter.Instance==null)return;s.Flow.LegacyHarness=true;foreach(var e in s.enemies)e.enabled=false;frame=Time.frameCount;return;}
 if(Time.frameCount<frame+30)return;
 if(phase==0){Need(s.Flow.Profile==null,"fixture must have no Profile, not a cleared real save");bag=string.Join(",",Enumerable.Range(0,6).Select(i=>s.hero.Inventory.Count((MaterialKind)i)));rng=JsonUtility.ToJson(UnityEngine.Random.state);s.hero.Inventory.OpenPage(PlayerInventory.Page.Items);phase=1;frame=Time.frameCount;return;}
 if(phase==1){Need(errors==0,"PRE_PROFILE_INVENTORY_NULL_REFERENCE count="+errors);Need(s.hero.Inventory.IsOpen&&CatInventoryPresenter.Instance.Visible,"waiting UI must stay visible");Need(UnityEngine.Object.FindObjectsByType<Text>(FindObjectsSortMode.None).Any(t=>t.text.StartsWith("背包数据尚未就绪")),"missing honest loading reason");Need(s.Flow.Profile==null&&rng==JsonUtility.ToJson(UnityEngine.Random.state)&&bag==string.Join(",",Enumerable.Range(0,6).Select(i=>s.hero.Inventory.Count((MaterialKind)i))),"waiting mutated profile/inventory/RNG");s.hero.Inventory.SetOpen(false);s.Flow.LegacyHarness=false;s.Saves.InitializeStorage(Root+"/readiness-store",false);s.Saves.autoSaveInterval=3600;Need(s.Saves.IsInitialized&&s.Flow.Profile!=null,"actual storage initialization");s.hero.Inventory.OpenPage(PlayerInventory.Page.Items);phase=2;frame=Time.frameCount;return;}
 Need(errors==0&&CatInventoryPresenter.Instance.Visible&&CatInventoryPresenter.Instance.RenderedPage==PlayerInventory.Page.Items,"ready view missing or errored");Need(!UnityEngine.Object.FindObjectsByType<Text>(FindObjectsSortMode.None).Any(t=>t.gameObject.activeInHierarchy&&t.text.StartsWith("背包数据尚未就绪")),"stale waiting view survived initialization");File.WriteAllText(Root+"/b10-readiness-result.txt","PASS Profile absent -> visible honest waiting UI -> actual InitializeStorage -> real inventory; no UI exception, no fabricated Profile/materials/RNG or swallowed log");Application.logMessageReceived-=Log;EditorApplication.update-=Tick;EditorApplication.Exit(0);
 }catch(Exception e){Application.logMessageReceived-=Log;Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
