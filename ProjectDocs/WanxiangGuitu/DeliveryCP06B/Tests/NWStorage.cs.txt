using System;using System.IO;using System.Linq;using System.Collections;using UnityEngine;
public static class NWStorage {
 [Serializable]class Expected{public string run,region,streams,profile,gear,content;public int seed;}
 static string Gear(ThirdPersonHero h)=>JsonUtility.ToJson(new Items{items=h.GetComponent<AdventureLoadout>().items.ToArray()});[Serializable]class Items{public GearItem[] items;}
 public static IEnumerator Run(NWProbe p){yield return p.Wait(()=>GameEntry.Instance!=null,15);var entry=GameEntry.Instance;entry.Refresh();bool resume=entry.Info.canContinue;if(resume)p.Check("resume-normal-continue",entry.Continue());else{entry.AskNew();p.Check("save-confirm-new",entry.ConfirmNew());}yield return p.Wait(()=>UnityEngine.Object.FindFirstObjectByType<CombatSession>()?.Saves?.IsInitialized==true,30);var s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s?.Saves?.IsInitialized!=true)throw new Exception("No initialized save");if(s.Flow.ActiveRun==null)p.Check("save-depart",s.Flow.Depart(AdventureMode.ForestExploration));yield return p.Wait(()=>s.WorldReady&&s.Flow.ActiveRun!=null,30);if(!s.WorldReady)throw new Exception("No ready world");foreach(var e in s.enemies){e.enabled=false;var b=e.GetComponent<ForestBossBrain>();if(b!=null)b.enabled=false;}var h=s.hero;var c=h.GetComponent<AdventureCombat>();var run=s.Flow.ActiveRun;
 if(resume){Time.timeScale=0;var expected=JsonUtility.FromJson<Expected>(File.ReadAllText(p.Root+"/expected.json"));p.Check("resume-run",run.runId==expected.run&&run.regionInstanceId==expected.region&&run.seed==expected.seed);p.Check("resume-content",run.contentRevision==expected.content);p.Check("resume-streams",JsonUtility.ToJson(run.randomStreams)==expected.streams);p.Check("resume-profile",JsonUtility.ToJson(s.Flow.Profile)==expected.profile);p.Check("resume-items",Gear(h)==expected.gear);p.Check("resume-no-action-replay",!c.NewWeapons.Busy&&c.PendingAttackCount==0);p.Check("resume-new-equipped",NewWeaponRegistry.Is(c.Gear.Weapon)&&c.Gear.Main.Definition!=null);p.Check("resume-v9",JsonUtility.FromJson<CombatSaveEnvelope>(File.ReadAllText(s.Saves.SavePath)).version==9);yield break;}
 p.Check("new-run-frozen-channel",run.contentRevision.StartsWith("NWD1:")&&NewWeaponDropRules.Read(run.contentRevision,out int enabled)&&enabled==NewWeaponRegistry.EnabledCount);p.Check("old-run-flags-not-enabled",!NewWeaponDrop.KnownRevision("B06-B4-R1")&&!NewWeaponDrop.KnownRevision("B05-1"));p.Check("unknown-channel-version-rejected",!NewWeaponDrop.KnownRevision("NWD2:3|B06-B4-R1")&&!NewWeaponDrop.KnownRevision("NWD1:4|B06-B4-R1"));p.Check("initial-channel-config",NewWeaponDropRules.MaximumPerDeath==1&&NewWeaponDropRules.Chance(EnemyArchetype.Melee)==.03&&NewWeaponDropRules.Chance(EnemyArchetype.Elite)==.08&&NewWeaponDropRules.Chance(EnemyArchetype.Boss)==0&&Enumerable.Range(0,3).All(i=>NewWeaponDropRules.Weight(i)==1));string streams=JsonUtility.ToJson(run.randomStreams),global=JsonUtility.ToJson(UnityEngine.Random.state);int drops=0;for(int i=0;i<1000;i++){var a=NewWeaponDrop.Preview(run,"fixture-"+i,EnemyArchetype.Melee);var b=NewWeaponDrop.Preview(run,"fixture-"+i,EnemyArchetype.Melee);if(a!=null){drops++;p.Check("drop-stable-"+i,JsonUtility.ToJson(a)==JsonUtility.ToJson(b));}}p.Check("drop-nonempty-sample",drops>0&&drops<100);p.Check("drop-readonly",streams==JsonUtility.ToJson(run.randomStreams)&&global==JsonUtility.ToJson(UnityEngine.Random.state));var oldRun=AdventureData.Clone(run);oldRun.contentRevision="B06-B4-R1";p.Check("old-run-no-new-drop",Enumerable.Range(0,100).All(i=>NewWeaponDrop.Preview(oldRun,"fixture-"+i,EnemyArchetype.Elite)==null));
 // Synthetic equipment fixture, acquired solely through the existing real ground pickup transaction.
 for(int i=0;i<NewWeaponRegistry.EnabledCount;i++){var item=AdventureItems.New((GearType)(11+i),GearQuality.Purple);var loot=LootPickup.SpawnGear(item,h.transform.position+Vector3.up*.3f);UnityEngine.SceneManagement.SceneManager.MoveGameObjectToScene(loot.gameObject,h.gameObject.scene);p.Check("fixture-ground-not-credited-"+i,!c.Gear.items.Any(g=>g.id==item.id));p.Check("fixture-real-pickup-"+i,loot.PickUp(h.Inventory));p.Check("new-main-"+i,c.Gear.SelectMain(c.Gear.items.Single(g=>g.id==item.id)));var reserve=c.Gear.items.First(g=>g.type==GearType.Sword&&g.InBag);p.Check("reserve-old-"+i,c.Gear.SelectReserve(reserve));p.Check("swap-to-old-"+i,c.Gear.SwapWeapons()&&c.Gear.Weapon==GearType.Sword);p.Check("swap-to-new-"+i,c.Gear.SwapWeapons()&&c.Gear.Weapon==(GearType)(11+i));}
 yield return new WaitForSeconds(.3f);p.Check("save-normal-atomic",s.Saves.SaveProgress(false));p.Check("save-v9",JsonUtility.FromJson<CombatSaveEnvelope>(File.ReadAllText(s.Saves.SavePath)).version==9);var state=new Expected{run=run.runId,region=run.regionInstanceId,seed=run.seed,streams=JsonUtility.ToJson(run.randomStreams),profile=JsonUtility.ToJson(s.Flow.Profile),gear=Gear(h),content=run.contentRevision};File.WriteAllText(p.Root+"/expected.json",JsonUtility.ToJson(state,true));Directory.CreateDirectory(p.Root+"/SeedPackage");foreach(var f in Directory.GetFiles(Path.GetDirectoryName(s.Saves.SavePath)))File.Copy(f,Path.Combine(p.Root+"/SeedPackage",Path.GetFileName(f)),false);
 }
}
