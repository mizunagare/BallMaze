using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class BAL01Baseline {
 static string root;static CombatSession s;static ThirdPersonHero h;static IEnumerator run;static int frame=-1;static double deadline;static List<Row> rows=new List<Row>();static List<GameObject> owned=new List<GameObject>();static Vector3 O=new Vector3(300,20.1f,300);
 [Serializable]class Row{public string name,status,detail;}
 [Serializable]class Config{public string id,type,boss;public float hp,damage,interval,windup,range,speed,detect;public int xp;}
 [Serializable]class Level{public int level,required;}
 [Serializable]class Snapshot{public string scope="actual isolated runtime configuration; controlled setup, not human or normal AI";public Config[] enemies;public Level[] levels;public float heroHP,heroDamage,heroDefense,heroInterval,potionHeal,potionCooldown,dodgeDuration,dodgeInvulnerability,dodgeCooldown;public int first,growth,seed;}
 [Serializable]class Report{public string scope="controlled synthetic fixtures using actual enabled enemy Update/attack; teleport/setup is not normal AI or native input";public Row[] rows;}
 static void C(bool ok,string n,string d=""){rows.Add(new Row{name=n,status=ok?"PASS":"FAIL",detail=d});}
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+200;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("baseline bounded deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f||frame==Time.frameCount)return;frame=Time.frameCount;if(run==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(root+"/Store",false);s.Saves.enabled=false;h=s.hero;run=Cases();}if(!run.MoveNext())End();}catch(Exception e){C(false,"fixture exception",e.ToString());End();}}
 static void End(){EditorApplication.update-=Tick;if(s!=null)s.Saves.SuspendEntry();File.WriteAllText(root+"/baseline-checks.json",JsonUtility.ToJson(new Report{rows=rows.ToArray()},true));EditorApplication.Exit(rows.Any(r=>r.status=="FAIL")?1:0);}
 static IEnumerator Wait(float t){float end=Time.time+t;while(Time.time<end)yield return null;}
 static void Box(string n,Vector3 p,Vector3 scale){var o=GameObject.CreatePrimitive(PrimitiveType.Cube);o.name=n;o.transform.position=p;o.transform.localScale=scale;owned.Add(o);}
 static void Clean(){foreach(var x in owned)if(x!=null)UnityEngine.Object.Destroy(x);owned.Clear();EnemyProjectile.Clear(s.gameObject.scene);h.Progression.Clear();h.Health.maxHealth=100;h.Health.ResetHealth();h.RestorePosition(O,0);h.SaveMenuOpen=h.HudPanelOpen=false;h.Inventory.SetOpen(false);Time.timeScale=1;}
 static CombatEnemy Enemy(EnemyArchetype type,Vector3 p,int id=0){var e=s.Waves.CreateEncounter(s.transform,"bal:negative:"+id,p,type);e.transform.position=p;e.transform.rotation=Quaternion.LookRotation(Vector3.ProjectOnPlane(O-p,Vector3.up));owned.Add(e.gameObject);return e;}
 static IEnumerator Cases(){AdventureRandom.SetNextSeedForValidation(418);C(s.Flow.Depart(AdventureMode.ForestExploration),"actual forest created for configuration");var levels=new List<Level>();for(int l=1;l<=30;l++){h.Progression.RestoreProgress(l,0,new int[6],new string[0],new int[0]);levels.Add(new Level{level=l,required=h.Progression.RequiredExperience});}h.Progression.Clear();h.Health.ResetHealth();
 var snap=new Snapshot{enemies=s.enemies.Select(e=>new Config{id=e.entityId,type=e.archetype.ToString(),boss=e.GetComponent<ForestBossBrain>()?.Kind.ToString(),hp=e.Health.maxHealth,damage=e.damage,interval=e.attackInterval,windup=e.windup,range=e.attackRange,speed=e.speed,detect=e.detectionRange,xp=e.experienceReward}).ToArray(),levels=levels.ToArray(),first=h.Progression.firstLevelExperience,growth=h.Progression.experienceGrowth,seed=s.Flow.ActiveRun.seed,heroHP=h.Health.maxHealth,heroDamage=h.AttackPower,heroDefense=h.Defense,heroInterval=h.AttackInterval,potionHeal=h.Inventory.PotionHealing,potionCooldown=h.Inventory.PotionCooldown,dodgeDuration=h.Dodge.duration,dodgeInvulnerability=h.Dodge.invulnerability,dodgeCooldown=h.Dodge.cooldown};File.WriteAllText(root+"/runtime-baseline.json",JsonUtility.ToJson(snap,true));foreach(var e in s.enemies)e.gameObject.SetActive(false);h.enabled=false;s.orbit.enabled=false;s.Saves.SuspendEntry();Box("owned floor",O+Vector3.down*.2f,new Vector3(40,.2f,40));var floor=owned.Last();owned.Remove(floor);Time.captureDeltaTime=.02f;
 Clean();yield return null;var enemy=Enemy(EnemyArchetype.Melee,O+Vector3.forward*1.3f);float end=Time.time+2;while(!enemy.WindingUp&&Time.time<end)yield return null;C(enemy.WindingUp,"real AI starts telegraph");var forward=enemy.transform.forward;float hp=h.Health.Current;h.RestorePosition(enemy.transform.position-forward*1.2f,0);var w=Wait(.8f);while(w.MoveNext())yield return null;C(h.Health.Current==hp,"locked-facing melee cannot hit player behind","hpBefore="+hp+" after="+h.Health.Current+" facing="+forward);
 Clean();yield return null;enemy=Enemy(EnemyArchetype.Ranged,O+Vector3.forward*2);Box("pocket left",O+new Vector3(-.9f,1.5f,2),new Vector3(.2f,4,4));Box("pocket right",O+new Vector3(.9f,1.5f,2),new Vector3(.2f,4,4));Box("pocket back",O+new Vector3(0,1.5f,3),new Vector3(2,4,.2f));Physics.SyncTransforms();bool warned=false;var shots=new HashSet<EnemyProjectile>();end=Time.time+4;hp=h.Health.Current;while(Time.time<end){warned|=enemy.WindingUp;foreach(var p in UnityEngine.Object.FindObjectsByType<EnemyProjectile>(FindObjectsSortMode.None))shots.Add(p);yield return null;}C(warned&&shots.Count>0,"close ranged blocked retreat still telegraphs and fires","warning="+warned+" shots="+shots.Count+" loss="+(hp-h.Health.Current));
 Clean();yield return null;var group=new[]{Enemy(EnemyArchetype.Melee,O+new Vector3(-1,0,1),1),Enemy(EnemyArchetype.Melee,O+new Vector3(0,0,1),2),Enemy(EnemyArchetype.Melee,O+new Vector3(1,0,1),3)};var scheduled=new Dictionary<string,float>();end=Time.time+3;while(Time.time<end&&scheduled.Count<3){foreach(var e in group)if(e.WindingUp&&!scheduled.ContainsKey(e.entityId))scheduled[e.entityId]=(float)typeof(CombatEnemy).GetField("strikeAt",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(e);yield return null;}var times=scheduled.Values.OrderBy(x=>x).ToArray();float min=times.Length>1?Enumerable.Range(1,times.Length-1).Min(i=>times[i]-times[i-1]):0;C(times.Length==3&&min>=.219f,"nearby three attacks have bounded release separation","count="+times.Length+" times="+string.Join(",",times)+" min="+min);
 Clean();yield return null;enemy=Enemy(EnemyArchetype.Melee,O+Vector3.forward*3);var guard=enemy.gameObject.AddComponent<EncounterGuard>();guard.Configure(s.GetComponent<ForestRegionRuntime>(),new ForestEncounter{id=enemy.entityId,poi="bal:wall",home=enemy.transform.position,type=0});Box("solid sight wall",O+new Vector3(0,1.5f,1.5f),new Vector3(8,4,.3f));Physics.SyncTransforms();w=Wait(.4f);while(w.MoveNext())yield return null;C(!guard.Engaged,"unseen undamaged encounter does not alert through wall","guardState="+guard.State);
 Clean();yield return null;enemy=Enemy(EnemyArchetype.Melee,O+Vector3.forward*1.2f);hp=h.Health.Current;end=Time.time+2;while(h.Health.Current==hp&&Time.time<end)yield return null;C(h.Health.Current<hp,"front attack actually hits");var pos=enemy.transform.position;h.RestorePosition(O+Vector3.back*4,0);w=Wait(.2f);while(w.MoveNext())yield return null;float moved=Vector3.ProjectOnPlane(enemy.transform.position-pos,Vector3.up).magnitude;C(moved<.02f,"release has readable stationary recovery window","movement in .2s="+moved);
 Clean();yield return null;enemy=Enemy(EnemyArchetype.Melee,O+Vector3.forward*1.2f);enemy.enabled=false;float before=h.Progression.Experience;enemy.Health.Damage(100000);yield return null;enemy.DropLootOnce();float first=h.Progression.Experience;int level=h.Progression.Level;var rng=JsonUtility.ToJson(UnityEngine.Random.state);enemy.DropLootOnce();C(first==h.Progression.Experience&&level==h.Progression.Level&&rng==JsonUtility.ToJson(UnityEngine.Random.state),"dead enemy XP and loot once");var saved=new SavedEnemy{position=enemy.transform.position,health=0,yaw=0};enemy.RestoreState(saved);enemy.DropLootOnce();C(first==h.Progression.Experience&&level==h.Progression.Level,"restored dead enemy no repeated XP");Time.captureDeltaTime=0;
 }
}
