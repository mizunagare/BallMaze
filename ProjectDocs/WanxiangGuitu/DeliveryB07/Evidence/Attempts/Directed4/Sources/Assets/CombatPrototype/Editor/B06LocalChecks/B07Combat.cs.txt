using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
// Controlled hitboxes/iframes, not a normal-AI victory or native-input acceptance.
public static class B07Combat {
 static CombatSession s;static int index,variant,phase,frames;static ForestBossBrain brain;static CombatEnemy enemy;static Vector3 home;static float before;static EditorWindow view;static double deadline;
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT");
 static ForestBossAttack Attack=>(ForestBossAttack)index;
 static void Need(bool ok,string why){if(!ok)throw new Exception(why);}static void Mark(string m){File.AppendAllText(Root+"/b07-combat.txt",m+"\n");Debug.Log("B07_COMBAT_"+m);}
 public static void Run(){ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+240;ShaderUtil.allowAsyncCompilation=false;view=UI01Before.Size(1280,720);EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();Application.runInBackground=true;view?.Repaint();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("combat deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.5f)return;
  if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(Root+"/combat-save",false);Need(s.Flow.Depart(AdventureMode.ForestExploration),"depart");s.hero.enabled=false;s.orbit.enabled=false;foreach(var e in s.enemies){e.enabled=false;if(e.GetComponent<ForestBossBrain>()!=null)e.GetComponent<ForestBossBrain>().enabled=false;}File.WriteAllText(Root+"/b07-hits.csv","attack,variant,before,after,state,releases\n");}
  if(index>=6){foreach(var boss in s.enemies.Where(e=>e.GetComponent<ForestBossBrain>()!=null)){var b=boss.GetComponent<ForestBossBrain>();boss.Health.ResetHealth();EnemyProjectile.Clear(s.gameObject.scene);s.hero.Dodge.ResetDodge();s.hero.Health.ResetHealth();s.hero.RestorePosition(boss.transform.position+Vector3.forward*3,180);Need(b.Begin(b.Kind==ForestBossKind.Warden?ForestBossAttack.RootSlam:ForestBossAttack.WaterFan),"interrupt prepare");boss.Health.Damage(1);Need(b.State==ForestBossState.Hurt&&!b.Release(),"actual hit failed to interrupt");boss.Health.Damage(boss.Health.maxHealth);b.SendMessage("Update");Need(b.State==ForestBossState.Dead&&!b.Begin(b.Attack)&&!b.Release(),"dead boss attacks");Mark("ACTUAL_HEALTH_INTERRUPT_AND_DEAD_STATE_"+b.Kind);}Mark("SIX_ATTACKS_HIT_DODGE_OUTSIDE_INTERRUPT_DEATH_PASS_CONTROLLED");EditorApplication.update-=Tick;EditorApplication.Exit(0);return;}
  if(phase==0){enemy=s.enemies.Single(e=>e.GetComponent<ForestBossBrain>()?.Kind==(index<3?ForestBossKind.Warden:ForestBossKind.Tide));brain=enemy.GetComponent<ForestBossBrain>();home=enemy.GetComponent<EncounterGuard>().home;enemy.RestoreState(new SavedEnemy{position=home,health=enemy.Health.maxHealth});s.hero.Dodge.ResetDodge();s.hero.Health.ResetHealth();s.hero.RestorePosition(home+Vector3.forward*3,180);EnemyProjectile.Clear(s.gameObject.scene);Physics.SyncTransforms();phase=1;frames=0;return;}
  if(phase==1){if(++frames<3)return;Need(brain.Begin(Attack),"begin "+Attack);before=s.hero.Health.Current;Camera.main.transform.position=home+new Vector3(12,10,15);Camera.main.transform.LookAt(home+Vector3.up);enemy.GetComponent<EncounterGuard>().Restore(1);frames=0;phase=2;return;}
  if(phase==2){frames++;if(frames==3&&variant==0)ScreenCapture.CaptureScreenshot(Root+"/b07-telegraph-"+Attack+".png");if(frames<8)return;
   if(variant==1)Need(s.hero.Dodge.TryDodge(Vector3.right),"iframe begin");
   if(variant==2)s.hero.RestorePosition(home+Vector3.forward*24,180);
   Physics.SyncTransforms();Need(brain.Release(),"release "+Attack);
   if(Attack==ForestBossAttack.AntlerCharge)for(int i=0;i<15;i++)brain.AdvanceCharge(.1f);
   foreach(var shot in UnityEngine.Object.FindObjectsByType<EnemyProjectile>(FindObjectsSortMode.None))shot.Advance(3);
   float after=s.hero.Health.Current;File.AppendAllText(Root+"/b07-hits.csv",Attack+","+variant+","+before+","+after+","+brain.State+","+brain.Releases+"\n");
   Need(variant==0?after<before:after==before,"hit/dodge/outside "+Attack+" variant="+variant+" before="+before+" after="+after);
   Mark(Attack+" variant="+variant+" actualDamage="+(before-after));if(variant==0)ScreenCapture.CaptureScreenshot(Root+"/b07-hit-"+Attack+".png");variant++;if(variant>=3){variant=0;index++;}phase=0;return;
  }
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
