using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class BAL02Rules{
 static CombatSession s;static ThirdPersonHero h;static IEnumerator run;static string root;static int frame=-1;static double deadline;static List<Row>rows=new List<Row>();
 [Serializable]class Row{public string name,status,detail;public float raw,defense,before,after,expected;public int events;}
 [Serializable]class Report{public string layer="controlled actual Damage entry and synthetic legal gear, not AI/native";public Row[] rows;}
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+200;EditorSceneManager.playModeStartScene=null;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>deadline)throw new Exception("rules deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f||frame==Time.frameCount)return;frame=Time.frameCount;if(run==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(root+"/Store",false);h=s.hero;BAL02Fixture.Init(h);run=Cases();}if(!run.MoveNext())End();}catch(Exception e){rows.Add(new Row{name="exception",status="FAIL",detail=e.ToString()});End();}}
 static void End(){EditorApplication.update-=Tick;if(s!=null)s.Saves.SuspendEntry();File.WriteAllText(root+"/rules.json",JsonUtility.ToJson(new Report{rows=rows.ToArray()},true));EditorApplication.Exit(rows.Any(r=>r.status=="FAIL")?1:0);}
 static IEnumerator Cases(){AdventureRandom.SetNextSeedForValidation(418);if(!s.Flow.Depart(AdventureMode.ForestExploration))throw new Exception(s.Flow.Error);s.Flow.BeginLegacyRestore();h.enabled=false;foreach(var e in s.enemies){e.enabled=false;e.target=null;var boss=e.GetComponent<ForestBossBrain>();if(boss!=null)boss.enabled=false;}
 File.WriteAllText(root+"/defense-sources.json",JsonUtility.ToJson(new Sources{tiers=Enumerable.Range(0,4).Select(t=>{BAL02Fixture.Defense(h,t);var g=h.GetComponent<AdventureLoadout>();return new Tier{tier=t,defense=h.Defense,gear=JsonUtility.ToJson(new Items{items=g.items.ToArray()}),skill=h.Progression.DefenseBonus,rig=g.Rig.defenseBonus,bag=g.Bag.defenseBonus};}).ToArray()},true));
 for(int tier=0;tier<4;tier++)foreach(float raw in new[]{9f,12,20,23,27,12,11,24,21}){BAL02Fixture.Defense(h,tier);yield return null;float before=h.Health.Current,d=h.Defense;int events=0;Action<HealthObservation> observe=x=>{if(x.kind==HealthObservationKind.Damage)events++;};h.Health.Observed+=observe;h.Health.Damage(raw);h.Health.Observed-=observe;float expected=BAL02Fixture.Expected(raw,d);rows.Add(new Row{name="actual armor tier="+tier+" raw="+raw,status=Mathf.Abs(before-h.Health.Current-expected)<.0001f&&events==1?"PASS":"FAIL",raw=raw,defense=d,before=before,after=h.Health.Current,expected=expected,events=events});}
 foreach(float raw in new[]{float.NaN,float.PositiveInfinity,-1,0}){BAL02Fixture.Defense(h,0);float before=h.Health.Current;h.Health.Damage(raw);rows.Add(new Row{name="invalid raw "+raw,status=h.Health.Current==before?"PASS":"FAIL"});}
 BAL02Fixture.Defense(h,3);h.Dodge.TryDodge(Vector3.right);float hp=h.Health.Current;h.Health.Damage(27);rows.Add(new Row{name="actual dodge prevents damage",status=h.Health.Current==hp?"PASS":"FAIL"});h.Dodge.ResetDodge();h.Health.Damage(40);float injured=h.Health.Current;h.Inventory.RestoreInventory(new int[6],2,true,0);bool heal=h.Inventory.UseHealthPotion();float healed=h.Health.Current;rows.Add(new Row{name="real potion50 and cooldown",status=heal&&Mathf.Abs(healed-Mathf.Min(100,injured+50))<.001f&&!h.Inventory.UseHealthPotion()?"PASS":"FAIL"});
 }
 [Serializable]class Items{public GearItem[]items;}[Serializable]class Tier{public int tier;public float defense,skill,rig,bag;public string gear;}[Serializable]class Sources{public Tier[]tiers;}
}
