using System;using System.IO;using System.Linq;using UnityEngine;using UnityEngine.AI;using UnityEditor;using UnityEditor.SceneManagement;
public static class B07Rules {
 static CombatSession s;static System.Collections.Generic.HashSet<int> acceptedSeeds=new System.Collections.Generic.HashSet<int>();static int seed=1;static double deadline;static string mode;
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT");
 static void Need(bool ok,string why){if(!ok)throw new Exception(why);}static void Mark(string m){File.AppendAllText(Root+"/b07-rules.txt",m+"\n");Debug.Log("B07_"+m);}
 public static void Run(){Start("run");}public static void Resume(){Start("resume");}public static void PendingResume(){Start("pending");}public static void CommittedResume(){Start("committed");}
 static void Start(string m){mode=m;ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+360;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Freeze(){foreach(var e in s.enemies){e.enabled=false;var b=e.GetComponent<ForestBossBrain>();if(b!=null)b.enabled=false;}}
 static CombatCheckpoint CanonicalWorld(CombatCheckpoint c){if(c==null)return null;c=AdventureData.Clone(c);c.randomStateJson="";c.groundLootPages=null;c.heroPosition=Vector3.zero;c.heroYaw=c.cameraYaw=c.cameraPitch=0;c.potionCooldown=c.dodgeCooldown=c.weaponSkillCooldown=c.weaponAttackCooldown=c.weaponSwitchCooldown=0;if(c.enemies!=null)c.enemies=c.enemies.OrderBy(e=>e.entityId).ToArray();if(c.events!=null)c.events=c.events.OrderBy(e=>e.entityId).ToArray();if(c.resources!=null)c.resources=c.resources.OrderBy(e=>e.entityId).ToArray();if(c.gear!=null)c.gear=c.gear.OrderBy(e=>e.id).ToArray();if(c.loot!=null)c.loot=c.loot.OrderBy(e=>e.entityId).ToArray();return c;}
 static ProfileState CanonicalProfile(ProfileState p){if(p==null)return null;p=AdventureData.Clone(p);p.camp=CanonicalWorld(p.camp);p.warehouse=p.warehouse.OrderBy(i=>i.id).ToArray();p.pending=p.pending.OrderBy(i=>i.pendingId).ToArray();p.badges=p.badges.OrderBy(x=>x).ToArray();p.regions=p.regions.OrderBy(x=>x).ToArray();return p;}
 static string Snapshot(){var run=AdventureData.Clone(s.Flow.ActiveRun);if(run!=null)run.checkpoint=CanonicalWorld(run.checkpoint);var result=AdventureData.Clone(s.Flow.Settlement);if(result!=null)result.plan=CanonicalProfile(result.plan);return JsonUtility.ToJson(CanonicalProfile(s.Flow.Profile))+JsonUtility.ToJson(run)+JsonUtility.ToJson(result);}

 static void Quit(){EditorApplication.update-=Tick;EditorApplication.Exit(0);}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();Application.runInBackground=true;if(EditorApplication.timeSinceStartup>deadline)throw new Exception("B07 deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.5f)return;
  if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.autoSaveInterval=3600;s.Saves.InitializeStorage(Root+"/b07-save",false);
   if(mode!="run"){Need(s.Saves.LoadProgress(false),"new process load");Freeze();File.WriteAllText(Root+"/b07-actual-"+mode+".txt",Snapshot());Need(Snapshot()==File.ReadAllText(Root+"/b07-expected.txt"),"cross process profile/run/settlement changed");if(s.Flow.ActiveRun!=null){var checkpoint=s.Flow.ActiveRun.checkpoint;foreach(var state in checkpoint.enemies.Where(e=>e.entityId.StartsWith("boss:"))){var actor=s.enemies.Single(e=>e.entityId==state.entityId);Need(Mathf.Approximately(actor.Health.Current,state.health)&&actor.GetComponent<EncounterGuard>().State==state.aiState,"runtime Boss health/guard not restored");if(state.health<=0)Need(!actor.GetComponent<CharacterController>().enabled&&!actor.GetComponent<ForestBossBrain>().Begin(ForestBossAttack.RootSlam)&&!actor.GetComponent<ForestBossBrain>().Release(),"restored dead Boss can act");}var loadout=s.hero.GetComponent<AdventureLoadout>();Need(loadout.items.Select(i=>i.id).OrderBy(x=>x).SequenceEqual(checkpoint.gear.Select(i=>i.id).OrderBy(x=>x))&&(loadout.mainWeapon??"")==(checkpoint.mainWeapon??"")&&(loadout.reserveWeapon??"")==(checkpoint.reserveWeapon??""),"runtime gear identity not restored");foreach(MaterialKind material in Enum.GetValues(typeof(MaterialKind)))Need(s.hero.Inventory.Count(material)==checkpoint.materials[(int)material],"runtime material balance not restored");Mark("ACTUAL_RUNTIME_BOSS_GEAR_MATERIAL_RESTORE_"+mode);var expectedIds=s.Flow.ActiveRun.checkpoint.loot.Select(l=>l.entityId).OrderBy(x=>x);var actualIds=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Select(l=>l.InteractionId).OrderBy(x=>x);Need(expectedIds.SequenceEqual(actualIds),"ground identity changed on process restart");}Mark("NEW_PROCESS_EXACT_"+mode);
    if(mode=="resume"){Need(s.Flow.Profile.bossHunts.All(r=>r.kills==2),"first/repeat kills");s.hero.Health.Damage(99999);s.Flow.ResolveTerminal();Need(s.Flow.HasResult&&s.Flow.Settlement.outcome==AdventureOutcome.Death,"death outcome");Need(s.Flow.Settlement.plan.badges.Contains("forest:dual-hunt"),"badge lost on death");File.WriteAllText(Root+"/b07-expected.txt",Snapshot());Mark("PENDING_BADGES_SURVIVE_DEATH");Quit();return;}
    if(mode=="pending"){s.Saves.WriteValidationHook=p=>{if(p=="before-publish")throw new IOException("B07_EXPECTED_COMMIT_FAIL");};Need(!s.Flow.CommitSettlement(),"fault commit succeeded");s.Saves.WriteValidationHook=null;Need(s.Flow.CommitSettlement(),"commit retry");string profile=JsonUtility.ToJson(s.Flow.Profile);Need(s.Flow.CommitSettlement()&&profile==JsonUtility.ToJson(s.Flow.Profile),"duplicate commit");File.WriteAllText(Root+"/b07-expected.txt",Snapshot());Mark("COMMIT_FAIL_RETRY_NO_DUPLICATE");Quit();return;}
    Need(s.Flow.Settlement.status==SettlementStatus.Committed&&s.Flow.Profile.badges.Contains("forest:dual-hunt"),"committed badges missing");Need(s.Flow.AcknowledgeSettlement(),"ack");Need(s.Flow.Depart(AdventureMode.ForestExploration),"depart after death");Need(s.Flow.Profile.bossHunts.All(r=>r.kills==2)&&s.GetComponent<ForestRegionRuntime>().Layout.bossVersion==1,"permanent/new forest");Mark("COMMITTED_ACK_REDEPART_BADGES_SURVIVE");Quit();return;
   }
   Need(ForestBossRules.HudTop(ForestBossKind.Tide)>=ForestBossRules.HudTop(ForestBossKind.Warden)+90,"two boss warning rows overlap");var legacy=new ProfileState{bossHunts=null};ForestBossRules.Validate(legacy);Need(!legacy.badges.Any(),"old badges invented");
   foreach(var order in new[]{new[]{ForestBossKind.Warden,ForestBossKind.Tide},new[]{ForestBossKind.Tide,ForestBossKind.Warden}}){var p=new ProfileState();foreach(var k in order){Need(ForestBossRules.Record(p,k,"run:region:"+k,out bool first)&&first,"first record");Need(!ForestBossRules.Record(p,k,"run:region:"+k,out first),"duplicate record");Need(ForestBossRules.Record(p,k,"run2:region:"+k,out first)&&!first,"repeat record");}ForestBossRules.Validate(p);Need(p.badges.Contains("forest:dual-hunt")&&p.bossHunts.All(r=>r.kills==2),"either order");var bad=AdventureData.Clone(p);bad.bossHunts[0].receipts[1]=bad.bossHunts[0].receipts[0];bool rejected=false;try{ForestBossRules.Validate(bad);}catch(InvalidDataException){rejected=true;}Need(rejected,"duplicate persisted receipt accepted");}
   foreach(ForestBossAttack a in Enum.GetValues(typeof(ForestBossAttack))){var d=ForestBossRules.Definition(a);Need(d.damage>0&&d.windup>=1&&d.recovery>=2&&d.cooldown>0,"definition counter windows");Need(ForestBossRules.InCircle(Vector3.right*d.radius,Vector3.zero,d.radius)&&!ForestBossRules.InCircle(Vector3.right*(d.radius+.01f),Vector3.zero,d.radius),"radius boundary");Need(ForestBossRules.InCharge(new Vector3(1,0,3),Vector3.zero,Vector3.forward*6,1)&&!ForestBossRules.InCharge(new Vector3(1.01f,0,3),Vector3.zero,Vector3.forward*6,1),"charge boundary");}
   Mark("PURE_DEFINITIONS_BOUNDARIES_FIRST_REPEAT_EITHER_ORDER_DUPLICATE_VALIDATION_PASS");Need(s.Flow.Depart(AdventureMode.ForestExploration),"depart");
  }
  var region=s.GetComponent<ForestRegionRuntime>();
  if(seed<=40){var pure=ForestLayout.Generate(seed,true);Need(JsonUtility.ToJson(pure)==JsonUtility.ToJson(ForestLayout.Generate(seed,true)),"seed unstable");pure.Validate();region.GeneratePlayable(seed,true);Freeze();var map=region.Capture();acceptedSeeds.Add(map.seed);Need(map.encounters.Count(e=>e.boss!=ForestBossKind.None)==2,"two bosses");foreach(var actor in s.enemies.Where(e=>e.GetComponent<ForestBossBrain>()!=null)){var guard=actor.GetComponent<EncounterGuard>();float baseline=(float)typeof(EncounterGuard).GetField("lastHealth",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic).GetValue(guard);Need(guard.State==0&&Mathf.Approximately(baseline,actor.Health.Current),"BOSS_INITIAL_HEALTH_BASELINE_FALSE_ALERT "+actor.name+" baseline="+baseline+" current="+actor.Health.Current);}foreach(var e in map.encounters.Where(e=>e.boss!=ForestBossKind.None)){Need(region.PathComplete(map.entry,e.home)&&!map.terrain.InLake(e.home)&&!map.terrain.InRidge(e.home),"boss spawn unreachable "+seed+e.id);var path=new NavMeshPath();Need(NavMesh.SamplePosition(e.home,out var a,2,NavMesh.AllAreas)&&NavMesh.SamplePosition(map.entry,out var b,2,NavMesh.AllAreas)&&NavMesh.CalculatePath(b.position,a.position,NavMesh.AllAreas,path)&&path.status==NavMeshPathStatus.PathComplete,"boss path");File.AppendAllText(Root+"/b07-nav.csv",seed+","+map.seed+","+e.id+","+e.home.x+","+e.home.z+","+path.corners.Length+"\n");}foreach(var exit in map.exits)Need(region.TryInteractionApproach(exit,out _),"exit obstructed");region.GeneratePlayable(seed);var old=region.Capture();region.Build(old);Need(JsonUtility.ToJson(region.Capture())==JsonUtility.ToJson(old),"accepted old layout altered on rebuild");Need(s.enemies.All(e=>e.GetComponent<ForestBossBrain>()==null),"old layout inserted boss");Freeze();seed++;return;}
  Need(acceptedSeeds.Count>=30,"fewer than thirty distinct accepted seeds");Mark("FORTY_REQUESTED_THIRTY_DISTINCT_NAV_FREE_EXIT_REPEAT_AND_OLD_NO_INSERTION_PASS distinct="+acceptedSeeds.Count);Need(s.Saves.LoadProgress(false),"legitimate departure restore");Freeze();Combat02RecoveryProbe.Dump(s,Root);region=s.GetComponent<ForestRegionRuntime>();
  // Controlled damage/fault cases are deliberately separate from normal-AI victories.
  foreach(var kind in new[]{ForestBossKind.Warden,ForestBossKind.Tide}){
   var boss=s.enemies.Single(e=>e.GetComponent<ForestBossBrain>()?.Kind==kind);var brain=boss.GetComponent<ForestBossBrain>();var home=boss.transform.position;brain.ResetBrain();s.hero.RestorePosition(home+Vector3.forward*10,0);Need(brain.Begin(kind==ForestBossKind.Warden?ForestBossAttack.RootSlam:ForestBossAttack.WaterFan)&&brain.Release(),"recovery setup");var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;float recovery=(float)typeof(ForestBossBrain).GetField("until",flags).GetValue(brain),ready=(float)typeof(ForestBossBrain).GetField("readyAt",flags).GetValue(brain);boss.Health.Damage(1);Need((float)typeof(ForestBossBrain).GetField("until",flags).GetValue(brain)>=recovery&&(float)typeof(ForestBossBrain).GetField("readyAt",flags).GetValue(brain)>=ready,"HIT_SHORTENED_RECOVERY_OR_COOLDOWN "+kind);EnemyProjectile.Clear(s.gameObject.scene);boss.RestoreState(new SavedEnemy{position=home,health=boss.Health.maxHealth});Mark("RECOVERY_HIT_PRESERVES_COUNTER_WINDOW_"+kind);
   s.hero.RestorePosition(home+Vector3.forward*3,0);s.hero.Health.ResetHealth();var attacks=kind==ForestBossKind.Warden?new[]{ForestBossAttack.RootSlam,ForestBossAttack.AntlerCharge,ForestBossAttack.SeedVolley}:new[]{ForestBossAttack.WaterFan,ForestBossAttack.TidalMark,ForestBossAttack.ShoreBite};
   foreach(var attack in attacks){s.hero.Health.ResetHealth();EnemyProjectile.Clear(s.gameObject.scene);Need(brain.Begin(attack),"begin "+attack);Need(brain.State==ForestBossState.Telegraph&&brain.Remaining>0,"telegraph "+attack);s.hero.RestorePosition(home+Vector3.forward*15,0);Need(brain.Release(),"release "+attack);if(attack==ForestBossAttack.AntlerCharge)for(int i=0;i<20;i++)brain.AdvanceCharge(.1f);Need(brain.State!=ForestBossState.Telegraph,"release stuck");Mark("ATTACK_TELEGRAPH_RELEASE_"+attack);}
   boss.Health.Damage(boss.Health.maxHealth);boss.DropLootOnce();Need(ForestBossRules.Find(s.Flow.Profile,kind)?.kills==1&&s.Flow.BossCheckpointPending,"kill receipt pending");int drops=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Length;string p=JsonUtility.ToJson(s.Flow.Profile);boss.DropLootOnce();Need(p==JsonUtility.ToJson(s.Flow.Profile)&&drops==UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Length,"duplicate drop");
   s.Saves.WriteValidationHook=point=>{if(point=="before-publish")throw new IOException("B07_EXPECTED_BEFORE_PUBLISH");};Need(!s.Flow.RetryBossCheckpoint()&&s.Flow.BossCheckpointPending,"expected save fault");Need(!CombatSaveSystem.InspectReadOnly(s.Saves.SavePath).profile.badges.Contains(ForestBossRules.Id(kind)),"failed save published badge");
   s.Saves.WriteValidationHook=point=>{if(point=="after-publish")throw new IOException("B07_EXPECTED_AFTER_PUBLISH");};Need(s.Flow.RetryBossCheckpoint()&&!s.Flow.BossCheckpointPending,"post publish reconcile");s.Saves.WriteValidationHook=null;Need(CombatSaveSystem.InspectReadOnly(s.Saves.SavePath).profile.badges.Contains(ForestBossRules.Id(kind)),"badge not persisted");Mark("KILL_GROUND_ONLY_ONCE_PREFAIL_POSTPUBLISH_RECONCILE_"+kind);
  }
  Need(s.Flow.Profile.badges.Contains("forest:dual-hunt"),"dual badge");var beforeGround=UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None).Where(l=>l.Gear!=null).ToArray();Need(beforeGround.Count(l=>l.Gear.quality==GearQuality.Purple)==2,"first gear count");
  // A second region incarnation within a separate controlled run fixture proves repeat policy.
  foreach(var k in new[]{ForestBossKind.Warden,ForestBossKind.Tide}){var r=ForestBossRules.Find(s.Flow.Profile,k);Need(ForestBossRules.Record(s.Flow.Profile,k,"controlled-repeat:"+s.Flow.ActiveRun.runId+":"+k,out bool first)&&!first&&r.kills==2,"repeat not first");}
  while(s.hero.Progression.PendingChoices>0){if(!s.hero.Progression.IsChoosing)s.hero.Progression.BeginChoice();Need(s.hero.Progression.SelectChoice(0),"resolve controlled pending upgrades before checkpoint");}Need(s.Saves.SaveProgress(false),"save checkpoint");File.WriteAllText(Root+"/b07-expected.txt",Snapshot());Mark("RUN_PROFILE_GROUND_STREAM_LEDGER_READY_FOR_NEW_PROCESS");Quit();
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
