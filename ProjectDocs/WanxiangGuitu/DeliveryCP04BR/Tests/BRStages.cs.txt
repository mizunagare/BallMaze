using System;using System.IO;using System.Linq;using System.Collections.Generic;using System.Reflection;using System.Security.Cryptography;using UnityEngine;using UnityEditor;using UnityEditor.Build;using UnityEditor.Build.Reporting;using UnityEditor.Rendering;using UnityEngine.Rendering;
[InitializeOnLoad]public static class BRStages {
 public static readonly string[] Files={"Assets/Settings/PC_RPAsset.asset","Assets/Settings/UniversalRenderPipelineGlobalSettings.asset","ProjectSettings/GraphicsSettings.asset","ProjectSettings/ProjectSettings.asset","ProjectSettings/UnityConnectSettings.asset"};static int seq;static bool firstShader;public static List<ShaderRow> shaders=new List<ShaderRow>();
 [Serializable]public class FileRow{public string path,sha256;public long bytes;}
 [Serializable]class Snapshot{public string phase,utc,unity,project;public FileRow[] files;public string[] packages,connect;}
 [Serializable]public class ShaderRow{public string shader,pass,type,utc;public int variants;}
 public static string Root=>ValidationStorage.Root(Environment.GetCommandLineArgs(),true);
 public static string Hash(byte[] b)=>BitConverter.ToString(SHA256.Create().ComputeHash(b)).Replace("-","").ToLowerInvariant();
 static BRStages(){try{if(ValidationStorage.Root(Environment.GetCommandLineArgs())!=null){SnapshotNow("managed-domain-initialized");EditorApplication.delayCall+=()=>SnapshotNow("editor-delay-after-domain");}}catch(Exception e){Debug.LogException(e);}}
 static string Read(object obj,Type type,string name){try{var p=type?.GetProperty(name,BindingFlags.Public|BindingFlags.NonPublic|BindingFlags.Static|BindingFlags.Instance);return name+"="+(p==null?"MISSING":p.GetValue(obj)?.ToString());}catch(Exception e){return name+"="+e.GetType().Name;}}
 public static void SnapshotNow(string phase){string dir=Path.Combine(Root,"Stages",(++seq).ToString("D4")+"-"+phase);Directory.CreateDirectory(dir);var rows=Files.Select(f=>{byte[] b=File.ReadAllBytes(f);File.WriteAllBytes(Path.Combine(dir,f.Replace('/','_')+".txt"),b);return new FileRow{path=f,sha256=Hash(b),bytes=b.Length};}).ToArray();var asm=typeof(EditorApplication).Assembly;var t=asm.GetType("UnityEditor.Connect.UnityConnect");var inst=t?.GetProperty("instance",BindingFlags.Static|BindingFlags.Public|BindingFlags.NonPublic)?.GetValue(null);var settings=AppDomain.CurrentDomain.GetAssemblies().Select(a=>a.GetType("UnityEngine.Connect.UnityConnectSettings")).FirstOrDefault(x=>x!=null);var connect=new[]{Read(inst,t,"online"),Read(inst,t,"loggedIn"),Read(inst,t,"projectValid"),Read(inst,t,"workingOffline"),Read(null,t,"preferencesEnabled"),Read(null,settings,"enabled")};var packages=UnityEditor.PackageManager.PackageInfo.GetAllRegisteredPackages().Where(p=>p.name.Contains("render-pipelines")||p.name.Contains("inputsystem")||p.name=="com.unity.ugui").Select(p=>p.name+"="+p.version+" source="+p.source).ToArray();File.WriteAllText(Path.Combine(dir,"snapshot.json"),JsonUtility.ToJson(new Snapshot{phase=phase,utc=DateTime.UtcNow.ToString("o"),unity=Application.unityVersion,project=Path.GetFullPath("."),files=rows,packages=packages,connect=connect},true));Debug.Log("CP04BR_PHASE "+phase+" "+DateTime.UtcNow.ToString("o"));}
 public static void Shader(Shader s,ShaderSnippetData pass,int count){if(!firstShader){firstShader=true;SnapshotNow("first-shader-preprocess");}if(shaders.Count>=50000)throw new Exception("bounded shader audit capacity exceeded");shaders.Add(new ShaderRow{shader=s.name,pass=pass.passName,type=pass.shaderType.ToString(),variants=count,utc=DateTime.UtcNow.ToString("o")});}
}
public class BRImports:AssetPostprocessor{static void OnPostprocessAllAssets(string[] imported,string[] deleted,string[] moved,string[] from){if(imported.Any(x=>BRStages.Files.Contains(x)))BRStages.SnapshotNow("tracked-assets-postimport");}}
public class BREarlyBuild:IPreprocessBuildWithReport{public int callbackOrder=>int.MinValue;public void OnPreprocessBuild(BuildReport r){BRStages.SnapshotNow("early-build-preprocess");}}
public class BRLateBuild:IPreprocessBuildWithReport{public int callbackOrder=>int.MaxValue;public void OnPreprocessBuild(BuildReport r){BRStages.SnapshotNow("late-build-preprocess");}}
public class BRPostBuild:IPostprocessBuildWithReport{public int callbackOrder=>int.MaxValue;public void OnPostprocessBuild(BuildReport r){BRStages.SnapshotNow("late-build-postprocess");}}
public class BRShaderAudit:IPreprocessShaders{public int callbackOrder=>int.MaxValue;public void OnProcessShader(Shader shader,ShaderSnippetData data,IList<ShaderCompilerData> variants){BRStages.Shader(shader,data,variants.Count);}}
public static class BRBuildAudit {
 [Serializable]class BuildResultRow{public string result,options,utc;public int errors,warnings;public double seconds;public string[] messages;public BRStages.ShaderRow[] shaders;}
 public static void OpenOnly(){BRStages.SnapshotNow("execute-open-only");int frames=0;EditorApplication.CallbackFunction tick=null;tick=()=>{if(++frames<8)return;EditorApplication.update-=tick;BRStages.SnapshotNow("open-only-before-exit");EditorApplication.Exit(0);};EditorApplication.update+=tick;}
 public static void Pure(){Build(false);}public static void Acceptance(){Build(true);}
 static void Build(bool acceptance){try{BRStages.SnapshotNow("execute-before-BuildPipeline");string output=Path.Combine(BRStages.Root,"Release/WanxiangGuitu.exe");BuildReport report;if(acceptance)report=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{GameEntry.BootPath,GameEntry.TitlePath,GameEntry.ForestPath},locationPathName=output,target=BuildTarget.StandaloneWindows64,options=BuildOptions.None,extraScriptingDefines=new[]{"CP04BR_ACCEPTANCE"}});else{typeof(WanxiangBuild).GetMethod("Build",BindingFlags.Static|BindingFlags.NonPublic).Invoke(null,new object[]{output});report=BuildReport.GetLatestReport();}BRStages.SnapshotNow("execute-after-BuildPipeline");var s=report.summary;File.WriteAllText(Path.Combine(BRStages.Root,"build.json"),JsonUtility.ToJson(new BuildResultRow{result=s.result.ToString(),options=s.options.ToString(),utc=DateTime.UtcNow.ToString("o"),errors=s.totalErrors,warnings=s.totalWarnings,seconds=s.totalTime.TotalSeconds,messages=report.steps.SelectMany(st=>st.messages.Select(m=>st.name+" | "+m.type+" | "+m.content)).ToArray(),shaders=BRStages.shaders.ToArray()},true));EditorApplication.Exit(s.result==UnityEditor.Build.Reporting.BuildResult.Succeeded?0:1);}catch(Exception e){File.WriteAllText(Path.Combine(BRStages.Root,"exception.txt"),e.ToString());EditorApplication.Exit(1);}}
}
