using System;using System.IO;using System.Linq;using System.Collections;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
// Actual altar interaction/choice/save and a separate-process restore. Controlled, not native.
public static class WorldPoi {
 static string root;static bool resume;static CombatSession s;static IEnumerator work;static double end;static int frame=-1;
 [Serializable]class Expected{public string profile,run,region,layout,streams,point;public int seed;}
 public static void Run()=>Start(false);public static void Resume()=>Start(true);
 static void Start(bool load){resume=load;root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);end=EditorApplication.timeSinceStartup+150;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Need(bool ok,string label){File.AppendAllText(root+"/poi-"+(resume?"resume":"run")+".txt",(ok?"PASS ":"FAIL ")+label+"\n");if(!ok)throw new Exception(label);}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();if(EditorApplication.timeSinceStartup>end)throw new Exception("poi deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f||frame==Time.frameCount)return;frame=Time.frameCount;if(work==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.InitializeStorage(root+"/Store",resume);s.Saves.autoSaveInterval=3600;s.Saves.enabled=false;work=Cases();}if(!work.MoveNext())Finish(0);}catch(Exception e){File.WriteAllText(root+"/poi-exception.txt",e.ToString());Finish(1);}}
 static IEnumerator Cases(){if(!resume){AdventureRandom.SetNextSeedForValidation(420);Need(s.Flow.Depart(AdventureMode.ForestExploration),"real departure");}foreach(var e in s.enemies){e.enabled=false;var b=e.GetComponent<ForestBossBrain>();if(b!=null)b.enabled=false;}s.hero.enabled=false;s.orbit.enabled=false;
 if(resume){var x=JsonUtility.FromJson<Expected>(File.ReadAllText(root+"/poi-expected.json"));Need(s.Flow.Profile.profileId==x.profile&&s.Flow.ActiveRun.runId==x.run&&s.Flow.ActiveRun.regionInstanceId==x.region&&s.Flow.ActiveRun.seed==x.seed,"new process same identities");Need(JsonUtility.ToJson(s.GetComponent<ForestRegionRuntime>().Capture())==x.layout&&JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams)==x.streams,"same layout and streams");Need(s.Events.Points.Single(p=>p.entityId==x.point).Used,"actual altar remains consumed");Need(WorldAudit.Restored(s),"actual event and actor state");yield break;}
 var point=s.Events.Points.Single(p=>p.entityId=="poi:1:altar");var region=s.GetComponent<ForestRegionRuntime>();bool reached=false;for(int i=0;i<16&&!reached;i++){float a=i*Mathf.PI/8;if(!region.TryGround(point.transform.position+new Vector3(Mathf.Cos(a),0,Mathf.Sin(a))*1.8f,1,out var at))continue;s.hero.RestorePosition(at+Vector3.up*.1f,0);Physics.SyncTransforms();yield return null;reached=point.CanInteract(s.hero);}Need(reached&&point.Interact(s.hero),"actual altar offers skill");Need(point.Used&&s.hero.Progression.PendingChoices==1,"one real choice");Need(s.hero.Progression.SelectChoice(0),"select actual offered skill");yield return null;int choices=s.hero.Progression.PendingChoices;string rng=JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams);Need(!point.Interact(s.hero)&&choices==s.hero.Progression.PendingChoices&&rng==JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams),"repeat after unpause cannot grant or reroll");Need(s.Saves.SaveProgress(false),"publish used event and choice");var run=s.Flow.ActiveRun;File.WriteAllText(root+"/poi-expected.json",JsonUtility.ToJson(new Expected{profile=s.Flow.Profile.profileId,run=run.runId,region=run.regionInstanceId,seed=run.seed,layout=JsonUtility.ToJson(region.Capture()),streams=JsonUtility.ToJson(run.randomStreams),point=point.entityId},true));}
 static void Finish(int code){if(s!=null)s.Saves.SuspendEntry();EditorApplication.update-=Tick;EditorApplication.Exit(code);}
}
