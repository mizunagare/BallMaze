using System;using System.IO;using System.Linq;using System.Collections;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEngine.AI;using UnityEditor;using UnityEditor.SceneManagement;
public static class EncaiMatrix {
 [Serializable]class Config{public int seed,rep;}static Config config;static bool policyOverride;static string root;static CombatSession s;static ThirdPersonHero h;static AdventureLoadout g;static IEnumerator cases;static int frame=-1;static double deadline;static List<Row> rows=new List<Row>();static List<RunRecord> runs=new List<RunRecord>();
 [Serializable]public class RunRecord{public int seed,rep;public string run,profile,layout,streams;}
 [Serializable]public class Row{public string sample,run,profile,type,policy,outcome,gear,initialPositions,initialStreams,finalStreams,initialRanks,finalRanks;public int tier,window,potionRejected;public float finalHP,playerDistance,enemyDistance,blockedSeconds;public int repositionEpisodes,slotDenials;public int seed,rep,initialLevel,finalLevel,kills,enemyCount,heals,dodges,playerAttacks,windups,releases,hitActions,damageEvents,dodgeBlocks,experience,projectilesAtCutoff;public float startedTime,initialHP,initialDefense,initialPower,elapsed,ttk=-1,damage,hitRate;public bool xpMatchesWhenSurviving;}
 [Serializable]class Report{public string scope="controlled enabled actual AI, real combat/drop/XP; Flow disabled during fixture resets so draws use explicitly reseeded per-sample legacy global RNG, NOT Run-owned streams; setup teleports and fixture resets are NOT normal AI/native/human. Same frozen conditional policy, actual frame events archived; 30/90 active-second window; passive Stationary, attacking Walk/CorrectDodge, Triple rep2 intentionally early dodge; tier0/white/green/purple+IronSkin, fixed level20 to avoid mid-case level-up stat changes; no full Cartesian claim, uncompleted kills censored, not zero TTK";public RunRecord[] runs;public Row[] rows;}
 public static void Run(){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);config=JsonUtility.FromJson<Config>(File.ReadAllText(root+"/enc-config.json"));deadline=EditorApplication.timeSinceStartup+1000;EditorSceneManager.playModeStartScene=null;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();Application.runInBackground=true;if(EditorApplication.timeSinceStartup>deadline)throw new Exception("bounded matrix deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.6f||frame==Time.frameCount)return;frame=Time.frameCount;if(cases==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;h=s.hero;g=h.GetComponent<AdventureLoadout>();BAL02Fixture.Init(h);h.enabled=false;s.orbit.enabled=false;cases=Cases();}if(!cases.MoveNext())End(0);}catch(Exception e){File.WriteAllText(root+"/matrix-exception.txt",e.ToString());End(1);}}
 static void End(int code){EditorApplication.update-=Tick;Time.captureDeltaTime=0;if(s!=null)s.Saves.SuspendEntry();File.WriteAllText(root+"/matrix.json",JsonUtility.ToJson(new Report{runs=runs.ToArray(),rows=rows.ToArray()},true));EditorApplication.Exit(code);}
 static IEnumerator Cases(){Time.captureDeltaTime=.04f;PlayerInput.Focus(true);
 foreach(int seed in new[]{config.seed})for(int rep=config.rep;rep<config.rep+1;rep++){
  s.Saves.InitializeStorage(root+"/Runs/"+seed+"-"+rep,false);s.Saves.enabled=false;AdventureRandom.SetNextSeedForValidation(seed);if(!s.Flow.Depart(AdventureMode.ForestExploration))throw new Exception("real forest departure "+s.Flow.Error);var region=s.GetComponent<ForestRegionRuntime>();var id=s.Flow.ActiveRun.runId;var profile=s.Flow.Profile.profileId;var runRecord=new RunRecord{seed=seed,rep=rep,run=id,profile=profile,layout=JsonUtility.ToJson(region.Capture()),streams=JsonUtility.ToJson(s.Flow.ActiveRun.randomStreams)};runs.Add(runRecord);
  // The valid initial v9 Run is saved. Arena resets below are deliberately not persisted.
  s.Flow.BeginLegacyRestore();var all=s.enemies.ToArray();var homes=all.ToDictionary(e=>e,e=>e.transform.position);var initialGear=g.items.Select(i=>i.Clone()).ToArray();string main=g.mainWeapon,reserve=g.reserveWeapon;
  foreach(int tier in new[]{0,1,2,3})foreach(string kind in new[]{"Melee","Ranged","MeleeRanged","MeleeElite","Triple","BossMixed","MeleePair","MultiMeleeRanged","Warden","Tide"})foreach(string policy in new[]{rep==0?"Stationary":rep==1?"Walk":"CorrectDodge"}){
   policyOverride=false;UnityEngine.Random.InitState(seed*10000+rep*1000+Array.IndexOf(new[]{"Melee","Ranged","MeleeRanged","MeleeElite","Triple","BossMixed","MeleePair","MultiMeleeRanged","Warden","Tide"},kind)*100+(policy=="Stationary"?1:policy=="CorrectDodge"?2:policy=="Walk"?3:policy=="WrongDodge"?4:5));foreach(var e in all)e.gameObject.SetActive(false);EnemyProjectile.Clear(s.gameObject.scene);foreach(var p in UnityEngine.Object.FindObjectsByType<LootPickup>(FindObjectsSortMode.None))UnityEngine.Object.Destroy(p.gameObject);yield return null;
   BAL02Fixture.Defense(h,tier);h.Progression.RestoreProgress(20,0,h.Progression.CaptureRanks(),new string[0],new int[0]);h.Inventory.RestoreInventory(new int[6],2,true,0);
   CombatEnemy primary=(kind=="BossMixed"||kind=="Warden"||kind=="Tide")?all.Single(e=>e.GetComponent<ForestBossBrain>()?.Kind==(kind=="Warden"?ForestBossKind.Warden:ForestBossKind.Tide)):all.First(e=>e.GetComponent<ForestBossBrain>()==null&&e.archetype==(kind=="Ranged"?EnemyArchetype.Ranged:EnemyArchetype.Melee));
   var cohort=new List<CombatEnemy>{primary};if(kind=="MeleePair"||kind=="MultiMeleeRanged")cohort.Add(all.First(e=>e!=primary&&e.archetype==EnemyArchetype.Melee));if(kind=="BossMixed"){cohort.Add(all.First(e=>e.archetype==EnemyArchetype.Melee&&e.GetComponent<EncounterGuard>()?.poi==primary.GetComponent<EncounterGuard>()?.poi));}if(kind=="MeleeRanged"||kind=="Triple"||kind=="MultiMeleeRanged")cohort.Add(all.First(e=>e.archetype==EnemyArchetype.Ranged));if(kind=="MeleeElite"||kind=="Triple")cohort.Add(all.First(e=>e.archetype==EnemyArchetype.Elite));policyOverride=rep==2&&kind=="Triple";
   Vector3 center=homes[primary];if(!NavMesh.SamplePosition(center+Vector3.back*2.1f,out var nav,4,NavMesh.AllAreas))throw new Exception("hero sample nav");h.RestorePosition(nav.position+Vector3.up*.1f,0);
   for(int i=0;i<cohort.Count;i++){var e=cohort[i];e.gameObject.SetActive(true);e.enabled=true;var where=i==0?homes[e]:h.transform.position+Quaternion.Euler(0,(i-1)*130+65,0)*Vector3.forward*2.8f;if(NavMesh.SamplePosition(where,out var p,3,NavMesh.AllAreas))where=p.position+Vector3.up*.1f;e.SetSpawn(where);e.ResetEnemy();e.transform.position=where;var guard=e.GetComponent<EncounterGuard>();if(guard!=null){guard.Configure(null,new ForestEncounter{id=e.entityId,home=where,poi="controlled",type=(int)e.archetype});guard.enabled=false;}var old=e.GetComponent<BossBrain>();if(old!=null)old.enabled=false;}
   Physics.SyncTransforms();g.Refresh();h.GetComponent<AdventureCombat>().ResetCombat();foreach(var e in cohort)e.target=null;float warmUntil=Time.time+2.12f;while(Time.time<warmUntil)yield return null;foreach(var e in cohort)e.target=h;
   string sample=seed+"-"+rep+"-"+tier+"-"+kind+"-"+policy;BAL01Trace.Reset(root+"/Samples/"+sample);var row=new Row{sample=sample,tier=tier,window=rep==0?30:90,seed=seed,rep=rep,run=id,profile=profile,type=kind,policy=policyOverride?"WrongDodge":policy,enemyCount=cohort.Count,initialLevel=h.Progression.Level,initialHP=h.Health.Current,initialDefense=h.Defense,initialPower=h.GetComponent<AdventureCombat>().BasicPower,initialRanks=string.Join(",",h.Progression.CaptureRanks()),initialPositions=string.Join("|",cohort.Select(e=>e.entityId+":"+e.transform.position))+"/hero:"+h.transform.position,gear=string.Join("|",g.items.Select(i=>i.id+":"+i.type+":"+i.quality+":"+i.board)),initialStreams=JsonUtility.ToJson(UnityEngine.Random.state)};
   float started=Time.time,stop=started+(rep==0?30:90);row.startedTime=started;float taken=h.Health.TotalDamageTaken;var combat=h.GetComponent<AdventureCombat>();
   while(Time.time<stop&&!h.Health.Dead&&cohort.Any(e=>!e.Health.Dead)){
    if(h.Progression.PendingChoices>0){if(!h.Progression.IsChoosing)h.Progression.BeginChoice();if(h.Progression.IsChoosing)h.Progression.SelectChoice(0);}
    if(h.Health.Current<h.Health.maxHealth*.4f){if(h.Inventory.UseHealthPotion())row.heals++;else if(h.Inventory.PotionCooldownRemaining>0)row.potionRejected++;}
    var closest=cohort.Where(e=>!e.Health.Dead).OrderBy(e=>Vector3.Distance(e.transform.position,h.transform.position)).First();var toward=Vector3.ProjectOnPlane(closest.transform.position-h.transform.position,Vector3.up);if(toward.sqrMagnitude>.01f&&!h.Dodge.IsRolling)h.transform.rotation=Quaternion.LookRotation(toward);
    bool correct=policy=="CorrectDodge"&&!policyOverride&&cohort.Any(e=>{var boss=e.GetComponent<ForestBossBrain>();return boss!=null?boss.State==ForestBossState.Telegraph&&boss.Remaining<=.18f:e.WindingUp&&e.TelegraphRemaining<=.18f;});
    bool wrong=(policy=="WrongDodge"||policyOverride)&&cohort.Any(e=>{var boss=e.GetComponent<ForestBossBrain>();return boss!=null?boss.State==ForestBossState.Telegraph:e.WindingUp;});
    if((correct||wrong)&&h.Dodge.TryDodge(correct?h.transform.right:toward))row.dodges++;
    if(h.Dodge.IsRolling)h.MoveCharacter(h.Dodge.FrameDisplacement(Time.deltaTime));else if(policy=="Walk"){var direction=toward.magnitude>1.4f?toward.normalized:Vector3.Cross(Vector3.up,toward.normalized);h.MoveCharacter(direction*h.moveSpeed*h.MoveMultiplier*Time.deltaTime);}
    if(policy!="Stationary"&&combat.Strike())row.playerAttacks++;BAL01Trace.Frame(cohort.ToArray(),h);File.AppendAllText(root+"/Samples/"+sample+"/hp.csv",Time.time+","+Time.frameCount+","+h.Health.Current+","+h.Defense+","+h.Inventory.PotionCooldownRemaining+","+h.Progression.Level+","+h.Progression.Experience+"\n");yield return null;
   }
   // Let actual enemy Update perform the death/Xp payout; no direct reward submission.
   yield return null;row.playerDistance=BAL01Trace.PlayerDistance;row.enemyDistance=BAL01Trace.EnemyDistance;row.blockedSeconds=BAL01Trace.BlockedSeconds;row.repositionEpisodes=BAL01Trace.RepositionEpisodes;row.slotDenials=BAL01Trace.SlotDenials;row.finalHP=h.Health.Current;row.elapsed=Time.time-started;row.kills=cohort.Count(e=>e.Health.Dead);if(row.kills==cohort.Count)row.ttk=row.elapsed;row.outcome=h.Health.Dead?"PLAYER_DEAD":row.kills==cohort.Count?"ENEMIES_DEAD":"WINDOW_END_CENSORED";row.finalLevel=h.Progression.Level;row.damage=h.Health.TotalDamageTaken-taken;row.windups=BAL01Trace.Starts;row.releases=BAL01Trace.Releases;row.hitActions=BAL01Trace.HitActions;row.hitRate=row.releases==0?-1:(float)row.hitActions/row.releases;row.damageEvents=BAL01Trace.DamageEvents;row.dodgeBlocks=BAL01Trace.Dodges;row.experience=BAL01Trace.XP;row.xpMatchesWhenSurviving=h.Health.Dead||row.experience==cohort.Where(e=>e.Health.Dead).Sum(e=>e.experienceReward);row.finalRanks=string.Join(",",h.Progression.CaptureRanks());row.finalStreams=JsonUtility.ToJson(UnityEngine.Random.state);row.projectilesAtCutoff=UnityEngine.Object.FindObjectsByType<EnemyProjectile>(FindObjectsSortMode.None).Length;rows.Add(row);File.WriteAllText(root+"/matrix-progress.json",JsonUtility.ToJson(new Report{runs=runs.ToArray(),rows=rows.ToArray()},true));if(!row.xpMatchesWhenSurviving)throw new Exception("XP source ledger mismatch "+sample);BAL01Trace.Folder=null;
  }
 }
 }
}
