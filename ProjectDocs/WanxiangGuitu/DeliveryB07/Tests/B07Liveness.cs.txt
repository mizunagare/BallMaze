using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;
// Fixed-seed, controlled AI liveness/phase fixtures. Other NPCs are disabled;
// no kill, victory, normal adventure or native UI result is inferred from this suite.
public static class B07Liveness {
 static CombatSession s;static Keyboard kb;static ForestBossBrain brain;static CombatEnemy boss;static int test,phase;static float at,caseAt;static bool returned;static double deadline;
 static string Root=>Environment.GetEnvironmentVariable("UNITY_TEST_ROOT");static void Need(bool ok,string why){if(!ok)throw new Exception(why);}static void Mark(string m){File.AppendAllText(Root+"/b07-liveness.txt",m+"\n");Debug.Log("B07_LIVENESS_"+m);}
 public static void Run(){ValidationStorage.Root(Environment.GetCommandLineArgs(),true);deadline=EditorApplication.timeSinceStartup+420;EditorSceneManager.OpenScene("Assets/CombatPrototype/ForestRuins.unity");EditorApplication.update+=Tick;EditorApplication.EnterPlaymode();}
 static void Tick(){try{EditorApplication.QueuePlayerLoopUpdate();Application.runInBackground=true;if(EditorApplication.timeSinceStartup>deadline)throw new Exception("liveness deadline");if(!EditorApplication.isPlaying||Time.unscaledTime<.5f)return;
  if(s==null){s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s==null)return;s.Saves.autoSaveInterval=3600;s.Saves.InitializeStorage(Root+"/b07-liveness-save",false);Need(s.Flow.Depart(AdventureMode.ForestExploration),"depart");kb=InputSystem.AddDevice<Keyboard>();InputSystem.settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;InputSystem.settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;File.WriteAllText(Root+"/b07-liveness.csv","case,seed,kind,phase,ai,attack,releases,heroHP,bossHP,x,z\n");}
  InputSystem.QueueStateEvent(kb,new KeyboardState());if(test>=8){Mark("EIGHT_FIXED_SEED_HEALTH_PHASE_AI_RETURN_CASES_PASS_CONTROLLED_NOT_VICTORY");InputSystem.RemoveDevice(kb);EditorApplication.update-=Tick;EditorApplication.Exit(0);return;}
  if(phase==0){int seed=test<4?1:17;var region=s.GetComponent<ForestRegionRuntime>();region.GeneratePlayable(seed,true);foreach(var e in s.enemies){e.enabled=false;if(e.GetComponent<ForestBossBrain>()!=null)e.GetComponent<ForestBossBrain>().enabled=false;}var kind=test%4<2?ForestBossKind.Warden:ForestBossKind.Tide;boss=s.enemies.Single(e=>e.GetComponent<ForestBossBrain>()?.Kind==kind);brain=boss.GetComponent<ForestBossBrain>();boss.enabled=brain.enabled=true;bool low=test%2==1;if(low)boss.Health.RestoreHealth(boss.Health.maxHealth*.45f);s.hero.Health.ResetHealth();s.hero.Dodge.ResetDodge();s.hero.Inventory.SetOpen(false);s.hero.RestorePosition(boss.transform.position+Vector3.forward*4,180);at=caseAt=Time.time;phase=1;returned=false;return;}
  File.AppendAllText(Root+"/b07-liveness.csv",test+","+(test<4?1:17)+","+brain.Kind+","+phase+","+brain.State+","+brain.Attack+","+brain.Releases+","+s.hero.Health.Current+","+boss.Health.Current+","+boss.transform.position.x+","+boss.transform.position.z+"\n");Need(!s.hero.Health.Dead,"controlled dodge fixture died case="+test);Need(!boss.Health.Dead,"liveness fixture must not claim victory");
  if(phase==1){Need(Time.time-at<55,"AI did not release two attacks case="+test);if(brain.State==ForestBossState.Telegraph&&brain.Remaining<.3f){var away=Vector3.ProjectOnPlane(s.hero.transform.position-brain.LockedPoint,Vector3.up);if(brain.Attack==ForestBossAttack.AntlerCharge||brain.Attack==ForestBossAttack.WaterFan||brain.Attack==ForestBossAttack.SeedVolley)away=Vector3.Cross(Vector3.up,boss.transform.forward);s.hero.Dodge.TryDodge(away);}if(brain.Releases>=2){s.hero.RestorePosition(boss.GetComponent<EncounterGuard>().home+Vector3.left*36,0);phase=2;at=Time.time;}return;}
  if(brain.State==ForestBossState.Returning)returned=true;Need(Time.time-at<22,"return did not finish case="+test);if(brain.State==ForestBossState.Idle&&(returned||Vector3.Distance(boss.transform.position,boss.GetComponent<EncounterGuard>().home)<1.3f)){Need(Vector3.Distance(boss.transform.position,boss.GetComponent<EncounterGuard>().home)<1.3f,"idle away from home");float healthBefore=boss.Health.Current;boss.Health.Damage(1);Need(boss.Health.Current<healthBefore&&boss.GetComponent<EncounterGuard>().Engaged,"returned boss cannot re-engage");Mark("case="+test+" seed="+(test<4?1:17)+" kind="+brain.Kind+" enraged="+brain.Enraged+" releases="+brain.Releases+" seconds="+(Time.time-caseAt));test++;phase=0;}
 }catch(Exception e){Debug.LogException(e);EditorApplication.update-=Tick;EditorApplication.Exit(1);}}
}
