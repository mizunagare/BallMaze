using System;using System.IO;using UnityEngine;using UnityEngine.SceneManagement;

public class GameEntry:MonoBehaviour {
 public const string TitlePath="Assets/CombatPrototype/Title.unity",ForestPath="Assets/CombatPrototype/ForestRuins.unity";
 public static bool Loading {get;private set;}
 public static string Error {get;private set;}="";
 static bool requested,fresh;static string expected,selection,expectedLegacy;static EntryTransition transition;
 public string CurrentDirectory {get;private set;}public string LegacyDirectory {get;private set;}
 public SaveInfo Info {get;private set;}public string ConfirmationStamp {get;private set;}
 public static GameEntry Instance {get;private set;}
 public static bool LaunchRejected {get;private set;}
 public static string ValidateLaunch(string[] args)=>ValidationStorage.Root(args,Array.IndexOf(args,"-wanxiang-test-gameplay")>=0||Array.IndexOf(args,"-wanxiang-entry-evidence")>=0);
 public static string TestRoot=>ValidateLaunch(Environment.GetCommandLineArgs());
 [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.BeforeSceneLoad)] static void GuardLaunch(){try{ValidateLaunch(Environment.GetCommandLineArgs());}catch(Exception e){LaunchRejected=true;Error="隔离启动被拒绝："+e.Message;Debug.LogError("B09_ISOLATION_REJECTED "+e.Message);if(!Application.isEditor)Application.Quit(2);}}

 public static bool ExplicitGameplay {
  get{var args=Environment.GetCommandLineArgs();bool yes=Array.IndexOf(args,"-wanxiang-test-gameplay")>=0;if(yes)ValidationStorage.Root(args,true);return yes;}
 }
 [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.SubsystemRegistration)] static void ResetEntry(){Loading=requested=fresh=LaunchRejected=false;expected=selection=expectedLegacy=null;transition=null;Error="";Instance=null;}
 void Awake(){if(LaunchRejected){enabled=false;return;}Instance=this;Time.timeScale=0;Cursor.lockState=CursorLockMode.None;Cursor.visible=true;var root=TestRoot;LegacyDirectory=root==null?SaveLocationMigration.LegacyDirectory:Path.Combine(root,"Legacy","ForestRuins");CurrentDirectory=root==null?SaveLocationMigration.NewDirectory:Path.Combine(root,"WanxiangGuitu","ForestRuins");if(GetComponent<TitleMenu>()==null)gameObject.AddComponent<TitleMenu>();}
 void Start(){if(ExplicitGameplay){Loading=true;SceneManager.LoadSceneAsync(ForestPath);return;}Refresh();GamePreferences.Load(PreferencesPath).Apply(true);Record("title-ready");if(TestRoot!=null&&Array.IndexOf(Environment.GetCommandLineArgs(),"-wanxiang-entry-evidence")>=0)StartCoroutine(CaptureTitle());}
 System.Collections.IEnumerator CaptureTitle(){for(int i=0;i<40;i++)yield return null;ScreenCapture.CaptureScreenshot(Path.Combine(TestRoot,"title-player.png"));Record("title-rendered");}
 public string PreferencesPath=>Path.Combine(Path.GetDirectoryName(CurrentDirectory),"preferences.json");
 string Selection()=>SaveCatalog.Fingerprint(CurrentDirectory)+"|"+SaveCatalog.Fingerprint(LegacyDirectory);
 public void Refresh(){if(requested)return;try{string recovery=NewProfileTransaction.Recover(CurrentDirectory);Info=SaveCatalog.InspectLocations(LegacyDirectory,CurrentDirectory);if(recovery!=""){Info.canContinue=false;Info.status=SaveAvailability.Uncertain;Info.reason=recovery;}selection=Selection();}catch(Exception e){Info=new SaveInfo{status=SaveAvailability.Blocked,reason=e.Message};selection=null;}Loading=false;}
 public void AskNew(){if(Loading||requested)return;Refresh();try{ConfirmationStamp=Info.status==SaveAvailability.Blocked?null:SaveCatalog.Fingerprint(CurrentDirectory);}catch(Exception e){ConfirmationStamp=null;Error="无法核对存档："+e.Message;}}
 public bool ConfirmNew(){if(Loading||requested)return false;try{if(ConfirmationStamp==null){Error="请先查看新游戏确认说明。";return false;}if(SaveCatalog.Fingerprint(CurrentDirectory)!=ConfirmationStamp||selection!=Selection()){Error="确认期间存档变化，请重新检查。";ConfirmationStamp=null;Refresh();return false;}return Launch(true,ConfirmationStamp);}catch(Exception e){Error="存档检查未完成，原档保留："+e.Message;requested=false;Loading=false;return false;}}
 public bool ReadSelected(){if(Loading||requested)return false;try{if(Info==null||!Info.canContinue||selection!=Selection()){Error="所示档案已经变化或不可读取，请重新检查。";return false;}return Launch(false,SaveCatalog.Fingerprint(CurrentDirectory));}catch(Exception e){Error="存档检查未完成，原档保留："+e.Message;requested=false;Loading=false;return false;}}
 public void CancelNew(){ConfirmationStamp=null;}
 public bool Continue(){if(Loading||requested)return false;try{Refresh();if(!Info.canContinue){Error=Info.reason;return false;}return Launch(false,SaveCatalog.Fingerprint(CurrentDirectory));}catch(Exception e){Error="存档检查未完成，原档保留："+e.Message;requested=false;Loading=false;return false;}}
 static void Cover(){if(transition!=null)return;var g=new GameObject("场景过渡");DontDestroyOnLoad(g);transition=g.AddComponent<EntryTransition>();}
 bool Launch(bool create,string fingerprint){if(Loading||requested)return false;fresh=create;requested=true;expected=fingerprint;expectedLegacy=SaveCatalog.Fingerprint(LegacyDirectory);Error="";Loading=true;Cover();Record(create?"new-confirmed":"continue-selected");SceneManager.LoadSceneAsync(ForestPath);return true;}
 public bool ChooseCurrent(){try{var p=new SaveLocationResult{status=SaveLocationStatus.Conflict,canUseNew=SaveCatalog.Inspect(CurrentDirectory).canContinue,legacyFingerprint=SaveLocationMigration.CollectionFingerprint(LegacyDirectory),currentFingerprint=SaveLocationMigration.CollectionFingerprint(CurrentDirectory)};SaveLocationMigration.ChooseNew(LegacyDirectory,CurrentDirectory,p);Refresh();return Info.canContinue;}catch(Exception e){Error=e.Message;return false;}}
 public static bool HasRequest=>requested;
 public static void DirectReady(){Loading=false;}
 public static void Boot(CombatSaveSystem saves){try{var root=TestRoot;string current=root==null?SaveLocationMigration.NewDirectory:Path.Combine(root,"WanxiangGuitu","ForestRuins"),legacy=root==null?SaveLocationMigration.LegacyDirectory:Path.Combine(root,"Legacy","ForestRuins");if(SaveCatalog.Fingerprint(legacy)!=expectedLegacy)throw new IOException("确认后旧目录变化，请重新检查所选档案。");bool ok=saves.InitializeFromEntry(current,legacy,fresh,expected);if(!ok)throw new IOException(saves.Status);requested=false;Loading=false;var pref=GamePreferences.Load(Path.Combine(Path.GetDirectoryName(current),"preferences.json"));pref.Apply(false);Record("game-ready");}catch(Exception e){Error="进入游戏失败，原档保留："+e.Message;saves.SuspendEntry();requested=false;Loading=false;SceneManager.LoadSceneAsync(TitlePath);}}
 public static bool ReturnToTitle(CombatSession s){if(s==null||!s.Saves.IsInitialized)return false;s.Flow.ResolveTerminal();if(!s.Saves.SaveProgress(false))return false;s.Saves.SuspendEntry();Loading=true;Cover();Time.timeScale=0;SceneManager.LoadSceneAsync(TitlePath);return true;}
 void OnApplicationQuit(){Record("title-quit");}
 public bool Exit(){Record("title-exit-request");if(Application.isEditor){Error="编辑器中请使用停止 Play；不会关闭 Unity Editor。";return false;}Application.Quit(0);return true;}
 static void Record(string phase){if(LaunchRejected)return;var root=TestRoot;if(root==null)return;var file=Path.Combine(root,"entry-audit.jsonl");Directory.CreateDirectory(root);File.AppendAllText(file,JsonUtility.ToJson(new EntryAudit{phase=phase,utc=DateTime.UtcNow.ToString("o"),scene=SceneManager.GetActiveScene().name,status=Instance!=null?Instance.Info?.status.ToString():null,canContinue=Instance!=null&&Instance.Info?.canContinue==true,savePhase=Instance!=null?Instance.Info?.phase:null,profile=Instance!=null?Instance.Info?.profileId:null,run=Instance!=null?Instance.Info?.runId:null,gameObjects=UnityEngine.Object.FindObjectsByType<CombatSession>(FindObjectsSortMode.None).Length})+"\n");}
 [Serializable] class EntryAudit {public string phase,utc,scene,status,savePhase,profile,run;public bool canContinue;public int gameObjects;}
}
public sealed class EntryTransition:MonoBehaviour {
 void OnGUI(){if(!GameEntry.Loading){Destroy(gameObject);return;}int d=GUI.depth;var matrix=GUI.matrix;var color=GUI.color;GUI.depth=-1000;GUI.matrix=Matrix4x4.identity;GUI.color=Color.black;GUI.DrawTexture(new Rect(0,0,Screen.width,Screen.height),Texture2D.whiteTexture,ScaleMode.StretchToFill);GUI.color=Color.white;GUI.Label(new Rect(Screen.width/2-120,Screen.height/2-20,240,40),"正在读取完整进度…");GUI.color=color;GUI.matrix=matrix;GUI.depth=d;}
}
