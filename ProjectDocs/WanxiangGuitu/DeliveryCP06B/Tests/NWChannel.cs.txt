using System;using System.IO;using System.Linq;using System.Reflection;using System.Collections;using System.Collections.Generic;using UnityEngine;
public static class NWChannel {
 [Serializable]class Drop{public bool gear;public int kind,amount,type,quality;public string definition,actualID;public int[] affixes;public float attack,defense;}[Serializable]class Result{public string run,region,entity,content,before,after,globalBefore,globalAfter;public int seed,xpBefore,xpAfter;public Drop[] legacy;public int additional;public string newID;}
 [Serializable]class Expected{public string run,entity,newID,interaction,streams,gear,profile;public bool picked;}
 static CombatSession s;static ThirdPersonHero h;static AdventureCombat c;
 static string Items()=>string.Join("|",c.Gear.items.Select(x=>x.id).OrderBy(x=>x));
 static void Disable(){foreach(var e in s.enemies){e.enabled=false;var b=e.GetComponent<ForestBossBrain>();if(b!=null)b.enabled=false;}s.Events.enabled=false;}
 static void Package(string name,NWProbe p,Expected expected){string dest=p.Root+"/"+name;Directory.CreateDirectory(dest);string source=Path.GetDirectoryName(s.Saves.SavePath);foreach(var f in Directory.GetFiles(source,"*",SearchOption.AllDirectories)){string rel=f.Substring(source.Length+1);string target=Path.Combine(dest,rel);Directory.CreateDirectory(Path.GetDirectoryName(target));File.Copy(f,target,false);}File.WriteAllText(p.Root+"/"+name+"-expected.json",JsonUtility.ToJson(expected,true));}
 public static IEnumerator Run(NWProbe p,string mode){yield return p.Wait(()=>GameEntry.Instance!=null,15);GameEntry.Instance.Refresh();p.Check(mode+"-continue",GameEntry.Instance.Continue());yield return p.Wait(()=>UnityEngine.Object.FindFirstObjectByType<CombatSession>()?.Saves?.IsInitialized==true&&UnityEngine.Object.FindFirstObjectByType<CombatSession>()?.WorldReady==true&&UnityEngine.Object.FindFirstObjectByType<CombatSession>()?.Flow?.ActiveRun!=null,30);s=UnityEngine.Object.FindFirstObjectByType<CombatSession>();if(s?.WorldReady!=true||s.Flow.ActiveRun==null)throw new Exception("Actual initialized forest required");h=s.hero;c=h.GetComponent<AdventureCombat>();Disable();var run=s.Flow.ActiveRun;
 if(mode=="channel-ground"||mode=="channel-picked"){var expected=JsonUtility.FromJson<Expected>(File.ReadAllText(p.Root+"/channel-expected.json"));for(int pass=0;pass<3;pass++){p.Check(mode+"-same-run-"+pass,run.runId==expected.run&&JsonUtility.ToJson(run.randomStreams)==expected.streams);var dead=s.enemies.Single(e=>e.entityId==expected.entity);p.Check(mode+"-dead-restored-"+pass,dead.Health.Dead);dead.DropLootOnce();dead.DropLootOnce();var gear=UnityEngine.Object.FindObjectsByType<LootPickup>().Where(x=>x.Gear?.id==expected.newID).ToArray();p.Check(mode+"-unique-location-"+pass,expected.picked?gear.Length==0&&c.Gear.items.Count(x=>x.id==expected.newID)==1:gear.Length==1&&gear[0].InteractionId==expected.interaction&&!c.Gear.items.Any(x=>x.id==expected.newID));p.Check(mode+"-same-items-"+pass,Items()==expected.gear);p.Check(mode+"-same-profile-"+pass,JsonUtility.ToJson(s.Flow.Profile)==expected.profile);if(pass<2){p.Check(mode+"-real-reload-"+pass,s.Saves.LoadProgress(false));yield return p.Wait(()=>s.WorldReady,20);Disable();run=s.Flow.ActiveRun;}}yield break;}
 var target=s.enemies.FirstOrDefault(e=>!e.Health.Dead&&e.GetComponent<ForestBossBrain>()==null&&NewWeaponDrop.Preview(run,e.entityId,e.archetype)!=null);if(target==null)throw new Exception("Frozen checkpoint has no eligible natural target; never silently reroll");var expectedGear=NewWeaponDrop.Preview(run,target.entityId,target.archetype);p.Check(mode+"-natural-target",!string.IsNullOrEmpty(target.entityId),"real saved entity "+target.entityId);p.Check(mode+"-old-sword-equip",c.Gear.SelectMain(c.Gear.items.First(x=>x.type==GearType.Sword&&x.InBag)));yield return new WaitForSeconds(.3f);var cc=target.GetComponent<CharacterController>();cc.enabled=false;target.transform.position=h.transform.position+h.transform.forward*1.3f;cc.enabled=true;PlayerInput.Focus(true);yield return null;yield return null;var ids=new HashSet<string>(UnityEngine.Object.FindObjectsByType<LootPickup>().Select(x=>x.InteractionId));var result=new Result{run=run.runId,region=run.regionInstanceId,entity=target.entityId,content=run.contentRevision,seed=run.seed,before=JsonUtility.ToJson(run.randomStreams),globalBefore=JsonUtility.ToJson(UnityEngine.Random.state),xpBefore=h.Progression.Experience};int attempt=0;while(!target.Health.Dead&&attempt<8){p.Check(mode+"-real-strike-"+attempt,c.Strike());attempt++;yield return new WaitForSeconds(c.BasicInterval+.1f);}p.Check(mode+"-actual-death",target.Health.Dead&&target.Health.TotalDamageTaken>0);if(!target.Health.Dead)throw new Exception("Physical attacks failed to kill frozen target");if(mode=="channel-control"){typeof(CombatEnemy).GetMethod("DropLootCore",BindingFlags.Instance|BindingFlags.NonPublic).Invoke(target,new object[]{null});p.Check(mode+"-reference-scope",true,"exact baseline-verified real product legacy core; controlled reference only, not normal AI or shadow rewards");}target.enabled=true;yield return null;yield return null;for(int choice=0;choice<8&&(h.Progression.IsChoosing||h.Progression.PendingChoices>0);choice++){if(h.Progression.IsChoosing)h.Progression.SelectChoice(0);yield return null;}var all=UnityEngine.Object.FindObjectsByType<LootPickup>().Where(x=>!ids.Contains(x.InteractionId)).ToArray();result.legacy=all.Where(x=>x.Gear==null||!NewWeaponRegistry.Is(x.Gear.type)).Select(x=>new Drop{gear=x.Gear!=null,kind=(int)x.Kind,amount=x.Amount,type=x.Gear!=null?(int)x.Gear.type:-1,quality=x.Gear!=null?(int)x.Gear.quality:-1,definition=x.Gear?.definitionId,actualID=x.Gear?.id??x.InteractionId,affixes=x.Gear?.affixes,attack=x.Gear?.Attack??0,defense=x.Gear?.Defense??0}).OrderBy(x=>x.gear).ThenBy(x=>x.kind).ThenBy(x=>x.type).ToArray();var added=all.Where(x=>x.Gear!=null&&NewWeaponRegistry.Is(x.Gear.type)).ToArray();result.additional=added.Length;result.after=JsonUtility.ToJson(run.randomStreams);result.globalAfter=JsonUtility.ToJson(UnityEngine.Random.state);result.xpAfter=h.Progression.Experience;result.newID=expectedGear.id;File.WriteAllText(p.Root+"/channel-pair.json",JsonUtility.ToJson(result,true));p.Check(mode+"-global-unchanged",result.globalBefore==result.globalAfter);p.Check(mode+"-additional-count",mode=="channel-control"?added.Length==0:added.Length==1&&added[0].Gear.id==expectedGear.id);if(mode=="channel-control")yield break;
 p.Check("channel-ground-real-model",added[0].transform.Find("猫族世界掉落模型")!=null&&added[0].GetComponentsInChildren<Collider>().All(x=>!x.enabled),"equipped model also used as nonblocking real drop visual");p.Check("channel-ground-not-credited",!c.Gear.items.Any(x=>x.id==expectedGear.id)&&!s.Flow.Profile.warehouse.Any(x=>x.id==expectedGear.id));int count=all.Length;string states=result.after;target.DropLootOnce();target.DropLootOnce();p.Check("channel-repeat-death",UnityEngine.Object.FindObjectsByType<LootPickup>().Count(x=>!ids.Contains(x.InteractionId))==count&&states==JsonUtility.ToJson(run.randomStreams));p.Check("channel-save-ground",s.Saves.SaveProgress(false));Package("GroundPackage",p,new Expected{run=run.runId,entity=target.entityId,newID=expectedGear.id,interaction=added[0].InteractionId,streams=states,gear=Items(),profile=JsonUtility.ToJson(s.Flow.Profile),picked=false});p.Check("channel-real-pickup",added[0].PickUp(h.Inventory));p.Check("channel-pickup-twice-rejected",!added[0].PickUp(h.Inventory));p.Check("channel-save-picked",s.Saves.SaveProgress(false));Package("PickedPackage",p,new Expected{run=run.runId,entity=target.entityId,newID=expectedGear.id,interaction=added[0].InteractionId,streams=JsonUtility.ToJson(run.randomStreams),gear=Items(),profile=JsonUtility.ToJson(s.Flow.Profile),picked=true});
 }
}
