using System;using System.IO;using System.Linq;using System.Collections.Generic;using System.Diagnostics;using System.Reflection;using System.Security.Cryptography;using UnityEngine;using UnityEngine.SceneManagement;using UnityEditor;
// Editor-only observer. It never restores or mutates product state. The only input
// override is the existing pre-departure validation-seed hook, explicitly recorded.
public static class C02R1Trace {
 static string root,phase="boot";static int sequence,depth,ticks;static bool attached;
 [Serializable]class Ref {public string id,sceneHandle,name,scene;public bool active,enabled,exists;}
 [Serializable]class Actor {public string enemy,brain,sceneHandle,regionRoot,parentRoot,id,kind,scene,regionInstance;public bool active,enemyEnabled,brainEnabled,brainInitialized,controllerEnabled,dead,inSessionArray;public float hp;}
 [Serializable]class Forest {public Ref component,root;public bool ready,active,matchesActors;public ForestLayout layout;}
 [Serializable]class Disk {public string path,sha256,error;public bool exists;public long envelopeRevision;public int version;public bool hasRun,hasSettlement;public RunState run;}
 [Serializable]class Envelope {public int version;public long revision;public string payload;}
 [Serializable]class Snapshot {public int sequence,pid,frame,depth,tick;public string utc,unity,method,phase,randomBefore,randomAfter,streamsBefore,streamsAfter;public double editorTime;public float gameTime;public Ref session;public bool savesReady,savesEnabled,heroEnabled,heroDead,flowEnabled,tracking,applying,atCamp;public Vector3 heroPosition;public TutorialState tutorial;public string saveCaller;public long revision;public RunState run;public Ref[] sessions;public Forest[] forests;public Actor[] sessionEnemies,actualEnemies;public Disk[] disks;}
 static Ref Identity(Component c){if(c==null)return new Ref();return new Ref{exists=true,id=c.GetEntityId().ToString(),name=c.name,scene=c.gameObject.scene.name,sceneHandle=c.gameObject.scene.handle.GetRawData().ToString(),active=c.gameObject.activeInHierarchy,enabled=!(c is Behaviour)||((Behaviour)c).enabled};}
 static Ref Identity(GameObject g){if(g==null)return new Ref();return new Ref{exists=true,id=g.GetEntityId().ToString(),name=g.name,scene=g.scene.name,sceneHandle=g.scene.handle.GetRawData().ToString(),active=g.activeInHierarchy};}
 static bool FieldExists(object o,string name)=>o.GetType().GetField(name,BindingFlags.Instance|BindingFlags.NonPublic)?.GetValue(o) is UnityEngine.Object value&&value!=null;
 static Actor State(CombatEnemy e,CombatSession s){if(e==null)return new Actor{id="<destroyed>"};var b=e.GetComponent<ForestBossBrain>();var regions=UnityEngine.Object.FindObjectsByType<ForestRegionRuntime>(FindObjectsInactive.Include,FindObjectsSortMode.None);var owner=regions.FirstOrDefault(f=>f.PresentationRoot!=null&&e.transform.IsChildOf(f.PresentationRoot.transform));var cc=e.GetComponent<CharacterController>();return new Actor{parentRoot=e.transform.root.GetEntityId().ToString(),enemy=e.GetEntityId().ToString(),brain=b==null?"none":b.GetEntityId().ToString(),id=e.entityId,kind=b==null?"none":b.Kind.ToString(),scene=e.gameObject.scene.name,sceneHandle=e.gameObject.scene.handle.GetRawData().ToString(),active=e.gameObject.activeInHierarchy,enemyEnabled=e.enabled,brainEnabled=b!=null&&b.enabled,brainInitialized=b!=null&&FieldExists(b,"enemy")&&FieldExists(b,"guard")&&FieldExists(b,"controller"),controllerEnabled=cc!=null&&cc.enabled,dead=e.Health!=null&&e.Health.Dead,hp=e.Health==null?-1:e.Health.Current,inSessionArray=s!=null&&s.enemies!=null&&s.enemies.Contains(e),regionRoot=owner?.PresentationRoot==null?"none":owner.PresentationRoot.GetEntityId().ToString(),regionInstance=owner==null?"unowned":owner.GetComponent<AdventureFlow>()?.ActiveRun?.regionInstanceId};}
 static Disk ReadDisk(string path){var d=new Disk{path=path,exists=File.Exists(path)};if(!d.exists)return d;try{var bytes=File.ReadAllBytes(path);d.sha256=BitConverter.ToString(SHA256.Create().ComputeHash(bytes)).Replace("-","").ToLowerInvariant();var e=JsonUtility.FromJson<Envelope>(System.Text.Encoding.UTF8.GetString(bytes).TrimStart('\uFEFF'));d.version=e.version;d.envelopeRevision=e.revision;var data=JsonUtility.FromJson<CombatSaveData>(e.payload);d.hasRun=data.hasActiveRun;d.hasSettlement=data.hasSettlement;d.run=data.activeRun;}catch(Exception e){d.error=e.ToString();}return d;}
 public static void Boot(string mode){root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);Directory.CreateDirectory(root);File.AppendAllText(root+"/observer.jsonl",JsonUtility.ToJson(new BootRow{utc=DateTime.UtcNow.ToString("o"),pid=Process.GetCurrentProcess().Id,mode=mode,unity=Application.unityVersion,method="B07Rules.Run",input="Existing SetNextSeedForValidation(975140455); fresh Run identity; not exact historical process/RNG/timing replay"})+"\n");SceneManager.sceneLoaded+=(s,m)=>Event("loaded",s);SceneManager.sceneUnloaded+=s=>Event("unloaded",s);SceneManager.activeSceneChanged+=(a,b)=>Event("active-change",b);Application.logMessageReceived+=(message,stack,type)=>{if(type==LogType.Exception)File.AppendAllText(root+"/exceptions.jsonl",JsonUtility.ToJson(new ErrorRow{utc=DateTime.UtcNow.ToString("o"),phase=phase,message=message,stack=stack})+"\n");};}
 [Serializable]class BootRow{public string utc,unity,mode,method,input;public int pid;}
 [Serializable]class ErrorRow{public string utc,phase,message,stack;}
 [Serializable]class SceneRow{public string utc,op,name,phase;public string handle;public int frame;public bool loaded;}
 static void Event(string op,Scene s){File.AppendAllText(root+"/scenes.jsonl",JsonUtility.ToJson(new SceneRow{utc=DateTime.UtcNow.ToString("o"),op=op,name=s.name,phase=phase,handle=s.handle.GetRawData().ToString(),frame=Time.frameCount,loaded=s.isLoaded})+"\n");}
 public static void Enter(){depth++;ticks++;}public static void Leave(){depth--;}
 public static void Attach(CombatSession s){if(attached)return;attached=true;var region=s.GetComponent<ForestRegionRuntime>();if(region!=null)region.ValidationHook+=p=>Take("region-hook:"+p,s);s.Flow.ValidationHook+=p=>Take("flow-hook:"+p,s);s.Saves.WriteValidationHook+=p=>Take("save-hook:"+p,s);Take("attached",s);}
 public static void Take(string label,CombatSession s){phase=label;try{string before=JsonUtility.ToJson(UnityEngine.Random.state),streams=JsonUtility.ToJson(s?.Flow?.ActiveRun?.randomStreams);var forests=UnityEngine.Object.FindObjectsByType<ForestRegionRuntime>(FindObjectsInactive.Include,FindObjectsSortMode.None);var actors=UnityEngine.Object.FindObjectsByType<CombatEnemy>(FindObjectsInactive.Include,FindObjectsSortMode.None);var row=new Snapshot{sequence=++sequence,pid=Process.GetCurrentProcess().Id,frame=Time.frameCount,depth=depth,tick=ticks,utc=DateTime.UtcNow.ToString("o"),unity=Application.unityVersion,method="B07Rules.Run",phase=label,editorTime=EditorApplication.timeSinceStartup,gameTime=Time.time,session=Identity(s),savesReady=s?.Saves?.IsInitialized==true,savesEnabled=s?.Saves?.enabled==true,heroEnabled=s?.hero?.enabled==true,heroDead=s?.hero?.Health?.Dead==true,heroPosition=s?.hero==null?Vector3.zero:s.hero.transform.position,tutorial=Onboarding.State(s),saveCaller=label.StartsWith("save-hook:")?Environment.StackTrace:"",flowEnabled=s?.Flow?.Enabled==true,tracking=s?.Flow?.Tracking==true,applying=s?.Flow?.Applying==true,atCamp=s?.Flow?.AtCamp==true,revision=s?.Flow?.Revision??-1,run=s?.Flow?.ActiveRun,sessions=UnityEngine.Object.FindObjectsByType<CombatSession>(FindObjectsInactive.Include,FindObjectsSortMode.None).Select(Identity).ToArray(),forests=forests.Select(f=>new Forest{component=Identity(f),root=Identity(f.PresentationRoot),ready=f.Ready,active=f.Active,layout=f.Layout,matchesActors=s!=null&&f.MatchesActors(s.enemies)}).ToArray(),sessionEnemies=s?.enemies?.Select(e=>State(e,s)).ToArray(),actualEnemies=actors.Select(e=>State(e,s)).ToArray(),disks=s?.Saves==null?new Disk[0]:new[]{ReadDisk(s.Saves.SavePath),ReadDisk(s.Saves.SavePath+".bak")},randomBefore=before,streamsBefore=streams};row.randomAfter=JsonUtility.ToJson(UnityEngine.Random.state);row.streamsAfter=JsonUtility.ToJson(s?.Flow?.ActiveRun?.randomStreams);File.AppendAllText(root+"/snapshots.jsonl",JsonUtility.ToJson(row)+"\n");}catch(Exception error){File.AppendAllText(root+"/observer-errors.txt",DateTime.UtcNow.ToString("o")+" "+label+" "+error+"\n");}}
 public static void Failure(Exception e,CombatSession s){Take("harness-exception",s);File.WriteAllText(root+"/first-harness-exception.json",JsonUtility.ToJson(new ErrorRow{utc=DateTime.UtcNow.ToString("o"),phase=phase,message=e.Message,stack=e.ToString()},true));}
}
