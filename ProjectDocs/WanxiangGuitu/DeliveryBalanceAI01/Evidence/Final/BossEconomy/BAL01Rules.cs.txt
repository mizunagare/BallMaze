 s.Flow.BeginLegacyRestore();h.Progression.Clear();h.Inventory.SetOpen(false);h.SaveMenuOpen=h.HudPanelOpen=false;h.Health.ResetHealth();h.Dodge.ResetDodge();Time.timeScale=1;yield return null;var g=h.GetComponent<AdventureLoadout>();var e2=s.Waves.CreateEncounter(s.transform,"bal:lifecycle",h.transform.position+Vector3.forward*1.2f,EnemyArchetype.Melee);e2.enabled=false;e2.transform.rotation=Quaternion.LookRotation(-Vector3.forward);float hp=h.Health.Current;e2.ReleaseAttack();C(hp==h.Health.Current,"no unprepared public release damage");var signal=e2.GetComponentInChildren<EnemyAttackSignal>();C(signal!=null&&signal.GetComponentsInChildren<Collider>().Length==0,"signal has no colliders");C(before==JsonUtility.ToJson(UnityEngine.Random.state),"warning/queue do not advance global random");e2.ReleaseAttack();C(h.Health.Current==hp,"no early public release");float until=Time.time+e2.windup+.04f;while(Time.time<until)yield return null;e2.ReleaseAttack();float after=h.Health.Current;C(after<hp,"actual valid front release damage");e2.ReleaseAttack();C(h.Health.Current==after,"duplicate same attack release causes no repeated hit");e2.ResetEnemy();C(!signal.GetComponent<LineRenderer>().enabled,"reset hides signal");var clone=UnityEngine.Object.Instantiate(e2.gameObject).GetComponent<CombatEnemy>();clone.enabled=false;clone.target=h;clone.transform.position=h.transform.position+Vector3.forward*1.2f;clone.ResetEnemy();method.Invoke(clone,null);C(clone.GetComponentsInChildren<EnemyAttackSignal>(true).Length==1,"pooled clone reuses one signal component");C(clone.GetComponentInChildren<EnemyAttackSignal>().GetComponent<LineRenderer>().sharedMaterial!=signal.GetComponent<LineRenderer>().sharedMaterial,"cloned warning warning");UnityEngine.Object.Destroy(clone.gameObject);UnityEngine.Object.Destroy(e2.gameObject);yield return null;C(e2==null&&clone==null,"warning owners destroy without residue");
 }
