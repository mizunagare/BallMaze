using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.Build.Reporting;
public static class BR2ProbeBuild{public static void Run(){try{var root=ValidationStorage.Root(Environment.GetCommandLineArgs(),true);var report=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{GameEntry.BootPath,GameEntry.TitlePath,GameEntry.ForestPath},locationPathName=root+"/Release/WanxiangGuitu.exe",target=BuildTarget.StandaloneWindows64,options=BuildOptions.None,extraScriptingDefines=new[]{"CP04BR2_PROBE"}});File.WriteAllText(root+"/probe-build.json",JsonUtility.ToJson(new BuildData{result=report.summary.result.ToString(),options=report.summary.options.ToString(),warnings=report.summary.totalWarnings,errors=report.summary.totalErrors,messages=report.steps.SelectMany(x=>x.messages.Select(y=>y.type+" "+y.content)).ToArray()},true));EditorApplication.Exit(report.summary.result==BuildResult.Succeeded?0:1);}catch(Exception e){Debug.LogException(e);EditorApplication.Exit(1);}}[Serializable]class BuildData{public string result,options;public int warnings,errors;public string[] messages;}}
