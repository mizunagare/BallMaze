public static class Perf02OS{
 [StructLayout(LayoutKind.Sequential)]struct FT{public uint low,high;public double Ms=>((ulong)high<<32|low)/10000.0;}
 [StructLayout(LayoutKind.Sequential)]struct MEM{public uint cb,faults;public UIntPtr peak,working,quotaPeakPaged,quotaPaged,quotaPeakNonPaged,quotaNonPaged,pagefile,peakPagefile,privateUsage;}
 [DllImport("kernel32.dll")]static extern IntPtr GetCurrentProcess();
 [DllImport("kernel32.dll")]static extern bool GetProcessTimes(IntPtr p,out FT created,out FT exited,out FT kernel,out FT user);
 [DllImport("psapi.dll")]static extern bool GetProcessMemoryInfo(IntPtr p,ref MEM m,uint cb);
 public struct Data{public long workingSet,privateBytes;public double kernelMs,userMs;}
 public static Data Read(){var d=new Data{workingSet=-1,privateBytes=-1,kernelMs=-1,userMs=-1};var m=new MEM{cb=(uint)Marshal.SizeOf(typeof(MEM))};var p=GetCurrentProcess();if(GetProcessMemoryInfo(p,ref m,m.cb)){d.workingSet=(long)m.working.ToUInt64();d.privateBytes=(long)m.privateUsage.ToUInt64();}if(GetProcessTimes(p,out var c,out var e,out var k,out var u)){d.kernelMs=k.Ms;d.userMs=u.Ms;}return d;}
}
