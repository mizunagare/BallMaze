 static void ReadPackageMetadata()
 {
  var flags=System.Reflection.BindingFlags.Static|System.Reflection.BindingFlags.Public|System.Reflection.BindingFlags.NonPublic;var lines=new List<string>();
  try{Type type=null;foreach(var name in new[]{"UnityEditor.CoreModule","UnityEditor.AssetStoreModule","UnityEditor"}){try{type=System.Reflection.Assembly.Load(name).GetType("UnityEditor.PackageInfo");}catch(System.IO.FileNotFoundException){}if(type!=null)break;}
   if(type==null){lines.Add("Native local PackageInfo type unavailable");}else{var method=type.GetMethod("GetPackageInfo",flags,null,new[]{typeof(string)},null);var info=method?.Invoke(null,new object[]{"I:/WGCombat/MAPTOOLS-1104-A/Inputs/MapMagic 2.unitypackage"});
    if(info==null)lines.Add("Native metadata unavailable");else{
     var instanceFlags=System.Reflection.BindingFlags.Public|System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Instance;
     lines.Add("metadata-field-schema="+string.Join(",",type.GetFields(instanceFlags).Select(f=>f.Name)));
     foreach(var key in new[]{"id","packageId","packageName","version","versionString","versionId","uploadId"}){var member=type.GetField(key,instanceFlags);var property=type.GetProperty(key,instanceFlags);if(member!=null)lines.Add(key+"="+Convert.ToString(member.GetValue(info),CultureInfo.InvariantCulture));else if(property!=null)lines.Add(key+"="+Convert.ToString(property.GetValue(info),CultureInfo.InvariantCulture));}
     // Local package header only. Never output whole jsonInfo or download/auth fields.
     string json=type.GetField("jsonInfo",instanceFlags)?.GetValue(info) as string;
     if(json!=null)foreach(var key in new[]{"id","version","version_id","title","upload_id","unity_version"}){var match=System.Text.RegularExpressions.Regex.Match(json,"\\\""+key+"\\\"\\s*:\\s*(\\\"[^\\\"]*\\\"|[0-9]+)");if(match.Success)lines.Add("local-header-"+key+"="+match.Groups[1].Value);}
    }
   }
  }catch(Exception e){lines.Add("Read-only metadata query unavailable: "+e.GetBaseException().GetType().Name);}
  if(lines.Count==0)lines.Add("NOT_RUN: no whitelisted metadata fields obtained");File.WriteAllLines(MT02Runtime.Output+"/native-package-metadata.txt",lines);
 }
